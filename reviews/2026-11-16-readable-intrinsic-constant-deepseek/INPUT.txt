\section{A variational formula for the second-moment limit}
\label{S:intrinsic-second-moment}

Retain every original covariance and amplitude. Write $a = \alpha_0$, $I =
(0,1)$, $K(s,t) = |s-t|^{-a}$ and $H_I = H_t\otimes H_{\mathrm{sp}}$ for the
original covariance. Keep the uniform $E$ mean $\mathcal{T}_R = T_R(C_t\otimes
I)$ and $C_{\rm mean} = M_0\|C_t\|$ of
Section~\ref{S:uniform-physical-products}, with exactly the fixed $r$ chosen
there. In particular, $2 < r < 2/a$. To distinguish it from the map below, write
$J_R = T_R$. For a bounded multiplier $V:H^{1/2} \to H^{-1/2}$ write
$B_E(V)(u,v) = (0,Vu)$, the $E$ operator of Lemma~\ref{L:uniform-E-lift}. The
physical mask is $m_R = \eta(\cdot/R)^2$, used once. Let
$\mathcal{C}_{\mathrm{pos}}$ be the full nonseparable class
in~\eqref{E:variational-positive-class}. For $\Psi$ in that class, retain its
unmasked mean $h_{\Psi,s}$, original variance $v_\Psi$, integrated full-gradient
rate $I_\Psi$ and variational values for nonnegative controls
$S_c^{\mathrm{pos}}$ from Section~\ref{S:unmasked-variational}. The value with
coefficient one is not identified with a first-moment limit or a constant in the
cited source. We assume $\Gamma \ne 0$ through the following comparison lemmas.
The theorem treats zero covariance directly.

\LeanMarker{L:positive-raw-embedding}
\begin{lemma}[Inclusion into the covariance space and the smooth mean]
  \label{L:positive-raw-embedding}
  For any real Hilbert space $G$ there is a bounded map $J_G:L^2(I;G) \to
  H_t\otimes G$ with
  \begin{equation}\label{E:positive-bochner-inclusion}
    \|J_GF\|^2
    =\int_{I^2}K(s,t)\langle F(s),F(t)\rangle_G\ud s\ud t
    \le K_0\|F\|_{L^2(I;G)}^2,\qquad K_0=\frac{2^a}{1-a}.
  \end{equation}
  The natural inclusion puts every $\Psi \in \mathcal{C}_{\mathrm{pos}}$ in
  $H_I$, with
  \begin{equation}\label{E:positive-raw-mean}
    \|\Psi\|_{H_I}^2=v_\Psi,\qquad
    \mathcal{T}_R\Psi(s)=B_E(m_Rh_{\Psi,s})\quad\hbox{a.e. }s.
  \end{equation}
  For each fixed smooth $\Psi$ the right side gives a canonical version for
  every $R \ge 1$ and every $s \in I$.
\end{lemma}

\begin{proof}
  Direct integration and concavity of $x^{1-a}$ give
  \[
    \int_I K(s,t)\ud t
    =\frac{s^{1-a}+(1-s)^{1-a}}{1-a}\le K_0 .
  \]
  For scalar $f \ge 0$, Cauchy--Schwarz against $K(s,t)\ud t$ and then Tonelli
  give $\|Kf\|_2 \le K_0\|f\|_2$. The same estimate for absolute values bounds
  the bilinear integral by $K_0\|f\|_2\|g\|_2$. Approximate scalar $L^2(I)$
  functions by real $C_c^\infty(I)$ functions. This bound makes their original
  test-null classes Cauchy in $H_t$, defines the bounded inclusion, and
  identifies its pairing with the integral. Finite $G$-valued tensors then give
  \eqref{E:positive-bochner-inclusion}. Simple fields are dense in Bochner
  $L^2(I;G)$; the same bound on differences extends $J_G$. Absolute bilinear
  continuity extends the integral identity as well.

  For $G = H_{\mathrm{sp}}$, the Bochner integral $\int_IK(s,t)F(t)\ud t$ exists
  almost everywhere: its norm is bounded by $K\|F\|(s)$, which belongs to $L^2$.
  Simple-field approximation and the Schur estimate give a strongly measurable
  $L^2$ version and a bounded map of norm at most $K_0$. On smooth finite
  tensors it agrees with $(C_t\otimes I)J_G$ by the covariance definition.
  Approximation by such tensors converges in $L^2(I;G)$. The integral converges
  there by Schur; the completed tensor map converges in $L^r(I;G)$ by
  Lemma~\ref{L:uniform-mean-tensor} and~\eqref{E:positive-bochner-inclusion},
  hence also in $L^2$. Uniqueness of the $L^2$ limit identifies these versions.
  In particular, the integral belongs to $L^r$.

  Choose a closed ball $B$ containing all spatial supports of $\Psi$. For real
  smooth $q$ supported there, the physical covariance gives
  \begin{equation}\label{E:positive-local-spatial-bound}
    \|q\|_{H_{\mathrm{sp}}}^2
    \le |B|\,\Gamma(B-B)\|q\|_\infty^2.
  \end{equation}
  Indeed, in $\int\ud x\int\Gamma(\ud y)q(x)q(x-y)$ only $x \in B,y \in B-B$
  contribute. Absolute values and local finiteness prove the bound, including
  white and singular Radon covariances. Thus, $t\mapsto\Psi_t$ is continuous,
  bounded and compactly supported as an $H_{\mathrm{sp}}$-valued field, and
  belongs to $L^2$. Its image under $J_{H_{\mathrm{sp}}}$ is the original test
  class of $\Psi$. Indeed, the separated compact smooth approximants in
  Lemma~\ref{L:cameron-supported-tensor} converge uniformly on one common
  compact support. The local bound above and the temporal Schur estimate give
  convergence both in $L^2(I;H_{\mathrm{sp}})$ and in the original covariance
  norm. Their finite tensor identifications therefore have this same limit.
  Formula~\eqref{E:positive-bochner-inclusion} and absolute Fubini give
  \[
    \|\Psi\|_{H_I}^2
    =\int_{I^2}K(s,t)\langle\Psi_s,\Psi_t\rangle_{H_{\mathrm{sp}}}
    \ud s\ud t
    =\int_{I\times\R^d}\Psi(s,x)h_{\Psi,s}(x)\ud x\ud s=v_\Psi.
  \]
  The compact spatial supports and~\eqref{E:positive-local-spatial-bound} make
  all these absolute integrals finite.

  Test the spatial row~\eqref{E:uniform-mean-row} first with displacement $f$
  and dual second component $\Lambda g$, with $f,g$ compact smooth. The row
  product is $m_Rfg$, so the preceding temporal identity gives
  \[
    \left\langle\int_IK(s,t)\Psi_t\ud t,m_Rfg\right\rangle_{H_{\mathrm{sp}}}
    =\int m_Rh_{\Psi,s}fg .
  \]
  This Fubini step has two fixed compact supports and integrable time kernel.
  Both resulting operators are bounded on $E$: the left by the mean theorem, the
  right by the bounded smooth $h_{\Psi,s}$ and compact smooth mask. Dense states
  extend the identity to $E$. For this fixed $\Psi$ the temporal integral exists
  at every $s$, with norm at most $K_0\sup_t\|\Psi_t\|_{H_{\mathrm{sp}}}$.
  Applying $J_R$ gives the explicit version in~\eqref{E:positive-raw-mean} for
  every $R,s$. For each $R$ it agrees with the completed $L^r$ mean class. This
  makes no simultaneous-version assertion over all rough controls. This proves
  Lemma~\ref{L:positive-raw-embedding}.
\end{proof}

\LeanMarker{L:positive-physical-gram}
\begin{lemma}[Finite covariance vectors and entrywise order]
  \label{L:positive-physical-gram}
  Take a finite partition of $I$ with nonempty cells $I_i$, lengths $\ell_i > 0$
  and $\sum_i\ell_i = 1$, and real $f_i \in C_c^\infty(\R^d)$ with $\|f_i\|_2 =
  1$. Put
  \[
    h_i=\mathbf1_{I_i}/\ell_i,\quad b_i=\|\nabla f_i\|_2^2,\quad
    k_i^R=h_i\otimes(m_Rf_i^2),\quad
    k_i^\infty=h_i\otimes f_i^2 .
  \]
  These are elements of $H_I$. If $v_L$ is the multiplier form
  of~\eqref{E:uniform-E-pairing}, then for every $\Phi \in H_I$,
  \begin{equation}\label{E:positive-label-row}
    \langle\Phi,k_i^R\rangle_{H_I}
    =\ell_i^{-1}\int_{I_i}v_{\mathcal{T}_R\Phi(s)}(f_i,f_i)\ud s .
  \end{equation}
  The Grams $C^R_{ij} = \langle k_i^R,k_j^R\rangle$ and $D_{ij} = \langle
  k_i^\infty,k_j^\infty\rangle$ satisfy $0 \le C^R_{ij} \le D_{ij}$ entrywise.
\end{lemma}

\begin{proof}
  Lemma~\ref{L:positive-raw-embedding} puts $h_i$ in $H_t$. The compact smooth
  spatial factors lie in $H_{\mathrm{sp}}$ by its test definition
  and~\eqref{E:positive-local-spatial-bound}. On a smooth finite tensor,
  use~\eqref{E:uniform-mean-row} with $z = (f_i,0)$ and $w = (0,\Lambda f_i)$.
  Its row is $(\eta_Rf_i)(\eta_R\Lambda^{-1}\Lambda f_i) = m_Rf_i^2$. Temporal
  integration gives the pairing with $h_i$. This
  proves~\eqref{E:positive-label-row} on dense tensors. Its right side is a
  bounded original functional, with bound $\ell_i^{-1/r}C_{\rm
  mean}\|\Phi\|\|f_i\|_{H^{1/2}}^2$ by Hölder
  and~\eqref{E:uniform-completed-mean}. The left side is bounded by Hilbert
  Cauchy--Schwarz. Thus, the identity extends to the completed $H_I$. The form
  is the displacement form by Lemma~\ref{L:uniform-E-lift}.

  Each Gram is symmetric positive semidefinite. Its entries factor into the
  temporal pairing of nonnegative $h_i,h_j$ and the physical spatial covariance
  pairing. The kernels $K,\Gamma$ and $f_i^2,f_j^2$ are nonnegative; $0 \le m_R
  \le 1$. Tonelli therefore gives the entrywise order. The physical pairing
  already contains $\Gamma$, and the row cancellation introduces no additional
  Fourier factor. Zero labels are permitted. All hypotheses of
  Proposition~\ref{P:gram-entrywise-comparison} are met, without assuming
  $D-C^R$ is positive semidefinite. This proves
  Lemma~\ref{L:positive-physical-gram}.
\end{proof}

By Proposition~\ref{P:gram-positive-optimizer}, the unmasked finite problem has
a maximizer built from the $k_i^\infty$. Their time factors $h_i$ are scaled
indicators, so this maximizer is in general not a test in
$\mathcal{C}_{\mathrm{pos}}$. We also need a bound on the controls that holds
uniformly in $R$.

\begin{lemma}[Uniform optimizer norm and positive smoothing]
  \label{L:positive-uniform-optimizer}
  Put $M = 4C_{\rm mean}+4$. The supremum defining $\mathcal{S}_R$ may be
  restricted to $\|\Phi\| \le M$, uniformly in $R$. For every finite trial
  family in Lemma~\ref{L:positive-physical-gram}, the unmasked finite
  variational expression has a maximizer $\Phi_D$ with $\|\Phi_D\| \le M$. It is
  zero or is a positive conic combination of the $k_i^\infty$. In the latter
  case it is approximable in $H_I$ by tests in $\mathcal{C}_{\mathrm{pos}}$.
\end{lemma}

\begin{proof}
  Put $C = C_{\rm mean}$ and $q = \|\Phi\|$.
  Proposition~\ref{P:uniform-rate-averaging} gives
  \[
    2\operatorname{Rate}_R(\Phi)-q^2/2
    \le2+2Cq-q^2/2<2-2q<0\quad(q>M).
  \]
  The zero control gives zero, proving the restriction.

  Denote the finite unmasked objective by $\mathfrak{F}_D$.
  Proposition~\ref{P:gram-positive-optimizer} supplies an attained maximum at
  zero or a positive conic sum $\Phi_D$. Choose a finite $R' \ge 1$ whose
  plateau contains all supports of $f_i$. Then $k_i^{R'} = k_i^\infty$. Write
  $W_i = \ell_i^{-1}\int_{I_i}\mathcal{T}_{R'}\Phi(s)\ud s$. The row identity
  and real Rayleigh formula imply, for every $\Phi$,
  \begin{align*}
    \mathfrak{F}_D(\Phi)
    &\le2\sum_i\ell_i\lambda(W_i)-q^2/2\\
    &\le2+2\sum_i\ell_i\|W_i\|-q^2/2
    \le2+2Cq-q^2/2 .
  \end{align*}
  For fixed $f_i$, the map $L\mapsto v_L(f_i,f_i)$ is bounded linear; hence the
  form of the Bochner average is the average form. Here $\lambda(L) \le
  \max(1,\|L\|) \le 1+\|L\|$, the norm of an average is bounded by the average
  norm, and the $L^1$ mean norm is at most its $L^r$ norm. No time-averaging
  error or concavity of the positive-part square root enters this estimate.
  Since the maximum is nonnegative, its optimizer lies in the same ball of
  radius $M$, independently of the partition, profiles and $R'$.

  On a nonzero branch write $\Phi_D = \sum_{i\in A}w_i h_i\otimes f_i^2$, with
  $w_i > 0$. Extend $h_i$ by zero, convolve with a nonnegative compact smooth
  approximate identity, and multiply by smooth cutoffs $0 \le \chi_j \le 1$
  compactly supported in $I$, tending pointwise to one. The resulting
  nonnegative $h_{i,j} \in C_c^\infty(I)$ converge to $h_i$ in $L^2$: the error
  is bounded by the convolution error plus $\|(\chi_j-1)h_i\|_2$, which tends to
  zero by domination. Set $\Psi_j = \sum_{i\in A}w_i h_{i,j}f_i^2$. Finite
  tensor estimates and~\eqref{E:positive-bochner-inclusion} give $\Psi_j \to
  \Phi_D$ in $H_I$. The positive conic sum has positive squared norm, since at
  least one eligible diagonal is positive and all Gram entries are nonnegative.
  Thus, $\Psi_j \ne 0$ eventually, so $\Psi_j \in \mathcal{C}_{\mathrm{pos}}$.
  Costs converge. The integrated rate continuity proved in
  Proposition~\ref{P:uniform-rate-averaging} gives
  \begin{equation}\label{E:positive-rate-continuity}
    |\operatorname{Rate}_{R'}(\Psi_j)-\operatorname{Rate}_{R'}(\Phi_D)|
    \le\sqrt{3+C\|\Psi_j\|+C\|\Phi_D\|}
    \sqrt{C\|\Psi_j-\Phi_D\|}\longrightarrow0 .
  \end{equation}
  This estimate is uniform in $R'$. It asserts no uniform derivative bound on
  the smooth approximants. The zero branch will use $S_2^{\mathrm{pos}} \ge 0$
  directly. This proves Lemma~\ref{L:positive-uniform-optimizer}.
\end{proof}

Lemma~\ref{L:positive-physical-gram} orders the masked and unmasked Grams
entrywise. Lemma~\ref{L:positive-uniform-optimizer} keeps the unmasked optimizer
in a fixed ball and, when it is nonzero, approximates it by nonnegative tests.
The time averaging of Proposition~\ref{P:uniform-rate-averaging} links these
finite problems to the rate.

\begin{proposition}[Signed variational values bounded by the variational value over nonnegative controls]
  \label{P:positive-signed-comparison}
  For every $R \ge 1$,
  \begin{equation}\label{E:positive-signed-comparison}
    \mathcal{S}_R\le S_2^{\mathrm{pos}} .
  \end{equation}
\end{proposition}

\begin{proof}
  Use uniform partitions with $n \ge 2$ cells of length $1/n$. They are the $B =
  1,\delta = 1/n$ partitions in Proposition~\ref{P:uniform-rate-averaging}, with
  the empty last cell omitted. That proposition gives $e_n \to 0$, independent
  of $R$, such that
  \[
    \left|\operatorname{Rate}_R(\Phi)
    -\sum_i\ell_i\lambda\left(\ell_i^{-1}
    \int_{I_i}\mathcal{T}_R\Phi(s)\ud s\right)\right|\le e_n
    \quad(\|\Phi\|\le M).
  \]
  Fix $R,\Phi,n$, with $\|\Phi\| \le M$, and denote these averages by $W_i$.
  Real compact smooth unit trials are dense in the real $H^1$ unit sphere:
  truncate by cutoffs tending to one, whose gradient errors tend to zero, then
  mollify in $H^1$ and normalize in $L^2$. Continuity of the multiplier form in
  $H^{1/2}$ and hence $H^1$ proves
  \[
    \lambda(W_i)=
    \sup_{\substack{f\in C_c^\infty(\R^d;\R)\\\|f\|_2=1}}
    \sqrt{(v_{W_i}(f,f)-\|\nabla f\|_2^2)_+}.
  \]
  The real Rayleigh supremum is nonnegative by
  Proposition~\ref{P:uniform-E-spectral}, so the positive part does not change
  its supremum. For $\varepsilon > 0$ choose one near-maximizing $f_i$ per cell
  with error at most $\varepsilon$ in the square root. This is a finite choice,
  not a measurable continuum optimizer. Since $\sum_i\ell_i = 1$, the weighted
  trial error contributes $2\varepsilon$ to the next inequality.
  Lemma~\ref{L:positive-physical-gram} gives
  \[
    2\operatorname{Rate}_R(\Phi)-\|\Phi\|^2/2
    \le\mathfrak{F}_{C^R}(\Phi)+2e_n+2\varepsilon .
  \]
  Optimize this finite objective and use entrywise Gram comparison:
  \[
    \mathfrak{F}_{C^R}(\Phi)
    \le\mathscr{S}(C^R)\le\mathscr{S}(D)=\mathfrak{F}_D(\Phi_D).
  \]
  This compares optimized values, not the same individual control.
  Lemma~\ref{L:positive-uniform-optimizer} proves $\|\Phi_D\| \le M$ before time
  averaging is used again. Choose $R'$ covering the finite supports as in that
  lemma. The label row, Rayleigh bound and the same uniform averaging estimate
  give
  \begin{equation}\label{E:positive-finite-optimizer-bound}
    \mathfrak{F}_D(\Phi_D)
    \le2\operatorname{Rate}_{R'}(\Phi_D)-\|\Phi_D\|^2/2+2e_n .
  \end{equation}

  If $\Phi_D = 0$, its finite variational expression is zero. Otherwise use its
  positive smooth approximants $\Psi_j$. Lemma~\ref{L:positive-raw-embedding}
  identifies their mean with $B_E(m_{R'}h_{\Psi_j,s})$ and their Gaussian cost
  with $v_{\Psi_j}$. Since $h_{\Psi_j,s} \ge 0$ and $0 \le m_{R'} \le 1$, real
  $H^1$ form order gives $\operatorname{Rate}_{R'}(\Psi_j) \le I_{\Psi_j}$.
  Thus, $2\operatorname{Rate}_{R'}(\Psi_j)-\|\Psi_j\|^2/2 \le
  S_2^{\mathrm{pos}}$. Rate and cost convergence pass this inequality to
  $\Phi_D$. Together with~\eqref{E:positive-finite-optimizer-bound} and the
  separate zero branch this yields
  \[
    2\operatorname{Rate}_R(\Phi)-\|\Phi\|^2/2
    \le S_2^{\mathrm{pos}}+4e_n+2\varepsilon .
  \]
  All finite choices were made after fixing $n,R,\Phi$. The right side is
  independent of those choices. Let $\varepsilon\downarrow0$, take the supremum
  over the original ball, use Lemma~\ref{L:positive-uniform-optimizer}, and then
  let $n \to \infty$. This proves~\eqref{E:positive-signed-comparison}.
  Finiteness of the variational value over nonnegative controls was already
  proved from the independent canonical upper bound in
  Proposition~\ref{P:variational-positive-scores}, not from this conclusion.
  This proves Proposition~\ref{P:positive-signed-comparison}.
\end{proof}

Proposition~\ref{P:positive-signed-comparison} bounds $\mathcal{S}_R$ above for
every $R$. For the limit we still need a lower bound as the mask is removed.

\begin{proposition}[Cutoff removal for the optimized variational expression]
  \label{P:positive-score-cutoff}
  The possibly nonmonotone mask family satisfies
  \[
    \lim_{R\to\infty}\mathcal{S}_R=S_2^{\mathrm{pos}}.
  \]
\end{proposition}

\begin{proof}
  Fix $\Psi \in \mathcal{C}_{\mathrm{pos}}$ before taking $R$ or moment limits.
  Use the explicit mean version of Lemma~\ref{L:positive-raw-embedding}. Its
  rate $\lambda_R(s)$ obeys $0 \le \lambda_R(s) \le \lambda_{\Psi,s}$ by real
  form order. For each real compact smooth unit $f$, $m_R = 1$ on its support
  for every sufficiently large $R$. The Rayleigh formula therefore gives
  \[
    \liminf_{R\to\infty}\lambda_R(s)^2
    \ge\int h_{\Psi,s}f^2-\|\nabla f\|_2^2.
  \]
  Compact trial density and~\eqref{E:variational-real-rayleigh} identify the
  supremum of the right side as $\lambda_{\Psi,s}^2$. Hence, $\lambda_R(s) \to
  \lambda_{\Psi,s}$ for these versions. The global smooth-source bound $H_\Psi =
  \sup_{s,x}h_{\Psi,s}(x) < \infty$ gives $\lambda_R(s) \le \sqrt{H_\Psi}$.
  Dominated convergence yields $\operatorname{Rate}_R(\Psi) \to I_\Psi$. Neither
  monotonicity of $m_R$ nor an unmasked ground eigenfunction was used.

  The original representative and cost $\|\Psi\|^2 = v_\Psi$ are independent of
  $R$. Thus, $\liminf_R\mathcal{S}_R \ge 2I_\Psi-v_\Psi/2$ for each fixed test.
  Given $\varepsilon > 0$, choose one fixed test with variational expression
  greater than $S_2^{\mathrm{pos}}-\varepsilon$; the finite supremum admits such
  a test. Take the liminf and then $\varepsilon\downarrow0$.
  Proposition~\ref{P:positive-signed-comparison} supplies the reverse limsup.
  This proves Proposition~\ref{P:positive-score-cutoff}.
\end{proof}

This passage from cutoff scores to their suprema needs the comparison
just proved. No-go~\ref{P:nogo-30} shows why pointwise convergence and
bounded control costs alone would not justify it.


\begin{theorem}[Intrinsic original second-moment constant]
  \label{T:intrinsic-second-moment}
  For every original admissible model with data $(1,0)$ and fixed $x$,
  \begin{equation}\label{E:intrinsic-second-moment-limit}
    \lim_{T\to\infty}T^{-(4-\alpha-a)/(3-\alpha)}
    \log\E|u_\Gamma(T,x)|^2
    =C_2(\Gamma,a)=S_2^{\mathrm{pos}}
    =2^{(4-\alpha)/(3-\alpha)}S_1^{\mathrm{pos}}.
  \end{equation}
  Its complete variational formula over nonnegative controls is
  \begin{align}
    C_2(\Gamma,a)
    &=\sup_{\Psi\in\mathcal{C}_{\mathrm{pos}}}
    \left\{2\int_I
    \sqrt{\sup_{\substack{f\in H^1(\R^d;\R)\\\|f\|_2=1}}
    \left(\int h_{\Psi,s}f^2-\|\nabla f\|_2^2\right)}
    \ud s-\frac{v_\Psi}{2}\right\},
    \label{E:intrinsic-positive-formula}\\
    h_{\Psi,s}(x)
    &=\int_I|s-t|^{-a}(\Gamma*\Psi_t)(x)\ud t,\qquad
    v_\Psi=\int_{I\times\R^d}\Psi(s,x)h_{\Psi,s}(x)\ud x\ud s.
    \notag
  \end{align}
  Here $\mathcal{C}_{\mathrm{pos}}$ consists of all nonzero nonnegative real
  $C_c^\infty(I\times\R^d)$ tests, including nonseparable ones. The covariance
  retains its amplitude, the kinetic term has coefficient one, and the Gaussian
  action is $v_\Psi/2$. For nonzero covariance $0 < C_2 < \infty$. For $\Gamma =
  0$ the constant and moment rate are zero.
\end{theorem}

\begin{proof}
  Assume first $\Gamma \ne 0$. Proposition~\ref{P:original-signed-point-upper},
  followed by Proposition~\ref{P:positive-score-cutoff} with $R =
  \delta^{-1}+2$, bounds the original point limsup by $S_2^{\mathrm{pos}}$.
  Proposition~\ref{P:variational-positive-scores} bounds the original point
  liminf below by this same value. Both are for the same canonical Stratonovich
  solution, exponent, original covariance and variance-half action. Their with
  the regularization removed first identifications retain all mixed, unequal,
  self and empty chains and both phase components with initial data $(1,0)$. The
  original spatial stationarity proved in
  Proposition~\ref{P:raw-block-original-lower} extends both bounds to every
  fixed $x$. Thus, the limit exists and equals $S_2^{\mathrm{pos}}$. Its
  homogeneity, positivity and finiteness are
  Proposition~\ref{P:variational-positive-scores}.
  Formula~\eqref{E:variational-real-rayleigh} and the definition of the original
  variance give~\eqref{E:intrinsic-positive-formula}.

  If $\Gamma = 0$, $H_I = \{0\}$, all means, variances and rates vanish,
  $\mathcal{S}_R = S_c^{\mathrm{pos}} = 0$, and $u = 1$. Both formulas are
  direct zero identities. This proves Theorem~\ref{T:intrinsic-second-moment}.
\end{proof}

The variational value with coefficient one is auxiliary; the theorem does not
identify it with a first-moment limit or with Chen's printed constants. It
neither repairs a source energy normalization nor asserts a Jensen improvement.
The exponent formally tends to $(4-\alpha)/(3-\alpha)$ as $a\downarrow0$; no
interchange with the static moment limit is claimed. The explicit bounds and
full calibration are separate from this identification.

\section{A variational formula for the second-moment limit}
\label{S:intrinsic-second-moment}

Retain every original covariance and amplitude. Write $a = \alpha_0$, $I =
(0,1)$, $K(s,t) = |s-t|^{-a}$ and $H_I = H_t\otimes H_{\mathrm{sp}}$ for the
original covariance. Keep the uniform $E$ mean $\mathcal{T}_R = T_R(C_t\otimes
I)$ and $C_{\rm mean} = M_0\|C_t\|$ of
Section~\ref{S:uniform-physical-products}, with exactly the fixed $r$ chosen
there. In particular, $2 < r < 2/a$. To distinguish it from the map below, write
$J_R = T_R$. For a bounded multiplier $V:H^{1/2} \to H^{-1/2}$ write
$B_E(V)(u,v) = (0,Vu)$, the $E$ operator of Lemma~\ref{L:uniform-E-lift}. The
physical mask is $m_R = \eta(\cdot/R)^2$, used once. Let
$\mathcal{C}_{\mathrm{pos}}$ be the full nonseparable class
in~\eqref{E:variational-positive-class}. For $\Psi$ in that class, retain its
unmasked mean $h_{\Psi,s}$, original variance $v_\Psi$, integrated full-gradient
rate $I_\Psi$ and variational values for nonnegative controls
$S_c^{\mathrm{pos}}$ from Section~\ref{S:unmasked-variational}. The value with
coefficient one is not identified with a first-moment limit or a constant in the
cited source. We assume $\Gamma \ne 0$ through the following comparison lemmas.
The theorem treats zero covariance directly.

\LeanMarker{L:positive-raw-embedding}
\begin{lemma}[Inclusion into the covariance space and the smooth mean]
  \label{L:positive-raw-embedding}
  For any real Hilbert space $G$ there is a bounded map $J_G:L^2(I;G) \to
  H_t\otimes G$ with
  \begin{equation}\label{E:positive-bochner-inclusion}
    \|J_GF\|^2
    =\int_{I^2}K(s,t)\langle F(s),F(t)\rangle_G\ud s\ud t
    \le K_0\|F\|_{L^2(I;G)}^2,\qquad K_0=\frac{2^a}{1-a}.
  \end{equation}
  The natural inclusion puts every $\Psi \in \mathcal{C}_{\mathrm{pos}}$ in
  $H_I$, with
  \begin{equation}\label{E:positive-raw-mean}
    \|\Psi\|_{H_I}^2=v_\Psi,\qquad
    \mathcal{T}_R\Psi(s)=B_E(m_Rh_{\Psi,s})\quad\hbox{a.e. }s.
  \end{equation}
  For each fixed smooth $\Psi$ the right side gives a canonical version for
  every $R \ge 1$ and every $s \in I$.
\end{lemma}

\begin{proof}
  We construct the covariance inclusion from the temporal Schur bound and
  identify its completed temporal mean. We then use common compact spatial
  supports to identify a smooth positive test with its original covariance
  class and physical masked mean.


  \medskip
  \noindent\textit{Step 1.~}We construct the Hilbert-valued covariance
  inclusion.
  Direct integration and concavity of $x^{1-a}$ give
  \[
    \int_I K(s,t)\ud t
    =\frac{s^{1-a}+(1-s)^{1-a}}{1-a}\le K_0 .
  \]
  For scalar $f \ge 0$, Cauchy--Schwarz against $K(s,t)\ud t$ and then Tonelli
  give $\|Kf\|_2 \le K_0\|f\|_2$. The same estimate for absolute values bounds
  the bilinear integral by $K_0\|f\|_2\|g\|_2$. Approximate scalar $L^2(I)$
  functions by real $C_c^\infty(I)$ functions. This bound makes their original
  test-null classes Cauchy in $H_t$, defines the bounded inclusion, and
  identifies its pairing with the integral. Finite $G$-valued tensors then give
  \eqref{E:positive-bochner-inclusion}. Simple fields are dense in Bochner
  $L^2(I;G)$; the same bound on differences extends $J_G$. Absolute bilinear
  continuity extends the integral identity as well.


  \medskip
  \noindent\textit{Step 2.~}We identify the temporal integral with the
  completed tensor map.
  For $G = H_{\mathrm{sp}}$, the Bochner integral $\int_IK(s,t)F(t)\ud t$ exists
  almost everywhere: its norm is bounded by $K\|F\|(s)$, which belongs to $L^2$.
  Simple-field approximation and the Schur estimate give a strongly measurable
  $L^2$ version and a bounded map of norm at most $K_0$. On smooth finite
  tensors it agrees with $(C_t\otimes I)J_G$ by the covariance definition.
  Approximation by such tensors converges in $L^2(I;G)$. The integral converges
  there by Schur; the completed tensor map converges in $L^r(I;G)$ by
  Lemma~\ref{L:uniform-mean-tensor} and~\eqref{E:positive-bochner-inclusion},
  hence also in $L^2$. Uniqueness of the $L^2$ limit identifies these versions.
  In particular, the integral belongs to $L^r$.


  \medskip
  \noindent\textit{Step 3.~}We identify the original norm of a smooth
  spacetime test.
  The two approximation limits agree in $L^2(I;H_{\mathrm{sp}})$: Schur
  controls the integral limit there, while the tensor estimate gives an $L^r$
  limit and $r>2$ implies $L^2$ convergence on the unit interval. Thus, the
  completed mean is identified without selecting pointwise representatives for
  arbitrary rough controls.
  Choose a closed ball $B$ containing all spatial supports of $\Psi$. For real
  smooth $q$ supported there, the physical covariance gives
  \begin{equation}\label{E:positive-local-spatial-bound}
    \|q\|_{H_{\mathrm{sp}}}^2
    \le |B|\,\Gamma(B-B)\|q\|_\infty^2.
  \end{equation}
  Indeed, in $\int\ud x\int\Gamma(\ud y)q(x)q(x-y)$ only $x \in B,y \in B-B$
  contribute. Absolute values and local finiteness prove the bound, including
  white and singular Radon covariances. Thus, $t\mapsto\Psi_t$ is continuous,
  bounded and compactly supported as an $H_{\mathrm{sp}}$-valued field, and
  belongs to $L^2$. Its image under $J_{H_{\mathrm{sp}}}$ is the original test
  class of $\Psi$. Indeed, the separated compact smooth approximants in
  Lemma~\ref{L:cameron-supported-tensor} converge uniformly on one common
  compact support. The local bound above and the temporal Schur estimate give
  convergence both in $L^2(I;H_{\mathrm{sp}})$ and in the original covariance
  norm. Their finite tensor identifications therefore have this same limit.
  Formula~\eqref{E:positive-bochner-inclusion} and absolute Fubini give
  \[
    \|\Psi\|_{H_I}^2
    =\int_{I^2}K(s,t)\langle\Psi_s,\Psi_t\rangle_{H_{\mathrm{sp}}}
    \ud s\ud t
    =\int_{I\times\R^d}\Psi(s,x)h_{\Psi,s}(x)\ud x\ud s=v_\Psi.
  \]
  The compact spatial supports and~\eqref{E:positive-local-spatial-bound} make
  all these absolute integrals finite.


  \medskip
  \noindent\textit{Step 4.~}We identify its masked operator mean on all phase
  states.
  Test the spatial row~\eqref{E:uniform-mean-row} first with displacement $f$
  and dual second component $\Lambda g$, with $f,g$ compact smooth. The row
  product is $m_Rfg$, so the preceding temporal identity gives
  \[
    \left\langle\int_IK(s,t)\Psi_t\ud t,m_Rfg\right\rangle_{H_{\mathrm{sp}}}
    =\int m_Rh_{\Psi,s}fg .
  \]
  This Fubini step has two fixed compact supports and integrable time kernel.
  Both resulting operators are bounded on $E$: the left by the mean theorem, the
  right by the bounded smooth $h_{\Psi,s}$ and compact smooth mask. Dense states
  extend the identity to $E$. For this fixed $\Psi$ the temporal integral exists
  at every $s$, with norm at most $K_0\sup_t\|\Psi_t\|_{H_{\mathrm{sp}}}$.
  Applying $J_R$ gives the explicit version in~\eqref{E:positive-raw-mean} for
  every $R,s$. For each $R$ it agrees with the completed $L^r$ mean class. This
  makes no simultaneous-version assertion over all rough controls. This proves
  Lemma~\ref{L:positive-raw-embedding}.
\end{proof}

Proposition~\ref{P:original-signed-point-upper} bounds the original point
limsup by a limsup of the signed values $\mathcal{S}_R$.
Proposition~\ref{P:variational-positive-scores} bounds the original point
liminf below by $S_2^{\mathrm{pos}}$. To close this gap in
Theorem~\ref{T:intrinsic-second-moment} we compare $\mathcal{S}_R$ with
$S_2^{\mathrm{pos}}$. We start with finite Gram matrices, with and without the
mask.

\LeanMarker{L:positive-physical-gram}
\begin{lemma}[Finite covariance vectors and entrywise order]
  \label{L:positive-physical-gram}
  Take a finite partition of $I$ with nonempty cells $I_i$, lengths $\ell_i > 0$
  and $\sum_i\ell_i = 1$, and real $f_i \in C_c^\infty(\R^d)$ with $\|f_i\|_2 =
  1$. Put
  \[
    h_i=\mathbf1_{I_i}/\ell_i,\quad b_i=\|\nabla f_i\|_2^2,\quad
    k_i^R=h_i\otimes(m_Rf_i^2),\quad
    k_i^\infty=h_i\otimes f_i^2 .
  \]
  These are elements of $H_I$. If $v_L$ is the multiplier form
  of~\eqref{E:uniform-E-pairing}, then for every $\Phi \in H_I$,
  \begin{equation}\label{E:positive-label-row}
    \langle\Phi,k_i^R\rangle_{H_I}
    =\ell_i^{-1}\int_{I_i}v_{\mathcal{T}_R\Phi(s)}(f_i,f_i)\ud s .
  \end{equation}
  The Grams $C^R_{ij} = \langle k_i^R,k_j^R\rangle$ and $D_{ij} = \langle
  k_i^\infty,k_j^\infty\rangle$ satisfy $0 \le C^R_{ij} \le D_{ij}$ entrywise.
\end{lemma}

\begin{proof}
  We realize the cell labels in the original tensor Hilbert space and identify
  their rows by completion. We then compare their Gram entries through the
  nonnegative temporal and physical spatial integrands.


  \medskip
  \noindent\textit{Step 1.~}We identify the finite label rows.
  Lemma~\ref{L:positive-raw-embedding} puts $h_i$ in $H_t$. The compact smooth
  spatial factors lie in $H_{\mathrm{sp}}$ by its test definition
  and~\eqref{E:positive-local-spatial-bound}. On a smooth finite tensor,
  use~\eqref{E:uniform-mean-row} with $z = (f_i,0)$ and $w = (0,\Lambda f_i)$.
  Its row is $(\eta_Rf_i)(\eta_R\Lambda^{-1}\Lambda f_i) = m_Rf_i^2$. Temporal
  integration gives the pairing with $h_i$. This
  proves~\eqref{E:positive-label-row} on dense tensors. Its right side is a
  bounded original functional, with bound $\ell_i^{-1/r}C_{\rm
  mean}\|\Phi\|\|f_i\|_{H^{1/2}}^2$ by Hölder
  and~\eqref{E:uniform-completed-mean}. The left side is bounded by Hilbert
  Cauchy--Schwarz. Thus, the identity extends to the completed $H_I$. The form
  is the displacement form by Lemma~\ref{L:uniform-E-lift}.


  \medskip
  \noindent\textit{Step 2.~}We compare the physical Gram integrands entrywise.
  The masked spatial pairing contains one factor $m_R$ at each of its two
  spatial positions. Both lie in $[0,1]$, so their product is at most one. The
  other factors are nonnegative squares, the temporal label functions and the
  original positive kernels. Removing those two mask factors therefore orders
  each entry separately, without an operator-order claim for the matrix
  difference.
  Each Gram is symmetric positive semidefinite. Its entries factor into the
  temporal pairing of nonnegative $h_i,h_j$ and the physical spatial covariance
  pairing. The kernels $K,\Gamma$ and $f_i^2,f_j^2$ are nonnegative; $0 \le m_R
  \le 1$. Tonelli therefore gives the entrywise order. The physical pairing
  already contains $\Gamma$, and the row cancellation introduces no additional
  Fourier factor. Zero labels are permitted. All hypotheses of
  Proposition~\ref{P:gram-entrywise-comparison} are met, without assuming
  $D-C^R$ is positive semidefinite. This proves
  Lemma~\ref{L:positive-physical-gram}.
\end{proof}

By Proposition~\ref{P:gram-positive-optimizer}, the unmasked finite problem has
a maximizer built from the $k_i^\infty$. Their time factors $h_i$ are scaled
indicators, so this maximizer is in general not a test in
$\mathcal{C}_{\mathrm{pos}}$. We also need a bound on the controls that holds
uniformly in $R$.

\begin{lemma}[Uniform optimizer norm and positive smoothing]
  \label{L:positive-uniform-optimizer}
  Put $M = 4C_{\rm mean}+4$. The supremum defining $\mathcal{S}_R$ may be
  restricted to $\|\Phi\| \le M$, uniformly in $R$. For every finite trial
  family in Lemma~\ref{L:positive-physical-gram}, the unmasked finite
  variational expression has a maximizer $\Phi_D$ with $\|\Phi_D\| \le M$. It is
  zero or is a positive conic combination of the $k_i^\infty$. In the latter
  case it is approximable in $H_I$ by tests in $\mathcal{C}_{\mathrm{pos}}$.
\end{lemma}

\begin{proof}
  We first restrict the signed score to a uniform control ball. We prove that
  the finite unmasked optimizer lies in that same ball before using any
  averaging estimate. Finally, its positive conic structure permits
  nonnegative smooth approximation with convergent cost and rate.


  \medskip
  \noindent\textit{Step 1.~}We restrict the signed optimization to a fixed
  ball.
  Put $C = C_{\rm mean}$ and $q = \|\Phi\|$.
  Proposition~\ref{P:uniform-rate-averaging} gives
  \[
    2\operatorname{Rate}_R(\Phi)-q^2/2
    \le2+2Cq-q^2/2<2-2q<0\quad(q>M).
  \]
  The zero control gives zero, proving the restriction.


  \medskip
  \noindent\textit{Step 2.~}We bound the finite unmasked optimizer uniformly.
  Denote the finite unmasked objective by $\mathfrak{F}_D$.
  Proposition~\ref{P:gram-positive-optimizer} supplies an attained maximum at
  zero or a positive conic sum $\Phi_D$. Choose a finite $R' \ge 1$ whose
  plateau contains all supports of $f_i$. Then $k_i^{R'} = k_i^\infty$. Write
  $W_i = \ell_i^{-1}\int_{I_i}\mathcal{T}_{R'}\Phi(s)\ud s$. The row identity
  and real Rayleigh formula imply, for every $\Phi$,
  \begin{align*}
    \mathfrak{F}_D(\Phi)
    &\le2\sum_i\ell_i\lambda(W_i)-q^2/2\\
    &\le2+2\sum_i\ell_i\|W_i\|-q^2/2
    \le2+2Cq-q^2/2 .
  \end{align*}
  For fixed $f_i$, the map $L\mapsto v_L(f_i,f_i)$ is bounded linear; hence the
  form of the Bochner average is the average form. Here $\lambda(L) \le
  \max(1,\|L\|) \le 1+\|L\|$, the norm of an average is bounded by the average
  norm, and the $L^1$ mean norm is at most its $L^r$ norm. No time-averaging
  error or concavity of the positive-part square root enters this estimate.
  Since the maximum is nonnegative, its optimizer lies in the same ball of
  radius $M$, independently of the partition, profiles and $R'$.


  \medskip
  \noindent\textit{Step 3.~}We smooth the positive temporal factors without
  changing the limit class.
  On a nonzero branch write $\Phi_D = \sum_{i\in A}w_i h_i\otimes f_i^2$, with
  $w_i > 0$. Extend $h_i$ by zero, convolve with a nonnegative compact smooth
  approximate identity, and multiply by smooth cutoffs $0 \le \chi_j \le 1$
  compactly supported in $I$, tending pointwise to one. The resulting
  nonnegative $h_{i,j} \in C_c^\infty(I)$ converge to $h_i$ in $L^2$: the error
  is bounded by the convolution error plus $\|(\chi_j-1)h_i\|_2$, which tends to
  zero by domination. Set $\Psi_j = \sum_{i\in A}w_i h_{i,j}f_i^2$. Finite
  tensor estimates and~\eqref{E:positive-bochner-inclusion} give $\Psi_j \to
  \Phi_D$ in $H_I$. The positive conic sum has positive squared norm, since at
  least one eligible diagonal is positive and all Gram entries are nonnegative.
  Thus, $\Psi_j \ne 0$ eventually, so $\Psi_j \in \mathcal{C}_{\mathrm{pos}}$.
  Costs converge. The integrated rate continuity proved in
  Proposition~\ref{P:uniform-rate-averaging} gives
  \begin{equation}\label{E:positive-rate-continuity}
    |\operatorname{Rate}_{R'}(\Psi_j)-\operatorname{Rate}_{R'}(\Phi_D)|
    \le\sqrt{3+C\|\Psi_j\|+C\|\Phi_D\|}
    \sqrt{C\|\Psi_j-\Phi_D\|}\longrightarrow0 .
  \end{equation}
  The smoothing changes only finitely many temporal factors, each converging
  in $L^2(I)$ and hence in its covariance norm. The spatial factors stay fixed
  and compactly supported. Consequently, the tensor norm convergence controls
  both the quadratic cost and, by the displayed rate estimate, the integrated
  rate; no uniform bound on derivatives of the approximants is needed.
  This estimate is uniform in $R'$. It asserts no uniform derivative bound on
  the smooth approximants. The zero branch will use $S_2^{\mathrm{pos}} \ge 0$
  directly. This proves Lemma~\ref{L:positive-uniform-optimizer}.
\end{proof}

Lemma~\ref{L:positive-physical-gram} orders the masked and unmasked Grams
entrywise. Lemma~\ref{L:positive-uniform-optimizer} keeps the unmasked optimizer
in a fixed ball and, when it is nonzero, approximates it by nonnegative tests.
The time averaging of Proposition~\ref{P:uniform-rate-averaging} links these
finite problems to the rate.

\begin{proposition}[Signed variational values bounded by the variational value over nonnegative controls]
  \label{P:positive-signed-comparison}
  For every $R \ge 1$,
  \begin{equation}\label{E:positive-signed-comparison}
    \mathcal{S}_R\le S_2^{\mathrm{pos}} .
  \end{equation}
\end{proposition}

\begin{proof}
  We approximate each integrated signed rate by a finite collection of smooth
  Rayleigh trials. Entrywise Gram comparison replaces the optimized masked
  value by an unmasked positive conic optimizer. Its uniform norm bound allows
  the same averaging error to be used again, after which positive smoothing
  gives the nonnegative-control bound.


  \medskip
  \noindent\textit{Step 1.~}We approximate the integrated rate uniformly on
  the fixed control ball.
  Use uniform partitions with $n \ge 2$ cells of length $1/n$. They are the $B =
  1,\delta = 1/n$ partitions in Proposition~\ref{P:uniform-rate-averaging}, with
  the empty last cell omitted. That proposition gives $e_n \to 0$, independent
  of $R$, such that
  \[
    \left|\operatorname{Rate}_R(\Phi)
    -\sum_i\ell_i\lambda\left(\ell_i^{-1}
    \int_{I_i}\mathcal{T}_R\Phi(s)\ud s\right)\right|\le e_n
    \quad(\|\Phi\|\le M).
  \]

  \medskip
  \noindent\textit{Step 2.~}We choose finitely many near-maximizing spatial
  trials.
  Fix $R,\Phi,n$, with $\|\Phi\| \le M$, and denote these averages by $W_i$.
  Real compact smooth unit trials are dense in the real $H^1$ unit sphere:
  truncate by cutoffs tending to one, whose gradient errors tend to zero, then
  mollify in $H^1$ and normalize in $L^2$. Continuity of the multiplier form in
  $H^{1/2}$ and hence $H^1$ proves
  \[
    \lambda(W_i)=
    \sup_{\substack{f\in C_c^\infty(\R^d;\R)\\\|f\|_2=1}}
    \sqrt{(v_{W_i}(f,f)-\|\nabla f\|_2^2)_+}.
  \]
  The real Rayleigh supremum is nonnegative by
  Proposition~\ref{P:uniform-E-spectral}, so the positive part does not change
  its supremum. For $\varepsilon > 0$ choose one near-maximizing $f_i$ per cell
  with error at most $\varepsilon$ in the square root. This is a finite choice,
  not a measurable continuum optimizer. Since $\sum_i\ell_i = 1$, the weighted
  trial error contributes $2\varepsilon$ to the next inequality.
  Lemma~\ref{L:positive-physical-gram} gives
  \[
    2\operatorname{Rate}_R(\Phi)-\|\Phi\|^2/2
    \le\mathfrak{F}_{C^R}(\Phi)+2e_n+2\varepsilon .
  \]

  \medskip
  \noindent\textit{Step 3.~}We compare optimized Gram values and return to an
  integrated rate.
  Optimize this finite objective and use entrywise Gram comparison:
  \[
    \mathfrak{F}_{C^R}(\Phi)
    \le\mathscr{S}(C^R)\le\mathscr{S}(D)=\mathfrak{F}_D(\Phi_D).
  \]
  This compares optimized values, not the same individual control.
  Lemma~\ref{L:positive-uniform-optimizer} proves $\|\Phi_D\| \le M$ before time
  averaging is used again. Choose $R'$ covering the finite supports as in that
  lemma. The label row, Rayleigh bound and the same uniform averaging estimate
  give
  \begin{equation}\label{E:positive-finite-optimizer-bound}
    \mathfrak{F}_D(\Phi_D)
    \le2\operatorname{Rate}_{R'}(\Phi_D)-\|\Phi_D\|^2/2+2e_n .
  \end{equation}

  The first averaging comparison contributes $2e_n$, and the return from the
  finite optimizer to its integrated rate contributes another $2e_n$. The
  optimizer norm was bounded independently of the chosen profiles and covering
  radius, so the second use has the same error bound. The finite trial choice
  contributes $2\varepsilon$ because the cell lengths sum to one.

  \medskip
  \noindent\textit{Step 4.~}We pass through nonnegative smooth controls and
  remove the two finite errors.
  If $\Phi_D = 0$, its finite variational expression is zero. Otherwise use its
  positive smooth approximants $\Psi_j$. Lemma~\ref{L:positive-raw-embedding}
  identifies their mean with $B_E(m_{R'}h_{\Psi_j,s})$ and their Gaussian cost
  with $v_{\Psi_j}$. Since $h_{\Psi_j,s} \ge 0$ and $0 \le m_{R'} \le 1$, real
  $H^1$ form order gives $\operatorname{Rate}_{R'}(\Psi_j) \le I_{\Psi_j}$.
  Thus, $2\operatorname{Rate}_{R'}(\Psi_j)-\|\Psi_j\|^2/2 \le
  S_2^{\mathrm{pos}}$. Rate and cost convergence pass this inequality to
  $\Phi_D$. Together with~\eqref{E:positive-finite-optimizer-bound} and the
  separate zero branch this yields
  \[
    2\operatorname{Rate}_R(\Phi)-\|\Phi\|^2/2
    \le S_2^{\mathrm{pos}}+4e_n+2\varepsilon .
  \]
  All finite choices were made after fixing $n,R,\Phi$. The right side is
  independent of those choices. Let $\varepsilon\downarrow0$, take the supremum
  over the original ball, use Lemma~\ref{L:positive-uniform-optimizer}, and then
  let $n \to \infty$. This proves~\eqref{E:positive-signed-comparison}.
  Finiteness of the variational value over nonnegative controls was already
  proved from the independent canonical upper bound in
  Proposition~\ref{P:variational-positive-scores}, not from this conclusion.
  This proves Proposition~\ref{P:positive-signed-comparison}.
\end{proof}

Proposition~\ref{P:positive-signed-comparison} bounds $\mathcal{S}_R$ above for
every $R$. For the limit we still need a lower bound as the mask is removed.

\begin{proposition}[Cutoff removal for the optimized variational expression]
  \label{P:positive-score-cutoff}
  The possibly nonmonotone mask family satisfies
  \[
    \lim_{R\to\infty}\mathcal{S}_R=S_2^{\mathrm{pos}}.
  \]
\end{proposition}

\begin{proof}
  We fix one smooth positive control and use compactly supported Rayleigh
  tests to remove the mask from its rate. We then approximate the already
  finite supremum by one such fixed control; the uniform signed comparison
  supplies the opposite bound.

  Fix $\Psi \in \mathcal{C}_{\mathrm{pos}}$ before taking $R$ or moment limits.
  Use the explicit mean version of Lemma~\ref{L:positive-raw-embedding}. Its
  rate $\lambda_R(s)$ obeys $0 \le \lambda_R(s) \le \lambda_{\Psi,s}$ by real
  form order. For each real compact smooth unit $f$, $m_R = 1$ on its support
  for every sufficiently large $R$. The Rayleigh formula therefore gives
  \[
    \liminf_{R\to\infty}\lambda_R(s)^2
    \ge\int h_{\Psi,s}f^2-\|\nabla f\|_2^2.
  \]
  Compact trial density and~\eqref{E:variational-real-rayleigh} identify the
  supremum of the right side as $\lambda_{\Psi,s}^2$. Hence, $\lambda_R(s) \to
  \lambda_{\Psi,s}$ for these versions. The global smooth-source bound $H_\Psi =
  \sup_{s,x}h_{\Psi,s}(x) < \infty$ gives $\lambda_R(s) \le \sqrt{H_\Psi}$.
  Dominated convergence yields $\operatorname{Rate}_R(\Psi) \to I_\Psi$. Neither
  monotonicity of $m_R$ nor an unmasked ground eigenfunction was used.

  For each fixed compact trial, all sufficiently large plateaus give exactly
  the unmasked potential integral on that trial. Taking the supremum of these
  trial lower bounds determines the limiting rate, even though the mask family
  itself need not be monotone. The uniform upper bound by the unmasked rate
  then supplies domination for its time integral.
  The original representative and cost $\|\Psi\|^2 = v_\Psi$ are independent of
  $R$. Thus, $\liminf_R\mathcal{S}_R \ge 2I_\Psi-v_\Psi/2$ for each fixed test.
  Given $\varepsilon > 0$, choose one fixed test with variational expression
  greater than $S_2^{\mathrm{pos}}-\varepsilon$; the finite supremum admits such
  a test. Take the liminf and then $\varepsilon\downarrow0$.
  Proposition~\ref{P:positive-signed-comparison} supplies the reverse limsup.
  This proves Proposition~\ref{P:positive-score-cutoff}.
\end{proof}

This passage from cutoff scores to their suprema needs the comparison
just proved. No-go~\ref{P:nogo-30} shows why pointwise convergence and
bounded control costs alone would not justify it.


\begin{theorem}[Intrinsic original second-moment constant]
  \label{T:intrinsic-second-moment}
  For every original admissible model with data $(1,0)$ and fixed $x$,
  \begin{equation}\label{E:intrinsic-second-moment-limit}
    \lim_{T\to\infty}T^{-(4-\alpha-a)/(3-\alpha)}
    \log\E|u_\Gamma(T,x)|^2
    =C_2(\Gamma,a)=S_2^{\mathrm{pos}}
    =2^{(4-\alpha)/(3-\alpha)}S_1^{\mathrm{pos}}.
  \end{equation}
  Its complete variational formula over nonnegative controls is
  \begin{align}
    C_2(\Gamma,a)
    &=\sup_{\Psi\in\mathcal{C}_{\mathrm{pos}}}
    \left\{2\int_I
    \sqrt{\sup_{\substack{f\in H^1(\R^d;\R)\\\|f\|_2=1}}
    \left(\int h_{\Psi,s}f^2-\|\nabla f\|_2^2\right)}
    \ud s-\frac{v_\Psi}{2}\right\},
    \label{E:intrinsic-positive-formula}\\
    h_{\Psi,s}(x)
    &=\int_I|s-t|^{-a}(\Gamma*\Psi_t)(x)\ud t,\qquad
    v_\Psi=\int_{I\times\R^d}\Psi(s,x)h_{\Psi,s}(x)\ud x\ud s.
    \notag
  \end{align}
  Here $\mathcal{C}_{\mathrm{pos}}$ consists of all nonzero nonnegative real
  $C_c^\infty(I\times\R^d)$ tests, including nonseparable ones. The covariance
  retains its amplitude, the kinetic term has coefficient one, and the Gaussian
  action is $v_\Psi/2$. For nonzero covariance $0 < C_2 < \infty$. For $\Gamma =
  0$ the constant and moment rate are zero.
\end{theorem}

\begin{proof}
  We compare the upper and lower rates for the same canonical solution, then
  identify the variational expression and its homogeneity. The zero-covariance
  case is handled directly.


  \medskip
  \noindent\textit{Step 1.~}We match the two finite bounds for nonzero
  covariance.
  Assume first $\Gamma \ne 0$. Proposition~\ref{P:original-signed-point-upper},
  followed by Proposition~\ref{P:positive-score-cutoff} with $R =
  \delta^{-1}+2$, bounds the original point limsup by $S_2^{\mathrm{pos}}$.
  Proposition~\ref{P:variational-positive-scores} bounds the original point
  liminf below by this same value. Both are for the same canonical Stratonovich
  solution, exponent, original covariance and variance-half action. Their with
  the regularization removed first identifications retain all mixed, unequal,
  self and empty chains and both phase components with initial data $(1,0)$. The
  original spatial stationarity proved in
  Proposition~\ref{P:raw-block-original-lower} extends both bounds to every
  fixed $x$. Thus, the limit exists and equals $S_2^{\mathrm{pos}}$. Its
  homogeneity, positivity and finiteness are
  Proposition~\ref{P:variational-positive-scores}.
  Formula~\eqref{E:variational-real-rayleigh} and the definition of the original
  variance give~\eqref{E:intrinsic-positive-formula}.

  The radius $\delta^{-1}+2$ tends to infinity, so cutoff-score convergence
  applies along this radius without a separate exchange of limits. The upper
  bound is therefore at most $S_2^{\mathrm{pos}}$, while the lower bound is at
  least that same finite value. Since every liminf is at most the
  corresponding limsup, these inequalities force equality and the full limit.
  The factor relating $S_2^{\mathrm{pos}}$ to $S_1^{\mathrm{pos}}$ is the
  control-dilation factor, not an identification with a first-moment constant.

  \medskip
  \noindent\textit{Step 2.~}We verify zero covariance directly.
  If $\Gamma = 0$, $H_I = \{0\}$, all means, variances and rates vanish,
  $\mathcal{S}_R = S_c^{\mathrm{pos}} = 0$, and $u = 1$. Both formulas are
  direct zero identities. This proves Theorem~\ref{T:intrinsic-second-moment}.
\end{proof}

The variational value with coefficient one is auxiliary; the theorem does not
identify it with a first-moment limit or with Chen's printed constants. It
neither repairs a source energy normalization nor asserts a Jensen improvement.
The exponent formally tends to $(4-\alpha)/(3-\alpha)$ as $a\downarrow0$; no
interchange with the static moment limit is claimed. The explicit bounds and
full calibration are separate from this identification.

\section{Model and standing assumptions}\label{S:model}

We study the hyperbolic Anderson model
\cite{chen:author2026:time-dependency}*{(1.1)}
\begin{equation}\label{E:HAM}
  \frac{\partial^2 u}{\partial t^2}(t,x)=\Delta u(t,x)+\W(t,x)u(t,x),
  \qquad u(0,x)=1,\quad \frac{\partial u}{\partial t}(0,x)=0,
\end{equation}
for $(t,x) \in \R_+\times\R^d$, $d \in \{1,2,3\}$, interpreted in the
Stratonovich sense (limit of classical wave equations driven by smoothed noises
$\W_{\epsilon,\delta}$). The noise is centered Gaussian with
\begin{equation}\label{E:cov}
  \operatorname{Cov}\big(\W(s,x),\W(t,y)\big)=|s-t|^{-\alpha_0}\gamma(x-y),
  \qquad 0<\alpha_0<1,
\end{equation}
where $\gamma \ge 0$ is nonnegative definite with spectral measure $\mu$.

\begin{hypothesis}[Homogeneous covariance]\label{H:homog}
  Let $d \in \{1,2,3\}$ and $0 < \alpha_0 < 1$. The spatial covariance is a
  nonnegative, even, positive-definite tempered Radon measure $\Gamma$ on
  $\R^d$, with positive even tempered Radon spectral measure $\mu$ in the
  convention
  \[
    \Gamma=\int e^{i\xi\cdot x}\mu(\ud\xi)
    \quad\text{in the sense of tempered distributions}.
  \]
  The symbol $\gamma$ in~\eqref{E:cov} denotes this covariance as a function
  when one exists, and as a distribution otherwise. For every Borel set $A$ and
  $c > 0$, $\Gamma(cA) = c^{d-\alpha}\Gamma(A)$, and
  \begin{equation}\label{E:threshold}
    \alpha_0+\alpha<2,
  \end{equation}
  with $0 < \alpha < d$, or $\alpha = d = 1$. In particular, the last case
  includes spatial white noise $\Gamma = \delta_0$.
\end{hypothesis}

\paragraph{Zero covariance.}
Hypothesis~\ref{H:homog} allows $\Gamma = 0$ by design. In this case the
centered Gaussian noise is $\W = 0$, and the solution with initial data
$(1,0)$ is $u(t,x) = 1$. Thus $\E|u(t,x)|^p = 1$ for every $p > 0$, and
all normalized logarithmic moment rates are zero. Results asserting strict
positivity of covariance-dependent quantities impose $\Gamma \ne 0$ locally.

\section{Uniform spectral rates and phase products}
\label{S:uniform-physical-products}

Retain every original covariance and amplitude. Set $E = H^{1/2}\times
H^{-1/2}$, $F = E\oplus\R^2$, and $m_R = \eta(\cdot/R)^2$, $R \ge 1$, with the
same fixed cutoff profile. Choose
\[
  \begin{aligned}
    &\frac12<\theta<\min(1-a/2,(3-a-\alpha)/2),\qquad \nu=1/\theta,\\
    &(1-\theta)^{-1}<r<2/a,\qquad \nu\le p<2<r .
  \end{aligned}
\]
Choose auxiliary exponents $0 < \gamma < \min(1/2,(2-\alpha)/2)$ and $\theta <
\beta < \min(1-a/2,(3-a-\alpha)/2)$ for the phase construction. The original
strict threshold makes these intervals nonempty. Write $i_I:H_I \to H_W$ for the
supported closed inclusion. For nonzero covariance the field is
\[
  \mathcal{T}_R=T_R(C_t\otimes I_{H_{\rm sp}}),\qquad
  \|\mathcal{T}_R\|_{H_I\to L^r(I;\mathcal{K}(E))}
  \le C_{\rm mean}=M_0\|C_t\| ,
\]
by Proposition~\ref{P:uniform-mean-regulator}. For zero covariance set $H_{\rm
sp} = H_I = \{0\}$ and $M_0 = C_{\rm mean} = 0$. Means and their rates are then
zero, and mean flows are free; no nonzero-noise estimate is needed for that
case. The residual comparisons below still act on the nonzero phase $E$.

\subsection{The operator class on E and its spectral constants}

For real smooth compact $q$ let $V_q = m_R(\Gamma*q)$, $M_q:H^{1/2} \to
H^{-1/2}$ be its multiplier and $L_q^E(u,v) = (0,M_qu)$. Let $\mathcal{V}_{E,R}$
be their operator-norm closure in $\mathcal{K}(E)$.

\begin{lemma}[Unique potential and fixed-radius lift]\label{L:uniform-E-lift}
  Each $L \in \mathcal{V}_{E,R}$ has the form $L(u,v) = (0,Mu)$, with $\|L\| =
  \|M\| = :k$, and determines a unique real $V \in H^{-1/2}$ supported in
  $\operatorname{supp}m_R$. The symmetric pairing satisfies
  \begin{equation}\label{E:uniform-E-pairing}
    |v_V(u,w)|\le k\|u\|_{H^{1/2}}\|w\|_{H^{1/2}} .
  \end{equation}
  It has a lift $L_F$ in the exact class of Section~\ref{S:signed-static-wave},
  with $L_Fi = iL$. Lift constants may depend on $R$ and are used only for
  identification. Here $v_V(u,w) = \langle Vu,w\rangle_{H^{-1/2},H^{1/2}}$ on
  real states, with its Hermitian extension on complex states.
\end{lemma}

\begin{proof}
  For zero covariance, $L = M = V = L_F = 0$ gives the claims directly. If
  $L_n^E \to L$, the zero columns persist and $M_n \to M$ in
  $\mathcal{L}(H^{1/2},H^{-1/2})$, with the stated norm equality. At fixed $R$
  choose real compact smooth $\chi_R = 1$ near $\operatorname{supp}m_R$. Since
  $M_n\chi_R = V_n$, $\|V_n-V_j\|_{H^{-1/2}} \le
  \|\chi_R\|_{H^{1/2}}\|M_n-M_j\|$. Thus, $V_n \to V$, with the same support by
  testing outside it. Smooth multiplication on $H^{-1/2}$ identifies $Mu = Vu$
  on compact tests, then on $H^{1/2}$ by density. The identity $M\chi_R = V$
  gives uniqueness; symmetry and~\eqref{E:uniform-E-pairing} pass to the limit.

  The original $F$ operators are $(u,v,c_1,c_2)\mapsto(0,M_nu+c_1V_n,0,0)$.
  Their differences are bounded by $(1+\|\chi_R\|_{H^{1/2}})\|M_n-M_j\|$. They
  therefore converge in the old $F$ class and give the lift. This bound is
  allowed to grow with $R$; it will enter no uniform estimate. Dense smooth
  original controls and bounded completion of $T_R$ also show that all completed
  mean fields lie in $\mathcal{V}_{E,R}$ almost everywhere. Bochner averages
  preserve this closed linear class. Complexification preserves $k$ by the
  Hilbert real/imaginary splitting. The real symmetric pairing becomes Hermitian
  and its diagonal splits into the two real diagonals. This proves
  Lemma~\ref{L:uniform-E-lift}.
\end{proof}

\begin{proposition}[Spectral rate in E]\label{P:uniform-E-spectral}
  For $L \in \mathcal{V}_{E,R}$ put
  \[
    \beta(V)=\inf_{\substack{u\in H^1\\\|u\|_2=1}}
    \{\|\nabla u\|_2^2-v_V(u,u)\},\qquad \lambda(L)=\sqrt{-\beta(V)} .
  \]
  If $K \ge \max(1,k)$, the physical wave satisfies, for $t \ge 0$,
  \begin{equation}\label{E:uniform-E-static-bound}
    -K^2\le\beta(V)=\inf\operatorname{spec}H_V\le0,\qquad
    \|P_V^E(t)\|\le8(1+K)^2(1+t)e^{\lambda(L)t}.
  \end{equation}
  If two such operators have norms at most $K$, then
  \begin{equation}\label{E:uniform-E-rate-continuity}
    |\beta(V)-\beta(W)|\le(K+2)\|L_V-L_W\|,\qquad
    |\lambda(L_V)-\lambda(L_W)|
    \le\sqrt{(K+2)\|L_V-L_W\|}.
  \end{equation}
  Every quantitative constant here uses the $E$ norm alone.
\end{proposition}

\begin{proof}
  We derive the bounds without transferring a quantitative $F$ bound. With $b =
  \|u\|_2$, $z = \|u\|_{H^1}$, Fourier Cauchy--Schwarz gives $\|u\|_{H^{1/2}}^2
  \le bz$. Equation~\eqref{E:uniform-E-pairing} and the half-weight
  arithmetic--geometric mean inequality give $|v_V(u,u)| \le Kbz \le
  z^2/2+K^2b^2/2$. The weights are $1/2,1/2$, bases $z^2,(Kb)^2 \ge 0$; positive
  powers include zero norms. For $q_V = \|\nabla u\|_2^2-v_V(u,u)$ and $c =
  2+K^2$,
  \begin{equation}\label{E:uniform-E-form-budget}
    \tfrac12z^2+(1+K^2/2)b^2
    \le q_V(u)+cb^2
    \le\tfrac32z^2+(1+3K^2/2)b^2
    \le3(1+K^2)z^2 .
  \end{equation}
  This is a densely defined closed semibounded Hermitian form on full $H^1$: its
  shifted norm is equivalent to the complete $H^1$ norm. The lower bound
  dominates $2b^2$, since $z \ge b$ and $K \ge 1$.
  From~\cite{teschl:14:mathematical}*{Theorem~2.14, p.~81, (2.63)} and
  Theorem~3.6, p.~109, (3.51)--(3.53), it represents a unique self-adjoint $H_V$
  with $A = H_V+c \ge 2$ and $D(A^{1/2}) = H^1$. Closedness uses the definition
  on that book's p.~80. The spectral meaning of $A \ge 2$ follows from bounded
  spectral slices: a nonzero slice below two would give a unit vector in $D(A)$
  contradicting the form lower bound. The bottom argument below uses the same
  representation.

  Put $B_0 = 1-\Delta$ and $L_K = 3(1+K^2)$. The common forms obey $\tfrac12B_0
  \le A \le L_KB_0$. For positive self-adjoint $T,S \ge 1$ with ordered forms,
  completing the square gives
  \[
    \langle f,(T+t)^{-1}f\rangle
    =\sup_u\{2\operatorname{Re}\langle f,u\rangle-q_T(u)-t\|u\|^2\}
    \ge\langle f,(S+t)^{-1}f\rangle .
  \]
  The identity $\sqrt x = \pi^{-1}\int_0^\infty t^{-1/2}x/(x+t)\ud t$ follows by
  $t = xv^2$. Positive spectral integration and Tonelli give square-root form
  order, with domain inclusions, from $I-t(T+t)^{-1} \le I-t(S+t)^{-1}$. Apply
  this to $(B_0,2A)$ and $(A,L_KB_0)$, both pairs at least one:
  \[
    D(A^{1/4})=H^{1/2},\qquad
    2^{-1/4}\|u\|_{H^{1/2}}\le\|A^{1/4}u\|_2
    \le L_K^{1/4}\|u\|_{H^{1/2}} .
  \]
  Under the $L^2$ pairing the dual norm is $\|A^{-1/4}v\|_2$, by Cauchy--Schwarz
  and the equality test $u = A^{-1/2}v$. Completion gives $H^{-1/2}$. Thus, with
  $M_K = L_K^{1/4}$, the phase norm $\|(u,v)\|_{E_A}^2 =
  \|A^{1/4}u\|_2^2+\|A^{-1/4}v\|_2^2$ lies between $M_K^{-2}\|(u,v)\|_E^2$ and
  $M_K^2\|(u,v)\|_E^2$. The condition cost is $M_K^2 = \sqrt{3(1+K^2)} \le
  2(1+K)$.

  For unit $L^2$ data, $b = 1$ and $z \ge 1$. The form bound gives $q_V \ge
  z^2/2-1-K^2/2 \ge -K^2$. A dilated compact unit bump translated wholly off the
  potential support has zero potential pairing and kinetic energy tending to
  zero, so $\beta \le 0$. Complex form infima agree with real ones: dividing
  each complex diagonal by its squared norm is the convex combination of the two
  normalized real diagonals, omitting zero parts. The spectral form formulas
  identify $\beta$ with the spectral bottom by bounded spectral slices. There
  are support points because scalar spectral measures of unit vectors have mass
  one; their arbitrarily small bounded slices give approximate eigenvectors and
  the reverse form bound. If $\Sigma$ denotes the closed PVM support and real
  $z\notin\Sigma$, the Borel function $f_z(h) = (h-z)^{-1}$ on $\Sigma$, zero
  outside, is bounded, as is $hf_z(h)$. Its calculus has range in $D(H_V)$ and
  supplies both inverse identities for $H_V-z$ on their respective domains.
  Thus, no such $z$ is spectral; the approximate eigenvectors give the other
  inclusion. Hence, $\lambda \le K$ and $h+c \ge 2$ on the spectrum.

  Define $\operatorname{Cos}_t(h),\operatorname{Sin}_t(h)$ by $\cos(t\sqrt
  h),\sin(t\sqrt h)/\sqrt h$ for $h > 0$, by $1,t$ at zero and by
  $\cosh(t\sqrt{-h}),\sinh(t\sqrt{-h})/\sqrt{-h}$ for $h < 0$. In the $E_A$
  isometry the matrix is
  \[
    \begin{pmatrix}
      \operatorname{Cos}_t(h)&\sqrt{h+c}\operatorname{Sin}_t(h)\\
      -h\operatorname{Sin}_t(h)/\sqrt{h+c}&\operatorname{Cos}_t(h)
    \end{pmatrix}.
  \]
  For $h \ge 0$, its off-diagonals have bounds $1+\sqrt c\,t$ and one; at zero
  their values are $\sqrt c\,t,0$. For $\beta \le h < 0$, the entries are
  bounded using $|\operatorname{Cos}_t| \le e^{\lambda t}$,
  $|\operatorname{Sin}_t| \le te^{\lambda t}$, upper off-diagonal at most $\sqrt
  c\,te^{\lambda t}$, and lower at most $Ke^{\lambda t}$. Since $\sqrt c \le
  2K$, each entry is at most $2(1+K)(1+t)e^{\lambda t}$. Set these entries to
  zero off the closed spectral support before applying bounded Borel calculus.
  From~\cite{teschl:14:mathematical}*{Theorems~3.1--3.2, pp.~102--104,
  (3.18)--(3.33)}, their matrix norm is at most twice that entry bound, and the
  condition cost gives \eqref{E:uniform-E-static-bound}. Spectral dominated
  convergence and scalar addition give the group.

  Denote the spectral matrix evolution by $P_V^E(t)$. It is the physical $E$
  wave. The fixed-$R$ lift in Lemma~\ref{L:uniform-E-lift} belongs to the
  precise class in Proposition~\ref{P:signed-physical-wave}. Its form is this
  same $q_V$ and has the same self-adjoint representative by uniqueness,
  regardless of the lift's larger shift. On dense bounded spectral cutoffs the
  two waves have the same $\operatorname{Cos},\operatorname{Sin}$ functions of
  $H_V$. Boundedness extends equality to all $E$. The old graph-core Duhamel
  identification therefore applies; no lift norm or $\chi_R$ constant entered
  the bounds above. Conjugation commutes with the real structures, so real
  restrictions are the phases. For compact $L$, finite-rank approximation and
  strong continuity of the free group and its adjoint make $v\mapsto
  S_E(-v)LS_E(v)$ norm continuous on finite intervals. Its primitive is
  Lipschitz. Its Young equation is therefore the ordinary bounded-interaction
  equation by Proposition~\ref{P:constant-young-linear}. This identifies the
  averaged static baseline without transferring any lift norm into the uniform
  bounds.

  Finally, take a unit $\varepsilon$-minimizer for $W$, $0 < \varepsilon \le 1$.
  Since $\beta(W) \le 0$, it has $q_W \le 1$. Thus, $z^2-1-Kz \le 1$, which
  forces $z \le K+2$. The multiplier difference changes its form by at most
  $\|L_V-L_W\|\|u\|_{H^{1/2}}^2 \le (K+2)\|L_V-L_W\|$. Take infima, let
  $\varepsilon\downarrow0$ and reverse $V,W$. The square-root inequality on
  nonnegative $-\beta$ gives \eqref{E:uniform-E-rate-continuity}. For zero
  covariance the class is $\{0\}$, $H_V = -\Delta$ and $\beta = \lambda = 0$;
  the free wave gives the claims directly. This proves
  Proposition~\ref{P:uniform-E-spectral}.
\end{proof}

\subsection{Scalar-time averaging and mean bounds}

Proposition~\ref{P:uniform-E-spectral} bounds the wave of one fixed operator
$L \in \mathcal{V}_{E,R}$ through the rate $\lambda(L)$. The mean field
$\mathcal{T}_R\Phi$ depends on the time variable. We now average it over the
cells of the partition and bound the resulting change in the integrated rate,
uniformly in $R \ge 1$.

\begin{proposition}[Uniform completed-field averaging]\label{P:uniform-rate-averaging}
  Use the true slow partition $I_m$ with mesh at most $\delta B$. Let
  $\mathcal{A}_\delta$ average over each positive-length cell; an empty final
  cell is ignored. Put $\omega_\delta = \|(\mathcal{A}_\delta-I)C_t\|_{H_t\to
  L^r(I)}$. At fixed $B$, $\omega_\delta \to 0$. For $M \ge 0$, if
  $\|\Phi\|_{H_I} \le M$, $W_R = \mathcal{T}_R\Phi$, $\overline{W}_R =
  \mathcal{A}_\delta W_R$ and $Z = C_{\rm mean}M$, then
  \begin{align}
    \|W_R\|_{L^r},\ \|\overline{W}_R\|_{L^r}&\le Z,\qquad
    \|\overline{W}_R-W_R\|_{L^r}\le M_0M\omega_\delta=:\Omega_\delta(M),
    \label{E:uniform-field-averaging}\\
    \operatorname{Rate}_R(\Phi)&=\int_I\lambda(\mathcal{T}_R\Phi(s))\ud s,
    \label{E:uniform-mean-rate}\\
    \int_I|\lambda(\overline{W}_R)-\lambda(W_R)|\ud s
    &\le\sqrt{3+2Z}\sqrt{\Omega_\delta(M)}=:\Lambda_\delta(M)\to0 .
    \label{E:uniform-rate-averaging}
  \end{align}
  These errors are uniform in $R \ge 1$. The rate is finite, nonnegative and
  continuous on $H_I$.
\end{proposition}

\begin{proof}
  Bochner averaging is a contraction on $L^r(I;\mathscr{Z})$ for any Banach
  $\mathscr{Z}$, by Jensen. For continuous fields on $[0,1]$, its error is
  bounded by the modulus of continuity at the mesh. Simple-field approximation,
  followed by continuous approximation of scalar indicators in $L^r$, proves
  density of those fields. The contraction gives strong convergence for every
  Bochner field. In the scalar case it is uniform on the compact closure of
  $C_t$'s unit-ball image, by a finite net and $\|\mathcal{A}_\delta-I\| \le 2$.
  This proves $\omega_\delta \to 0$.

  On finite tensors, integration commutes with the fixed bounded $T_R$:
  \[
    \mathcal{A}_\delta^{\mathcal{K}(E)}
    T_R(C_t\otimes I)\Phi
    =T_R((\mathcal{A}_\delta C_t)\otimes I)\Phi .
  \]
  The dimension-free Hilbert tensor bound in Lemma~\ref{L:uniform-mean-tensor},
  and $\|T_R\| \le M_0$, extend this identity to the completed $H_I$. Applying
  it to the difference gives~\eqref{E:uniform-field-averaging}. Averages remain
  in the same closed $E$ operator class. Only scalar-time compactness was used;
  the tensor map is not claimed compact. Nor does this assert noncompactness of
  a composite spatial family: each fixed-radius completed mean is compact.

  For any two such fields, apply~\eqref{E:uniform-E-rate-continuity} with the
  pointwise bound
  \[
    K(s)=\max(1,\|W_1(s)\|,\|W_2(s)\|).
  \]
  Cauchy--Schwarz gives
  \[
    \int_I|\lambda(W_1)-\lambda(W_2)|
    \le\sqrt{3+\|W_1\|_{L^1}+\|W_2\|_{L^1}}\,
    \sqrt{\|W_1-W_2\|_{L^1}} .
  \]
  Continuity of $\lambda$ on the class makes these quantities measurable. Since
  $|I| = 1$, this proves~\eqref{E:uniform-rate-averaging}. It also proves rate
  continuity, while $\lambda(L) \le \max(1,\|L\|)$ gives
  $\operatorname{Rate}_R(\Phi) \le 1+C_{\rm mean}\|\Phi\|$. For zero covariance
  the fields and rate errors are zero directly. This proves
  Proposition~\ref{P:uniform-rate-averaging}.
\end{proof}

For the remaining proofs use the full parameter choices and the
$B,\delta,N,\ell_m,I_m$ partition of Section~\ref{S:actual-cameron-mean}. For
$W_R = \mathcal{T}_R\Phi$ define
\[
  H_m(u)=\int_0^{\min(u,\ell_m)}
  S_E(-v)W_R(s_m+\delta v)S_E(v)\ud v,\qquad
  P_m^{\rm mean}=S_E(\ell_m)U_{H_m}(\ell_m,0).
\]

\begin{lemma}[Explicit physical primitive and growth bounds]
  \label{L:uniform-product-budgets}
  Write $G_B = \sup_{|v|\le B}\|S_E(v)\|$ and $C_B =
  G_B^2[B^{1-1/r}+B^{1-1/r-\theta}]$. Then $1 \le G_B \le 3B$, $C_B \le
  18B^{3-1/r} \le 32B^{3-1/r}$ and $C_BB^{1/r} \ge B$. The stopped primitives
  lie in the little H\"older space of paths starting at zero and
  \begin{equation}\label{E:uniform-product-primitive}
    \|H_m\|_\theta\le a_m:=C_B\delta^{-1/r}\|W_R\|_{L^r(I_m)},
    \qquad\delta\sum_m a_m^r\le C_B^r\|W_R\|_{L^r}^r .
  \end{equation}
  There is a fixed $C_{\rm gr} = C_{\rm gr}(\theta) > 0$ such that
  \begin{equation}\label{E:uniform-product-growth}
    \log\|S_E(\ell)U_D(\ell,0)\|
    \le C_{\rm gr}B(1+\|D\|_\theta^\nu),\qquad 0\le\ell\le B .
  \end{equation}
  Here $D$ is any $C^\theta([0,B];\mathcal{L}(E))$ path starting at zero; no
  Gaussian or Hamiltonian hypothesis is imposed on this growth bound. The mean
  endpoints have $\|P_m^{\rm mean}\| \ge 1$.
\end{lemma}

\begin{proof}
  The balanced free matrix has upper entry $\langle k\rangle\sin(vk)/k$ and
  lower $-k\sin(vk)/\langle k\rangle$. Splitting $k \le 1$ and $k \ge 1$ bounds
  the upper by $\sqrt2 B$, while the lower and diagonals have bound one. At zero
  the off-diagonals are $v,0$. The Frobenius bound gives $G_B \le \sqrt{3+2B^2}
  \le 3B$; $S_E(0) = I$ gives $G_B \ge 1$. The stated $C_B$ inequalities follow,
  including $C_BB^{1/r} = G_B^2(B+B^{1-\theta}) \ge B$. H\"older in time with
  $1-1/r > \theta$, the affine time change, compact conjugation and constant
  stopping prove \eqref{E:uniform-product-primitive}; the true partition sums
  the $r$th powers, with zero on an empty cell.

  For the explicit Young bound put $d = \|D\|_\theta$ and $c_\theta =
  2\max(1,C_{\rm Young})$, as in the local estimate proving
  Lemma~\ref{L:young-upper-growth}. If $c_\theta dB^\theta \le 1/4$,
  whole-interval contraction gives $\log\sup\|U_D\| \le \log(4/3) < 1/2$.
  Otherwise at most $2B(4c_\theta d)^\nu$ contraction pieces have norm at most
  $4/3$. Combine their logarithms with $\log G_B \le \log(4B)$. For example
  $C_{\rm gr} = 4+\log4+2(4c_\theta)^\nu$ gives
  \eqref{E:uniform-product-growth}, since $B \ge 1$. This is the generic algebra
  argument on $\mathcal{L}(E)$ with multiplication norm one, not an unspecified
  radius-dependent bound.

  Finally, Proposition~\ref{P:uniform-mean-regulator} gives $W_R \in
  L^r(I;\mathcal{K}(E))$ for each fixed $R,\Phi$. The compact conjugation and
  stopped Bochner primitive used above therefore make each $H_m$ a
  compact-valued $C^\theta$ path starting at zero. Lemma~\ref{L:symplectic-free}
  gives $\|P_m^{\rm mean}\| \ge \|S_E(\ell_m)\| \ge 1$, deterministically for
  each $\Phi$. This norm bound needs no fast covariance direction or action
  estimate. An empty endpoint is $I$; zero covariance gives the free endpoint.
  This proves Lemma~\ref{L:uniform-product-budgets}.
\end{proof}

\subsection{Full blocks, the true last cell and polynomial control balls}

\begin{proposition}[Finite-mesh physical mean-product bound]
  \label{P:uniform-mean-product}
  Fix $B$ as above, $M \ge 0$ and $K \ge 1$, and put $Z = C_{\rm mean}M$.
  Uniformly for $R \ge 1$, $\|\Phi\| \le M$,
  \begin{align}
    \delta\sum_{m=0}^N\log\|P_m^{\rm mean}\|
    -\operatorname{Rate}_R(\Phi)
    \le{}&\frac{L_{B,K}}B+
    C_{\rm gr}32^r Z^rB^{3r}(K^{-r}+K^{\nu-r})\notag\\
    &+\Lambda_\delta(M)+C(\theta,B,K)E_\delta+
    \operatorname{Last}_\delta,\label{E:uniform-mean-product}
  \end{align}
  where $L_{B,K} = \log16+4\log(1+K)+3\log(1+B)$ and
  \begin{align*}
    E_\delta&=(2/B)^{1-1/r}C_B\Omega_\delta(M)
    +(2/B)^{1-\nu/r}C_B^\nu\Omega_\delta(M)^\nu,\\
    \operatorname{Last}_\delta&=
    C_{\rm gr}B[\delta+C_B^\nu Z^\nu\delta^{1-\nu/r}].
  \end{align*}
  The last three terms vanish as $\delta\downarrow0$ at fixed $B,K,M$.
\end{proposition}

\begin{proof}
  On each positive-length cell use its own Bochner average $\overline{M}_m$; set
  it to zero on an empty cell and form the constant-field primitive
  $\overline{H}_m$. Put $e_m = C_B\delta^{-1/r}
  \|W_R-\overline{W}_R\|_{L^r(I_m)}$. Jensen
  and~\eqref{E:uniform-product-primitive} give
  \[
    \|\overline{H}_m\|_\theta\le a_m,\quad
    \|H_m-\overline{H}_m\|_\theta\le e_m,\quad
    \delta\sum e_m^r\le C_B^r\Omega_\delta(M)^r,\quad
    \delta\sum a_m^r\le D_M:=C_B^rZ^r .
  \]
  Finite power mean at exponents $r$ and $r/\nu > 1$, with nonnegative inputs
  $e_m,e_m^\nu$, gives $\delta\sum_m(e_m+e_m^\nu) \le E_\delta$. Here $n = N+1
  \ge 2$, $\delta > 0$ and $\delta n \le 2/B$; all subsequent powers are
  positive and zero errors are allowed.

  On full cells only, $\|\overline{M}_m\| \le (\delta
  B)^{-1/r}\|W_R\|_{L^r(I_m)} = a_m/(C_BB^{1/r}) \le a_m/B$. Split full indices
  $m < N$ into good $a_m \le K$ and bad $a_m > K$. On a good block,
  Proposition~\ref{P:young-upper-multiplicative} applies with base
  $\overline{H}_m$, perturbation $H_m-\overline{H}_m$ and the left multiplier
  $S_E(B)$. The baseline is the static wave of $\overline{M}_m$ by
  Proposition~\ref{P:uniform-E-spectral}. Here $\|\overline{M}_m\| \le K/B \le
  K$, whereas $\|\overline{H}_m\|_\theta \le a_m \le K$. The spectral estimate
  gives
  \[
    \|S_E(B)U_{\overline{H}_m}(B,0)\|
    \le8(1+K)^2(1+B)e^{B\lambda(\overline{M}_m)}.
  \]
  Consequently,
  \[
    \log\|P_m^{\rm mean}\|
    \le L_{B,K}+B\lambda(\overline{M}_m)
    +C(\theta,B,K)(e_m+e_m^\nu).
  \]
  This sufficient prefactor follows from the bound in $E$, $8(1+K)^2(1+B)$, by
  enlarging it. Since $\delta N \le 1/B$, the prefactors sum to at most
  $L_{B,K}/B$. Nonnegativity of $\lambda$ gives $\delta\sum_{\rm
  good}B\lambda(\overline{M}_m) \le \int_I\lambda(\overline{W}_R) \le
  \operatorname{Rate}_R(\Phi)+\Lambda_\delta(M)$. Bad and last rate cells can be
  dropped in this upper bound.

  On bad full cells~\eqref{E:uniform-product-growth} gives $C_{\rm
  gr}B(1+a_m^\nu)$. Because $a_m > K \ge 1$ and $r > \nu$, $1+a_m^\nu \le
  (K^{-r}+K^{\nu-r})a_m^r$. Their weighted sum is bounded by $C_{\rm
  gr}BC_B^rZ^r(K^{-r}+K^{\nu-r})$. The inequality $C_B \le 32B^{3-1/r}$ yields
  the displayed bad-block term.

  If $\ell_N = 0$, the last endpoint is $I$ and $\log\|I\| = 0$. Otherwise use
  the stopped path, not its frozen full-cell formula. Since $a_N \le
  C_B\delta^{-1/r}Z$, its weighted logarithm is at most
  $\operatorname{Last}_\delta$ by~\eqref{E:uniform-product-growth}. This tends
  to zero because $\nu < r$, even for arbitrarily short last cells; no division
  by that cell length occurs. Combining these estimates
  proves~\eqref{E:uniform-mean-product}. All constants are independent of $R$.
  This proves Proposition~\ref{P:uniform-mean-product}.
\end{proof}

\begin{corollary}[Every polynomial control radius]\label{C:uniform-polynomial-balls}
  For $c_{\rm ball} > 0$, $\kappa_{\rm ball} \ge 0$ put $M_B = c_{\rm
  ball}B^{\kappa_{\rm ball}}$. Then
  \[
    \limsup_{\substack{B\to\infty\\B\in\mathbb{N}}}
    \limsup_{\delta\downarrow0}
    \sup_{\substack{R\ge1\\\|\Phi\|\le M_B}}
    \left\{\delta\sum_m\log\|P_m^{\rm mean}\|
    -\operatorname{Rate}_R(\Phi)\right\}\le0 .
  \]
\end{corollary}

\begin{proof}
  Choose $s > r(3+\kappa_{\rm ball})/(r-\nu)$ and $K_B = B^s$. At each fixed
  $B$, the inner $\delta$ limit has a fixed finite radius and threshold, so the
  last three errors in~\eqref{E:uniform-mean-product} vanish first. Its bad
  terms become
  \[
    C_{\rm gr}32^r C_{\rm mean}^r c_{\rm ball}^r
    \{B^{r(3+\kappa_{\rm ball})-sr}
    +B^{r(3+\kappa_{\rm ball})+s(\nu-r)}\}.
  \]
  Both exponents are negative: the second by choice and the first because $s >
  3+\kappa_{\rm ball}$. Also $L_{B,K_B}/B \le [\log16+7\log2+(4s+3)\log B]/B \to
  0$. Thus, the outer integer-$B$ limit gives the claim. No diagonal choice
  $B_\delta,M_\delta$ is asserted. At fixed parameters all primitive maps are
  bounded linear into separable little H\"older products. Young local dependence
  and positive endpoint norms make the displayed logarithms continuous; the rate
  is continuous by Proposition~\ref{P:uniform-rate-averaging}. Growth bounds
  make each control-ball supremum finite. A countable dense subset of each
  closed Hilbert ball has the same supremum. Supremum over $R$ is an ordinary
  deterministic real supremum, not a random measurability or compact-family
  claim. This proves Corollary~\ref{C:uniform-polynomial-balls}.
\end{proof}

\subsection{Arbitrary small residual paths and the ordered product}

Corollary~\ref{C:uniform-polynomial-balls} compares the sum of the logarithms
$\log\|P_m^{\rm mean}\|$ with the rate, for the mean primitives $H_m$ alone. We
now add compact-operator paths $x_m$ to the $H_m$ and bound the norm of the
ordered product of the perturbed endpoints when these paths are small.

\begin{proposition}[Residual comparison and the integrated rate]
  \label{P:uniform-arbitrary-residual}
  Let $x_m$ be arbitrary little $C^\theta$ compact-operator paths starting at
  zero on $[0,B]$, put $v_m = \|x_m\|_\theta$ and $Y = (\delta\sum_m
  v_m^p)^{1/p}$. Define
  \[
    P_m^{\rm pert}=S_E(\ell_m)U_{H_m+x_m}(\ell_m,0),\qquad
    z_{\rm rat}=\prod_m\frac{\|P_m^{\rm pert}\|}{\|P_m^{\rm mean}\|}.
  \]
  For fixed integer $B \ge 1$, $M \ge 0$ and every $K_{\rm res} \ge 1$,
  \begin{align}
    \delta\log\max(1,z_{\rm rat})\le{}&
    C_{K_{\rm res}}[(2/B)^{1-1/p}Y+(2/B)^{1-\nu/p}Y^\nu]\notag\\
    &+c_BD_M(K_{\rm res}^{-r}+K_{\rm res}^{\nu-r}),
    \label{E:uniform-residual-ratio}
  \end{align}
  uniformly in $R \ge 1$, $\|\Phi\| \le M$. The finite constants may be written
  $C_{K_{\rm res}} = C(\theta,B,K_{\rm res})$ and $c_B = 2^{\nu-1}C_{\rm gr}B$.
  Here $D_M = C_B^r(C_{\rm mean}M)^r$. If $b_\delta \ge 0$ is any deterministic
  function of $\delta$ tending to zero at each fixed $B$, $M_B$ is as in
  Corollary~\ref{C:uniform-polynomial-balls}, and $\mathcal{F} =
  2\delta\log\max(1,\|P_N^{\rm pert}\cdots P_0^{\rm pert}\|)$, then
  \begin{equation}\label{E:uniform-residual-integrated-rate}
    \limsup_{\substack{B\to\infty\\B\in\mathbb{N}}}
    \limsup_{\delta\downarrow0}
    \sup_{\substack{R\ge1,\ \|\Phi\|\le M_B\\Y\le b_\delta}}
    \{\mathcal{F}-2\operatorname{Rate}_R(\Phi)\}\le0 .
  \end{equation}
  No Gaussian, Cameron or Hamiltonian condition on the residuals is imposed.
\end{proposition}

\begin{proof}
  All endpoint norms are positive by the Young and free inverses. On every cell
  with $a_m \le K_{\rm res}$ apply
  Proposition~\ref{P:young-upper-multiplicative} with base $H_m$, perturbation
  $x_m$ and left multiplier $S_E(\ell_m)$. It gives logarithmic ratio at most
  $C(\theta,B,K_{\rm res})(v_m+v_m^\nu)$. On cells with $a_m > K_{\rm res}$ use
  \eqref{E:uniform-product-growth}, $(a_m+v_m)^\nu \le
  2^{\nu-1}(a_m^\nu+v_m^\nu)$ and the mean denominator $\|P_m^{\rm mean}\| \ge
  1$. The logarithmic ratio is at most $c_B(1+a_m^\nu+v_m^\nu)$, with $c_B$
  independent of $K_{\rm res},R,\delta,\Phi,M$. Increase $C_{K_{\rm res}}$ to
  dominate both good and bad residual constants. Summation gives a nonnegative
  bound
  \[
    \log\max(1,z_{\rm rat})
    \le C_{K_{\rm res}}\sum_m(v_m+v_m^\nu)
    +c_B\sum_{a_m>K_{\rm res}}(1+a_m^\nu).
  \]
  The bad sum times $\delta$ is at most $D_M(K_{\rm res}^{-r}+K_{\rm
  res}^{\nu-r})$. Finite power mean at exponents $p \ge 1$ and $p/\nu \ge 1$,
  with nonnegative inputs $v_m,v_m^\nu$, gives $\delta\sum v_m \le
  (2/B)^{1-1/p}Y$ and $\delta\sum v_m^\nu \le (2/B)^{1-\nu/p}Y^\nu$. This
  includes $p = \nu$, zero entries and the true empty block, with no singular
  powers. It proves~\eqref{E:uniform-residual-ratio}.

  The product $A_{\rm prod} = \prod_m\|P_m^{\rm mean}\|$ is at least one.
  Submultiplicativity, without commutation, gives
  \[
    \begin{aligned}
      \max(1,\|P_N^{\rm pert}\cdots P_0^{\rm pert}\|)
      &\le A_{\rm prod}\max(1,z_{\rm rat}),\\
      \mathcal{F}-2\delta\sum_m\log\|P_m^{\rm mean}\|
      &\le2\delta\log\max(1,z_{\rm rat}).
    \end{aligned}
  \]
  At fixed $B,M,K_{\rm res}$ the residual terms vanish on $Y \le b_\delta$. Let
  $\delta\downarrow0$ first and then $K_{\rm res} \to \infty$. Since $r > \nu$,
  the residual excess has limsup at most zero, uniformly in $R,\Phi$. Now take
  $M = M_B$ at each fixed $B$ and combine with
  Corollary~\ref{C:uniform-polynomial-balls}; a supremum of sums is no larger
  than the sum of the suprema. The outer integer-$B$ limit
  proves~\eqref{E:uniform-residual-integrated-rate}. The endpoint functional is
  continuous on all tuples by generic Young local dependence, including
  nonphysical residuals. Nothing was multiplied as scalar displacement moments.
  This proves Proposition~\ref{P:uniform-arbitrary-residual}.
\end{proof}

These are uniform deterministic $E$-phase product estimates. Their Gaussian
Rayleigh application, original point transfer and signed-variational value over
nonnegative controls comparison are proved separately in
Sections~\ref{S:actual-gaussian-tubes}--\ref{S:intrinsic-second-moment}.

\section{Mean and Cameron bounds uniform in the spatial radius}
\label{S:uniform-mean-regulation}

Retain the original nonzero covariance, amplitude and real spaces of
Section~\ref{S:actual-cameron-mean}. Fix $\alpha/4 < s < 1/2$ and put $\beta =
1/2-s > 0$. Write $A_s = C_s^2\sqrt{R_{\mathrm{sp}}}$ for the constant
in~\eqref{E:cameron-fractional-product}. Choose a real even smooth compact
$\eta$, equal to one on the unit ball, with $0 \le \eta \le 1$ and support in
the closed radius-two ball. For $R \ge 1$ set $\eta_R(x) = \eta(x/R)$ and $m_R =
\eta_R^2$. In this section the mean acts on $E = H^{1/2}\times H^{-1/2}$. The
constant data columns belong to $F$, outside $E$. All uniform-radius claims
below concern $E$. The spatial map $T_R:H_{\mathrm{sp}} \to \mathcal{K}(E)$ is
the $E$ restriction of Proposition~\ref{P:cameron-spatial-map}. Its explicit row
is
\begin{equation}\label{E:uniform-mean-row}
  \langle T_Rqz,w\rangle_E
  =\langle q,k_R(z,w)\rangle_{H_{\mathrm{sp}}},
  \qquad
  k_R(z,w)=(\eta_Rz_1)(\eta_R\Lambda^{-1}w_2).
\end{equation}
Thus, the physical mask occurs once, after covariance convolution.

\subsection{Uniform multiplier and spatial tails}

We first bound the row $k_R$ in~\eqref{E:uniform-mean-row} uniformly in
$R \ge 1$. Both factors of this row carry the cutoff $\eta_R$, so we start with
its multiplier norm on $H^\sigma$.

\begin{lemma}[Uniform cutoff and physical mean]\label{L:uniform-mean-cutoff}
  For every real $\sigma$, multiplication by $\eta_R$ on $H^\sigma$ has norm at
  most
  \[
    M_\sigma=2^{|\sigma|/2}(2\pi)^{-d/2}
    \int_{\R^d}|\widehat{\eta}(p)|\langle p\rangle^{|\sigma|}\ud p,
  \]
  independently of $R \ge 1$. With $M = M_{1/2}$ and $M_0 = A_sM^2$,
  \begin{equation}\label{E:uniform-spatial-mean}
    \|k_R(z,w)\|_{H_{\mathrm{sp}}}
    \le M_0\|z\|_E\|w\|_E,\qquad
    \|T_R\|_{H_{\mathrm{sp}}\to\mathcal{L}(E)}\le M_0 .
  \end{equation}
\end{lemma}

\begin{proof}
  Peetre's inequality bounds the weighted Fourier kernel for multiplication by
  $\eta_R$ by convolution with
  \[
    2^{|\sigma|/2}(2\pi)^{-d/2}
    |\widehat{\eta_R}(q)|\langle q\rangle^{|\sigma|}.
  \]
  Here $\widehat{\eta_R}(q) = R^d\widehat{\eta}(Rq)$. Substitution $p = Rq$ and
  $\langle p/R\rangle \le \langle p\rangle$ bound its $L^1$ norm by $M_\sigma$.
  The $L^1*L^2$ inequality proves the assertion on Schwartz functions and
  density extends it to $H^\sigma$. The map $w_2\mapsto\Lambda^{-1}w_2$ is an
  isometry from $H^{-1/2}$ to $H^{1/2}$, and $H^{1/2}\hookrightarrow H^s$ has
  norm one. Apply the cutoff bound at $\sigma = 1/2$ and
  Lemma~\ref{L:cameron-fractional-products} to the two factors
  in~\eqref{E:uniform-mean-row}. Hilbert Cauchy--Schwarz then gives the operator
  bound. This is the existing completed physical map: the row agrees on smooth
  controls and hence on their completion. Its compact values at each fixed $R$
  are retained. This proves Lemma~\ref{L:uniform-mean-cutoff}.
\end{proof}

Lemma~\ref{L:uniform-mean-cutoff} bounds $T_R$ on all of $H_{\mathrm{sp}}$.
The approximation in Proposition~\ref{P:uniform-mean-regulator} also needs
$T_R$ to be small on high frequencies, uniformly in $R \ge 1$. We obtain this
from a bound on $T_R(I-C_\varepsilon)$ for the heat contraction
$C_\varepsilon$.

\begin{lemma}[Uniform heat and frequency tails]
  \label{L:uniform-mean-spatial-tails}
  Let $C_\varepsilon$ be the covariance-Hilbert heat contraction with Fourier
  multiplier $e^{-\varepsilon|\xi|^2/2}$. There is a real orthogonal projection
  $P_K$ on $H_{\mathrm{sp}}$ corresponding to the closed frequency ball
  $\{|\xi|\le K\}$. It commutes with heat. For $K \ge 1$, $\varepsilon > 0$ and
  $R \ge 1$,
  \begin{align}
    \|T_R(I-C_\varepsilon)\|
    &\le M_h\varepsilon^{\beta/2},
    &M_h&=4A_sM^2\E|G_d|^\beta,\label{E:uniform-mean-heat}\\
    \|T_R(I-P_K)\|
    &\le M_fK^{-\beta},
    &M_f&=\frac{M_h}{1-e^{-1/2}},\label{E:uniform-mean-frequency}
  \end{align}
  where $G_d$ is a standard real $d$-dimensional normal vector. Operator norms
  are from $H_{\mathrm{sp}}$ to $\mathcal{L}(E)$. The projection $P_K$ is
  allowed to have infinite rank.
\end{lemma}

\begin{proof}
  Translations $\tau_hf(x) = f(x-h)$ extend to strongly continuous isometries of
  the $H_{\mathrm{sp}}$. Indeed, in its covariance Fourier isometry they
  multiply by $e^{-i\xi\cdot h}$; dominated convergence gives continuity on
  tests, then density and isometry give it on the completion. The real subspace
  is preserved. The inequalities $|e^{-it}-1| \le 2\min(1,|t|^\beta)$ and
  $|\xi|^\beta\langle\xi\rangle^s \le \langle\xi\rangle^{1/2}$ show that
  \[
    \|\tau_hf-f\|_{H^s}
    \le2\min(1,|h|^\beta)\|f\|_{H^{1/2}} .
  \]
  Use this for $f = \eta_Rz_1$ and $g = \eta_R\Lambda^{-1}w_2$. The product
  identity
  \[
    \tau_h(fg)-fg=(\tau_hf-f)\tau_hg+f(\tau_hg-g)
  \]
  holds in $H_{\mathrm{sp}}$, first for smooth factors and then by the
  continuous fractional product extension. Lemma~\ref{L:uniform-mean-cutoff}
  gives
  \begin{equation}\label{E:uniform-mean-row-translation}
    \|\tau_hk_R(z,w)-k_R(z,w)\|_{H_{\mathrm{sp}}}
    \le4A_sM^2\min(1,|h|^\beta)\|z\|_E\|w\|_E .
  \end{equation}

  The Bochner integral $\int p_\varepsilon(h)\tau_hk\ud h$ exists, since the
  translation orbit is strongly continuous and bounded in the separable
  covariance Hilbert space. Its Fourier image is
  $e^{-\varepsilon|\xi|^2/2}\widehat{k}$: ordinary Gaussian integration proves
  this on tests and contraction extends it to all $k$. It is therefore the
  $C_\varepsilon$ of the source construction. Heat has variance $\varepsilon$
  here; there is no additional Fourier normalization factor. Integrating
  \eqref{E:uniform-mean-row-translation} with $h = \sqrt\varepsilon G_d$ bounds
  $\|(I-C_\varepsilon)k_R(z,w)\|$ by $M_h\varepsilon^{\beta/2}\|z\|_E\|w\|_E$.
  Selfadjointness in~\eqref{E:uniform-mean-row} proves
  \eqref{E:uniform-mean-heat}.

  To construct $P_K$, let $V$ be the closed image of the covariance Fourier
  isometry into $L^2(\mu)$, including its fixed $(2\pi)^{d/2}$ factor. No
  surjectivity is assumed. Its formula extends to Schwartz functions: compact
  cutoff approximation converges in Schwartz seminorms, and temperedness of
  $\mu$ controls the Fourier error in $L^2(\mu)$. Choose real even radial smooth
  $\chi_n$, equal to one on the closed ball of radius $K$, zero outside radius
  $K+1/n$, and between zero and one. For real Schwartz $f$, the function
  $\chi_n\widehat{f}$ is the transform of another real Schwartz function and so
  lies in $V$. Dominated convergence against $|\widehat{f}|^2\mu$ gives
  $1_{\{|\xi|\le K\}}\widehat{f} \in V$. The pointwise limit includes the
  boundary sphere, even if $\mu$ charges it. Test density and contraction extend
  this assertion to all of $V$. Indicator multiplication is selfadjoint and
  idempotent on $V$; pulling it back defines the asserted real orthogonal $P_K$.
  All Fourier multipliers here commute.

  On $|\xi| > K$ one has $1-e^{-|\xi|^2/(2K^2)} \ge 1-e^{-1/2}$, whence
  \[
    \|(I-P_K)k\|_{H_{\mathrm{sp}}}
    \le(1-e^{-1/2})^{-1}\|(I-C_{K^{-2}})k\|_{H_{\mathrm{sp}}}.
  \]
  Apply the row heat bound and selfadjointness of $P_K$
  in~\eqref{E:uniform-mean-row} to obtain \eqref{E:uniform-mean-frequency}. No
  density of $\mu$, periodic state space or finite rank was used. This proves
  Lemma~\ref{L:uniform-mean-spatial-tails}.
\end{proof}

\subsection{Tensor products and mean approximation}

We first record a tensor extension bound for maps into $L^r(I)$. The proof
of Proposition~\ref{P:uniform-mean-regulator} uses it to pass from a scalar
time map to the Hilbert-valued time map without a dimension factor.

\begin{lemma}[Dimension-free Hilbert tensor extension]
  \label{L:uniform-mean-tensor}
  If $H_0,G$ are real Hilbert spaces, $2 \le r < \infty$ and $A:H_0 \to L^r(I)$
  is bounded, then
  \[
    \|A\otimes I_G\|_{H_0\otimes G\to L^r(I;G)}\le\|A\|.
  \]
  The tensor extension need not be compact when $A$ is compact.
\end{lemma}

\begin{proof}
  For a finite tensor write $v = \sum_{j=1}^n f_j\otimes e_j$ with $(e_j)$
  orthonormal. Its squared norm is $\sum_j\|f_j\|^2$. Let $\gamma_j$ be
  independent standard real normals and $m_r = \E|\gamma_1|^r$. Finite
  covariance diagonalization gives
  \[
    \E\left\|\sum_j\gamma_j f_j\right\|_{H_0}^r
    \le m_r\left(\sum_j\|f_j\|_{H_0}^2\right)^{r/2}.
  \]
  In fact, if $\lambda_l$ are the covariance eigenvalues and $L =
  \sum_l\lambda_l$, the squared norm has law $\sum_l\lambda_l\zeta_l^2$ for
  independent standard $\zeta_l$. For $L > 0$, convexity at exponent $r/2 \ge 1$
  bounds $(\sum_l(\lambda_l/L)\zeta_l^2)^{r/2}$ by
  $\sum_l(\lambda_l/L)|\zeta_l|^r$. Taking expectations proves the inequality;
  $L = 0$ is immediate. Also $L = \sum_j\|f_j\|^2$.

  Choose measurable representatives of the finitely many $Af_j$. Scalar Gaussian
  moments and Tonelli give
  \[
    m_r\int_I\left(\sum_j|Af_j(t)|^2\right)^{r/2}\ud t
    =\E\left\|A\sum_j\gamma_jf_j\right\|_{L^r}^r
    \le\|A\|^rm_r\left(\sum_j\|f_j\|^2\right)^{r/2}.
  \]
  Cancel the finite positive $m_r$ and take the $r$th root. Finite tensor
  density and Bochner completeness give the extension. The zero space is
  trivial. If $Af \ne 0$ and $G$ has an infinite orthonormal sequence $(e_j)$,
  the images of the bounded sequence $f\otimes e_j$ have pairwise distance
  $\sqrt2\|Af\|_{L^r}$. They have no convergent subsequence, even when the
  scalar map $A$ is compact. Only finite auxiliary Gaussians were used in the
  estimate. This proves Lemma~\ref{L:uniform-mean-tensor}.
\end{proof}

Recall the supported identity $H_I = H_t\otimes H_{\mathrm{sp}}$, the compact
scalar covariance map $C_t:H_t \to L^r(I)$, $2 < r < 2/a$, and its completed
tensor $U = C_t\otimes I_{H_{\mathrm{sp}}}$ from
Section~\ref{S:actual-cameron-mean}. On finite smooth tensors $U$ agrees with
the earlier $\mathcal{U}$; density gives equality of their completed maps. For
block applications fix the same $r$ as in~\eqref{E:cameron-block-parameters}.
All $c_t,\varepsilon_J$ and mean norms below use that one choice. Choose a
smooth-test orthonormal basis of $H_t$ and let $\pi_J$ be its first-$J$
orthogonal projection, with $\pi_0 = 0$. Put
\[
  c_t=\|C_t\|,\qquad
  \varepsilon_J=\|C_t(I-\pi_J)\|_{H_t\to L^r(I)},\qquad
  \mathcal{T}_R=T_RU .
\]

\begin{proposition}[Uniform approximation of the completed mean]
  \label{P:uniform-mean-regulator}
  On $H_I$ the operators
  \[
    P_{\mathrm{ret}}=\pi_J\otimes P_K,\qquad
    Q_{J,K}=I-P_{\mathrm{ret}}
  \]
  are real orthogonal projections, and $\varepsilon_J \to 0$. The mean is a
  strongly measurable $L^r(I;\mathcal{K}(E))$ field and satisfies, uniformly for
  $R \ge 1$,
  \begin{equation}\label{E:uniform-completed-mean}
    \|\mathcal{T}_R\|\le M_0c_t,\qquad
    \|\mathcal{T}_RQ_{J,K}\|
    \le M_0\varepsilon_J+M_fc_tK^{-\beta}.
  \end{equation}
  These norms have domain $H_I$ and target $L^r(I;\mathcal{L}(E))$.
\end{proposition}

\begin{proof}
  The tensor projections are selfadjoint, idempotent and contractive on finite
  tensors and hence on their Hilbert completion. For the scalar tail, failure of
  convergence would supply unit $v_J$ with $\|C_t(I-\pi_J)v_J\|$ bounded below
  on a subsequence. But $(I-\pi_J)v_J$ tends weakly to zero, since its pairing
  with $h$ is bounded by $\|(I-\pi_J)h\|$. Compactness supplies a
  norm-convergent image subsequence; bounded linear scalar observables show its
  limit is zero, a contradiction. Lemma~\ref{L:uniform-mean-tensor} transfers
  this scalar tail norm to the Hilbert-valued time map without a dimension
  factor.

  For fixed $R$, $T_R$ is bounded into the separable compact-operator space on
  $E$. Composition with the Bochner field $U\Phi$ gives the stated mean and its
  first bound. The exact decomposition is
  \[
    Q_{J,K}=(I-\pi_J)\otimes I+\pi_J\otimes(I-P_K).
  \]
  The first term is bounded by $M_0\varepsilon_J$. For the second term use the
  intertwining identity
  \[
    U(\pi_J\otimes(I-P_K))
    =(I_{L^r}\otimes(I-P_K))(C_t\pi_J\otimes I).
  \]
  It holds first on finite tensors, then by continuity.
  Apply~\eqref{E:uniform-mean-frequency} pointwise, followed by
  Lemma~\ref{L:uniform-mean-tensor} for $C_t\pi_J$, whose norm is at most $c_t$.
  This proves~\eqref{E:uniform-completed-mean}. Infinite spatial rank is allowed
  throughout; neither tensor compactness nor uniform-radius compactness has been
  asserted. This proves Proposition~\ref{P:uniform-mean-regulator}.
\end{proof}

\subsection{The block Cameron embedding for the given noise}

Use $B,\delta,N,s_m,\ell_m,I_m$ and $\theta,r,p$ exactly as
in~\eqref{E:cameron-block-parameters} and its preceding block partition. Let
$i_I:H_I \to H_W$ be the isometric closed inclusion. On the full original noise
space set
\[
  P_{\mathrm{full}}=i_IP_{\mathrm{ret}}i_I^*,\qquad
  Q_{\mathrm{full}}=I-P_{\mathrm{full}} .
\]
These are orthogonal projections on $H_W$. Define the residual fast noise and
stopped $E$ operator path by
\begin{equation}\label{E:uniform-actual-residual}
  Y_m(f)=W(Q_{\mathrm{full}}T_{\delta,m}f),\qquad
  X_m(u)=g_\delta A_{m_R}(Y_m)(\min(u,\ell_m)),\qquad
  g_\delta=\delta^{(1-a)/2}.
\end{equation}
Here $T_{\delta,m}$ is the explicit original isometry
\eqref{E:raw-block-pullback}, and $A_{m_R}$ takes its $E$ restriction. Its
isometry is proved in Lemma~\ref{L:raw-block-inputs}. The symbol
$H_{\mathrm{fast}}$ below denotes this same original covariance norm in fast
coordinates, carried isometrically into $H_W$. The $E$ little H\"older space
means the closure of Lipschitz compact-operator paths starting at zero in
$C^\theta$ norm. Compact operators on separable $E$ and polygonal approximation
make this closure separable. The last interval can be empty, in which case $X_N
= 0$.

\begin{proposition}[Uniform-radius Cameron bound]
  \label{P:uniform-actual-cameron}
  The tuple $X = (X_m)_{m=0}^N$ is a centered Gaussian variable in the separable
  little H\"older compact-operator path space on $E$. Its Bochner covariance $Lh
  = \E[XW(h)]$, $h \in H_W$, is
  \begin{equation}\label{E:uniform-actual-covariance}
    (Lh)_m(u)=\sqrt\delta\int_0^{\min(u,\ell_m)}
    S_E(-v)\mathcal{T}_R(Q_{J,K}i_I^*h)(s_m+\delta v)
    S_E(v)\ud v .
  \end{equation}
  In the exact norm $\|x\|_{\delta,p} =
  (\delta\sum_m\|x_m\|_{C^\theta}^p)^{1/p}$, its Cameron embedding has the
  uniform-radius bound
  \begin{equation}\label{E:uniform-actual-cameron-bound}
    \|Lh\|_{\delta,p}
    \le\sqrt\delta\,C_{B,E}(2/B)^{1/p-1/r}
    [M_0\varepsilon_J+M_fc_tK^{-\beta}]\|h\|_{H_W},
    \qquad C_{B,E}\le18B^{3-1/r}.
  \end{equation}
  All spatial constants are independent of $R \ge 1$.
\end{proposition}

\begin{proof}
  Orthogonality and original pullback isometry give $\operatorname{Var}Y_m(f)
  \le \|f\|_{H_{\mathrm{fast}}}^2$. Gaussian transfer,
  Proposition~\ref{P:source-gaussian-transfer}, therefore constructs the
  operator paths on the given $W$, with finite path moments at each fixed
  $R,B,\delta,J,K$. Choose an admissible H\"older exponent strictly above
  $\theta$ as in Lemma~\ref{L:cameron-actual-residual}; constant stopping puts
  the paths in the separable little $C^\theta$ closure. Finite-coefficient
  constructions and their $L^2$ limits show that every continuous linear
  coordinate is in the first chaos of the given $W$.

  Every explicit observed row, pulled back to slow time, belongs to $H_I$. This
  is the supported-row approximation from Lemma~\ref{L:cameron-actual-residual},
  also valid before any projection; approximate endpoints inside $I$ against the
  integrable time kernel and locally finite spatial covariance. For such a row
  $k$, exact orthogonality gives
  \[
    \langle Q_{\mathrm{full}}T_{\delta,m}k,h\rangle_{H_W}
    =\langle T_{\delta,m}k,i_IQ_{J,K}i_I^*h\rangle_{H_W}.
  \]
  Directions perpendicular to $H_I$ disappear from this pairing. Apply the
  completed $J = 0$ identity \eqref{E:cameron-actual-covariance} to
  $Q_{J,K}i_I^*h$, and restrict its input and output to $E$. The free $F$ group
  preserves $E$ and restricts to $S_E$; the mean restricts to $\mathcal{T}_R$.
  This proves~\eqref{E:uniform-actual-covariance}. Its amplitude is exactly
  $g_\delta\delta^{a/2} = \sqrt\delta$, on the noise $W$. Arbitrary projected
  controls are covered by the completed mean identity; no smoothness or finite
  rank of $P_K$ is required. Compact operator values and strong continuity of
  $S_E$ and its adjoint make conjugation norm continuous on compact operators,
  as in Proposition~\ref{P:cameron-actual-embedding}. Thus, the integral is
  Bochner measurable. Fixed-parameter $L^2$ covariance continuity, dense states
  and rational times identify the entire operator path.

  For explicit free constants, the Bessel-unitary Fourier matrix of $S_E(v)$ is
  the matrix displayed in that proposition. For $|v| \le B$, $B \ge 1$, the
  upper off-diagonal entry is at most $\sqrt2 B$: use $|\sin(vk)| \le Bk$ for $k
  \le 1$ and $\langle k\rangle/k \le \sqrt2$ for $k \ge 1$. The lower entry is
  at most one, as are both diagonal entries; at zero the sine quotient has value
  $v$. Its Frobenius bound is $\sqrt{3+2B^2} \le 3B$. Hence, $G_{B,E} =
  \sup_{|v|\le B}\|S_E(v)\| \le 3B$. With
  \[
    \mathcal{M}_h(t)=\mathcal{T}_R(Q_{J,K}i_I^*h)(t),\qquad
    t_m=\|\mathcal{M}_h\|_{L^r(I_m;\mathcal{L}(E))},
  \]
  the primitive increment bound and the affine time change give
  \[
    \|(Lh)_m\|_{C^\theta}
    \le\sqrt\delta\,C_{B,E}\delta^{-1/r}t_m,\qquad
    C_{B,E}=G_{B,E}^2[B^{1-1/r}+B^{1-1/r-\theta}]
    \le18B^{3-1/r}.
  \]
  Here $1-1/r > \theta$, constant stopping preserves the bound, and the empty
  block contributes zero.

  Put $n = N+1 \ge 2$. The finite power-mean inequality at exponent $r/p > 1$
  with nonnegative inputs $t_m^p$ gives
  \[
    \left(\sum_m t_m^p\right)^{r/p}
    \le n^{r/p-1}\sum_m t_m^r,\qquad
    \left(\sum_m t_m^p\right)^{1/p}
    \le n^{1/p-1/r}\left(\sum_m t_m^r\right)^{1/r}.
  \]
  Raising nonnegative terms uses only positive powers, including when $t_m = 0$.
  The true partition gives $\sum_m t_m^r = \|\mathcal{M}_h\|_{L^r(I)}^r$ and
  $\delta n \le 2/B$. The exact $\delta$ weight therefore yields
  \[
    \|Lh\|_{\delta,p}
    \le\sqrt\delta\,C_{B,E}(\delta n)^{1/p-1/r}\|\mathcal{M}_h\|_{L^r}
    \le\sqrt\delta\,C_{B,E}(2/B)^{1/p-1/r}
    [M_0\varepsilon_J+M_fc_tK^{-\beta}]\|h\|_{H_W}.
  \]

  Finally, let $H_{\mathrm{seen}}$ be the closed span in $H_W$ of all
  coefficient tests $k_\ell$ with $\ell(X) = W(k_\ell)$. The first Wiener chaos
  Cameron identification in Proposition~\ref{P:cameron-actual-embedding} applies
  on $H_W$: $L$ kills $H_{\mathrm{seen}}^\perp$ and its Cameron space is
  $L(H_{\mathrm{seen}})$ with minimum norm of a preimage in the original
  covariance space. Thus, $\|Lh\|_{\mathrm{CM}} = \|P_{\mathrm{seen}}h\|_{H_W}$.
  Applying the bound to this minimum-norm representative gives precisely the
  Cameron embedding bound, without an inverse covariance or finite-dimensional
  density. This proves Proposition~\ref{P:uniform-actual-cameron}.
\end{proof}

The orthogonal $P_{\mathrm{full}}$ and $Q_{\mathrm{full}}$ Gaussian fields are
independent; residual time blocks can remain correlated. The retained spatial
head may have infinite rank. Heat filtering alone would not give this orthogonal
decomposition. These linear bounds do not bound the expected residual path norm
uniformly on growing masks, nor perform conditioning or integration over the
infinite sum. The remaining stochastic, product and pointwise upper estimates
require their own proofs.

\section{The compact mean and Cameron map on time blocks}
\label{S:actual-cameron-mean}

Retain the nonzero original covariance in Hypothesis~\ref{H:homog}, its
amplitude, $a = \alpha_0$, and the real phases $E = H^{1/2}\times H^{-1/2}$ and
$F = E\oplus\R^2$. Write $\Lambda = (1-\Delta)^{1/2}$. The form $B_\Gamma$ and
constant $R_{\mathrm{sp}}$ are those of
Section~\ref{S:spatial-occupation-foundation}. Let $H_{\mathrm{sp}}$ be the real
covariance-null completion of $C_c^\infty(\R^d)$ with inner product $B_\Gamma$,
and $H_t$ the analogous completion of $C_c^\infty(I)$, $I = (0,1)$, with time
kernel $|s-t|^{-a}$. Let $H = H_I$ be the completion of smooth tests supported
inside $I\times\R^d$. It embeds isometrically as a closed subspace of the given
$H_W$. We retain the given $W$ restricted to it.

Fix $R > 0$ and real even $\eta_R \in C_c^\infty(\R^d)$ with $0 \le \eta_R \le
1$, equal to one on the radius-$R$ ball and supported in the closed radius-$2R$
ball. The physical mask is $m_R = \eta_R^2$. Every operator path below takes its
mask argument equal to $m_R$: at this fixed radius the base profile $\zeta(y) =
m_R(Ry)$ meets the earlier hypotheses. The root $\eta_R$ factors state tests; it
does not multiply an original noise class. All compactness conclusions here fix
$R$.

\subsection{Fractional products and the compact spatial map}

The four-linear estimate~\eqref{E:spatial-four-linear} needs $H^1$ factors.
The first component of a state in $E$ lies only in $H^{1/2}$, so we first
extend that estimate to fractional Sobolev factors.

\begin{lemma}[Fractional covariance products]
  \label{L:cameron-fractional-products}
  For $\alpha/4 < s < 1/2$, products of real $H^s$ functions define elements of
  $H_{\mathrm{sp}}$, with
  \begin{equation}\label{E:cameron-fractional-product}
    \|fg\|_{H_{\mathrm{sp}}}
    \le C_s^2\sqrt{R_{\mathrm{sp}}}\|f\|_{H^s}\|g\|_{H^s}.
  \end{equation}
  Their Hilbert pairings equal their physical integrals. In particular,
  \begin{equation}\label{E:cameron-fractional-absolute}
    \int|f_1(x)f_2(x)f_3(x-y)f_4(x-y)|\ud x\,\Gamma(\ud y)
    \le C_s^4R_{\mathrm{sp}}\prod_{j=1}^4\|f_j\|_{H^s}.
  \end{equation}
\end{lemma}

\begin{proof}
  First approximate real $H^1$ factors by compact smooth ones. The absolute
  four-linear bound~\eqref{E:spatial-four-linear} and covariance PSD make their
  products Cauchy in $H_{\mathrm{sp}}$, with norm at most
  $\sqrt{R_{\mathrm{sp}}}N(f)N(g)$. Expanding product differences proves
  independence of approximation; physical continuity identifies all pairings.
  PSD for products also follows from $fg = ((f+g)^2-(f-g)^2)/4$.

  Use real radial Fourier bands $D_0 = 1_{\{|\xi|<1\}}$ and $D_j =
  1_{\{2^{j-1}\le|\xi|<2^j\}}$, $j \ge 1$. For $f \in H^s$ put $f_j = D_jf$.
  Each block is real and in $H^1$. Frequency support, Parseval and weighted
  Cauchy--Schwarz give
  \begin{equation}\label{E:cameron-dyadic-product}
    N(f_j)\le2^{j\alpha/4}\|f_j\|_2,\qquad
    \sum_{j\ge0}N(f_j)\le C_s\|f\|_{H^s}.
  \end{equation}
  Indeed, $\sum_j2^{-2j(s-\alpha/4)} < \infty$, while $\sum_j2^{2js}\|f_j\|_2^2
  \le 2^{2s}\|f\|_{H^s}^2$. The low block has factor one. No $L^p$ multiplier
  result is used. Also $\sum_j\|f_j\|_2 < \infty$ since $s > 0$. Thus,
  $\sum_j|f_j|$ lies in $L^2$ by Minkowski and monotone convergence, and the
  series is absolutely convergent a.e. with sum $f$.

  Apply the absolute $H^1$ four-linear estimate to each band quadruple. The sum
  of its absolute integrals is at most
  $R_{\mathrm{sp}}\prod_{\ell=1}^4\sum_jN((f_\ell)_j) < \infty$. Tonelli and the
  null-set transfer at both spatial positions in
  Lemma~\ref{L:spatial-physical-pair} identify the physical product series,
  proving~\eqref{E:cameron-fractional-absolute}. The double series
  $\sum_{j,k}f_jg_k$ is also absolutely convergent in $H_{\mathrm{sp}}$ by the
  $H^1$ product bound. Its partial sums are products of $H^1$ functions. The
  absolute quadruple calculation identifies their limiting physical pairings and
  gives~\eqref{E:cameron-fractional-product}. Compact smooth functions are dense
  in $H^s$: approximate in Fourier space by real Schwartz functions, then cut
  each off compactly in $H^1$ and hence in $H^s$. This also proves uniqueness of
  the bilinear extension. This proves Lemma~\ref{L:cameron-fractional-products}.
\end{proof}

With these products in $H_{\mathrm{sp}}$, we can extend the spatial mean map
from smooth tests to all of $H_{\mathrm{sp}}$.

\begin{proposition}[Compact spatial mean map]\label{P:cameron-spatial-map}
  The smooth-test map
  \[
    T_{\mathrm{sp}}q=iB_F(m_R(\Gamma*q))
  \]
  extends to a compact linear map $T_{\mathrm{sp}}:H_{\mathrm{sp}} \to
  \mathcal{K}(F)$, where $\mathcal{K}(F)$ has the operator norm. For $z =
  (z_1,z_2,c_1,c_2)$ and $w = (w_1,w_2,d_1,d_2)$ set $w' = \Lambda^{-1}w_2$. Its
  exact row identity is
  \begin{equation}\label{E:cameron-spatial-row}
    \langle T_{\mathrm{sp}}qz,w\rangle_F
    =\langle q,k(z,w)\rangle_{H_{\mathrm{sp}}},
    \qquad
    k(z,w)=(\eta_Rz_1+c_1\eta_R)(\eta_Rw').
  \end{equation}
\end{proposition}

\begin{proof}
  Fix $s$ with $\alpha/4 < s < 1/2$; this interval is nonempty because the
  original threshold implies $\alpha < 2$. For smooth compact $q$, $\Gamma*q$ is
  locally smooth by differentiation under the compactly restricted Radon
  integral. The masked potential is smooth compact. The velocity pairing in $E$
  has weight $\Lambda^{-1}$; hence physical Fubini
  gives~\eqref{E:cameron-spatial-row} on smooth states. Evenness of $\Gamma$
  changes $B_\Gamma(k,q)$ to $B_\Gamma(q,k)$. For general states the localized
  products are $L^1$, and $\Gamma*q$ is bounded on their compact support. $L^2$
  state approximation passes the physical integral to the limit, while
  Lemma~\ref{L:cameron-fractional-products} passes the $H_{\mathrm{sp}}$
  pairing. This proves the row.

  Multiplication by $\eta_R$ maps bounded $H^{1/2}$ sets to precompact $H^s$
  sets by Proposition~\ref{P:cutoff-singular-values}, with gain $1/2-s > 0$.
  This applies to $z_1$ and $w'$. The $c_1\eta_R$ family is finite dimensional
  and compact. The closures of both factor families for unit $z,w$ are compact,
  so their products have compact closure $\mathcal{L}$ in $H_{\mathrm{sp}}$ by
  the fractional product bound. The row identity gives $\|T_{\mathrm{sp}}q\| \le
  \sup_{k\in\mathcal{L}}\|k\|\|q\|$. Null covariance classes therefore map to
  zero.

  For smooth $q$, multiplication by its potential is compact $H^{1/2} \to
  H^{-1/2}$ by the same whole-space cutoff result with gain one. Its extra $c_1$
  column is finite rank, and the instantaneous $c_2$ column is zero. Thus, the
  smooth images lie in $\mathcal{K}(F)$, and so does the bounded extension by
  operator-norm closure.

  Choose a complete smooth-test orthonormal basis of $H_{\mathrm{sp}}$ by
  countable compact-support approximation and Gram--Schmidt, omitting null
  classes. Let $Q_n$ project onto its first $n$ terms. The row identity yields
  \[
    \|T_{\mathrm{sp}}(I-Q_n)\|
    \le\sup_{k\in\mathcal{L}}\|(I-Q_n)k\|_{H_{\mathrm{sp}}}
    \longrightarrow0.
  \]
  Strong projection convergence is uniform on compact $\mathcal{L}$ by a
  finite-net argument and $\|I-Q_n\| \le 1$. Each $T_{\mathrm{sp}}Q_n$ has
  finite-dimensional range; their operator-norm limit is a compact map. This is
  compactness of the map, in addition to compact values. This proves
  Proposition~\ref{P:cameron-spatial-map}.
\end{proof}

\subsection{The supported tensor and temporal mean}

Proposition~\ref{P:cameron-spatial-map} treats the spatial variable alone. A
control in $H$ depends on time and space, so we first split $H$ into a time
factor and a spatial factor.

\begin{lemma}[Supported original tensor space]\label{L:cameron-supported-tensor}
  Canonically $H = H_t\otimes H_{\mathrm{sp}}$ as real Hilbert spaces. It is
  separable and admits a complete orthonormal basis $e_j$ with smooth test
  representatives supported inside $I\times\R^d$. Projection $P_J$ onto its
  first $J$ elements is also an orthogonal projection on $H_W$ by the same
  finite formula.
\end{lemma}

\begin{proof}
  On separated smooth tests, absolute Fubini makes the original inner product
  exactly the tensor product after both null quotients. Finite sums are dense:
  approximate a smooth compact spacetime test uniformly by smooth time
  partitions times its spatial slices, on one common compact support inside
  $I\times\R^d$. A uniform error $\epsilon$ on spatial compact $A$ has squared
  original covariance norm at most
  \[
    \epsilon^2|A|\Gamma(A-A)
    \int_I\int_I|s-t|^{-a}\ud s\ud t<\infty.
  \]
  This also gives countable smooth-test density on exhausting compact sets, both
  here and in $H_{\mathrm{sp}}$. Gram--Schmidt skipping zeros preserves smooth
  compact support. The isometric embedding into $H_W$ has closed image by
  completeness; its finite orthonormal projection formula remains orthogonal
  there. It is not multiplication by a time indicator and gives no independence
  of disjoint time intervals. This proves
  Lemma~\ref{L:cameron-supported-tensor}.
\end{proof}

It remains to treat the time factor $H_t$, whose kernel is $|s-t|^{-a}$.
Proposition~\ref{P:cameron-full-mean} then combines the two maps.

\begin{proposition}[The compact temporal mean, without trace class]
  \label{P:cameron-temporal-map}
  Let $K$ on $L^2(I)$ have kernel $|s-t|^{-a}$. It is compact, self-adjoint,
  strictly positive and injective. The map $\iota:f\mapsto K^{1/2}f$ extends
  isometrically from $H_t$ onto $L^2(I)$, and the temporal mean
  \begin{equation}\label{E:cameron-temporal-map}
    C_t=K^{1/2}\iota:H_t\longrightarrow L^r(I)
  \end{equation}
  is bounded and compact for every $2 < r < 2/a$. The displayed formula first
  has its $L^2(I)$ meaning; the same completed map is then shown to take values
  compactly in $L^r(I)$. The operator $K$ is not trace class and $C_t:H_t \to
  L^2(I)$ is not Hilbert--Schmidt.
\end{proposition}

\begin{proof}
  Schur bounds $K$ by its finite row and column integrals. The Gamma
  representation of $|s-t|^{-a}$ and zero extension of $f \in L^2(I)$ give
  strict positivity directly: the kernel $e^{-\lambda|v|}$ has quadratic form
  \[
    \int_\R\frac{2\lambda}{\lambda^2+\xi^2}
    |\widehat{f}(\xi)|^2\ud\xi>0
    \quad(f\ne0,\ \lambda>0)
  \]
  in the unitary Fourier convention. Absolute Fubini is justified by Schur
  applied to $|f(s)f(t)|$. Integrating with positive weight
  $\lambda^{a-1}/\Gamma_E(a)$ proves positivity and injectivity. Symmetry gives
  self-adjointness. The deterministic truncated kernel $K_\epsilon =
  K1_{\{|s-t|\ge\epsilon\}}$ is square integrable, so finite separated-kernel
  approximations make it compact. The Schur norm of $K-K_\epsilon$ is at most
  $2\epsilon^{1-a}/(1-a)$, proving compactness of $K$. These truncations are not
  used as Gaussian covariances. Its positive square root is compact by the
  compact spectral theorem.

  On tests $\|K^{1/2}f\|_2 = \|f\|_{H_t}$. Smooth-test $L^2$ density and $\ker K
  = 0$ imply that its range is dense, so the isometry extends onto $L^2(I)$. On
  tests $C_tf = Kf$, and~\eqref{E:cameron-temporal-map} first gives compactness
  into $L^2$.

  Extend $K_a(v) = |v|^{-a}1_{\{|v|\le1\}}$ to the line. It lies in $L^{r/2}$.
  Scalar Young convolution gives
  \[
    \|g\|_{H_t}^2\le\|K_a\|_{r/2}\|g\|_{r'}^2,\qquad
    r'=r/(r-1).
  \]
  To verify the needed Young inequality, take $u = r/2$, $v = r'$, so $u,v < r$
  and $1/u+1/v = 1+1/r$. Split the absolute convolution integrand into
  $(|f|^u|h|^v)^{1/r}$, $|f|^{1-u/r}$ and $|h|^{1-v/r}$. H\"older with exponents
  $r$, $(1/u-1/r)^{-1}$ and $(1/v-1/r)^{-1}$, followed by integration and
  Tonelli, gives $\|f*h\|_r \le \|f\|_u\|h\|_v$. Covariance Cauchy--Schwarz and
  $L^r$ duality now imply $\|Kf\|_r \le C_{a,r}\|f\|_{H_t}$ on tests. Their $Kf$
  are bounded by the integrable-kernel row, making duality legitimate before
  completion. Density extends the estimate, with $L^2$ value equal
  to~\eqref{E:cameron-temporal-map}. Choose $r < r_2 < 2/a$. Compactness into
  $L^2$ and boundedness into $L^{r_2}$ give compactness into $L^r$ by $\|u\|_r
  \le \|u\|_2^\tau\|u\|_{r_2}^{1-\tau}$ with $1/r = \tau/2+(1-\tau)/r_2$.

  Finally, the orthonormal cell indicators $N^{1/2}1_{(j/N,(j+1)/N)}$, $0 \le j
  < N$, have $K$ quadratic forms summing to $2N^a/((1-a)(2-a)) \to \infty$.
  Thus, positive $K$ has infinite trace, and $K^{1/2}\iota$ is not
  Hilbert--Schmidt into $L^2$. No instantaneous $L^2$-valued Gaussian has been
  introduced. This proves Proposition~\ref{P:cameron-temporal-map}.
\end{proof}

No-go~\ref{P:nogo-27} explains the upper endpoint of this compactness
range: temporal concentration makes the mean map unbounded above
$2/\alpha_0$ and noncompact at $2/\alpha_0$. The endpoint witness does not
refute boundedness there.


\begin{proposition}[The compact time-space mean]\label{P:cameron-full-mean}
  Fix one $2 < r < 2/a$ throughout this proposition. For finite tensors with
  orthonormal spatial labels define
  \[
    \mathcal{U}\left(\sum_j f_j\otimes q_j\right)(s)
    =\sum_j(Kf_j)(s)q_j.
  \]
  It extends to a bounded map $H \to L^r(I;H_{\mathrm{sp}})$ with dimension-free
  bound. The map
  \begin{equation}\label{E:cameron-full-mean}
    T\Phi(s)=T_{\mathrm{sp}}(\mathcal{U}\Phi(s))
    :H\longrightarrow L^r(I;\mathcal{K}(F))
  \end{equation}
  is compact. For smooth supported $\Phi$, it is the multiplier
  $iB_F(m_Rh_\Phi(s))$, where $h_\Phi = |\cdot|^{-a}*(\Gamma*\Phi)$. For the
  supported projections of Lemma~\ref{L:cameron-supported-tensor},
  \begin{equation}\label{E:cameron-compact-tail}
    \epsilon_J:=\|T(I-P_J)\|_{H\to L^r(\mathcal{K}(F))}
    \longrightarrow0.
  \end{equation}
\end{proposition}

\begin{proof}
  For a finite-dimensional spatial field $g$, positivity of the time kernel and
  scalar Young give
  \[
    \|g\|_{H_t\otimes H_{\mathrm{sp}}}^2
    \le\int_I\int_I|s-t|^{-a}\|g_s\|\|g_t\|\ud s\ud t
    \le C_{a,r}^2\|g\|_{L^{r'}(H_{\mathrm{sp}})}^2.
  \]
  Hilbert Cauchy--Schwarz and finite-dimensional $L^r$ duality give
  $\|\mathcal{U}\Phi\|_{L^r(H_{\mathrm{sp}})} \le C_{a,r}\|\Phi\|_H$. The
  duality can be checked by the pointwise direction
  $\|\mathcal{U}\Phi\|^{r-2}\mathcal{U}\Phi$ and smooth coefficient
  approximation in $L^{r'}$, controlled by the displayed estimate. No dimension
  factor occurs. Tensor density and completeness extend this to a strongly
  measurable Bochner field. This time-only map still contains spatial vectors in
  the covariance space. Spatial covariance is applied once, by
  $T_{\mathrm{sp}}$.

  The physical mean identity on separated tests follows
  from~\eqref{E:cameron-spatial-row}. For a smooth spacetime test use the
  separated compact-support approximations in the tensor proof. Their uniform
  errors give uniformly small masked mean errors, using the finite time row and
  locally finite spatial covariance on the relevant compact differences. The
  multiplier bound by $L^\infty+L^2$ then gives uniform operator convergence,
  identifying the $L^r$ mean in~\eqref{E:cameron-full-mean}. General $H$
  controls denote completed operator fields, without a pointwise-potential
  assertion.

  Let $Q_n$ be the spatial projections in the compact-map proof. The maps
  $T_n\Phi = T_{\mathrm{sp}}Q_n\mathcal{U}\Phi$ approximate $T$ in operator norm
  by the dimension-free bound and $\|T_{\mathrm{sp}}(I-Q_n)\| \to 0$. For fixed
  $n$, each spatial coordinate extraction $H \to H_t$ has norm at most one and
  its image under $C_t$ is precompact in $L^r$. Finite combination with the
  compact operators $T_{\mathrm{sp}}q_j$ makes $T_n$ compact. Hence, $T$ is
  compact. Scalar time compactness alone tensored with an infinite identity
  would not prove this.

  To prove~\eqref{E:cameron-compact-tail}, if it failed choose unit $v_J$ on a
  subsequence with $\|T(I-P_J)v_J\|$ bounded below. The vectors $(I-P_J)v_J$
  tend weakly to zero, since their pairings with fixed $h$ are bounded by
  $\|(I-P_J)h\|$. Compactness gives a norm-convergent image subsequence with
  weak limit zero, a contradiction. This proves
  Proposition~\ref{P:cameron-full-mean}.
\end{proof}

The radius remains fixed in this approximation. No-go~\ref{P:nogo-25}
shows that translated white-noise controls prevent these fixed smooth
projections from giving tails uniform over growing radii.


\subsection{The noise residual and its Cameron operator}

Choose integer $B \ge 1$ and $0 < \delta < \min(1,1/B)$. Put $N =
\lfloor1/(\delta B)\rfloor$, $s_m = m\delta B$, $\ell_m = B$ for $m < N$ and
$\ell_N = \delta^{-1}-NB \in [0,B)$. The intervals $I_m =
[s_m,s_m+\delta\ell_m]$ partition $[0,1]$ up to endpoints. With $Q_J = I-P_J$ on
$H_W$ and the explicit pullback~\eqref{E:raw-block-pullback}, define
\[
  Y_{m,J}(f)=W(Q_JT_{\delta,m}f),\qquad
  g_\delta=\delta^{(1-a)/2}.
\]
Fix parameters
\begin{equation}\label{E:cameron-block-parameters}
  \frac12<\theta<\min\left(1-\frac a2,\frac{3-a-\alpha}{2}\right),
  \qquad \frac1{1-\theta}<r<\frac2a,\qquad
  \frac1\theta\le p<2<r .
\end{equation}
These intervals are nonempty.

\begin{lemma}[Supported residual paths]\label{L:cameron-actual-residual}
  Gaussian transfer with mask $m_R$ and its $F$ extension constructs the
  centered Gaussian paths
  \[
    X_{m,J}(u)=g_\delta A_{m_R}(Y_{m,J})(\min(u,\ell_m)),
    \qquad 0\le u\le B.
  \]
  Their finite tuple lies in the separable space $\mathscr{B}_0^{N+1}$, where
  $\mathscr{B}_0 = h_0^\theta([0,B];\mathcal{K}(F))$. Every continuous linear
  statistic $\ell(X)$ is $W(k_\ell)$ for some $k_\ell \in H_I$. An empty last
  interval gives zero.
\end{lemma}

\begin{proof}
  Isometry of the pullback and orthogonality of $Q_J$ give
  $\operatorname{Var}Y_{m,J}(f) \le \|f\|_{H_W}^2$.
  Proposition~\ref{P:source-gaussian-transfer} therefore applies. Its finite
  sums retain both deterministic localizations, and their limits retain the true
  $F$ extension and constant columns. Choose $\theta < \beta < \beta_2$ below
  the upper threshold in~\eqref{E:cameron-block-parameters}; read the
  operator-path construction at exponent $\beta$ with auxiliary $\beta_2$.
  Constant stopping preserves compact values and $C^\beta$ bounds, hence gives
  paths in the little $C^\theta$ closure by polygonal approximation. Its
  separability is Lemma~\ref{L:raw-block-inputs}.

  As in Lemma~\ref{L:raw-block-inputs}, every observed row's slow pullback
  belongs to $H_I$: its time support is in $[0,1]$, and approximating its
  indicator inside $I$ converges in original covariance norm by dominated
  convergence against the integrable time kernel and locally finite $\Gamma$ on
  the common compact support. This includes endpoints. Schwartz $E$ states and
  both true constant columns supply the identities, then boundedness extends
  them to $F$. Let $P_H$ be the orthogonal projection onto closed $H_I$.
  Transfer for the noise $f\mapsto W(P_HQ_JT_{\delta,m}f)$ leaves each observed
  row unchanged, since its residual slow test already lies in $H_I$. Countable
  states and rational times identify the same operator paths. This
  projected-transfer series has coefficient tests in $H_I$. Applying $\ell$
  gives an $L^2$ limit in the closed isonormal image $W(H_I)$, proving the
  assertion. Original global-basis tests were not assumed individually
  supported. No time-support independence or Hilbert-valued noise is used. This
  proves Lemma~\ref{L:cameron-actual-residual}.
\end{proof}

\begin{proposition}[Cameron covariance and averaged bound]
  \label{P:cameron-actual-embedding}
  The covariance map $L_{\delta,J}\Phi = \E[XW(\Phi)]$ is
  \begin{equation}\label{E:cameron-actual-covariance}
    (L_{\delta,J}\Phi)_m(u)=\sqrt\delta
    \int_0^{\min(u,\ell_m)}
    S_F(-v)T(Q_J\Phi)(s_m+\delta v)S_F(v)\ud v ,
    \qquad \Phi\in H.
  \end{equation}
  This is a compact-operator Bochner integral. Define the exact norm and
  constants
  \begin{equation}\label{E:cameron-averaged-norm}
    \|x\|_{\delta,p}
    =\left(\delta\sum_{m=0}^N\|x_m\|_{C^\theta}^p\right)^{1/p},
    \quad G_B=\sup_{|v|\le B}\|S_F(v)\|,
    \quad C_B=G_B^2[B^{1-1/r}+B^{1-1/r-\theta}].
  \end{equation}
  Then the Gaussian Cameron embedding satisfies
  \begin{equation}\label{E:cameron-averaged-bound}
    \|L_{\delta,J}\Phi\|_{\delta,p}
    \le\sqrt\delta\,C_B(2/B)^{1/p-1/r}
    \epsilon_J\|\Phi\|_H .
  \end{equation}
  Convergence $\epsilon_J \to 0$ fixes $R$ and is independent of $\delta$.
\end{proposition}

\begin{proof}
  For smooth supported $\Phi$, $Q_J\Phi$ is another such test. The explicit
  pullback covariance~\eqref{E:raw-block-mean-jacobian} gives
  \[
    \E[Y_{m,J}(f)W(\Phi)]
    =\delta^{a/2}\int f(v,x)
    h_{Q_J\Phi}(s_m+\delta v,x)\ud x\ud v .
  \]
  Absolute Fubini also holds for signed compact tests. Multiplying by $g_\delta$
  gives $\sqrt\delta$. Explicit rows and~\eqref{E:cameron-spatial-row} give
  \eqref{E:cameron-actual-covariance}. The constant-velocity input acquires
  displacement $v$ under $S_F(v)$; the constant displacement remains one. Both
  columns are included.

  Both $S_F$ and its adjoint are strongly continuous on Hilbert $F$. In
  Bessel-unitary $E$ coordinates the Fourier matrix is
  \[
    \begin{pmatrix}
      \cos(v|\xi|)&\langle\xi\rangle\sin(v|\xi|)/|\xi|\\
      -|\xi|\sin(v|\xi|)/\langle\xi\rangle&\cos(v|\xi|)
    \end{pmatrix}.
  \]
  Its continuous zero quotient and common bound $C(1+B)$ apply also to its
  adjoint; dominated convergence proves the assertion. The constant block is
  finite dimensional. For compact $A$, conjugation $S_F(-v)AS_F(v)$ is norm
  continuous: strong convergence is uniform on compact images, and the right
  factor follows by adjoints. Joint norm continuity in $(v,A)$ and uniform
  bounds give Bochner measurability for the conjugated strongly measurable
  compact-operator field. No norm continuity of the free group itself is
  assumed.

  Write $M = TQ_J\Phi$. H\"older and the primitive increment estimate give
  \begin{equation}\label{E:cameron-local-primitive-bound}
    \|(L_{\delta,J}\Phi)_m\|_{C^\theta}
    \le\sqrt\delta\,C_B\delta^{-1/r}\|M\|_{L^r(I_m)}.
  \end{equation}
  Its primitive exponent is $1-1/r > \theta$. Simple compact-operator integrand
  approximation in $L^r$ puts the primitive in $\mathscr{B}_0$; constant
  stopping does not increase the H\"older norm. Affine pullback preserves null
  sets. For fixed $\delta,J$ the covariance map is bounded by Bochner
  Cauchy--Schwarz and finite tuple $L^2$ moments. The integral is bounded by the
  displayed estimate, so equality on smooth controls extends by density to all
  $H$. Dense states and rational times make it a full operator-path identity.

  Let $H_{\mathrm{seen}}$ be the closed span of the $k_\ell$ from
  Lemma~\ref{L:cameron-actual-residual}. Then $\ell(L_{\delta,J}\Phi) = \langle
  k_\ell,\Phi\rangle_H$ and $\ker L_{\delta,J} = H_{\mathrm{seen}}^\perp$. The
  first Wiener chaos definition of the Cameron space is therefore
  $L_{\delta,J}(H_{\mathrm{seen}})$ with the minimum $H$ preimage norm. Extra
  directions are removed by orthogonal projection, without an inverse
  covariance. A bound in the original $H$ norm is thus the embedding bound.

  Put $t_m = \|M\|_{L^r(I_m)} \ge 0$ and $n = N+1 \ge 2$. Finite power mean at
  exponent $r/p > 1$, applied to $t_m^p$, gives
  \[
    \left(\sum_m t_m^p\right)^{1/p}
    \le n^{1/p-1/r}\left(\sum_m t_m^r\right)^{1/r}.
  \]
  The true partition gives $\sum_m t_m^r = \|M\|_{L^r(I)}^r$, including an empty
  last interval, and $n \le 2/(\delta B)$. Hence, the exact $\delta$-weighted
  norm gives
  \[
    \|L_{\delta,J}\Phi\|_{\delta,p}
    \le\sqrt\delta\,C_B(\delta n)^{1/p-1/r}\|M\|_{L^r}
    \le\sqrt\delta\,C_B(2/B)^{1/p-1/r}
    \epsilon_J\|\Phi\|_H.
  \]
  The $B$ factor remains explicit; covariance and mask occur once. This proves
  Proposition~\ref{P:cameron-actual-embedding}.
\end{proof}

These conclusions fix the radius and make no uniform-radius projection tail
assertion. No concentration estimate or original moment upper bound follows from
this section alone.

\section{The unmasked Rayleigh rate and nonnegative controls}
\label{S:unmasked-variational}

Fix a nonzero covariance satisfying Hypothesis~\ref{H:homog}, including its
original amplitude, and set $a = \alpha_0$. Define the full class
\begin{equation}\label{E:variational-positive-class}
  \mathcal{C}_{\mathrm{pos}}
  =\{\Psi\in C_c^\infty((0,1)\times\R^d;\R):
  \Psi\ge0,\ \Psi\ne0\}.
\end{equation}

\input{notes/figures/nonseparable-controls}
It includes nonseparable tests. It is nonempty: if $b(z) = e^{-1/(1-z^2)}$ on
$|z| < 1$ and zero otherwise, then $b(4s-2)\prod_{j=1}^db(x_j)$ belongs to this
class. All boundary derivatives vanish because exponential decay dominates every
power of $(1-z^2)^{-1}$. For each $\Psi$ retain the original mean $h_{\Psi,s}$
and original variance $v_\Psi = \E W(\Psi)^2$ from~\eqref{E:compact-raw-mean}--%
\eqref{E:compact-raw-cost}. Neither the test nor its covariance is changed by a
physical mask. We write $h_s$ when $\Psi$ is fixed.

\subsection{Global bounds and the unmasked spectral bottom}

We first record a bound for the covariance smoothed by the heat kernel. The
proof of Lemma~\ref{L:variational-global-mean} uses it to bound
$\Gamma*|D\Psi(t,\cdot)|$ uniformly in $t,x$.

\LeanMarker{L:variational-heat-envelope}
\begin{lemma}[Gaussian envelope for the original covariance]
  \label{L:variational-heat-envelope}
  For the normalized heat kernel $p_\epsilon(x) =
  (2\pi\epsilon)^{-d/2}e^{-|x|^2/(2\epsilon)}$,
  \begin{equation}\label{E:variational-heat-envelope}
    (\Gamma*p_\epsilon)(x)
    =\int e^{i\xi\cdot x}e^{-\epsilon|\xi|^2/2}\mu(\ud\xi),
    \qquad
    0\le(\Gamma*p_\epsilon)(x)
    \le(\Gamma*p_\epsilon)(0)<\infty.
  \end{equation}
  The value at zero is strictly positive.
\end{lemma}

\begin{proof}
  A positive tempered Radon measure has polynomial ball growth. Indeed, evaluate
  its tempered bound on a nonnegative smooth cutoff equal to one on $B_1$,
  dilated by $R \ge 1$. Its finitely many Schwartz seminorms are bounded by
  $CR^N$, so positivity gives a ball-mass bound $CR^N$. Summing dyadic annuli
  shows that $(1+|x|)^{-L}$ is integrable for $L > N$. Apply this separately to
  $\Gamma$ and $\mu$. Gaussian weights are therefore integrable against both
  measures. The classical convolution on the left is finite and smooth locally
  in $x$: on compact $x$ sets its translated Gaussian derivatives are bounded by
  any fixed sufficiently fast polynomial weight in the integration variable.

  Pair the convolution with a Schwartz test $\varphi$. Absolute Fubini is
  justified by these weights. The resulting pairing is that of $\Gamma$ with
  $p_\epsilon*\varphi$. Applying the spectral convention in
  Hypothesis~\ref{H:homog} and the identity $\int p_\epsilon(y)e^{-i\xi\cdot
  y}\ud y = e^{-\epsilon|\xi|^2/2}$ gives exactly the right side
  of~\eqref{E:variational-heat-envelope} as a distribution. The heat kernel is
  even, so either Fourier sign gives that same Gaussian multiplier. Its spectral
  integral is absolutely finite and continuous in $x$ by dominated convergence.
  Thus, equality of the two continuous distributions gives equality at every
  $x$, without an extra normalization factor. The physical convolution is
  nonnegative. Evenness of $\mu$ makes the spectral integral real, and its
  absolute value is bounded by its total Gaussian mass, the value at zero.
  Finally, $\Gamma \ne 0$ and $p_\epsilon > 0$ give strict positivity there.
  This proves Lemma~\ref{L:variational-heat-envelope}.
\end{proof}

\LeanMarker{L:variational-global-mean}
\begin{lemma}[Global regularity of the original mean]
  \label{L:variational-global-mean}
  Every mixed time and space derivative of $h_s$ is globally bounded in space,
  uniformly on compact time intervals. In particular, for $I = [0,1]$ there are
  finite constants $M,C_\Psi$ such that
  \begin{equation}\label{E:variational-global-mean}
    0\le h_s(x)\le M,\qquad
    \|h_s-h_t\|_\infty\le C_\Psi|s-t|,
    \qquad s,t\in I.
  \end{equation}
\end{lemma}

\begin{proof}
  Extend $\Psi$ by zero in time; its compact support inside $(0,1)$ makes this a
  smooth extension. For any mixed derivative $D\Psi$, a fixed $\epsilon > 0$ and
  compactness of its common spatial support give $|D\Psi(t,x)| \le
  C_Dp_\epsilon(x)$ for all $t,x$. Here the heat kernel has a positive minimum
  on that support. Lemma~\ref{L:variational-heat-envelope} consequently bounds
  $\Gamma*|D\Psi(t,\cdot)|$ uniformly in $t,x$. Ordinary local differentiation
  against the Radon measure gives the corresponding derivatives of the
  convolution. After the change $r = s-t$,
  \[
    h_s(x)=\int_\R |r|^{-a}
    (\Gamma*\Psi(s-r,\cdot))(x)\ud r.
  \]
  For $s$ in a compact interval the relevant $r$ lie in a fixed bounded
  interval. The preceding global bound times $|r|^{-a}$ is integrable since $a <
  1$. Every derivative thus falls on the smooth test and passes through the
  integral by dominated convergence, with the claimed global spatial bounds.
  Nonnegativity follows from positive kernels. The bounded time derivative
  proves the Lipschitz estimate. No directional decay of the mean has been
  inferred. This proves Lemma~\ref{L:variational-global-mean}.
\end{proof}

Lemma~\ref{L:variational-global-mean} shows that $h_s$ is bounded by $M$ for $s
\in I$ and is Lipschitz in $s$ in the supremum norm. We now use these two bounds
to define the rate $\lambda_{\Psi,s}$ of the potential $h_s$ without a mask and
to show that it is Lipschitz on $I$.

\LeanMarker{P:variational-unmasked-rate}
\begin{proposition}[The unmasked Rayleigh rate]
  \label{P:variational-unmasked-rate}
  On complex $L^2(\R^d)$, the operator $H_s = -\Delta-h_s$ is self-adjoint on
  $H^2$, with form domain $H^1$ and form $q_s(f) = \|\nabla f\|_2^2-\int
  h_s|f|^2$. Its spectral bottom $E_s$ is uniformly negative on $I$. The
  positive rate $\lambda_{\Psi,s} = \sqrt{-E_s}$ satisfies
  \begin{equation}\label{E:variational-real-rayleigh}
    \lambda_{\Psi,s}^2
    =\sup_{\substack{f\in H^1(\R^d;\R)\\\|f\|_2=1}}
    \left(\int h_sf^2-\|\nabla f\|_2^2\right).
  \end{equation}
  It is Lipschitz on $I$, and $I_\Psi: = \int_0^1\lambda_{\Psi,s}\ud s$ is
  positive and finite. No eigenstate at the unmasked bottom is asserted.
\end{proposition}

\begin{proof}
  The Fourier Laplacian $A = -\Delta$ is self-adjoint on $H^2$, being
  multiplication by $|\xi|^2$ with its maximal domain. Multiplication $B_s =
  -h_s$ is bounded self-adjoint with norm at most $M$. Thus, $H_s = A+B_s$ is
  densely defined, symmetric and closed on $H^2$; the two graph norms are
  equivalent under a bounded perturbation. For real $b > M$,
  \[
    H_s-ib=(I+B_s(A-ib)^{-1})(A-ib).
  \]
  The first factor has a Neumann inverse because $\|B_s(A-ib)^{-1}\| \le M/b <
  1$, and the same holds with $+ib$. To check self-adjointness directly, for $u
  \in D(H_s^*)$ solve $(H_s-ib)v = (H_s^*-ib)u$ with $v \in D(H_s)$. Then $u-v
  \in \ker(H_s^*-ib) = \operatorname{Ran}(H_s+ib)^\perp = \{0\}$. Hence,
  $D(H_s^*) = D(H_s)$.

  Set $c_{\mathrm{shift}} = 1+M$ and $J_s = H_s+c_{\mathrm{shift}} \ge 1$.
  Integration by parts on $H^2$ gives
  \[
    \|J_s^{1/2}f\|_2^2=q_s(f)+c_{\mathrm{shift}}\|f\|_2^2,\qquad
    \|f\|_{H^1}^2\le\|J_s^{1/2}f\|_2^2
    \le c_{\mathrm{shift}}\|f\|_{H^1}^2.
  \]
  Spectral cutoffs $1_{[1,n]}(J_s)$ show that $H^2 = D(J_s)$ is a core for the
  half-power norm; Fourier cutoffs give its density in $H^1$. Completion in the
  equivalent norms identifies $D(J_s^{1/2})$ and $H^1$ as $L^2$ subspaces, since
  both norms dominate $L^2$. This proves the stated form domain.

  The spectral lower bound gives one direction of the Rayleigh identity. For the
  reverse, nonzero spectral projections in $[E_s,E_s+\epsilon]$ give unit $H^2$
  vectors with Rayleigh values at most $E_s+\epsilon$. The bottom is finite
  because the operator is bounded below and has finite Rayleigh tests. Let
  $\epsilon\downarrow0$. Real tests give the same infimum: $q_s(u+iv) =
  q_s(u)+q_s(v)$, so the quotient is the norm-weighted average of the nonzero
  real quotients. This proves the Rayleigh identity with $-E_s$ on the left. The
  following negative upper bound justifies writing that value as
  $\lambda_{\Psi,s}^2$ in~\eqref{E:variational-real-rayleigh}.

  Proposition~\ref{P:compact-original-common-trial} supplies, for this original
  $\Psi$, one unit compactly supported $w$ and a finite plateau mask with a
  common strictly negative upper bound $\epsilon_0 > 0$. Its trial has support
  in $B_{2L}$; choose its plateau radius $R_0 \ge 2L$. The unmasked form on this
  same $w$ is no larger. Consequently, $-M \le E_s \le -\epsilon_0 < 0$, and
  $\sqrt{\epsilon_0} \le \lambda_{\Psi,s} \le \sqrt M$. Comparing identical unit
  form tests gives $|E_s-E_t| \le \|h_s-h_t\|_\infty$ and therefore
  \[
    |\lambda_{\Psi,s}-\lambda_{\Psi,t}|
    \le \frac{C_\Psi}{2\sqrt{\epsilon_0}}|s-t|.
  \]
  This also proves positivity and finiteness of $I_\Psi$. No compact-potential
  spectral theorem was applied to $H_s$. This proves
  Proposition~\ref{P:variational-unmasked-rate}.
\end{proof}

\subsection{Removing masks from the numerical lower rate}

Proposition~\ref{P:raw-block-original-lower} gives the lower bound with the rate
of the masked potential $m_Rh_s$, for one fixed mask. We now let the mask radius
$R$ tend to infinity and replace this rate by the rate $\lambda_{\Psi,s}$ of
Proposition~\ref{P:variational-unmasked-rate}.

\LeanMarker{P:variational-remove-mask}
\begin{proposition}[Monotone rates and the original point moment]
  \label{P:variational-remove-mask}
  There is an increasing family of admissible radial masks $m_R$ tending to one,
  for which the compact ground rates satisfy
  $\lambda_{R,s}\uparrow\lambda_{\Psi,s}$ uniformly on $I$ as $R \to \infty$,
  with the compact ground statements used for $R \ge R_0(\Psi)$. For the
  original canonical solution and any fixed $x$,
  \begin{equation}\label{E:variational-unmasked-point-lower}
    \liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,x)|^2
    \ge2I_\Psi-\frac{v_\Psi}{2}.
  \end{equation}
\end{proposition}

\begin{proof}
  Put $F(t) = e^{-1/t}$ for $t > 0$ and zero otherwise, and define
  \[
    \chi(r)=\frac{F(2-r)}{F(2-r)+F(r-1)},\qquad
    m_R(x)=\chi(|x|/R).
  \]
  The denominator is positive. The numerator decreases and the second
  denominator term increases; thus $\chi$ is nonincreasing, equals one on $r \le
  1$, and zero on $r \ge 2$. Flatness of $F$ at zero gives smoothness. Since
  $\chi$ is constant near zero, $m_R$ is smooth also at the spatial origin.
  These even $[0,1]$ masks increase pointwise to one.

  Choose $R_0$ so the preceding common trial lies in the plateau. Every $R \ge
  R_0$ has the same negative margin. For each fixed $R$, the compact-family
  theorem applies to $m_Rh_s$. Its real Rayleigh formula follows from the same
  real/complex quotient argument as~\eqref{E:variational-real-rayleigh}. The
  rates are continuous, positive, at most $\sqrt M$ and monotone in $R$. For
  every fixed real unit $f \in H^1$, dominated convergence gives $\int m_Rh_sf^2
  \to \int h_sf^2$. Hence,
  \[
    \sup_{R\ge R_0}\lambda_{R,s}^2
    =\sup_{\|f\|_2=1,\ f\in H^1(\R^d;\R)}
    \sup_{R\ge R_0}
    \left(\int m_Rh_sf^2-\|\nabla f\|_2^2\right)
    =\lambda_{\Psi,s}^2.
  \]
  Only two numerical suprema were interchanged. Positive square roots give the
  pointwise rate convergence. For $R_n = R_0+n$ and $\epsilon > 0$, the
  increasing open sets $\{s:\lambda_{\Psi,s}-\lambda_{R_n,s}<\epsilon\}$ cover
  compact $I$. A finite subcover lies in the largest one, proving uniform
  convergence. Monotonicity extends it to all real $R \to \infty$. Therefore,
  $\int_I\lambda_{R,s}\ud s\uparrow I_\Psi$.

  At each fixed $R$ use this very same mask in the operator path and mean when
  applying Proposition~\ref{P:raw-block-original-lower}. It gives the lower
  bound with $2\int_I\lambda_{R,s}\ud s-v_\Psi/2$ for the same original point
  liminf. Taking the numerical supremum over these individual bounds proves
  \eqref{E:variational-unmasked-point-lower}. There is no $R(T)$, unmasked noisy
  wave, uniform-in-mask Gaussian estimate or exchange of source and time limits.
  This proves Proposition~\ref{P:variational-remove-mask}.
\end{proof}

\subsection{Full-class dilation and finite variational values for nonnegative controls}

The bound~\eqref{E:variational-unmasked-point-lower} holds for each $\Psi \in
\mathcal{C}_{\mathrm{pos}}$ separately. Before we take the supremum over $\Psi$,
we record how $h_{\Psi,s}$, $v_\Psi$, $\lambda_{\Psi,s}$ and $I_\Psi$ change
under a spatial dilation of the control. The proof of
Proposition~\ref{P:variational-positive-scores} uses this to relate the
variational values for different coefficients $c$.

\LeanMarker{L:variational-control-dilation}
\begin{lemma}[Radon and Rayleigh dilation]\label{L:variational-control-dilation}
  For $r > 0$ the map
  \begin{equation}\label{E:variational-control-dilation}
    (D_r\Psi)(s,x)=r^{d+2-\alpha}\Psi(s,rx)
  \end{equation}
  is a bijection of $\mathcal{C}_{\mathrm{pos}}$, with inverse $D_{1/r}$. With
  the same original covariance,
  \begin{equation}\label{E:variational-dilation-values}
    h_{D_r\Psi,s}(x)=r^2h_{\Psi,s}(rx),\qquad
    v_{D_r\Psi}=r^{4-\alpha}v_\Psi,\qquad
    \lambda_{D_r\Psi,s}=r\lambda_{\Psi,s},\quad
    I_{D_r\Psi}=rI_\Psi.
  \end{equation}
\end{lemma}

\begin{proof}
  Positive dilation preserves the full smooth positive nonzero class;
  multiplication of its amplitude powers gives the stated inverse. Homogeneity
  of the physical Radon measure gives, for every nonnegative measurable $G$,
  \[
    \int G(ry)\Gamma(\ud y)
    =r^{\alpha-d}\int G(z)\Gamma(\ud z).
  \]
  Indeed, its left pushforward measure on a Borel set $A$ is $\Gamma(A/r) =
  r^{\alpha-d}\Gamma(A)$. Positive Tonelli applies this identity inside the
  temporal convolution, giving $h_{D_r\Psi,s}(x) = r^2h_{\Psi,s}(rx)$. Now use
  the \emph{Lebesgue} substitution $z = rx$ in the original pairing:
  \[
    v_{D_r\Psi}=r^{d+2-\alpha}r^2r^{-d}v_\Psi
    =r^{4-\alpha}v_\Psi.
  \]
  This distinguishes the Radon pushforward from the spatial volume factor; it
  includes singular covariances and white noise.

  The map $U_rf(x) = r^{d/2}f(rx)$ is a unitary bijection on real $L^2$,
  preserving $H^1,H^2$ as sets under its inverse. Weak derivatives give
  $\|\nabla U_rf\|_2^2 = r^2\|\nabla f\|_2^2$ and $\int h_{D_r\Psi,s}(U_rf)^2 =
  r^2\int h_{\Psi,s}f^2$. The real Rayleigh bijection proves the squared rate
  identity; the rates are positive, so their square roots and integrals give
  \eqref{E:variational-dilation-values}. On $H^2$ this is also $H_{D_r\Psi,s}U_r
  = r^2U_rH_{\Psi,s}$. Time has not been rescaled. For compact masks the
  corresponding identity uses $m_R(rx) = m_{R/r}(x)$; a fixed mask radius is not
  invariant. This proves Lemma~\ref{L:variational-control-dilation}.
\end{proof}

\LeanMarker{P:variational-positive-scores}
\begin{proposition}[Intrinsic variational lower bounds]
  \label{P:variational-positive-scores}
  For $c > 0$, the set of variational values
  $\{cI_\Psi-v_\Psi/2:\Psi\in\mathcal{C}_{\mathrm{pos}}\}$ is nonempty and
  bounded above. Its finite real supremum $S_c^{\mathrm{pos}}$ is positive and
  satisfies
  \begin{equation}\label{E:variational-score-homogeneity}
    S_c^{\mathrm{pos}}=c^{(4-\alpha)/(3-\alpha)}S_1^{\mathrm{pos}},
    \qquad
    Q_2\ge S_2^{\mathrm{pos}},
  \end{equation}
  where $Q_2$ is the original point liminf. The coefficient-one value is an
  auxiliary variational value; no first-moment or source-constant identification
  is asserted.
\end{proposition}

\begin{proof}
  Write $k = 3-\alpha > 0$, $q = 4-\alpha = k+1$ and $m = q/k > 1$. Fix one
  $\Psi_0$ from the nonempty class and put $I_0 = I_{\Psi_0} > 0$, $v_0 =
  v_{\Psi_0} > 0$. Positivity of the latter is
  Lemma~\ref{L:compact-raw-profile}. For $r_0 = (I_0/v_0)^{1/k} > 0$, which need
  not be less than one, the exact real-power identity gives
  \[
    I_{D_{r_0}\Psi_0}-v_{D_{r_0}\Psi_0}/2
    =r_0I_0-r_0^qv_0/2=r_0I_0/2>0.
  \]
  This positive value is obtained from a smooth nonnegative control.

  For $c > 0$ put $r_c = c^{1/k}$ and $K_c = c^m > 0$.
  Lemma~\ref{L:variational-control-dilation} and the exact scalar powers give,
  for every $\Psi$,
  \begin{equation}\label{E:variational-score-set-scaling}
    cI_{D_{r_c}\Psi}-v_{D_{r_c}\Psi}/2
    =K_c(I_\Psi-v_\Psi/2).
  \end{equation}
  Since $D_{r_c}$ is a bijection of the entire class, the whole set of
  variational values with coefficient $c$ is $K_c$ times the whole
  coefficient-one set of variational values.

  We prove boundedness before taking finite-real suprema. The independent
  Theorem~\ref{T:full-upper}, for the \emph{same canonical solution}, gives
  \[
    \E|u(T,0)|^2\le e(T+2)^d e^{C_{\mathrm{up}}T^\kappa},
    \qquad T\ge1,\quad C_{\mathrm{up}}<\infty.
  \]
  Its prefactor is negligible: choose $\epsilon = \kappa/2 > 0$, use $\log T \le
  (T^\epsilon-1)/\epsilon$ and $\log(T+2) \le \log T+\log3$ for $T \ge 1$, and
  divide by $T^\kappa$. Thus, the original point limsup is at most
  $C_{\mathrm{up}}$. One fixed test/mask lower bound gives a finite lower bound
  for $Q_2$, so $Q_2$ is finite. By~\eqref{E:variational-unmasked-point-lower}
  every coefficient-two variational expression is at most this same $Q_2 \le
  C_{\mathrm{up}}$, uniformly in $\Psi$.
  Equation~\eqref{E:variational-score-set-scaling} at $c = 2$ consequently
  bounds all coefficient-one variational values by $C_{\mathrm{up}}/K_2$. The
  nonempty set has a finite real supremum $S_1^{\mathrm{pos}} > 0$ by the
  positive test above.

  For general $c$, set equality and $K_c > 0$ show that $K_cS_1^{\mathrm{pos}}$
  is the supremum: it is an upper bound, and every smaller number is exceeded by
  $K_c$ times a coefficient-one variational expression within the corresponding
  positive distance of its supremum. This proves finiteness, positivity and the
  first identity in~\eqref{E:variational-score-homogeneity}. Taking the
  numerical supremum of the coefficient-two lower bounds proves the second.
  Although $2^m > 2$, the resulting $S_2^{\mathrm{pos}} > 2S_1^{\mathrm{pos}}$
  is only a comparison of auxiliary variational values. It is not a Jensen
  improvement without a first-moment identification. This proves
  Proposition~\ref{P:variational-positive-scores}.
\end{proof}

For $\Gamma = 0$, the mean, original variance and Rayleigh rate are zero. Set
all these variational values to zero; the free canonical solution $u = 1$ has
the corresponding zero moment rate. Strict positivity above is restricted to
nonzero covariance. Matching an upper variational rate and identifying an exact
pointwise limit remain later steps.

\section{Finite optimization with nonnegative Gram entries}
\label{S:finite-positive-gram}

Let $H$ be any real Hilbert space, $\mathcal{I}$ a finite index set, $k_i \in
H$, $\ell_i > 0$ and $b_i \ge 0$. Write $C_{ij} = \langle k_i,k_j\rangle$ and
assume $C_{ij} \ge 0$ for every pair. The Gram matrix is symmetric and positive
semidefinite, since $\sum_{i,j}a_iC_{ij}a_j = \|\sum_i a_i k_i\|^2$. Put $t_+ =
\max(t,0)$ and define
\begin{equation}\label{E:finite-gram-score}
  \mathfrak{F}_C(\Phi)
  =2\sum_{i\in\mathcal{I}}\ell_i
  \sqrt{(\langle\Phi,k_i\rangle-b_i)_+}-\|\Phi\|^2/2,
  \qquad \mathscr{S}(C)=\sup_{\Phi\in H}\mathfrak{F}_C(\Phi).
\end{equation}
Let $\mathcal{J} = \{i:k_i\ne0\} = \{i:C_{ii}>0\}$. Zero vectors contribute zero
because $b_i \ge 0$. Empty index sets, singular Grams and $H = \{0\}$ are
allowed. For nonempty $A \subset \mathcal{J}$ and $w_i > 0$, set
\begin{equation}\label{E:finite-gram-dual}
  D_A^C(w)=\sum_{i\in A}\frac{\ell_i^2}{w_i}
  -\sum_{i\in A}b_iw_i
  +\frac12\sum_{i,j\in A}w_iC_{ij}w_j,
  \qquad d_A^C=\inf_{w_i>0}D_A^C(w).
\end{equation}
Set $d_\varnothing^C = 0$.

\LeanMarker{L:gram-scalar-algebra}
\begin{lemma}[Scalar and Hilbert identities]\label{L:gram-scalar-algebra}
  For real $\ell$, $w > 0$ and $t \ge 0$,
  \begin{equation}\label{E:gram-sqrt-dual}
    2\ell\sqrt t\le\ell^2/w+wt.
  \end{equation}
  For $\Phi,z \in H$,
  \begin{equation}\label{E:gram-hilbert-square}
    \langle\Phi,z\rangle-\|\Phi\|^2/2
    =\|z\|^2/2-\|\Phi-z\|^2/2\le\|z\|^2/2.
  \end{equation}
  For finite real $w_i$, $\|\sum_iw_ik_i\|^2 = \sum_{i,j}w_iC_{ij}w_j$.
\end{lemma}

\begin{proof}
  Since $\sqrt w > 0$ and $\sqrt{wt} = \sqrt w\sqrt t$,
  \[
    0\le(\ell/\sqrt w-\sqrt{wt})^2
    =\ell^2/w+wt-2\ell\sqrt t.
  \]
  This includes $t = 0$. If $\ell,t > 0$, equality holds for $w = \ell/\sqrt t$.
  Expanding $\|\Phi-z\|^2$ proves~\eqref{E:gram-hilbert-square}. Bilinearity of
  the real inner product proves the finite Gram identity. These identities use
  neither compactness of a Hilbert ball nor an inverse covariance matrix. This
  proves Lemma~\ref{L:gram-scalar-algebra}.
\end{proof}

\begin{lemma}[Interior dual minimum]\label{L:gram-dual-minimum}
  Each nonempty $A \subset \mathcal{J}$ has a unique positive minimizer $w^A$ of
  $D_A^C$. At this minimizer,
  \begin{equation}\label{E:gram-dual-stationarity}
    \sum_{j\in A}C_{ij}w_j^A-b_i
    =\frac{\ell_i^2}{(w_i^A)^2}>0,\qquad i\in A.
  \end{equation}
\end{lemma}

\begin{proof}
  Put $c_i = C_{ii} > 0$ and $K_A = \sum_{i\in A}b_i^2/c_i$. Nonnegative entries
  and positive weights give
  \[
    D_A^C(w)\ge
    \sum_i\left(\ell_i^2/w_i-b_iw_i+c_iw_i^2/2\right).
  \]
  Completing the square in $c_iw_i^2/4-b_iw_i$ yields
  \begin{equation}\label{E:gram-dual-coercivity}
    D_A^C(w)\ge
    \sum_i\ell_i^2/w_i+\frac14\sum_i c_iw_i^2-K_A.
  \end{equation}
  Thus, no coordinate can approach zero or infinity in a bounded sublevel while
  a different negative term hides its growth. Take $w^0 = (1,\ldots,1)$, $d_0 =
  D_A^C(w^0)$ and $L_A = d_0+K_A$. Equation~\eqref{E:gram-dual-coercivity} at
  $w^0$ gives $L_A \ge \sum_i\ell_i^2+\sum_i c_i/4 > 0$. Every $w$ with
  $D_A^C(w) \le d_0$ satisfies
  \[
    \ell_i^2/L_A\le w_i\le\sqrt{4L_A/c_i}.
  \]
  The sublevel is nonempty and closed inside this compact box, which lies in the
  open positive orthant. A minimizing sequence chosen in the sublevel has a
  convergent subsequence there. Continuity attains its infimum. Outside this
  sublevel the values exceed $d_0$, so the minimum is global on the entire
  positive orthant.

  The function is smooth in that orthant. Its Hessian on a nonzero vector $\xi$
  has value
  \[
    \sum_i\frac{2\ell_i^2}{w_i^3}\xi_i^2
    +\sum_{i,j}\xi_iC_{ij}\xi_j>0.
  \]
  The first term is strictly positive and the second is a squared Hilbert norm.
  Restriction to any nonconstant line segment is strictly convex, proving
  uniqueness. Each coordinate permits both signs of a small perturbation at the
  interior minimum. Differentiation gives $-\ell_i^2/w_i^2-b_i+\sum_jC_{ij}w_j =
  0$. This is~\eqref{E:gram-dual-stationarity}. This proves
  Lemma~\ref{L:gram-dual-minimum}.
\end{proof}

\LeanMarker{P:gram-positive-optimizer}
\begin{proposition}[Positive conic maximizer]\label{P:gram-positive-optimizer}
  The finite real value in~\eqref{E:finite-gram-score} satisfies
  \begin{equation}\label{E:gram-active-duality}
    \mathscr{S}(C)=\max_{A\subset\mathcal{J}}d_A^C .
  \end{equation}
  If a maximizing subset is empty, $\Phi = 0$ attains the supremum. If a
  maximizing subset $A$ is nonempty, a maximizer is $\Phi_A = \sum_{i\in A}w_i^A
  k_i$ with strictly positive coefficients. No uniqueness of the primal
  maximizer or maximizing subset is asserted.
\end{proposition}

\begin{proof}
  For arbitrary $\Phi$, let $A(\Phi) = \{i:\langle\Phi,k_i\rangle>b_i\}$. It is
  contained in $\mathcal{J}$. All other terms in the objective vanish, including
  equality at the threshold. If $A(\Phi)$ is empty, $\mathfrak{F}_C(\Phi) =
  -\|\Phi\|^2/2 \le 0$. Otherwise, for any positive $w$ on $A = A(\Phi)$, set
  $t_i = \langle\Phi,k_i\rangle-b_i > 0$ and $z = \sum_{i\in A}w_ik_i$.
  Lemma~\ref{L:gram-scalar-algebra} gives
  \[
    \mathfrak{F}_C(\Phi)
    \le\sum_i\ell_i^2/w_i-\sum_i b_iw_i
    +\langle\Phi,z\rangle-\|\Phi\|^2/2
    \le D_A^C(w).
  \]
  This holds for every positive $w$, so $\mathfrak{F}_C(\Phi) \le d_A^C$. There
  are finitely many subsets and their values are finite by
  Lemma~\ref{L:gram-dual-minimum}. Taking the supremum bounds $\mathscr{S}(C)$
  above by their maximum.

  Conversely, $\mathfrak{F}_C(0) = 0 = d_\varnothing^C$. For nonempty $A$, take
  its dual minimizer. Equation~\eqref{E:gram-dual-stationarity} makes every
  vector in $A$ strictly active at $\Phi_A$, with
  $\sqrt{\langle\Phi_A,k_i\rangle-b_i} = \ell_i/w_i^A$. Both inequalities in the
  preceding display are then equalities for the sum restricted to $A$, since $z
  = \Phi_A$. This restricted objective equals $d_A^C$. All omitted vectors
  contribute nonnegative terms to the full objective, hence
  $\mathfrak{F}_C(\Phi_A) \ge d_A^C$, even if this value is negative. Maximizing
  over $A$, including the empty set, proves \eqref{E:gram-active-duality} and
  the asserted attainment. For a maximizing nonempty subset the upper and lower
  bounds force $\mathfrak{F}_C(\Phi_A) = d_A^C$; no outside inactivity was
  assumed. All-zero vectors give $\mathscr{S}(C) = 0$ directly. This proves
  Proposition~\ref{P:gram-positive-optimizer}.
\end{proof}

No-go~\ref{P:nogo-29} gives the three finite counterexamples behind this
formulation: the empty active set cannot be omitted, the primal objective
need not be concave, and entrywise matrix order is not positive
semidefinite order.

\begin{lemma}[Real Gram realization]\label{L:gram-realization}
  Every finite real symmetric positive semidefinite matrix is the Gram matrix of
  vectors in a finite-dimensional real Hilbert space. When its entries are
  nonnegative, the value $\mathscr{S}(C)$ is independent of its realization.
\end{lemma}

\begin{proof}
  Induct on the size, with the empty matrix realized in $\{0\}$. Diagonal
  entries are nonnegative. If all vanish, testing on $e_i+t e_j$ gives $2tC_{ij}
  \ge 0$ for every real $t$, hence $C = 0$, realized by zero vectors. Otherwise
  permute a positive diagonal $c$ into the first position and write $C =
  \left[\begin{smallmatrix}c&v^{\mathsf{T}}\\v&A\end{smallmatrix}\right]$. The
  Schur complement $B = A-vv^{\mathsf{T}}/c$ is positive semidefinite: test the
  original quadratic form on $(-v^{\mathsf{T}}y/c,y)$. Realize $B$ by vectors
  $u_i$ using the induction hypothesis. Then $k_1 = (\sqrt c,0)$ and $k_i =
  (v_i/\sqrt c,u_i)$ for the other indices realize $C$. Undo the permutation.
  Entries of $B$ need not be nonnegative; this construction requires neither
  nonnegative coordinates nor complete positivity.

  In any realization, $C_{ii} = 0$ means $k_i = 0$, so its entire row vanishes.
  Formula~\eqref{E:gram-active-duality} depends only on $C,\ell,b$ and these
  eligible indices. It therefore proves realization independence whenever the
  entries are nonnegative, which is the scope in which $\mathscr{S}(C)$ is used.
  This proves Lemma~\ref{L:gram-realization}.
\end{proof}

By Lemma~\ref{L:gram-realization}, $\mathscr{S}(C)$ depends only on the matrix
$C$ and the data $\ell,b$. We can therefore compare two Gram matrices directly.

\LeanMarker{P:gram-entrywise-comparison}
\begin{proposition}[Entrywise Gram comparison]\label{P:gram-entrywise-comparison}
  Let $C,D$ be real symmetric positive semidefinite matrices on the same finite
  index set, with $0 \le C_{ij} \le D_{ij}$. With the same $\ell_i > 0,b_i \ge
  0$,
  \[
    \mathscr{S}(C)\le\mathscr{S}(D).
  \]
  No positive-semidefinite order on $D-C$ is required.
\end{proposition}

\begin{proof}
  Lemma~\ref{L:gram-realization} defines both values. Their eligible sets
  satisfy $\mathcal{J}_C \subset \mathcal{J}_D$. For nonempty $A \subset
  \mathcal{J}_C$ and positive weights,
  \[
    D_A^D(w)-D_A^C(w)
    =\frac12\sum_{i,j\in A}w_i(D_{ij}-C_{ij})w_j\ge0.
  \]
  Taking infima gives $d_A^C \le d_A^D$. The empty values are both zero. Taking
  maxima and using the inclusion of eligible families proves the claim by
  Proposition~\ref{P:gram-positive-optimizer}. This includes the appearance of
  previously zero vectors. This proves
  Proposition~\ref{P:gram-entrywise-comparison}.
\end{proof}

\begin{remark}[Exact checks and scope]\label{R:gram-controls}

\section{An upper bound from a signed variational expression}
\label{S:original-signed-score-upper}

Keep all auxiliary choices fixed as in the Gaussian tube construction, including
$\alpha/4 < s_{\rm sp} < 1/2$, the admissible $\theta,\nu,r,p$, and the three
fixed even-$Q$ conditions. The mean bound $C_{\rm mean}$ is independent of
$R,B,\delta$; it may depend on those fixed auxiliary parameters and the model.
The space $H_I$ has the original covariance $\Gamma$. Its norm is not reweighted
by $g$ and is not a sum of fast block costs. Let $R_\delta = \delta^{-1}+2$ and
$M_\delta = \E\|P_\delta\|_{\mathcal{L}(E)}^2$. For every fixed integer $B$ the
Gaussian block tuple has typical radius $b_{\delta,B} =
2\E\|X_{\delta,B}\|_{\delta,p} \to 0$. Its endpoint is the same $P_\delta$ for
every $B$ by Lemma~\ref{L:tube-actual-endpoint}, not a $B$-dependent surrogate.
Write $\mathcal{M}_{\delta,B,R}$ for the normalized mean tuple
in~\eqref{E:tube-actual-mean}, with the indicated block length and radius. Write

$F_\delta$ of Lemma~\ref{L:tube-actual-endpoint}.

\begin{lemma}[Signed variational expression and quantitative radial envelope]
  \label{L:signed-score-envelope}
  For $R \ge 1$ define the variational expression
  \begin{equation}\label{E:signed-raw-score}
    \mathcal{S}_R=\sup_{\Phi\in H_I}
    \{2\operatorname{Rate}_R(\Phi)-\|\Phi\|_{H_I}^2/2\},
    \qquad V_{\rm up}=\limsup_{\delta\downarrow0}\mathcal{S}_{R_\delta}.
  \end{equation}
  Then $0 \le \mathcal{S}_R,V_{\rm up} \le 2+2C_{\rm mean}^2 < \infty$. The tube
  envelope $G_{\delta,B}$ of Proposition~\ref{P:tube-coarse-envelope} satisfies,
  uniformly in $R \ge 1$ (in particular at the stochastic radius $R_\delta$),
  \begin{equation}\label{E:signed-polynomial-envelope}
    G_{\delta,B}(\rho)\le A_0+A_1B^{3\nu}\rho^\nu
    +\eta_B b_{\delta,B}^\nu ,
  \end{equation}
  for $0 < \delta < \min(1,1/B)$ and every $\rho \ge 0$, where
  \[
    \begin{aligned}
      A_0&=4C_{\rm gr},\\
      A_1&=2^{\nu+1-\nu/r}C_{\rm gr}32^\nu C_{\rm mean}^\nu,\\
      \eta_B&=2^{\nu+1-\nu/p}C_{\rm gr}B^{\nu/p}.
    \end{aligned}
  \]
  The variational expression is not identified here with a constant in the cited
  source or a variational value over nonnegative controls.
\end{lemma}

\begin{proof}
  The mean is linear, so $\mathcal{T}_R0 = 0$. Its free form has nonnegative
  gradient energy and spectral bottom zero; thus $\operatorname{Rate}_R(0) = 0$.
  Proposition~\ref{P:uniform-rate-averaging} gives $\operatorname{Rate}_R(\Phi)
  \le 1+C_{\rm mean}\|\Phi\|$. With $q = \|\Phi\| \ge 0$,
  \[
    2\operatorname{Rate}_R(\Phi)-q^2/2
    \le2+2C_{\rm mean}q-q^2/2
    =2+2C_{\rm mean}^2-(q-2C_{\rm mean})^2/2 .
  \]
  The coarser inequality also follows from half-weight arithmetic--geometric
  mean at bases $q^2,(2C_{\rm mean})^2$, with nonnegative weights and bases,
  including zeros. The zero control gives the lower bound. No optimizer, finite
  rank, positive control or convergence of $\mathcal{S}_{R_\delta}$ is assumed.
  The finite ceiling is auxiliary-parameter dependent and need not be sharp.

  For $\|a\|_{\delta,p} \le b_{\delta,B}$, $\|\Phi\| \le \rho$ set $v_m =
  \|a_m\|_\theta$ and $h_m = \|(\mathcal{M}_{\delta,B,R}\Phi)_m\|_\theta$. The
  quantitative physical bound~\eqref{E:uniform-product-growth}, already
  including the free factor, and convexity give
  \[
    \mathcal{F}_{\delta,B}(a+\mathcal{M}_{\delta,B,R}\Phi)
    \le2C_{\rm gr}B\delta n
    +2^\nu C_{\rm gr}B\delta\sum_m(v_m^\nu+h_m^\nu).
  \]
  Its upper expression is nonnegative, so it controls the maximum at one. The
  true primitive estimate and weighted power mean give
  \begin{align*}
    \delta\sum v_m^\nu&\le(2/B)^{1-\nu/p}b_{\delta,B}^\nu,\\
    \delta\sum h_m^\nu&\le(2/B)^{1-\nu/r}
    (32B^{3-1/r}C_{\rm mean}\rho)^\nu .
  \end{align*}
  The exponents are $p/\nu \ge 1$, $r/\nu > 1$, inputs nonnegative, $n = N+1 \ge
  2$, $\delta > 0$, and $\delta n \le 2/B$. Zeros and $p = \nu$ are legal. The
  first constant is $4C_{\rm gr}$. In the mean term the factors $B^{\nu/r}$ and
  $B^{-\nu/r}$ cancel exactly, leaving $B^{3\nu}$. These computations give the
  displayed coefficients and~\eqref{E:signed-polynomial-envelope}, uniformly for
  all radii. They use no unspecified $c_Y(B)$ or quantitative $F$ estimate. This
  proves Lemma~\ref{L:signed-score-envelope}.
\end{proof}

Lemma~\ref{L:signed-score-envelope} bounds the envelope $G_{\delta,B}$ at every
radius $\rho$. We now split the radial integral
in~\eqref{E:tube-coarse-envelope} at a radius $M$. Below $M$ we compare the
exponent with $\mathcal{S}_{R_\delta}$, and above $M$ we use
Lemma~\ref{L:tube-rayleigh-tail}.

\begin{proposition}[Finite-delta bulk and exact tail]
  \label{P:signed-moment-bulk-tail}
  For fixed integer $B \ge 1$, $0 < \delta < \min(1,1/B)$ and $M > 0$, define
  the deterministic number
  \[
    \mathcal{E}_{\delta,B,M}=
    \max\left(0,\sup_{\substack{R\ge1,\ \|\Phi\|\le M\\
    \|x\|_{\delta,p}\le b_{\delta,B}}}
    [\mathcal{F}_{\delta,B}(x+\mathcal{M}_{\delta,B,R}\Phi)
    -2\operatorname{Rate}_R(\Phi)]\right).
  \]
  It is finite. For every polynomial $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$,
  with $c_{\rm ball} > 0$, $\kappa_{\rm ball} \ge 0$, one has
  \begin{equation}\label{E:signed-error-limit}
    e_B:=\limsup_{\delta\downarrow0}\mathcal{E}_{\delta,B,M_B}
    \ \text{finite and nonnegative},\qquad e_B\longrightarrow0 .
  \end{equation}
  If $4A_1B^{3\nu} \le M^{2-\nu}$, then
  \begin{equation}\label{E:signed-finite-moment}
    M_\delta\le
    \frac{M^2}{2\delta}
    e^{(\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M})/\delta}
    +4e^{(A_0+\eta_Bb_{\delta,B}^\nu-M^2/4)/\delta}.
  \end{equation}
\end{proposition}

\begin{proof}
  The polynomial envelope and nonnegative rate make this supremum finite. It is
  an ordinary deterministic real number, not a random uncountable-radius
  supremum. At fixed $R$ the inner functions are continuous in controls and
  paths; no measurable maximizing selector or joint $\delta,R$ measurability is
  needed. The zero control and zero path have rate zero and
  $\mathcal{F}_{\delta,B}(0) \ge 0$, so the untruncated supremum is already
  nonnegative. The anchor $b_{\delta,B} \to 0$ at fixed $B$ satisfies
  Proposition~\ref{P:uniform-arbitrary-residual} exactly. Its polynomial-radius
  conclusion gives $\limsup_B e_B \le 0$. At fixed $B$, the polynomial envelope
  bounds the inner limsup by a finite number, and $e_B \ge 0$ then
  proves~\eqref{E:signed-error-limit}.

  The exact Gaussian-enlargement bound is
  \[
    M_\delta\le\int_0^\infty\frac{\rho}{\delta}
    e^{(G_{\delta,B}(\rho)-\rho^2/2)/\delta}\ud\rho .
  \]
  For every $\rho \le M$, every base-ball $a$ and $\|\Phi\| \le \rho$,
  \[
    \mathcal{F}_{\delta,B}(a+\mathcal{M}_{\delta,B,R_\delta}\Phi)-\rho^2/2
    \le2\operatorname{Rate}_{R_\delta}(\Phi)-\|\Phi\|^2/2
    +\mathcal{E}_{\delta,B,M}
    \le\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M}.
  \]
  Taking suprema and integrating $\rho/\delta$ over $[0,M]$ gives the first term
  of~\eqref{E:signed-finite-moment}, even if $M^2/(2\delta) < 1$. This used
  $\rho^2 \ge \|\Phi\|^2$, not equality or an optimizer. For the tail apply
  Lemma~\ref{L:tube-rayleigh-tail} with $C = A_1B^{3\nu}$ and intercept
  $A_0+\eta_Bb_{\delta,B}^\nu$. All cutoff conditions hold: $M > 0$, $\rho \ge
  M$, $C \ge 0$, $1 < \nu < 2$, $4C \le M^{2-\nu}$. Its analytic integration
  retains both the front factor four and the exponent $-M^2/4$. Adding bulk and
  tail proves the finite-delta bound. No infinite-dimensional density or Laplace
  principle is used. This proves Proposition~\ref{P:signed-moment-bulk-tail}.
\end{proof}

\begin{proposition}[Two limits on the same moment]
  \label{P:signed-operator-upper}
  The global operator moment obeys
  \[
    0\le\liminf_{\delta\downarrow0}\delta\log M_\delta
    \le\limsup_{\delta\downarrow0}\delta\log M_\delta
    \le V_{\rm up}\le2+2C_{\rm mean}^2 .
  \]
\end{proposition}

\begin{proof}
  Choose any $c_{\rm ball} > 0$ and $\kappa_{\rm ball} > 3\nu/(2-\nu)$, and set
  $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$. The cutoff in
  Proposition~\ref{P:signed-moment-bulk-tail} holds for all sufficiently large
  integers $B$. For $A_1 = 0$ this is immediate; otherwise its side ratio is
  \[
    \frac{M_B^{2-\nu}}{4A_1B^{3\nu}}
    =\frac{c_{\rm ball}^{2-\nu}}{4A_1}
    B^{\kappa_{\rm ball}(2-\nu)-3\nu}\longrightarrow\infty .
  \]
  Both constants and the strictly positive exponent are retained. At such a
  fixed $B$, a positive sum is at most twice its larger term. Taking
  $\delta\log$ of~\eqref{E:signed-finite-moment} therefore gives
  \begin{align*}
    \delta\log M_\delta\le\delta\log2+
    \max\{&
    \mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M_B}
    +\delta\log(M_B^2/(2\delta)),\\
    &A_0+\eta_Bb_{\delta,B}^\nu-M_B^2/4+\delta\log4\}.
  \end{align*}
  At fixed $B$, $b_{\delta,B} \to 0$. All logarithmic normalizers vanish,
  including $\delta\log(M_B^2/(2\delta)) = \delta\log(M_B^2/2)-\delta\log\delta
  \to 0$. The last limit follows from $te^{-t} \le 2/t$ for $t = -\log\delta >
  0$. Finite variational expression and error bounds permit limsup
  subadditivity, giving
  \[
    \limsup_{\delta\downarrow0}\delta\log M_\delta
    \le\max\{V_{\rm up}+e_B,\ A_0-M_B^2/4\}.
  \]
  The left side has no $B$: restart identifies every block product with this
  same $P_\delta$. Now let integer $B \to \infty$. We have $e_B \to 0$, $M_B \to
  \infty$ and finite $V_{\rm up} \ge 0$. This gives the asserted upper bound,
  while $M_\delta \ge 1$ gives the nonnegative lower bound. The order was
  $\delta$ first at fixed block length, ball and thresholds, then $B$; no
  diagonal or limit of $\mathcal{S}_{R_\delta}$ was assumed. This proves
  Proposition~\ref{P:signed-operator-upper}.
\end{proof}

\subsection{The common original noise and its exact Gaussian shift}

\LeanMarker{L:raw-block-inputs}
\begin{lemma}[Coupled Gaussian block inputs]\label{L:raw-block-inputs}
  Each $T_{\delta,m}$ is an isometry of the original test completion, respecting
  covariance-null tests. All $W_{\delta,m}$ and $Z$ are on the same given noise
  space, without an independence assertion. Let $\mathcal{K}(F)$ have the
  operator norm and let $\mathscr{B}_0 = h_0^\theta([0,B];\mathcal{K}(F))$ be
  the $C^\theta$ closure of zero-initial polygonal compact-operator paths. It is
  a separable Banach space. In $\mathscr{B}_\delta = \mathscr{B}_0^{N+1}$ with
  maximum norm, $(X,Z)$, $X = (X_m)_{m=0}^N$, is jointly centered Gaussian and
  \begin{equation}\label{E:raw-block-covariance}
    b:=\E[XZ]=\sqrt\delta\,H,\qquad H=(H_m)_{m=0}^N.
  \end{equation}
  Moreover,
  \begin{equation}\label{E:raw-block-moments}
    \|X_m\|_{L^p\mathscr{B}_0}
    \le C_{R,B}\sqrt p\,g_\delta,\qquad p\ge1.
  \end{equation}
\end{lemma}

\begin{proof}
  Substituting $s = s_m+\delta u$, $t = s_m+\delta r$ in the original covariance
  gives the factor $\delta^{a-2}\delta^2\delta^{-a} = 1$. Thus, the maps extend
  isometrically through the null quotient. Direct test substitution gives
  $W_{\delta,m}(f) = W_{\mathrm{fast}}(f(\cdot-mB,\cdot))$. These are explicit
  common-space identities.

  Finite-rank rational matrices in a countable real Hilbert basis are dense in
  $\mathcal{K}(F)$: for compact $T$, $P_nTP_n \to T$ in norm by compactness of
  $T,T^*$, and finite entries admit rational approximation. Polygonal paths with
  rational breakpoints and such matrices give separability of $\mathscr{B}_0$.
  To approximate breakpoints, use uniform convergence, uniform Lipschitz bounds
  and $[f]_\theta \le (2\|f\|_\infty)^{1-\theta}\operatorname{Lip}(f)^\theta$.
  The same inequality puts every zero-initial Lipschitz compact-valued path in
  $\mathscr{B}_0$ by polygonal approximation. The finite original coefficient
  paths $iA_jL_\psi$ have these properties. Their Gaussian sums converge in
  $L^pC^\theta$ and hence in the closed space $\mathscr{B}_0$. Their laws and
  the parent Gaussian norm bound give~\eqref{E:raw-block-moments}. Every finite
  joint coefficient list with $Z$ is Gaussian on the same $W$. Scalar continuous
  functionals pass through the $L^2$ limits, preserving Gaussianity by
  convergence of variances and characteristic functions. Thus, $(X,Z)$ is the
  asserted Banach Gaussian variable. Cauchy--Schwarz gives $\E\|XZ\| \le
  \|X\|_2\sqrt v < \infty$, so its Bochner covariance exists.

  The original covariance and the test pullback give
  \begin{equation}\label{E:raw-block-mean-jacobian}
    \E[W_{\delta,m}(f)Z]
    =\delta^{a/2}\int f(u,x)h_{s_m+\delta u}(x)\ud x\ud u.
  \end{equation}
  Absolute Fubini uses compact supports, local Radon mass and $a < 1$. This
  identity extends to the explicit matrix-element tests for the operator path on
  $E$: their time indicator is approximated by smooth bounded tests on one
  compact support, converging in the original covariance norm by dominated
  convergence against $|u-r|^{-a}\Gamma$; their mean pairings converge as well.
  The Sobolev adjoint is the one in~\eqref{E:driver-covariance-row}. Hence, the
  path covariance is $g_\delta\delta^{a/2}H_m = \sqrt\delta\,H_m$ on real
  Schwartz $E$ pairs. For constant inputs the two columns are the same $L_\psi$
  inputs $(\psi,0)$ and $(0,\psi)$, and total finite propagation identifies
  their deterministic response. The span of $E$ and both constant coordinates is
  dense in $F$; $E$ alone is not. Boundedness, rational times and path
  continuity prove~\eqref{E:raw-block-covariance} on all $F$. In particular, $H
  = b/\sqrt\delta \in \mathscr{B}_\delta$. This proves
  Lemma~\ref{L:raw-block-inputs}.
\end{proof}

The next identity is the covariance-direction Cameron--Martin formula;
see Janson~\cite{janson:97:gaussian}*{Theorem~14.1(v), pp.~216--217}.
We prove it on the norm-Borel space used here to verify the measurable
realization and its deterministic translate.

\begin{proposition}[Change of law on the whole norm-Borel space]
  \label{P:raw-block-gaussian-law}
  For every nonnegative norm-Borel $j$ on all of $\mathscr{B}_\delta$,
  \begin{equation}\label{E:raw-block-gaussian-law}
    \E j(X)=
    \E\left[e^{-Z/\sqrt\delta-v/(2\delta)}j(X+H)\right].
  \end{equation}
  The equality is valid also as an extended nonnegative integral.
\end{proposition}

\begin{proof}
  For real $s$, the density $e^{-sZ-s^2v/2}$ defines an equivalent probability
  measure $Q_s$ by completing the square in the $N(0,v)$ density. For any finite
  list $\ell$ of continuous real coordinates on $\mathscr{B}_\delta$, the
  Gaussian vector $(\ell(X),Z)$ is equal in law to $(TG,z\cdot G)$ for a finite
  standard Gaussian vector $G$. A positive covariance square root gives this
  representation even when singular; no inverse is used. Here $Tz = \ell(b)$ and
  $|z|^2 = v$. Substitution $G' = G+sz$ in its density gives
  $\E_{Q_s}f(\ell(X+sb)) = \E f(\ell(X))$ for every bounded Borel $f$.

  The matrix entries of each block in a real Hilbert basis at rational times,
  together with $0,B$, generate the norm-Borel sigma-field. To see the
  operator-norm point, for a countable dense set $\mathcal{D}$ of unit vectors
  obtained from normalized rational finite basis combinations, $\|T\| =
  \sup_{u,w\in\mathcal{D}}|\langle Tu,w\rangle|$. These pairings are finite
  linear combinations of matrix entries. Supremum and H\"older norms of
  continuous paths are suprema over rational times and pairs. Thus, distance to
  every fixed element of $\mathscr{B}_\delta$ is coordinate-measurable. Open
  balls are measurable by rational closed-ball approximation, and separability
  makes every open set a countable union of balls from dense centers. Conversely
  the coordinates are continuous. The cylinder sets form a generating pi-system;
  uniqueness of probability measures extends the finite-coordinate equality to
  all norm-Borel sets. Simple functions and monotone convergence give the
  identity for nonnegative $j$. Finally, take $s = \delta^{-1/2}$ and use $sb =
  H$ from~\eqref{E:raw-block-covariance}. The map and its deterministic
  translate are defined on the entire input space; there is no translated
  exceptional-domain assumption. This proves
  Proposition~\ref{P:raw-block-gaussian-law}.
\end{proof}

The identity retains the random shifted input $X+H$. Replacing its
expected contribution by evaluation at the deterministic mean can lose
the exponential constant, even in a scalar Gaussian calculation; see
No-go~\ref{P:nogo-16}. Positive semidefinite residual covariance does not
justify discarding that residual: No-go~\ref{P:nogo-19} gives an admissible
parent model whose rank-one projection has a larger small-coupling wave mean.


\subsection{Small noisy blocks and a cone in the displacement-velocity space}

For any $x \in \mathscr{B}_0$ define $D_x(u) = S_F(u)U_x(u,0)$ by the
everywhere-defined linear Young map of
Proposition~\ref{P:constant-young-linear}. Write $D_m^0 = D_{X_m}$ and $D_m^+ =
D_{X_m+H_m}$. The latter retains both the mean and the residual noise.

The shifted blocks $D_m^+$ are random. We now show that their largest distance
from the deterministic $P_{s_m}$ of~\eqref{E:compact-F-block}, over all blocks
$m \le N$, tends to zero in every finite $L^q$.

\begin{lemma}[Uniform block error]\label{L:raw-block-uniform-error}
  Let $M_\delta = \max_{m\le N}\|X_m\|_{\mathscr{B}_0}$, including its factor
  $g_\delta$. Then $M_\delta \to 0$ in every finite $L^q$, and, for some $c >
  0$,
  \begin{equation}\label{E:raw-block-maximum-exponential}
    \sup_\delta\E e^{cM_\delta^2}<\infty.
  \end{equation}
  With the deterministic $P_s$ in~\eqref{E:compact-F-block},
  \begin{equation}\label{E:raw-block-operator-error}
    E_{\mathrm{op},\delta}:=
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m^+(u)-P_{s_m}(u)\|_{\mathcal{B}(F)}
    \longrightarrow0
    \quad\hbox{in every finite }L^q.
  \end{equation}
\end{lemma}

\begin{proof}
  The maximum and~\eqref{E:raw-block-moments}, with $N+1 \le 2/\delta$, give
  \[
    \|M_\delta\|_p
    \le (N+1)^{1/p}C\sqrt p\,g_\delta
    \le C'\sqrt p\,\delta^{(1-a)/2-1/p}.
  \]
  For fixed $q$, choose $p > \max(q,2/(1-a))$ to prove the first limit. No
  independence is needed. Choose an integer $n_0$ with $n_0(1-a) \ge 1$. For $n
  \ge n_0$,
  \[
    \E M_\delta^{2n}
    \le2C^{2n}(2n)^n\delta^{n(1-a)-1}
    \le2C^{2n}(2n)^n.
  \]
  For $1 \le n < n_0$, Lyapunov bounds the moment by the appropriate power of
  the $2n_0$ moment. These finitely many terms are bounded. For the remaining
  exponential series, $n! \ge (n/e)^n$ gives a geometric tail $2\sum_{n\ge
  n_0}(2ecC^2)^n$ for small $c$. This
  proves~\eqref{E:raw-block-maximum-exponential}; no such assertion is made for
  the unscaled maximum.

  Let $H_m^{\mathrm{stat}}$ be the primitive for constant potential $V_{s_m}$.
  Compact mean smoothness on $[0,2]$ and the statewise primitive bound give
  \[
    \sup_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \sup_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C\delta.
  \]
  Indeed, the potential difference is at most $C\delta B$ in $L^\infty$ and
  $L^2$; integration of the bounded free groups gives its Lipschitz primitive
  bound on this fixed horizon. On $\{M_\delta\le1\}$, local Young stability on
  the fixed ball of radius $H_0+1$ gives $E_{\mathrm{op},\delta} \le
  C(M_\delta+\delta)$. This proves convergence in probability. The global Young
  bound also gives
  \[
    \max_m\sup_{u\le B}\|D_m^+(u)\|
    \le C_B\exp\big(C_B(1+H_0+M_\delta)^{1/\theta}\big).
  \]
  Since $1/\theta < 2$, every finite power of this bound is dominated by a
  constant times the exponential in~\eqref{E:raw-block-maximum-exponential}.
  Uniform higher moments yield uniform integrability, proving
  \eqref{E:raw-block-operator-error}. Suprema are measurable: on a countable
  dense set of unit states each difference is continuous in $u$, so rational
  times and those states suffice. Operator-norm continuity of the physical free
  group is not used. Only $s_m \le 1$ supplies the ground templates. This proves
  Lemma~\ref{L:raw-block-uniform-error}.
\end{proof}

Lemma~\ref{L:raw-block-uniform-error} bounds the distance of each shifted block
from $P_{s_m}$. It says nothing about the product of the blocks. The next lemma
is deterministic and holds for every $N \ge 1$. If each factor $D_m$ is within
$\epsilon_{\mathrm{op}}$ of $P_{\sigma_m}$, and adjacent indices $\sigma_m$ are
within $\omega$, then the product satisfies the lower bound
\eqref{E:raw-block-positive-product}.

\LeanMarker{L:raw-block-indexed-cone}
\begin{lemma}[Indexed cone and terminal displacement]
  \label{L:raw-block-indexed-cone}
  Fix $0 < \varrho < 1/2$. There are an integer $B \ge 1$ and positive
  $\omega,\epsilon_{\mathrm{op}},c_*$ such that the following holds for every $N
  \ge 1$. If $\sigma_0,\ldots,\sigma_N \in I$ have adjacent distances at most
  $\omega$ and real bounded operator families $D_m(u)$, $0 \le u \le B$, satisfy
  \[
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m(u)-P_{\sigma_m}(u)\|\le\epsilon_{\mathrm{op}},
  \]
  then, for $0 \le u \le B$,
  \begin{equation}\label{E:raw-block-positive-product}
    O_\phi D_N(u)D_{N-1}(B)\cdots D_0(B)e_0
    \ge c_*(1-\varrho)^{N-1}
    \exp\left(B\sum_{m<N}\lambda_{\sigma_m}
    +u\lambda_{\sigma_N}\right)>0.
  \end{equation}
  No positivity, composition or independence assumption on the $D_m$ is
  required.
\end{lemma}

\begin{proof}
  From~\eqref{E:compact-normalized-complement} fix $C_{\mathrm{gap}} \ge 1$,
  $\eta_{\mathrm{gap}} > 0$ so the normalized complementary norm is at most
  $C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}t}$. Define positive margins and
  finite bounds
  \[
    a_*=\inf_I\ell_s(e_0),\quad j_*=\inf_I O_\phi(e_s),\quad
    M_e=\max(1,\sup_I\|e_s\|),\quad
    M_\ell=\max(1,\sup_I\|\ell_s\|).
  \]
  Also put $M_P = \max(1,\sup_I\|\Pi_s\|)$, $M_Q = 1+M_P$, $M_z = M_e+1$, $E_0 =
  \|e_0\| = 1$ and $H_{\mathrm{obs}} = \max(1,\|O_\phi\|)$. Choose the aperture
  \emph{first}: $\vartheta = \min(1,j_*/(16H_{\mathrm{obs}}C_{\mathrm{gap}})) >
  0$. Next choose integer $B \ge 1$ with
  \[
    q_B:=C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}B}
    \le\min\left(\frac18,\frac{\vartheta a_*}{8M_QE_0}\right).
  \]
  Choose $\nu > 0$ such that $\nu M_z \le \min(\varrho/2,\vartheta,1/4)$.
  Uniform continuity of $\ell_s,\Pi_s$ supplies $\omega > 0$ for which adjacent
  distances at most $\omega$ give both norm differences at most $\nu$. Finally,
  choose $\epsilon_{\mathrm{op}} > 0$ satisfying all five smallness conditions
  \begin{align*}
    M_\ell\epsilon_{\mathrm{op}}M_z&\le\varrho/2,&
    M_Q\epsilon_{\mathrm{op}}M_z&\le\vartheta/4,\\
    M_\ell\epsilon_{\mathrm{op}}E_0&\le a_*/2,&
    M_Q\epsilon_{\mathrm{op}}E_0&\le\vartheta a_*/8,&
    H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z&\le j_*/8.
  \end{align*}
  All denominators are positive. This order of choices uses the normalized
  complement rather than a growing terminal norm.

  The cone at $s$ consists of $z = Ae_s+r$, with $A = \ell_s(z) > 0$, $\ell_s(r)
  = 0$ and $\|r\|_F \le \vartheta A$. In particular, $r = Q_sz$ and $\|z\| \le
  M_zA$. For entrance, $z_1 = D_0(B)e_0$, with $b = \sigma_0$, has normalized
  amplitude at least $a_*-M_\ell\epsilon_{\mathrm{op}}E_0 \ge a_*/2$. Its
  normalized residual is at most $q_BM_QE_0+M_Q\epsilon_{\mathrm{op}}E_0 \le
  \vartheta a_*/4$. Thus, it enters the cone with residual ratio at most
  $\vartheta/2$ and amplitude at least $e^{\lambda_bB}a_*/2$.

  For an old-cone state $z = Ae_b+r$ and an adjacent index $c$, rebase before
  using the complementary estimate:
  \[
    \ell_c(z)\ge(1-\nu M_z)A,\qquad
    \|Q_cz\|\le\|Q_bz\|+\|(\Pi_b-\Pi_c)z\|
    \le(\vartheta+\nu M_z)A.
  \]
  For the next full block $D$ close to $P_c(B)$, put $x =
  e^{-\lambda_cB}\ell_c(Dz)$ and $y = e^{-\lambda_cB}\|Q_cDz\|$. Since
  $\lambda_c > 0$, the perturbation and smallness conditions give
  \[
    x\ge(1-\nu M_z-M_\ell\epsilon_{\mathrm{op}}M_z)A
    \ge(1-\varrho)A>A/2,
  \]
  and
  \[
    y\le
    [q_B(\vartheta+\nu M_z)+M_Q\epsilon_{\mathrm{op}}M_z]A
    \le\vartheta A/2.
  \]
  The last inequality uses $q_B \le 1/8$, $0 \le \nu M_z \le \vartheta$ and
  $M_Q\epsilon_{\mathrm{op}}M_z \le \vartheta/4$. Together with $\varrho < 1/2$,
  these give $y \le \vartheta x$. These are precisely the finite residual and
  cone-conversion margins; the old residual was not assumed to lie in the new
  kernel. Each full block therefore preserves the new cone and multiplies its
  amplitude by at least $e^{\lambda_cB}(1-\varrho)$.

  For the terminal $0 \le u \le B$, rebase once more to $c = \sigma_N$. The same
  bounds give $\ell_c(z) \ge 3A/4$ and $\|Q_cz\| \le 2\vartheta A$.
  Consequently,
  \[
    e^{-\lambda_cu}O_\phi(P_c(u)z)/A
    \ge3j_*/4-2H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta.
  \]
  The perturbation changes this by at most
  $H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z \le j_*/8$. Since
  $16H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta \le j_*$, the result is at least
  $j_*/2$. Combine entrance, all $N-1$ intermediate full blocks and this
  terminal estimate to obtain~\eqref{E:raw-block-positive-product}, with $c_* =
  a_*j_*/4$. The induction is empty for $N = 1$ and the estimate includes $u =
  0$. This proves Lemma~\ref{L:raw-block-indexed-cone}.
\end{proof}

\subsection{Source restart and comparison with the original point}

So far the blocks $D_m^0$ and $D_m^+$ are defined one at a time, from the local
operator paths $X_m$ and $X_m+H_m$. We now show that each family forms the
successive pieces of its corresponding global flow, and that its finite
products are source limits in $L^2$ at fixed $\delta,N$. The proof of
Proposition~\ref{P:raw-block-point-comparison} uses this to verify the
source-limit premise of the masked comparison.

\LeanMarker{L:raw-block-restart}
\begin{lemma}[Composition over time blocks and regularized products]
  \label{L:raw-block-restart}
  Fix $\delta,B,N$ and let $L = \delta^{-1}$, $L_{\mathrm{ext}} = (N+1)B \ge L$.
  On this global fast horizon define $A_{\mathrm{glob}} = g_\delta
  A_R(W_{\mathrm{fast}})$. Its physical flow obeys
  \begin{equation}\label{E:raw-block-restart}
    P_{\mathrm{glob}}(mB+u,mB)=D_m^0(u),\qquad 0\le u\le B.
  \end{equation}
  The analogous identity holds with the deterministic mean $V_{\delta t}$ added
  globally and local operator paths $X_m+H_m$. Their finite block products,
  including the last remainder $\tau$, are source limits in $L^2$, at this fixed
  $\delta,N$.
\end{lemma}

\begin{proof}
  Take one regularized slow noise preserving the given noise $Y_{\epsilon,\eta}$
  on the extension of the given $W$ constructed in
  Section~\ref{S:source-driver-convergence}. Define the fast and local noises by
  \[
    Y_{\mathrm{fast}}(f)=Y_{\epsilon,\eta}(T_{\delta,0}f),\qquad
    Y_m(f)=Y_{\epsilon,\eta}(T_{\delta,m}f).
  \]
  The pullback isometry and~\eqref{E:source-two-dominations} give source and
  residual variance bounds by $\|f\|_{H_W}^2$. The residual tends to zero on
  each transformed smooth compact test at fixed $\delta,m$. Joint Gaussianity is
  retained. Proposition~\ref{P:source-gaussian-transfer} therefore gives the
  $L^pC^\theta$ operator-path limits on $[0,L_{\mathrm{ext}}]$ and on $[0,B]$,
  including both constant columns.

  Their random fields are $Y_{\mathrm{fast}}(u,x) =
  \delta^{a/2}Y_{\epsilon,\eta}(\delta u,x)$ and $Y_m(u,x) =
  Y_{\mathrm{fast}}(mB+u,x)$. Indeed, these follow by change of variables in
  their test integrals. The temporal law is $K_{\eta/\delta}$ because
  $\delta^aK_\eta(\delta r) = K_{\eta/\delta}(r)$; the spatial heat law and all
  joint width rules are unchanged. They are continuous in time and smooth in
  space. Their ordinary primitives equal their Gaussian operator paths: their
  matrix entries are $Y_m(h_R)$, by the indicator approximation used
  in~\eqref{E:raw-block-mean-jacobian} and the point-test integral identity.
  Total finite propagation extends this identity to both constant columns. This
  does not assume a separately chosen extension preserving the given noise
  automatically commutes with time dilation.

  Changing the ordinary source integration variable and composing the free
  groups gives the exact identity on $F$
  \[
    S_F(mB)\big[A_{\mathrm{glob}}^{\mathrm{src}}(mB+u)
    -A_{\mathrm{glob}}^{\mathrm{src}}(mB)\big]
    S_F(-mB)=X_m^{\mathrm{src}}(u).
  \]
  No physical mask is commuted through a wave group. Restriction, difference and
  conjugation are bounded maps on these fixed-horizon $C^\theta$ spaces. Passing
  the source limits gives the same identity for $A_{\mathrm{glob}},X_m$. For the
  finitely many $m$, a convergent subsequence or rational times and path
  continuity gives simultaneous identities in $u$. Conjugation norms need only
  be finite at this fixed $\delta$.

  Conjugate the global interaction Young equation on $[mB,mB+u]$ by $S_F(mB)$.
  Bounded linearity of its Riemann sums and that original identity give the
  local equation driven by $X_m$. Uniqueness identifies it with $U_{X_m}$,
  proving \eqref{E:raw-block-restart}. The ordinary deterministic primitive for
  $V_{\delta t}$ has the identical cocycle with local $H_m$. Composition now
  identifies the full product, with terminal $\tau$.

  For completeness, source convergence of these finite products requires no
  growing-product stability assertion. At fixed $\delta,N$ let $C \ge 1$ be the
  maximum norm of their source and original factors, and let $e_{\mathrm{src}}$
  be the maximum factor difference. The parent Young convergence gives every
  finite moment of $C$ uniformly in source widths and $e_{\mathrm{src}} \to 0$
  in every finite $L^p$. If $d(n)$ is the difference of the first $n$ ordered
  products, then
  \[
    d(n+1)\le C d(n)+C^n e_{\mathrm{src}},\qquad d(0)=0.
  \]
  Extend both finite sequences by identities so the recurrence holds for all
  natural $n$. The finite error-accumulation estimate gives $d(N+1) \le
  (N+1)C^{N+1}e_{\mathrm{src}}$. H\"older, using $C$ in $L^{4(N+1)}$ and the
  error in $L^4$, proves $L^2$ convergence. This is only a fixed-$\delta$ source
  limit. Shifted source products may use the fixed original mean plus the
  regularized noise; no finite-source Cameron mean is substituted. This proves
  Lemma~\ref{L:raw-block-restart}.
\end{proof}

\begin{proposition}[The square observable and canonical comparison]
  \label{P:raw-block-point-comparison}
  Define the signed averages without and with the deterministic mean
  \[
    U_{\delta,B}
    =O_\phi D_N^0(\tau)D_{N-1}^0(B)\cdots D_0^0(B)e_0,
    \qquad
    Y_{\delta,B}
    =O_\phi D_N^+(\tau)D_{N-1}^+(B)\cdots D_0^+(B)e_0.
  \]
  They have all finite unweighted moments, and
  \begin{align}
    \E U_{\delta,B}^2
    &=\E\left[e^{-Z/\sqrt\delta-v/(2\delta)}
    Y_{\delta,B}^2\right],
    \label{E:raw-block-square-identity}\\
    \E U_{\delta,B}^2
    &\le\E|V_\delta(1,0)|^2
    =\E|u(T,0)|^2,\qquad \delta=T^{-\kappa}.
    \label{E:raw-block-original-comparison}
  \end{align}
  The variance $v$ in the density is the original slow original variance.
\end{proposition}

\begin{proof}
  On all of $\mathscr{B}_\delta$ put
  \[
    j_\delta(x)=
    |O_\phi D_{x_N}(\tau)D_{x_{N-1}}(B)\cdots D_{x_0}(B)e_0|^2.
  \]
  The everywhere-defined continuous Young map, fixed-time bounded free groups,
  finite products and bounded $O_\phi$ make $j_\delta$ continuous and
  nonnegative on the whole input space. Apply
  Proposition~\ref{P:raw-block-gaussian-law} directly to these original paths.
  Since $X+H$ is its exact translate, this proves
  \eqref{E:raw-block-square-identity} with the single original $Z = W(\Psi)$ and
  its variance $v$. Finite products of the local Young flows have every finite
  unweighted moment by their Gaussian path bounds and H\"older, at fixed
  $\delta,N$. The Gaussian identity itself did not require prior integrability
  of $j_\delta$. No source convergence under its exponential density is used.

  To apply the masked comparison at the amplitude, put $\Gamma_\delta =
  g_\delta^2\Gamma$ and $W' = g_\delta W_{\mathrm{fast}}$. This is a nonzero
  member of the same admissible covariance class, with unchanged degree $\alpha$
  and temporal parameter $a$. Its integrated operator is $g_\delta$ times the
  original one: the original matrix forms are linear in noise, and
  basis-independent Gaussian transfer and the bounded $F$ extension retain both
  constant columns. The $n$th source chain is multiplied by $g_\delta^n$. Thus,
  this is exactly the model without the deterministic mean at that amplitude.

  We verify the source-limit premise of the masked comparison. Keep $\delta,B,N$
  fixed and use the single regularized noise preserving the given noise of
  Lemma~\ref{L:raw-block-restart}. For $m < N$ let $F_m = D_m^0(B)$ and put $F_N
  = D_N^0(\tau)$; write $F_{m,\rho}$ for the corresponding ordinary source
  factors, where both source widths in $\rho$ tend to zero.
  Lemma~\ref{L:raw-block-restart} and Proposition~\ref{P:constant-gaussian-flow}
  give factor convergence in every finite $L^q$ and uniform bounds for every
  finite moment of their norms. These statements are on the fixed finite
  horizons $[0,(N+1)B]$ and $[0,B]$, and include both constant-response columns.
  Set $n = N+1$ and
  \[
    C_\rho=\max\left(1,\max_{m\le N}\|F_{m,\rho}\|,
    \max_{m\le N}\|F_m\|\right),\qquad
    e_\rho=\max_{m\le N}\|F_{m,\rho}-F_m\|.
  \]
  Then $\sup_\rho\|C_\rho\|_q < \infty$ for every finite $q$, and $\|e_\rho\|_q
  \to 0$. Telescoping the ordered products gives
  \[
    \|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\|
    \le n C_\rho^n e_\rho.
  \]
  Therefore, H\"older with exponents $4n$ and $4$ gives
  \[
    \big\|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\big\|_{L^2}
    \le n\|C_\rho\|_{L^{4n}}^n\|e_\rho\|_{L^4}
    \longrightarrow0.
  \]
  No uniform estimate in the growing number $N$ is needed: this source limit is
  completed at each fixed $\delta,N$. The physical source identification in
  Lemma~\ref{L:constant-source-solution}, with the same $m_R$ and initial state
  $e_0$, identifies the source product with the tested ordinary masked solution
  at fast time $L = 1/\delta$. Boundedness of $O_\phi$ and $\|e_0\| = 1$ thus
  identify $U_{\delta,B}$ as its $L^2$ limit. The sources $g_\delta
  Y_{\mathrm{fast}}$ have the correct law, contraction and original-test limit
  by Lemma~\ref{L:raw-block-restart}.

  At positive source widths, $0 \le m_R \le 1$ multiplies each nonnegative
  displacement Wick integrand by the product of its vertex masks. Positive Green
  measures and covariance retain this comparison for unequal, self and empty
  chains as well. Proposition~\ref{P:masked-source-moments} gives the tested
  comparison, and the just-proved $L^2$ convergence discharges the premise of
  Proposition~\ref{P:masked-canonical-transfer} at $\Gamma_\delta$. Its
  canonical unmasked moment limit uses the fixed-order joint source limits and
  the absolutely summable order bounds. Since $\phi \ge 0$ and $\int\phi = 1$,
  these facts give
  \[
    \E U_{\delta,B}^2
    \le\E|u_{\Gamma_\delta}(L,0)|^2.
  \]
  Only the source products without the deterministic mean were needed for this
  passage. The shifted original product is already the everywhere-defined
  $j_\delta(X+H)$ in the Gaussian identity; no tilted source limit or change of
  the original slow $\Psi,Z,v$ has been made.

  The ordinary change from fast $u$ to slow $t = \delta u$, with phase
  $(V,\delta V_t)$, transforms the unmasked source equation into the regularized
  version of~\eqref{E:exact-fast-equation}, with slow terminal time one. Indeed,
  $g_\delta Y_{\mathrm{fast}}(u) = \sqrt\delta\,Y_{\epsilon,\eta}(\delta u)$,
  and its fast cutoff is $\eta/\delta$. The canonical fixed-order and absolutely
  convergent series construction passes this law identity at fixed $\delta$ to
  the original moment, preserving every mixed, unequal, self and empty chain.
  Lemma~\ref{L:exact-canonical-scaling} gives the final equality
  in~\eqref{E:raw-block-original-comparison}. The physical scale $r_T = T\delta$
  is distinct from the fixed mask radius $R$. The mask need not contain the
  growing backward cone: the comparison is a moment inequality, not equality of
  masked and full fields. All source limits preceded this large-time comparison.
  This proves Proposition~\ref{P:raw-block-point-comparison}.
\end{proof}

Proposition~\ref{P:raw-block-point-comparison} bounds $\E|u(T,0)|^2$ from below
by the weighted second moment of $Y_{\delta,B}$
in~\eqref{E:raw-block-square-identity}. We still need a lower bound for
$Y_{\delta,B}$ itself. The proof below obtains one on an event whose probability
tends to one. It uses the estimates of Lemma~\ref{L:raw-block-uniform-error} and
the cone bound of Lemma~\ref{L:raw-block-indexed-cone}.

\LeanMarker{P:raw-block-original-lower}
\begin{proposition}[Original point lower rate at the Gaussian quadratic cost]
  \label{P:raw-block-original-lower}
  For every original positive $\Psi$ and eligible fixed mask $R$ chosen above,
  and every fixed $x \in \R^d$,
  \begin{equation}\label{E:raw-block-original-lower}
    \liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,x)|^2
    \ge 2\int_0^1\lambda_s\ud s-\frac{v_\Psi}{2}.
  \end{equation}
\end{proposition}

\begin{proof}
  Fix $e_{\mathrm{rate}} > 0$. Choose $0 < \varrho < 1/2$ with $\log(1-\varrho)
  > -e_{\mathrm{rate}}/3$, then choose the cone constants, including an integer
  $B$ fixed independently of $\delta$, from
  Lemma~\ref{L:raw-block-indexed-cone}. The control $\Psi$ and eligible mask $R$
  remain fixed throughout.

  Choose a finite $p > 2/(1-a)$ and put $\gamma_p = p(1-a)/2-1 > 0$.
  From~\eqref{E:raw-block-moments} and $N+1 \le 2/\delta$, Markov and the union
  bound give, for $\eta > 0$,
  \[
    \begin{split}
      \mathbb{P}(M_\delta>\eta)
      &\le\eta^{-p}\sum_{m=0}^N
      \E\|X_m\|_{C^\theta}^p\\
      &\le A_p\eta^{-p}\delta^{\gamma_p},\qquad
      A_p=2(C_{R,B}\sqrt p)^p<\infty.
    \end{split}
  \]
  This estimate concerns the coupled blocks; no independence is needed. At these
  fixed data $A_p$ is independent of $\delta$.

  We also specify the deterministic tolerance used with this bound. The
  primitive estimates in Lemma~\ref{L:raw-block-uniform-error} give, for the
  constant-potential primitives $H_m^{\mathrm{stat}}$,
  \[
    \max_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \max_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C_B\delta.
  \]
  Apply Proposition~\ref{P:constant-young-linear} on the fixed ball of operator
  paths of radius $H_0+1$ and use the bounded free groups on $[0,B]$. There is
  $L_B \ge 1$, depending on these fixed $R,B,\Psi$ and the admissible exponents,
  such that on $\{M_\delta\le1\}$,
  \[
    E_{\mathrm{op},\delta}\le L_B(M_\delta+\delta).
  \]
  Set $\eta = \min(1,\epsilon_{\mathrm{op}}/(2L_B)) > 0$. If $\delta \le
  \epsilon_{\mathrm{op}}/(2L_B)$, then $M_\delta \le \eta$ implies
  $E_{\mathrm{op},\delta} \le \epsilon_{\mathrm{op}}$. Thus, for the measurable
  event on the original common space
  \[
    E_\delta=\{E_{\mathrm{op},\delta}\le\epsilon_{\mathrm{op}}\}
    \cap\{|Z|\le\delta^{-1/4}\},\qquad
    p_\delta=\mathbb{P}(E_\delta),
  \]
  Chebyshev for the same single $Z = W(\Psi)$, with $\E Z^2 = v$, gives
  \[
    p_\delta\ge1-A_p\eta^{-p}\delta^{\gamma_p}
    -v\delta^{1/2}\longrightarrow1.
  \]
  This is one union bound, with no independence between blocks or from $Z$ and
  no product of block probabilities.

  Let $C_\lambda$ be a Lipschitz bound for $s\mapsto\lambda_s$. Take $\delta$
  small enough, in addition to $0 < \delta < \min(1,1/B)$, that
  \[
    \begin{gathered}
      \delta B\le\omega,\qquad
      \delta\le\epsilon_{\mathrm{op}}/(2L_B),\qquad
      C_\lambda\delta B\le e_{\mathrm{rate}}/3,\\
      \delta|\log c_*|\le e_{\mathrm{rate}}/3,\qquad
      A_p\eta^{-p}\delta^{\gamma_p}+v\delta^{1/2}\le1/2.
    \end{gathered}
  \]
  Every constant has been fixed before this choice. The condition involving
  $C_\lambda$ is vacuous if that constant is zero. All source limits in
  Proposition~\ref{P:raw-block-point-comparison} have already been completed at
  each such fixed $\delta$.

  The indices $s_m = m\delta B$ lie in $I$ with adjacent gaps at most $\omega$.
  On $E_\delta$ the shifted full-$F$ blocks therefore satisfy
  Lemma~\ref{L:raw-block-indexed-cone}, retaining both phase entries and the
  final remainder $\tau$. In particular, $Y_{\delta,B} > 0$, and
  \[
    \delta\log Y_{\delta,B}
    \ge\delta\log c_*+\delta(N-1)\log(1-\varrho)
    +\delta B\sum_{m<N}\lambda_{s_m}
    +\delta\tau\lambda_{s_N}.
  \]
  Since $s_N+\delta\tau = 1$, the last two terms are the left Riemann sum for
  $I_R$ on the entire interval, including its last interval of length
  $\delta\tau$, with mesh at most $\delta B$. Its error is at most
  $C_\lambda\delta B$. Also $0 \le \delta(N-1) \le 1/B \le 1$, so the
  logarithmic loss is at least $\log(1-\varrho) > -e_{\mathrm{rate}}/3$. The two
  other losses are each at most $e_{\mathrm{rate}}/3$ by the choices above.
  Hence,
  \[
    Y_{\delta,B}\ge
    \exp((I_R-e_{\mathrm{rate}})/\delta)\quad\hbox{on }E_\delta,
  \]
  including the $N = 1$ and $\tau = 0$ cases of the cone estimate.

  The integrand in~\eqref{E:raw-block-square-identity} is nonnegative. On the
  same $E_\delta$ its linear Gaussian term satisfies $-Z/\sqrt\delta \ge
  -\delta^{-3/4}$. Restriction to this event
  and~\eqref{E:raw-block-original-comparison} therefore give the explicit bound
  for the original point moment
  \[
    \E|u(T,0)|^2\ge\E U_{\delta,B}^2
    \ge p_\delta\exp\left(
    \frac{2I_R-2e_{\mathrm{rate}}-v/2}{\delta}
    -\delta^{-3/4}\right),\qquad \delta=T^{-\kappa}.
  \]
  Here $p_\delta \ge 1/2$ for small $\delta$, so $-\delta\log2 \le \delta\log
  p_\delta \le 0$. Multiplying the logarithm of the preceding inequality by
  $\delta$ yields
  \[
    \delta\log\E|u(T,0)|^2
    \ge2I_R-2e_{\mathrm{rate}}-v/2
    -\delta^{1/4}+\delta\log p_\delta.
  \]
  Thus, the normalized linear penalty is $\delta\delta^{-3/4} = \delta^{1/4} \to
  0$, and $\delta\log p_\delta \to 0$. The quadratic Gaussian cost is precisely
  half the original variance $v$. Since $\kappa > 0$, $\delta = T^{-\kappa}$ is
  a bijection between all sufficiently large $T$ and all sufficiently small
  $\delta$. This gives the full continuous-time liminf $Q_2 \ge
  2I_R-2e_{\mathrm{rate}}-v/2$. For each $e_{\mathrm{rate}}$ the argument first
  fixed its own $B = B(e_{\mathrm{rate}})$ and then took $\delta\downarrow0$.
  The original point rate $Q_2$ is independent of $B$; only now let
  $e_{\mathrm{rate}}\downarrow0$. No choice $B = B(\delta)$ is made.

  Finally, spatial translation of all positive-width vertices preserves the
  original noise law, Green kernels and constant initial data. The fixed-order
  $L^2$ and absolute-series limits preserve equality of point second moments.
  Hence, zero can be replaced by every fixed $x$
  in~\eqref{E:raw-block-original-lower}. Every control and mask were fixed
  before asymptotics; a supremum over them may follow their individual
  inequalities. This proves Proposition~\ref{P:raw-block-original-lower}.
\end{proof}
