namespace StratonovichHAMAudit.VectorProjection

open GaussianChaos.GaussianProduct GaussianChaos.ChaosDensity
open MeasureTheory

universe u v w

variable {Mode : Type u} {E : Type v}
variable [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]

/-- An actual vector-valued Gaussian L2 element, not an arbitrary polynomial representative. -/
def atom (ν : Mode →₀ ℕ) (x : E) : Lp E 2 (noise Mode) :=
  (ContinuousLinearMap.toSpanSingleton ℝ x).compLp (hermiteNormalized ν)

/-- The finite vector-valued Hermite polynomial in the actual probability L2 space. -/
def realize (f : (Mode →₀ ℕ) →₀ E) : Lp E 2 (noise Mode) :=
  f.sum (fun ν x => atom ν x)

/-- The closed L2 subspace generated by all selected modes with arbitrary vector coefficients. -/
def cutoffSpace (p : (Mode →₀ ℕ) → Prop) : ClosedSubmodule ℝ (Lp E 2 (noise Mode)) :=
  (Submodule.span ℝ {z | ∃ (ν : Mode →₀ ℕ) (x : E), p ν ∧ z = atom ν x}).closure

/-- Coefficient deletion is a definition; its identification with orthogonal projection is the target. -/
def cut (p : (Mode →₀ ℕ) → Prop) (f : (Mode →₀ ℕ) →₀ E) : (Mode →₀ ℕ) →₀ E := by
  classical
  exact f.filter p

/-- Each actual coordinate has a color, without adjoining any unused Gaussian coordinates. -/
def colorDegree {Color : Type w} [DecidableEq Color]
    (color : Mode → Color) (ν : Mode →₀ ℕ) (c : Color) : ℕ :=
  ν.sum (fun i n => if color i = c then n else 0)

def colorLow {Color : Type w} [DecidableEq Color]
    (color : Mode → Color) (N : Color → ℕ) (ν : Mode →₀ ℕ) : Prop :=
  ∀ c, colorDegree color ν c ≤ N c

open scoped InnerProductSpace
attribute [local instance] Classical.propDecidable

omit [CompleteSpace E] in
private lemma atom_inner (ν μ : Mode →₀ ℕ) (x y : E) :
    ⟪atom ν x, atom μ y⟫_ℝ = if ν = μ then ⟪x,y⟫_ℝ else 0 := by
  classical
  calc
    ⟪atom ν x, atom μ y⟫_ℝ =
        ⟪hermiteNormalized ν, hermiteNormalized μ⟫_ℝ * ⟪x,y⟫_ℝ := by
      simp only [MeasureTheory.L2.inner_def, atom]
      rw [← integral_mul_const]
      apply integral_congr_ae
      filter_upwards [ContinuousLinearMap.coeFn_compLp
          (ContinuousLinearMap.toSpanSingleton ℝ x) (hermiteNormalized ν),
        ContinuousLinearMap.coeFn_compLp
          (ContinuousLinearMap.toSpanSingleton ℝ y) (hermiteNormalized μ)] with ω hx hy
      rw [hx, hy]
      simp only [ContinuousLinearMap.toSpanSingleton_apply, real_inner_smul_left,
        real_inner_smul_right, Real.inner_apply]
      ring
    _ = _ := by
      rw [orthonormal_iff_ite.mp (orthonormal_hermite (ι := Mode)) ν μ]
      split_ifs <;> simp

private lemma project_atom (p : (Mode →₀ ℕ) → Prop) (ν : Mode →₀ ℕ) (x : E) :
    (cutoffSpace (E := E) p).toSubmodule.starProjection (atom ν x) =
      if p ν then atom ν x else 0 := by
  classical
  split_ifs with h
  · apply Submodule.starProjection_eq_self_iff.mpr
    change atom ν x ∈ (Submodule.span ℝ
      {z | ∃ (μ : Mode →₀ ℕ) (y : E), p μ ∧ z = atom μ y}).topologicalClosure
    exact Submodule.le_topologicalClosure _ (Submodule.subset_span ⟨ν, x, h, rfl⟩)
  · let S : Submodule ℝ (Lp E 2 (noise Mode)) :=
      Submodule.span ℝ {z | ∃ (μ : Mode →₀ ℕ) (y : E), p μ ∧ z = atom μ y}
    have hs : S ≤ (Submodule.span ℝ {atom ν x})ᗮ := by
      apply Submodule.span_le.mpr
      rintro z ⟨μ, y, hm, rfl⟩
      apply (Submodule.mem_orthogonal_singleton_iff_inner_right
        (𝕜 := ℝ) (u := atom ν x) (v := atom μ y)).mpr
      rw [atom_inner]
      split_ifs with heq
      · exact (h (heq ▸ hm)).elim
      · rfl
    have hc : S.topologicalClosure ≤ (Submodule.span ℝ {atom ν x})ᗮ :=
      S.topologicalClosure_minimal hs (Submodule.isClosed_orthogonal _)
    apply Submodule.eq_starProjection_of_mem_of_inner_eq_zero (Submodule.zero_mem _)
    intro z hz
    simp only [sub_zero]
    exact Submodule.mem_orthogonal_singleton_iff_inner_right.mp (hc hz)

omit [CompleteSpace E] in
private lemma realize_cut (p : (Mode →₀ ℕ) → Prop) (f : (Mode →₀ ℕ) →₀ E) :
    realize (cut p f) = f.sum (fun ν x => if p ν then atom ν x else 0) := by
  classical
  simp [realize, cut, Finsupp.sum, Finset.sum_filter, Finsupp.filter_apply]
  apply Finset.sum_congr rfl
  intro ν _
  by_cases h : p ν <;> simp [h]

/-- Actual Hilbert orthogonal projection of a finite vector-valued Hermite polynomial.
The exact target and source/definition correspondence were independently reviewed before proof. -/
theorem projection_realize (p : (Mode →₀ ℕ) → Prop) (f : (Mode →₀ ℕ) →₀ E) :
    (cutoffSpace (E := E) p).toSubmodule.starProjection (realize f) = realize (cut p f) := by
  classical
  calc
    (cutoffSpace (E := E) p).toSubmodule.starProjection (realize f) =
        f.sum (fun ν x => (cutoffSpace (E := E) p).toSubmodule.starProjection (atom ν x)) := by
      simp only [realize, Finsupp.sum, map_sum]
    _ = f.sum (fun ν x => if p ν then atom ν x else 0) := by
      apply Finset.sum_congr rfl
      intro ν _
      exact project_atom p ν (f ν)
    _ = realize (cut p f) := (realize_cut p f).symm

-- Private nondegenerate probe: different Gaussian modes and different coefficient directions.
example :
    let ν₀ : Fin 2 →₀ ℕ := Finsupp.single 0 1
    let ν₁ : Fin 2 →₀ ℕ := Finsupp.single 1 1
    let x₀ : EuclideanSpace ℝ (Fin 2) := EuclideanSpace.single 0 1
    let x₁ : EuclideanSpace ℝ (Fin 2) := EuclideanSpace.single 1 1
    let P := (cutoffSpace (E := EuclideanSpace ℝ (Fin 2)) (fun ν => ν = ν₀)).toSubmodule.starProjection
    let v := atom ν₀ x₀ + atom ν₁ x₁
    P v = atom ν₀ x₀ ∧ ⟪P v, P v⟫_ℝ = 1 ∧ ⟪v - P v, v - P v⟫_ℝ = 1 := by
  classical
  dsimp only
  let ν₀ : Fin 2 →₀ ℕ := Finsupp.single 0 1
  let ν₁ : Fin 2 →₀ ℕ := Finsupp.single 1 1
  let x₀ : EuclideanSpace ℝ (Fin 2) := EuclideanSpace.single 0 1
  let x₁ : EuclideanSpace ℝ (Fin 2) := EuclideanSpace.single 1 1
  have hn : ν₁ ≠ ν₀ := by
    intro h
    have hh := congrArg (fun ν : Fin 2 →₀ ℕ => ν 0) h
    norm_num [ν₀, ν₁] at hh
  have hp : (cutoffSpace (E := EuclideanSpace ℝ (Fin 2)) (fun ν => ν = ν₀)).toSubmodule.starProjection
      (atom ν₀ x₀ + atom ν₁ x₁) = atom ν₀ x₀ := by
    rw [map_add, project_atom, project_atom]
    simp [hn]
  refine ⟨hp, ?_, ?_⟩
  · rw [hp, atom_inner]
    simp [x₀]
  · have hx : ⟪atom ν₁ x₁, atom ν₁ x₁⟫_ℝ = 1 := by
      rw [atom_inner]
      simp [x₁]
    have hv : atom ν₀ x₀ + atom ν₁ x₁ -
        (cutoffSpace (E := EuclideanSpace ℝ (Fin 2)) (fun ν => ν = ν₀)).toSubmodule.starProjection
          (atom ν₀ x₀ + atom ν₁ x₁) = atom ν₁ x₁ := by
      rw [hp]
      abel
    exact (congrArg₂ (inner ℝ) hv hv).trans hx
