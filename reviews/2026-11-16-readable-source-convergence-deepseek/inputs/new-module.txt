\section{Convergence of the regularized Gaussian operator paths}
\label{S:source-driver-convergence}

We match the heat normalization, Laplace prefix and joint source law in the
author version~\cite{chen:author2026:time-dependency}*{(1.20), (1.21), (2.9),
(2.11), pp.~5--6, 8--9}. Preservation of the given noise retention, the Radon
implementation and the operator-path conclusions are proved here. No
first-moment variational theorem is an input.

Retain Hypothesis~\ref{H:homog}, the fixed $R,T,\zeta_R,E$ and parameters
of~\eqref{E:driver-parameters}, and the original noise space $H_W$. Write
$\mathscr{B}_\theta = C^\theta([0,T];\mathcal{B}(E))$. Fix one original test
basis $f_j$ and its coefficient paths $A_j(t)$ from
Proposition~\ref{P:actual-operator-driver}. Thus, $M_N = \sum_{j\le N}W(f_j)A_j
\to M_R$ in every finite $L^p(\mathscr{B}_\theta)$. Zero covariance gives zero
noises and operator paths throughout; assume $\Gamma \ne 0$ in the construction.

\subsection{Gaussian transfer in the full path norm}

\LeanMarker{L:source-finite-comparison}
\begin{lemma}[Finite Gaussian linear images]\label{L:source-finite-comparison}
  Let $v_1,\ldots,v_N$ lie in a real Banach space and let $\xi$ be a centered
  Gaussian vector with covariance bounded by $CI$, $C \ge 0$. For independent
  standard real $g_j$ and $p \ge 1$,
  \begin{equation}\label{E:source-finite-comparison}
    \left(\E\left\|\sum_j\xi_jv_j\right\|^p\right)^{1/p}
    \le\sqrt C\left(\E\left\|\sum_jg_jv_j\right\|^p\right)^{1/p}.
  \end{equation}
\end{lemma}

\begin{proof}
  The function $\varphi(x) = \|\sum_jx_jv_j\|^p$ is continuous and convex, even
  when the linear image gives only a seminorm on $\R^N$. Its Gaussian
  expectations are finite by the finite-sum bound. On an auxiliary space take an
  independent centered Gaussian $Z$ of covariance $CI-\operatorname{Cov}\xi$.
  The covariance difference is positive semidefinite by hypothesis, so its
  diagonalization constructs such a Gaussian vector even if it is degenerate.
  Independence and centering give conditional mean $\xi$ for $\xi+Z$ given
  $\xi$. Convexity therefore applies to this conditional expectation.
  Conditional Jensen gives $\E\varphi(\xi) \le \E\varphi(\xi+Z)$. The latter
  vector has the law of $\sqrt C\,g$, proving the claim. The case $C = 0$ is
  immediate. This auxiliary comparison changes neither the coupling of $\xi$ nor
  any nonlinear wave solution. This proves
  Lemma~\ref{L:source-finite-comparison}.
\end{proof}

Lemma~\ref{L:source-finite-comparison} compares two finite Gaussian sums with
the same vectors. We apply it to the coefficient paths $A_j$, keeping the
original basis $f_j$ and replacing $W(f_j)$ by the evaluations of another
Gaussian noise.

\begin{proposition}[Gaussian transfer and stability]
  \label{P:source-gaussian-transfer}
  Let $Y$ be a centered Gaussian linear noise on smooth compact positive-time
  tests, on a common coupled space, with
  \begin{equation}\label{E:source-raw-variance-bound}
    \operatorname{Var}Y(F)\le C\|F\|_{H_W}^2.
  \end{equation}
  It extends to $H_W$. The finite paths $M_{Y,N} = \sum_{j\le N}Y(f_j)A_j$
  converge in every finite $L^p(\mathscr{B}_\theta)$ to a centered Gaussian path
  $M_Y$ with compact values at all times on one path event, and
  \begin{align}
    \|M_Y-M_{Y,N}\|_{L^p\mathscr{B}_\theta}
    &\le\sqrt C\|M_R-M_N\|_{L^p\mathscr{B}_\theta},\label{E:source-transfer-tail}\\
    \|M_Y\|_{L^p\mathscr{B}_\theta}
    &\le\sqrt C\|M_R\|_{L^p\mathscr{B}_\theta}.\label{E:source-transfer-norm}
  \end{align}
  For a fixed real Schwartz pair and time,
  \begin{equation}\label{E:source-transferred-row}
    \langle M_Y(t)z,w\rangle_E=Y(h_{R,t;z,w})\quad\text{in }L^2,
  \end{equation}
  where $h_{R,t;z,w}$ is the explicit original test class
  $\zeta_R1_{(0,t)}v_zG_w$ with the adjoint from~\eqref{E:variance-dual-wave}.
  The path is basis-independent up to indistinguishability, is linear in jointly
  Gaussia noise families, and complexifies sesquilinearly.

  If $(W,Y_n)$ are jointly centered Gaussian, $Y_n$ and $D_n = W-Y_n$ have
  uniform bounds of the form~\eqref{E:source-raw-variance-bound}, and
  $\operatorname{Var}D_n(F) \to 0$ on every smooth compact test, then $M_{Y_n}
  \to M_R$ in every finite $L^p(\mathscr{B}_\theta)$.
\end{proposition}

\begin{proof}
  We first transfer finite path sums and their tails using the Gaussian
  comparison. We then identify the transferred rows on the given coupling. For
  stability, we make the common tail small before sending the finitely many
  remaining noise errors to zero.

  \medskip
  \noindent\textit{Step 1.~}We extend the noise and compare its finite
  coordinate vectors.
  The variance bound annihilates covariance-null tests and extends $Y$ as a
  bounded linear map into its $L^2$ space. Completion preserves Gaussian
  finite-dimensional laws. For each finite original orthonormal family, the
  covariance of its $Y$ evaluations is bounded by $CI$, by applying the variance
  bound to every real linear combination. Independence of $Y(f_j)$ is not
  assumed.

  \medskip
  \noindent\textit{Step 2.~}We construct the transferred path and its bounds.
  Apply Lemma~\ref{L:source-finite-comparison} with $v_j = A_j(\cdot)$ in
  $\mathscr{B}_\theta$, also to finite tails. The standard comparison vector has
  the law of the original $W(f_j)$. The already-proved original path convergence
  makes $M_{Y,N}$ Cauchy in every $L^p$. It lies in the same separable closed
  span of the countably many deterministic paths. The elementary
  Cauchy/subsequence construction of Lemma~\ref{L:driver-holder-limit} gives a
  strongly measurable limit on the space of $Y$. Different $p$ limits agree by
  convergence in probability. Passing to limits in the finite inequalities gives
  both displayed bounds. An almost surely uniformly convergent subsequence gives
  compact values for all times. Functional $L^2$ limits of Gaussian linear
  combinations prove that the path is Gaussian.

  To obtain the tail inequality, first compare the sum from index $N+1$ to a
  finite upper index. Let that upper index tend to infinity in both $L^p$ path
  spaces. The norm inequality passes to the limit. The full-norm comparison is
  obtained in the same way starting at index one.

  \medskip
  \noindent\textit{Step 3.~}We identify the rows and remove dependence on the
  basis.
  For real Schwartz $z,w$, the coefficient identity is
  \[
    \langle A_j(t)z,w\rangle_E
    =\E[W(h_{R,t;z,w})W(f_j)]
    =\langle h_{R,t;z,w},f_j\rangle_{H_W}.
  \]
  Therefore, the finite transferred row is $Y(P_Nh)$, with $P_N$ the original
  $H_W$ orthogonal projection. Since $P_Nh \to h$, the noise bound gives
  $Y(P_Nh) \to Y(h)$ in $L^2$. Operator-path $L^2$ convergence
  proves~\eqref{E:source-transferred-row}. Long intervals use the same
  covariance completion and endpoint nullity as in
  Lemma~\ref{L:driver-interval-limit}. No filter is commuted through a physical
  mask or wave group. Four real pair identities give the complex assertion.
  Countably many dense Schwartz phases and rational times, followed by phase
  density and path continuity, prove basis indistinguishability.

  The row identity determines each bounded operator on a dense set of phase
  pairs. Taking rational times makes the collection countable; path continuity
  then determines all remaining times. Thus, comparing two bases requires only
  a countable intersection of events, followed by deterministic continuity.

  \medskip
  \noindent\textit{Step 4.~}We prove linearity and stability of the transfer.
  For jointly Gaussian $Y,Z$, their real linear combinations have an original
  variance bound by the $L^2$ triangle inequality. The finite sums are linear in
  the noise, hence so are the path limits. In the stability assertion $D_n$ is
  centered Gaussian by the joint hypothesis. Its uniform bound gives
  \[
    \|M_{D_n}-M_{D_n,N}\|_{L^p\mathscr{B}_\theta}
    \le\sqrt{C_D}\|M_R-M_N\|_{L^p\mathscr{B}_\theta}.
  \]
  The right side of this bound is independent of $n$ and tends to zero as $N$
  increases. We can therefore select the tail cutoff before considering
  convergence of the finite coordinate vector.
  For fixed $N$, the covariance matrix of $(D_n(f_j))_{j\le N}$ is positive and
  bounded by $s_N(n)I$, where $s_N(n) = \sum_{j\le N}\operatorname{Var}D_n(f_j)
  \to 0$. Every eigenvalue of a positive finite matrix is at most its trace,
  so the scalar $s_N(n)$ bounds that covariance matrix in positive order.
  Thus,
  the finite comparison gives $\|M_{D_n,N}\|_{L^p\mathscr{B}_\theta} \le
  \sqrt{s_N(n)}\|M_N\|_{L^p\mathscr{B}_\theta}$. Choose $N$ large for the
  uniform tail, then $n$ large for the finite head. Linearity identifies
  $M_{D_n} = M_R-M_{Y_n}$. The same argument proves a multiparameter limit
  whenever each fixed test error tends to zero, with no relation between
  parameter rates. This proves Proposition~\ref{P:source-gaussian-transfer}.
\end{proof}

\subsection{The spatial covariance space and an extension preserving the given noise}

Proposition~\ref{P:source-gaussian-transfer} reduces source-path convergence
to uniform variance domination and vanishing residual variance on each test,
for noises jointly Gaussian with $W$. We now construct such regularized
sources, beginning with the spatial covariance space and heat regularization.

\LeanMarker{L:source-spatial-heat}
\begin{lemma}[Original spatial space and heat]\label{L:source-spatial-heat}
  Let $H_\Gamma$ be the real test-null completion with
  \[
    B_\Gamma(f,g)=\int\Gamma(\ud y)\int f(x)g(x-y)\ud x
    =(2\pi)^d\int\widehat{f}(\xi)\overline{\widehat{g}(\xi)}\mu(\ud\xi).
  \]
  It is separable and includes Schwartz tests. With
  \begin{equation}\label{E:source-heat-definition}
    p_\varepsilon(x)=(2\pi\varepsilon)^{-d/2}
    e^{-|x|^2/(2\varepsilon)},\qquad C_\varepsilon f=p_\varepsilon*f,
  \end{equation}
  heat is a positive self-adjoint contraction on $H_\Gamma$, $I-C_\varepsilon$
  is a contraction, and $C_\varepsilon \to I$ strongly.
\end{lemma}

\begin{proof}
  The spectral normalization is the spatial computation in
  Proposition~\ref{P:masked-noise-completion}; its integral is real on real
  tests by evenness of $\mu$. The local bound $|B_\Gamma(f,g)| \le
  |K|\Gamma(K-K)\|f\|_\infty\|g\|_\infty$ on a common compact support gives
  separability by countable smooth approximation. Expanding smooth cutoffs of a
  Schwartz test converge in Schwartz seminorms, and its covariance correlation
  pairing converges against tempered $\Gamma$. Thus, Schwartz tests enter the
  same completion. The Fourier multiplier of $C_\varepsilon$ is
  $m_\varepsilon(\xi) = e^{-\varepsilon|\xi|^2/2}$. On dense Schwartz tests, the
  displayed norm represents heat by multiplication by the real function
  $m_\varepsilon$, which satisfies $0 \le m_\varepsilon \le 1$. The multiplier
  of $I-C_\varepsilon$ is $1-m_\varepsilon$, also between zero and one. Thus,
  the
  displayed norm gives positivity, self-adjointness and both contraction bounds.
  Dominated convergence gives strong convergence there; uniform boundedness
  extends it to the completion. This uses no physical mask operator on a noise
  quotient. This proves Lemma~\ref{L:source-spatial-heat}.
\end{proof}

Set $\nu_a(\ud\lambda) = \lambda^{a-1}\ud\lambda/\Gamma_E(a)$ and
\[
  \mathcal{G}=L^2((0,\infty)_\lambda\times\R_r,
  \nu_a(\ud\lambda)\ud r;H_\Gamma).
\]
This is a real separable Hilbert space: countable simple functions on rational
rectangles with a countable dense spatial collection suffice.

\begin{lemma}[Feature isometry preserving the given noise]\label{L:source-retaining-isometry}
  On finite smooth temporal-spatial tensors define
  \begin{equation}\label{E:source-feature-isometry}
    \Phi F(\lambda,r)=\sqrt{2\lambda}
    \int_{\substack{t>0\\t\ge r}}e^{-\lambda(t-r)}F(t,\cdot)\ud t.
  \end{equation}
  It extends to an isometry $\Phi:H_W \to \mathcal{G}$ with closed range, and
  the formula represents smooth compact tests almost everywhere. On a product
  extension retaining the original $W$, there is an isonormal $\mathcal{I}$ on
  $\mathcal{G}$ with
  \begin{equation}\label{E:source-given-W-retention}
    \mathcal{I}(\Phi h)=W(h)\quad\text{in }L^2
    \quad\text{for each }h\in H_W.
  \end{equation}
\end{lemma}

\begin{proof}
  We first prove the feature map is an isometry and identify its formula on
  completed tests. We then extend the given noise only on the orthogonal
  complement of the feature range, preserving every original evaluation.

  \medskip
  \noindent\textit{Step 1.~}We identify the feature norm with the original
  covariance norm.
  The identity
  \[
    \int_{-\infty}^{\min(t,u)}
    2\lambda e^{-\lambda(t+u-2r)}\ud r
    =e^{-\lambda|t-u|}
  \]
  and the original Laplace formula give $\|\Phi F\|_{\mathcal{G}}^2 =
  \|F\|_{H_W}^2$.
  Indeed, integrating the two exponential time features over $r$ gives the
  displayed kernel. Integrating that kernel against $\nu_a$ gives
  $|t-u|^{-a}$. The spatial inner product is the original $B_\Gamma$, so
  these two integrations reproduce the original covariance.
  All signed exchanges for finite tensors are absolutely
  integrable: compact temporal factors give integrals of
  $|\tau(t)\sigma(u)||t-u|^{-a}$, finite for $a < 1$, times fixed finite spatial
  inner products. No oscillatory infinite-measure Fubini is used. Smooth time
  partitions on one compact support approximate a general smooth spacetime test
  uniformly by finite tensors. The local covariance
  bound~\eqref{E:driver-test-continuity} proves their density in $H_W$. Hence,
  the isometry extends with closed range.

  The norm identity sends every covariance-null tensor to zero and preserves
  the distance between tensor classes. It therefore extends uniquely by
  completion. Since $H_W$ is complete, the range of the extended isometry is
  closed.

  \medskip
  \noindent\textit{Step 2.~}We identify the integral formula for general
  compact tests.
  For a general smooth compact $F$, those approximants also converge uniformly
  in spatial $H_\Gamma$ norm in time, by local Radon mass. For fixed
  $\lambda,r$, their integrals in~\eqref{E:source-feature-isometry} converge in
  $H_\Gamma$. Meanwhile the isometry gives convergence in $\mathcal{G}$, with an
  almost everywhere convergent subsequence. Its two limits agree, proving the
  stated formula. The coordinate $r$ is auxiliary; no value of original
  negative-time noise has been assigned.

  \medskip
  \noindent\textit{Step 3.~}We extend the isonormal process while retaining
  the given noise.
  Let $P$ project orthogonally onto $\Phi H_W$. On a product extension take an
  independent isonormal $J$ on $(\Phi H_W)^\perp$, built from independent real
  standard Gaussians on a finite or countable basis. Define
  \[
    \mathcal{I}(g)=W(\Phi^{-1}Pg)+J((I-P)g).
  \]
  The two summands use orthogonal feature components and independent Gaussian
  processes. Their variances are $\|Pg\|_{\mathcal G}^2$ and
  $\|(I-P)g\|_{\mathcal G}^2$, whose sum is $\|g\|_{\mathcal G}^2$. If $g =
  \Phi h$, the second component vanishes and the first is the original $W(h)$.
  Orthogonality and independence make it isonormal on $\mathcal{G}$ and
  prove~\eqref{E:source-given-W-retention} with the original random variable,
  not only its law. For the countable fixed original basis, these equalities
  hold on one common probability-one set. Thus, the original finite paths and
  their $L^p$ limits are precisely pullbacks of the already constructed original
  operator path. Cutoff fields may use $J$; they are not asserted to be
  functions of $W$ alone. This proves Lemma~\ref{L:source-retaining-isometry}.
\end{proof}

The auxiliary component can be nonzero: No-go~\ref{P:nogo-32}
constructs an orthogonal witness for every positive finite cutoff point
feature at positive time. Proposition~\ref{P:constant-raw-average} later
proves that the tested fixed-mask limit is $W$-measurable, using the
operator-path and Young-flow convergence.

\subsection{Exact joint source law and convergence}

Let $P_\eta$ multiply $\mathcal{G}$ by $1_{\{\lambda\le\eta^{-1}\}}$. Let
$C_\varepsilon$ act fiberwise on $H_\Gamma$. These commute. For $\rho =
(\varepsilon,\eta) > 0$, define
\begin{equation}\label{E:source-cutoff-noises}
  Q_\rho=C_\varepsilon P_\eta,\qquad
  Y_\rho(F)=\mathcal{I}(Q_\rho\Phi F),\qquad
  D_\rho(F)=\mathcal{I}((I-Q_\rho)\Phi F)=W(F)-Y_\rho(F).
\end{equation}
The last equality uses the coupling preserving the given noise. Define
\begin{equation}\label{E:source-covariance-functions}
  \Gamma_e(x)=(p_e*\Gamma)(x),\qquad
  K_\eta(v)=\frac1{\Gamma_E(a)}
  \int_0^{\eta^{-1}}e^{-\lambda|v|}\lambda^{a-1}\ud\lambda.
\end{equation}
The temporal operation is a Laplace prefix, not temporal convolution.

\begin{proposition}[Canonical sources on the extended space containing the given noise]
  \label{P:source-canonical-convergence}
  The noises in~\eqref{E:source-cutoff-noises} have random field realizations
  with joint covariance
  \begin{equation}\label{E:source-joint-covariance}
    \operatorname{Cov}(Y_{\varepsilon,\eta}(t,x),
    Y_{\varepsilon',\eta'}(s,y))
    =K_{\max(\eta,\eta')}(t-s)\Gamma_{\varepsilon+\varepsilon'}(x-y).
  \end{equation}
  Mean-square random field integration agrees with $Y_\rho(F)$ on tests. Both
  source and full residual obey
  \begin{equation}\label{E:source-two-dominations}
    \operatorname{Var}Y_\rho(F),\quad\operatorname{Var}D_\rho(F)
    \le\|F\|_{H_W}^2,
  \end{equation}
  and $\operatorname{Var}D_\rho(F) \to 0$ for every $F \in H_W$ as
  $\varepsilon,\eta\downarrow0$ at arbitrary relative rates. Consequently, for
  every finite $p \ge 1$,
  \begin{equation}\label{E:source-driver-limit}
    M_R^\rho:=M_{Y_\rho}\longrightarrow M_R
    \quad\text{in }L^p(\mathscr{B}_\theta),\qquad
    M_{D_\rho}=M_R-M_R^\rho.
  \end{equation}
  The source and residual have uniform local Gaussian operator-path moments and
  square-exponential norms, with constants for fixed $R,T$ and exponents.
\end{proposition}

\begin{proof}
  We compute the joint law from the point features, including the two heat
  widths and the common Laplace range. We then identify field integrals, bound
  both filters, and apply Gaussian transfer to the vanishing residual.

  \medskip
  \noindent\textit{Step 1.~}We compute the spatial covariance and
  point-feature covariance.
  Apply the spatial distribution identity to a single shifted heat test:
  \[
    \Gamma_e(x)=\int e^{i\xi\cdot x}e^{-e|\xi|^2/2}\mu(\ud\xi).
  \]
  Gaussian spectral moments of every order are finite for tempered $\mu$. Thus,
  this is smooth with bounded derivatives. Since
  $\widehat{p_\varepsilon(x-\cdot)} = c_de^{-i\xi\cdot
  x}e^{-\varepsilon|\xi|^2/2}$, the two heat factors in the spatial inner
  product cancel $(2\pi)^d$ exactly, giving
  \begin{equation}\label{E:source-heat-inner-product}
    B_\Gamma(p_\varepsilon(x-\cdot),p_{\varepsilon'}(y-\cdot))
    =\Gamma_{\varepsilon+\varepsilon'}(x-y).
  \end{equation}
  The product of the two heat multipliers is
  $e^{-(\varepsilon+\varepsilon')|\xi|^2/2}$. The product of their Fourier
  normalization factors is $c_d^2$, and $(2\pi)^dc_d^2 = 1$. This gives the
  sum of widths without any remaining Fourier factor.
  Evenness of $\mu$ removes the reflected argument.

  For each positive $\rho$ and point $(t,x)$ put
  \begin{equation}\label{E:source-point-feature}
    v_{\rho;t,x}(\lambda,r)
    =1_{\{\lambda\le\eta^{-1}\}}\sqrt{2\lambda}
    e^{-\lambda(t-r)}1_{\{r\le t\}}p_\varepsilon(x-\cdot).
  \end{equation}
  Its squared $\mathcal{G}$ norm is $K_\eta(0)\Gamma_{2\varepsilon}(0) <
  \infty$. Define $Y_\rho(t,x) = \mathcal{I}(v_{\rho;t,x})$. The common
  $\lambda$ range for two such vectors is $\lambda \le
  \min(\eta^{-1},\eta'^{-1}) = \max(\eta,\eta')^{-1}$. Integrating $r$ gives
  $e^{-\lambda|t-s|}$, and~\eqref{E:source-heat-inner-product} gives the spatial
  factor. This proves~\eqref{E:source-joint-covariance} with the prescribed
  widths and original amplitudes, also for singular Radon and white covariances.

  \medskip
  \noindent\textit{Step 2.~}We identify the mean-square field integral with
  the filtered noise.
  The representing vector is continuous in $\mathcal{G}$ norm, directly from
  this continuous covariance. Its norm is bounded on compact parameter sets.
  Hence, for a smooth compact spacetime $F$, its Bochner integral against
  $v_{\rho;t,x}$ exists. Integrate the heat translates in $x$, then the time
  feature, to obtain exactly $Q_\rho\Phi F$. Applying the bounded map
  $\mathcal{I}:\mathcal{G} \to L^2$ proves the random field-test identity on the
  same space. Common-support indicator tests follow by covariance completion and
  $L^1$ approximation; bounded point-feature norms and the domination below give
  the same $L^2$ limit.

  For the two domination bounds, the important point is that the omitted
  Laplace range and the retained range are orthogonal. The heat error acts
  only in the retained range. We use this decomposition before comparing the
  residual norm with the original feature norm.

  \medskip
  \noindent\textit{Step 3.~}We bound the source and the full residual.
  Orthogonality of the two $\lambda$ ranges gives, for $g \in \mathcal{G}$,
  \[
    \|(I-Q_\rho)g\|_{\mathcal{G}}^2
    =\|(I-P_\eta)g\|_{\mathcal{G}}^2
    +\|P_\eta(I-C_\varepsilon)g\|_{\mathcal{G}}^2
    \le\|g\|_{\mathcal{G}}^2.
  \]
  On the retained range, $I-C_\varepsilon$ is a contraction by the heat lemma.
  Hence, its squared norm is bounded by $\|P_\eta g\|_{\mathcal G}^2$. Adding
  the omitted-range norm gives at most the full squared norm, because these
  two ranges are orthogonal.
  Also $\|Q_\rho g\|_{\mathcal{G}} \le \|g\|_{\mathcal{G}}$. Together with the
  isometry these prove both variance bounds. Disjoint $\lambda$ ranges are
  independent; the full heat residual need not be independent of the source.

  \medskip
  \noindent\textit{Step 4.~}We send both regularization parameters to zero.
  The $L^2$ tail integral gives $P_\eta \to I$ strongly. Fiberwise heat
  converges strongly on simple $\mathcal{G}$ functions by
  Lemma~\ref{L:source-spatial-heat}, then on all of $\mathcal{G}$ by the
  contraction bound. The estimate
  \[
    \|(I-Q_\rho)g\|_{\mathcal{G}}
    \le\|(I-C_\varepsilon)g\|_{\mathcal{G}}
    +\|(I-P_\eta)g\|_{\mathcal{G}}\longrightarrow0
  \]
  holds at arbitrary relative rates.
  Indeed, the two terms on the right depend on separate parameters, and each
  tends to zero by strong convergence. Therefore, there is no need to compare
  the rates at which the heat width and the Laplace cutoff are removed.
  Thus, the residual variance tends to zero
  for every original $F$. Proposition~\ref{P:source-gaussian-transfer}
  gives~\eqref{E:source-driver-limit} without commuting a filter through
  $\zeta_R$ or a wave group.

  \medskip
  \noindent\textit{Step 5.~}We retain the uniform Gaussian path bounds.
  The same comparison gives $\|M_R^\rho\|_{L^p\mathscr{B}_\theta},
  \|M_{D_\rho}\|_{L^p\mathscr{B}_\theta} \le \|M_R\|_{L^p\mathscr{B}_\theta}$.
  The original $\sqrt p$ estimate and its even-moment exponential-series
  argument give uniform Gaussian operator path bounds. They are statements about
  $L^p$ equivalence classes on the extension preserving the given noise, not a
  simultaneous path event for all uncountably many cutoffs. This proves
  Proposition~\ref{P:source-canonical-convergence}.
\end{proof}

\subsection{Regularized random fields and the ordinary first response}

\begin{lemma}[Regularity for each fixed cutoff]\label{L:source-pointfield-regularity}
  For each fixed $\rho > 0$, the random field has a version continuous in time
  and smooth in space, with every spatial derivative jointly continuous on
  compact time-space rectangles.
\end{lemma}

\begin{proof}
  We first differentiate the point features and bound their increments. A
  dyadic construction gives jointly continuous versions for all spatial
  derivative fields. The feature-space fundamental theorem then identifies
  those fields as actual derivatives of one version.

  \medskip
  \noindent\textit{Step 1.~}We construct derivative fields and bound their
  increments.
  For each spatial multiindex $b$, differentiate \eqref{E:source-point-feature}
  in $x$ as a $\mathcal{G}$ vector. All derivatives exist because their spatial
  squared norms are the finite heat moments
  \[
    \int \xi^{2b}e^{-\varepsilon|\xi|^2}\mu(\ud\xi),
  \]
  and the next moments dominate difference quotients. Set $Z_b(t,x) =
  \mathcal{I}(\partial_x^bv_{\rho;t,x})$. At fixed $x$, use $1-e^{-\lambda h}
  \le \lambda h$ in the finite $\lambda$ integral; at fixed $t$, use
  $|e^{i\xi\cdot x}-e^{i\xi\cdot y}| \le |\xi||x-y|$ and the next heat moment.
  The squared triangle inequality gives
  \begin{equation}\label{E:source-derivative-increments}
    \E|Z_b(t,x)-Z_b(s,y)|^2
    \le C_{\rho,b}(|t-s|+|x-y|^2).
  \end{equation}
  For fixed cutoff, the temporal integral is over a bounded range of
  $\lambda$, so its extra factor $\lambda$ remains integrable. The heat factor
  controls the extra powers of $\xi$ needed for the spatial increment.
  Splitting a general increment into a time increment and a spatial increment
  gives the displayed sum of bounds.
  This computation uses~\eqref{E:source-joint-covariance}, including its spatial
  width, and applies to singular spectral measures without a density assumption.

  \medskip
  \noindent\textit{Step 2.~}We construct the jointly continuous versions.
  Here is a direct continuity construction. In $m = d+1$ parameter dimensions on
  a compact rectangle, even Gaussian $q$th increment moments have root bounded
  by $C\sqrt q$ times the parameter distance to the power $1/2$ at small
  distances. The dyadic grid has at most $C_m2^{nm}$ adjacent edges at level
  $n$. Their maximum has $L^q$ norm at most $C\sqrt q\,2^{-n(1/2-m/q)}$, with
  constants for this rectangle and fixed $\rho,b$. Choose $q$ large and $0 < h_0
  < 1/2-m/q$. The sum of these maxima weighted by $2^{nh_0}$ has finite $L^q$
  norm. Binary refinement, using a fixed number of adjacent edges per coordinate
  at each scale, gives a continuous extension from dyadic points. At each fixed
  parameter the extension agrees with the given Gaussian variable in $L^2$,
  by~\eqref{E:source-derivative-increments}; it is a modification. Use countably
  many nested compact rectangles and all multiindices. The versions agree on
  overlaps on countable dense sets, hence everywhere there by continuity, so
  they patch.

  At each level, the maximum of finitely many edge increments is bounded in
  $L^q$ by the $q$th root of the sum of their $q$th moments. This accounts for
  the factor $2^{nm/q}$. The strict choice of $h_0$ makes the weighted level
  series geometric and summable. The construction uses no independence of grid
  increments.

  \medskip
  \noindent\textit{Step 3.~}We identify the continuous fields as spatial
  derivatives.
  They are derivatives. The $\mathcal{G}$-valued fundamental theorem gives,
  along a coordinate segment,
  \[
    \partial_x^bv(t,x+he_i)-\partial_x^bv(t,x)
    =\int_0^h\partial_x^{b+e_i}v(t,x+ue_i)\ud u.
  \]
  Apply bounded linear $\mathcal{I}$ and $L^2$ integration. The continuous
  versions give the same ordinary integral by their Riemann sums, which converge
  both pathwise and in $L^2$. Enforce these identities first on countably many
  rational $t,x,h$ and multiindices, then use joint continuity to extend them to
  every segment in the compact rectangles. The ordinary fundamental theorem
  applies after the identities have been extended to every segment by
  continuity. The integral of the continuous field $Z_{b+e_i}$ is
  differentiable in the segment endpoint, with that field as its derivative.
  It therefore
  yields $\partial_iZ_b = Z_{b+e_i}$. Thus, all spatial derivatives exist and
  are jointly continuous with time. The versions are for this fixed $\rho$, not
  a common path event over uncountably many source parameters. This proves
  Lemma~\ref{L:source-pointfield-regularity}.
\end{proof}

\begin{proposition}[Ordinary source response]\label{P:source-ordinary-response}
  For each fixed $\rho > 0$, on its regularity event, put $b_\rho(t,x) =
  \zeta_R(x)Y_\rho(t,x)$. The statewise integral
  \begin{equation}\label{E:source-ordinary-response}
    L_\rho(t)z=\int_0^tS(-u)B(b_\rho(u))S(u)z\ud u
  \end{equation}
  defines an operator-norm Lipschitz path on $E$, and $L_\rho = M_R^\rho$ for
  all $t \in [0,T]$ on one probability-one set. In particular, it has the
  strongly measurable path version supplied by the Gaussian construction.
\end{proposition}

\begin{proof}
  We first construct the ordinary integral path using the samplewise smooth
  coefficient. We then identify its scalar rows with the transferred noise
  evaluations and use countability and continuity to obtain equality of paths.

  \medskip
  \noindent\textit{Step 1.~}We construct the ordinary response on its
  regularity event.
  The coefficient is compact smooth in space and continuous in time in every
  spatial derivative norm. Multiplication estimates and strong continuity of the
  full free group show that the statewise integral exists on every $E_r$. Its
  operator increments $E_r \to E_{r+1}$ are bounded by a finite sample constant
  times $|t-s|$, exactly by the statewise gain argument of
  Lemma~\ref{L:driver-gain-localization}. This proves Lipschitz continuity in
  $\mathcal{B}(E)$, without requiring norm continuity or an operator-norm
  derivative of $S(t)$.

  \medskip
  \noindent\textit{Step 2.~}We identify its rows by physical duality and test
  approximation.
  At each fixed $u$, the Sobolev duality calculation uses only spatial
  smoothness of $b_\rho(u,\cdot)$. From $S(-u)(0,f)$ and the two Bessel weights,
  for real Schwartz phases,
  \[
    \langle S(-u)B(b_\rho(u))S(u)z,w\rangle_E
    =\int b_\rho(u,x)[S(u)z]_1(x)G_w(u,x)\ud x.
  \]
  Thus, no prior hypothesis of smoothness in time is invoked for this
  instantaneous identity. For complex pairs conjugate the second factor.
  Integrating gives the ordinary source-field integral against the specified
  wave test for the original covariance $h_{R,t;z,w}$. For smooth tests, the
  point-feature integral has already been identified with $Y_\rho(F)$. The
  continuous random field version has the same integral: compact Riemann sums
  converge pathwise and in $L^2$ to the ordinary and feature integrals,
  respectively. Approximate the indicator wave test in the covariance completion
  with common compact spatial support and $L^2$-time spatial supremum error
  tending to zero. Its $L^1$ error also tends to zero. The point-feature norms
  are bounded on that support, so their integrals converge in $\mathcal{G}$; the
  original variance domination gives the same $L^2$ limit on the test side.
  On a finite time interval, the $L^1$ time error is bounded by the square
  root of the interval length times the $L^2$ time error. With common compact
  spatial support, the spatial integral has the same bound up to its fixed
  volume. This makes the feature-space integral converge to the same test
  limit as the variance domination gives.
  Consequently,
  \[
    \langle L_\rho(t)z,w\rangle_E
    =Y_\rho(h_{R,t;z,w})
    =\langle M_R^\rho(t)z,w\rangle_E
    \quad\text{in }L^2.
  \]

  \medskip
  \noindent\textit{Step 3.~}We obtain equality at every time on one event.
  Compare a countable dense Schwartz phase collection and rational times.
  Boundedness and the path continuity of both sides extend this equality to all
  states and all times on one probability-one set. The equality is on the
  coupled space, not only in law. The set is for fixed $\rho$; the $L^p$ limit
  in~\eqref{E:source-driver-limit} needs no simultaneous version for every
  source parameter. This proves Proposition~\ref{P:source-ordinary-response}.
\end{proof}
