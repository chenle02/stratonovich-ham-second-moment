\section{Physical spatial pairings and the full occupation class}
\label{S:spatial-occupation-foundation}

Retain a nonzero original covariance from Hypothesis~\ref{H:homog} with its
amplitude. Set $a = \alpha_0$, $k = 3-\alpha > 1$, $q = 4-\alpha = k+1$ and $D =
q-2a > 0$. The scalar $D$ is fixed. All profiles are real. The original positive
temporal exponent and threshold imply $\alpha < 2$; this is the strict
inequality used in the scalar powers and concavity. The finite variational value
over nonnegative controls $S_1^{\mathrm{pos}}$ is that of
Proposition~\ref{P:variational-positive-scores}.

\LeanMarker{L:spatial-physical-pair}
\begin{lemma}[The physical two-point integral]\label{L:spatial-physical-pair}
  The positive measure $\nu(\ud x,\ud y) = \ud x\,\Gamma(\ud y)$ is
  sigma-finite. For nonnegative functions define
  \begin{equation}\label{E:spatial-physical-pair}
    B_\Gamma(f,h)=\int f(x)h(x-y)\nu(\ud x,\ud y)
  \end{equation}
  initially as an extended nonnegative integral, using finite Borel
  representatives of Lebesgue classes. Its value is representative independent.
  Lebesgue-a.e. convergence transfers to $\nu$-a.e. convergence at both
  positions $x$ and $x-y$.
\end{lemma}

\begin{proof}
  Local finiteness of $\Gamma$ gives sigma-finiteness on exhausting balls, and
  Lebesgue measure is sigma-finite. If $N$ is Borel and Lebesgue null, Tonelli
  applied first in $x$ gives $\nu\{x\in N\} = 0$ and $\nu\{x-y\in N\} = 0$,
  because every fixed translate of $N$ is null. An arbitrary Lebesgue null set
  has a Borel null superset. Thus, arbitrary versions are measurable in the
  $\nu$ completion with the same value, and the same observation applies to the
  exceptional sets of a.e. convergence. Signed pairings will be used only after
  absolute convergence. This proves Lemma~\ref{L:spatial-physical-pair}.
\end{proof}

Put $A_{\mathrm{sp}}(g) = B_\Gamma(g^2,g^2)$ and $b(g) = \|\nabla g\|_2^2$ on
$H^1$; initially $A_{\mathrm{sp}}$ is used only for smooth compact profiles
before the extension below. For smooth compact $z$, write $U_z = \Gamma*z$ in
its usual pointwise sense.

\LeanMarker{L:spatial-static-controls}
\begin{lemma}[Smooth temporal controls]\label{L:spatial-static-controls}
  For every nonnegative nonzero $z \in C_c^\infty(\R^d)$, let
  \[
    \lambda_z(c)^2=
    \sup_{\substack{g\in H^1(\R^d;\R)\\\|g\|_2=1}}
    \left(c\int U_zg^2-b(g)\right),\qquad c>0.
  \]
  This is positive and finite, and $\lambda_z$ is Lipschitz on every compact
  positive $c$ interval. With the exact $M_a$
  in~\eqref{E:temporal-equilibrium-mass},
  \begin{equation}\label{E:spatial-static-score}
    S_1^{\mathrm{pos}}\ge
    \lambda_z(1)-\frac{M_a}{2}B_\Gamma(z,z).
  \end{equation}
\end{lemma}

\begin{proof}
  Choose nonnegative nonzero $f \in C_c^\infty(0,1)$ and $s_0 = 1/2$. Let
  $k_f(s_0) = \int_0^1|s_0-t|^{-a}f(t)\ud t > 0$. The smooth test $\Psi(t,x) =
  cf(t)z(x)/k_f(s_0)$ has mean $cU_z$ at $s_0$.
  Proposition~\ref{P:variational-unmasked-rate} gives boundedness of this
  potential, the displayed real Rayleigh formula, and a positive rate for every
  $c > 0$. Comparing identical unit tests makes $\lambda_z(c)^2$ Lipschitz with
  constant $\|U_z\|_\infty$. On a positive compact interval its rate has a
  positive floor by monotonicity, giving the square-root Lipschitz bound.

  Use the nonincreasing smooth cutoff $\chi$ constructed in the proof of
  Proposition~\ref{P:variational-remove-mask}. For integers $n \ge 4$, set on
  $(0,1)$
  \[
    f_n(t)=f_a(t)\chi(1/(nt))\chi(1/(n(1-t))),\qquad
    k_n(s)=\int_0^1|s-t|^{-a}f_n(t)\ud t.
  \]
  Extend $f_n$ by zero. It is smooth compact, vanishes near both endpoints, is
  nonzero, and increases to $f_a$. Thus, $\Psi_n = f_n\otimes z$ is a positive
  smooth control. Monotone convergence
  and~\eqref{E:temporal-equilibrium-potential} give $k_n(s)\uparrow1$ for every
  $s \in [0,1]$. Since the kernel is at least one on the unit square off its
  null diagonal, $0 < c_0: = \int_0^1f_4 \le k_n(s) \le 1$. The mean is
  $k_n(s)U_z$; hence its rate is $\lambda_z(k_n(s))$, converging to
  $\lambda_z(1)$ and bounded by that finite number. Dominated convergence gives
  $I_{\Psi_n} \to \lambda_z(1)$.

  The Gaussian cost is exactly the nonnegative Tonelli pairing
  \[
    v_{\Psi_n}=B_\Gamma(z,z)\int_0^1f_n(s)k_n(s)\ud s.
  \]
  Its time integrand increases to $f_a$, so the cost tends to
  $M_aB_\Gamma(z,z)$. The spatial factor is finite and positive: its smooth
  compact autocorrelation is positive near zero, and homogeneity scales any
  origin-centered ball of positive $\Gamma$ mass to every other positive radius.
  Such a ball exists by exhaustion. Take the limit of the individual
  inequalities $S_1^{\mathrm{pos}} \ge I_{\Psi_n}-v_{\Psi_n}/2$. The variational
  expression differences need not increase. No control depends on original large
  time and no rough stochastic evolution has been used. This proves
  Lemma~\ref{L:spatial-static-controls}.
\end{proof}

\LeanMarker{L:spatial-fourier-bound}
\begin{lemma}[Deterministic Fourier bound for the spatial energy]
  \label{L:spatial-fourier-bound}
  Under Hypothesis~\ref{H:homog}, allowing $\Gamma=0$, let
  $g\in C_c^\infty(\R^d;\R)$ satisfy $\|g\|_2=1$ and put
  $b=\|\nabla g\|_2^2$. Then $b>0$ and
  \begin{equation}\label{E:spatial-fourier-bound}
    0\le A_{\mathrm{sp}}(g)
    \le 2^{1+\alpha}R_\alpha b^{\alpha/2},
  \end{equation}
  where $R_\alpha$ is defined in~\eqref{E:intro-spectral-mass}.
\end{lemma}

\begin{proof}
  If $b=0$, smoothness makes $\nabla g=0$ everywhere. Thus, $g$ is constant,
  and compact support forces $g=0$, contrary to $\|g\|_2=1$. Put $Q=g^2$ and
  use the unnormalized Fourier transform
  \[
    \mathcal{F}_0Q(\xi)=\int_{\R^d}e^{-i\xi\cdot x}Q(x)\ud x.
  \]
  In the unitary convention of Section~\ref{S:masked-covariance}, this is
  $c_d^{-1}\widehat Q$, with $c_d=(2\pi)^{-d/2}$.
  The autocorrelation
  \[
    c_Q(y)=\int_{\R^d}Q(x)Q(x-y)\ud x
  \]
  is smooth and compactly supported. Apply the distributional spectral
  identity~\eqref{E:masked-spectral-convention} to this Schwartz function.
  Its inner Fourier integral is $|\mathcal{F}_0Q(\xi)|^2$, since $Q$ is real.
  Hence
  \[
    A_{\mathrm{sp}}(g)=\int c_Q(y)\Gamma(\ud y)
    =\int|\mathcal{F}_0Q(\xi)|^2\mu(\ud\xi).
  \]
  Both integrals are finite: $c_Q$ is compactly supported and $\Gamma$ is
  Radon, while $|\mathcal{F}_0Q|^2$ is Schwartz and $\mu$ is tempered.
  No density of either measure is required.

  Since $\|Q\|_1=1$, we have $|\mathcal{F}_0Q|\le1$. Integration by parts
  and the Cauchy--Schwarz inequality give
  \[
    |\xi|\,|\mathcal{F}_0Q(\xi)|
    \le\|\nabla Q\|_1
    \le2\|g\|_2\|\nabla g\|_2=2\sqrt b.
  \]
  Splitting into $|\xi|^2\le4b$ and $|\xi|^2>4b$ therefore yields
  \[
    |\mathcal{F}_0Q(\xi)|^2\le\frac{8b}{4b+|\xi|^2}.
  \]
  The deterministic resolvent calculation in the proof of
  Lemma~\ref{L:damped-homogeneous-norms}, together
  with~\eqref{E:intro-spectral-mass}, gives, for $q>0$,
  \[
    \mathcal{R}_\Gamma(q):=\int\frac{\mu(\ud\xi)}{q^2+|\xi|^2}
    =q^{\alpha-2}R_\alpha<\infty.
  \]
  Consequently,
  \[
    A_{\mathrm{sp}}(g)\le8b\mathcal{R}_\Gamma(2\sqrt b)
    =2^{1+\alpha}R_\alpha b^{\alpha/2}.
  \]
  Nonnegativity follows from the spectral identity. If $\Gamma=0$, then
  $\mu=0$ and both sides of~\eqref{E:spatial-fourier-bound} vanish. This proves
  Lemma~\ref{L:spatial-fourier-bound}.
\end{proof}

Lemma~\ref{L:spatial-static-controls} gives a lower bound for each nonzero,
nonnegative smooth spatial control of compact support. We now scale and
dilate a single profile and optimize. The resulting bound depends on the
profile only through the ratio of $A_{\mathrm{sp}}$ to a power of $b$.

\LeanMarker{P:spatial-smooth-ratio}
\begin{proposition}[The finite smooth spatial ratio]
  \label{P:spatial-smooth-ratio}
  For real unit $g \in C_c^\infty(\R^d)$ put $R(g) =
  A_{\mathrm{sp}}(g)/b(g)^{\alpha/2}$. Its supremum $R_0$ over these profiles is
  a positive finite real number, and
  \begin{equation}\label{E:spatial-smooth-score}
    S_1^{\mathrm{pos}}\ge
    k\left[
    \frac{R(g)\alpha^{\alpha/2}}{8M_aq^{q/2}}
    \right]^{1/k}.
  \end{equation}
\end{proposition}

\begin{proof}
  The gradient energy $b(g)$ is positive: a nonzero $L^2$ function with zero
  weak gradient would have Fourier transform supported only at zero, and hence
  would vanish. Its square is nonnegative nonzero smooth compact, so the
  autocorrelation argument in the proof of
  Lemma~\ref{L:spatial-static-controls} gives
  $0 < A:=A_{\mathrm{sp}}(g)<\infty$. Lemma~\ref{L:spatial-fourier-bound} gives
  $R(g)\le2^{1+\alpha}R_\alpha$ for every such profile. A normalized compact
  bump supplies a positive ratio. Thus, $0<R_0<\infty$, independently of the
  finiteness of $S_1^{\mathrm{pos}}$. It remains to prove the score inequality.
  With $U_\ell g(x)=\ell^{d/2}g(\ell x)$, the physical changes of variables give
  \begin{equation}\label{E:spatial-dilation}
    A_{\mathrm{sp}}(U_\ell g)=\ell^\alpha A,\qquad
    b(U_\ell g)=\ell^2b(g).
  \end{equation}
  The factors in the first identity are $\ell^{2d}$ from the two squares,
  $\ell^{-d}$ from $\ud x$, and $\ell^{\alpha-d}$ from the Radon pushforward.

  For $\theta,\ell > 0$ choose the smooth control $z = \theta(U_\ell g)^2$. The
  Rayleigh test $U_\ell g$ gives $\lambda_z(1)^2 \ge \theta\ell^\alpha
  A-\ell^2b$, while $B_\Gamma(z,z) = \theta^2\ell^\alpha A$. For any $L > 0$,
  put $u = \alpha/q > 0$ and choose
  \[
    \ell^2=uL^2/b,\qquad
    \theta=(1+u)L^2/(\ell^\alpha A).
  \]
  The radicand is $L^2$. Lemma~\ref{L:spatial-static-controls} consequently
  gives
  \[
    S_1^{\mathrm{pos}}\ge L-CL^q,\qquad
    C=\frac{M_a(1+u)^2}{2R(g)u^{\alpha/2}}>0.
  \]
  Its maximum occurs at $L^k = 1/(qC)$ and is $(k/q)(qC)^{-1/k}$. Since $1+u =
  4/q$, $1/(q^qC) = R(g)\alpha^{\alpha/2}/(8M_aq^{q/2})$,
  proving~\eqref{E:spatial-smooth-score}. This argument retains the finite
  real value $S_1^{\mathrm{pos}}$ fixed at the start of the section; only the
  proof of $R_0<\infty$ is independent of that input. This proves
  Proposition~\ref{P:spatial-smooth-ratio}.
\end{proof}

So far $A_{\mathrm{sp}}$ has been used only for smooth compact profiles.
Proposition~\ref{P:occupation-physical-class} needs it and the square pairings
for real $H^1$ profiles, so we extend them with the same constant $R_0$.

\LeanMarker{P:spatial-four-linear}
\begin{proposition}[The physical four-linear extension]
  \label{P:spatial-four-linear}
  For real $f \in H^1(\R^d)$ define $N(f) = \|f\|_2^{1-\alpha/4}\|\nabla
  f\|_2^{\alpha/4}$. Every physical square pairing is finite and
  \begin{align}
    A_{\mathrm{sp}}(f)&\le R_0N(f)^4,\label{E:spatial-homogeneous-bound}\\
    0\le B_\Gamma(f^2,h^2)
    &\le\sqrt{A_{\mathrm{sp}}(f)A_{\mathrm{sp}}(h)}.
    \label{E:spatial-square-CS}
  \end{align}
  The signed integral $C_4(f_1,f_2,f_3,f_4) = B_\Gamma(f_1f_2,f_3f_4)$ is
  absolutely convergent and jointly continuous on $(H^1)^4$, with
  \begin{equation}\label{E:spatial-four-linear}
    \int|f_1(x)f_2(x)f_3(x-y)f_4(x-y)|\ud\nu
    \le R_0\prod_{j=1}^4N(f_j).
  \end{equation}
  The ratio supremum $R_{\mathrm{sp}}$ over all real unit $H^1$ profiles equals
  $R_0$. Every nonzero $H^1$ profile has $A_{\mathrm{sp}}(f) > 0$.
\end{proposition}

\begin{proof}
  For smooth compact $f$, normalization in $R_0$ proves
  \eqref{E:spatial-homogeneous-bound}, including $f = 0$. All exponents in $N$
  are positive and $N(f) \le \|f\|_{H^1}$. For smooth compact signed $Q,P$, the
  absolute correlation is compactly supported and bounded by $\|Q\|_2\|P\|_2$.
  Local finiteness of $\Gamma$ therefore makes its physical integral finite.
  Smooth signed covariance is positive: with the ordinary transform
  $\widehat{Q}_{\mathrm{ord}}(\xi) = \int e^{i\xi\cdot x}Q(x)\ud x$, the model
  convention gives
  \[
    B_\Gamma(Q,Q)=\int|\widehat{Q}_{\mathrm{ord}}(\xi)|^2\mu(\ud\xi)
    \ge0,\qquad Q\in C_c^\infty(\R^d;\R).
  \]
  Pair $\Gamma$ with the smooth compact correlation to obtain this identity;
  rapid Fourier decay and tempered $\mu$ give finiteness. Evenness of $\Gamma$
  and $(x,y)\mapsto(x-y,-y)$ give symmetry. The quadratic polynomial for $Q+tP$
  gives covariance Cauchy--Schwarz. In particular, for smooth real $f,h$,
  \[
    0\le B_\Gamma(f^2,h^2)
    \le R_0N(f)^2N(h)^2.
  \]
  The squares of signed smooth profiles are already smooth and nonnegative.

  Smooth compact functions are dense in $H^1$. Cutoffs equal to one on the
  radius-$R$ ball, vanishing beyond $2R$ and with gradient at most $C/R$,
  converge by the function and gradient tails and the bound $C\|f\|_2/R$ for the
  extra gradient term. Mollification then converges in $H^1$ by $L^2$
  translation continuity of the function and its weak gradients. The space is
  separable through its closed isometric image $f\mapsto(f,\nabla f)$ in
  $L^2(\R^d)^{d+1}$.

  For arbitrary $f,h \in H^1$ take smooth approximants converging in $H^1$ and
  along a common subsequence Lebesgue-a.e. Lemma~\ref{L:spatial-physical-pair}
  transfers this convergence at both positions to $\nu$-a.e. convergence. Fatou
  and continuity of $N$ give $B_\Gamma(f^2,h^2) \le R_0N(f)^2N(h)^2 < \infty$,
  including the diagonal bound. At this stage Fatou supplies an inequality, not
  continuity.

  Apply ordinary Cauchy--Schwarz under the positive measure $\nu$ to
  $f_1(x)f_3(x-y)$ and $f_2(x)f_4(x-y)$. The two squared integrals are finite by
  the last bound; their square root is at most $R_0\prod_jN(f_j)$. This
  proves~\eqref{E:spatial-four-linear}, absolute convergence and multilinearity.
  Telescoping a difference one argument at a time and using $N(f) \le
  \|f\|_{H^1}$ proves joint continuity. Thus, $A_{\mathrm{sp}}$ and the square
  pairings are continuous. PSD for smooth compact profiles passes to finite
  linear combinations of $H^1$ squares by this continuity, and its two-profile
  Cauchy--Schwarz gives \eqref{E:spatial-square-CS}. This completes the physical
  pairing argument beyond Fatou.

  Every real unit $H^1$ profile has positive gradient energy by the Fourier
  argument above. The homogeneous bound gives $R_{\mathrm{sp}} \le R_0$, and
  smooth inclusion gives the reverse. Normalized smooth approximations preserve
  the ratio by continuity. Finally, for nonzero $f$, choose a bounded set $E$ of
  finite positive measure where $|f| \ge c > 0$. The correlation of $1_E$ is
  continuous and positive near zero by $L^2$ translation continuity. An
  origin-centered ball there has positive $\Gamma$ mass, so $A_{\mathrm{sp}}(f)
  \ge c^4B_\Gamma(1_E,1_E) > 0$. This uses a nonnegative integral, not an $H^1$
  claim for $1_E$. Finiteness was already proved. This proves
  Proposition~\ref{P:spatial-four-linear}.
\end{proof}

By Proposition~\ref{P:spatial-four-linear}, the ratio supremum is the same
over smooth and over $H^1$ profiles. We now express it through an energy and
restate the score bound in those terms.

\LeanMarker{P:spatial-energy-map}
\begin{proposition}[The spatial energy and static lower variational expression]
  \label{P:spatial-energy-map}
  Over all real unit $H^1$ profiles define the energy
  \[
    \mathcal{E}_{\mathrm{sp}}
    =\sup_{\|g\|_2=1}
    \left(A_{\mathrm{sp}}(g)-\frac{b(g)}2\right).
  \]
  Then
  \begin{equation}\label{E:spatial-energy-map}
    \mathcal{E}_{\mathrm{sp}}
    =\frac{2-\alpha}{2}\alpha^{\alpha/(2-\alpha)}
    R_{\mathrm{sp}}^{2/(2-\alpha)}
    \in(0,\infty).
  \end{equation}
  In particular,
  \begin{equation}\label{E:spatial-static-energy-score}
    S_1^{\mathrm{pos}}\ge M_a^{-1/k}F_0(\mathcal{E}_{\mathrm{sp}}),
    \qquad
    F_0(z)=k\left[
    \frac{(2z/(2-\alpha))^{(2-\alpha)/2}}{8q^{q/2}}
    \right]^{1/k},\quad z>0.
  \end{equation}
  The subscript in $F_0$ labels this scalar map; the covariance retains its
  original temporal exponent $a$.
\end{proposition}

\begin{proof}
  Physical change of variables, or smooth approximation and
  Proposition~\ref{P:spatial-four-linear}, extends \eqref{E:spatial-dilation} to
  real $H^1$ profiles. For fixed unit $g$, with $A,b > 0$, the maximum of
  $\ell^\alpha A-\ell^2b/2$ occurs at $\ell^{2-\alpha} = \alpha A/b$ and equals
  \[
    \frac{2-\alpha}{2}\alpha^{\alpha/(2-\alpha)}
    R(g)^{2/(2-\alpha)}.
  \]
  Dilation preserves the whole unit class and includes $\ell = 1$. Taking both
  numerical suprema, then profiles with ratio tending to $R_{\mathrm{sp}}$,
  proves~\eqref{E:spatial-energy-map}. No optimizer is assumed. Smooth ratios
  also approach $R_{\mathrm{sp}}$ by Proposition~\ref{P:spatial-four-linear}.
  Pass them through~\eqref{E:spatial-smooth-score} and substitute
  \[
    R_{\mathrm{sp}}\alpha^{\alpha/2}
    =\left(\frac{2\mathcal{E}_{\mathrm{sp}}}{2-\alpha}
    \right)^{(2-\alpha)/2}.
  \]
  This gives~\eqref{E:spatial-static-energy-score}. This proves
  Proposition~\ref{P:spatial-energy-map}.
\end{proof}

\subsection{The kernel row and full occupation energy}

Define, for $u \ne 0$,
\begin{equation}\label{E:occupation-damped-kernel}
  K_\star(u)=\frac1{\Gamma_E(a)}
  \int_0^\infty
  e^{-(\lambda+\lambda^2/(2D))|u|}
  \lambda^{a-1}\ud\lambda,
\end{equation}
and set $K_\star(0) = 0$ for time integration. The spatial estimates above
concern individual $H^1$ profiles. To apply them to time-dependent families,
we first compute the exact row bound for this time kernel.

\LeanMarker{L:occupation-kernel-row}
\begin{lemma}[Exact row maximum]\label{L:occupation-kernel-row}
  The kernel is even, positive off zero, locally integrable, decreasing on
  $(0,\infty)$, and $K_\star(u) \le |u|^{-a}$ off zero. Its exact row maximum on
  $[0,1]$ is
  \begin{equation}\label{E:occupation-kernel-row}
    \max_{s\in[0,1]}\int_0^1K_\star(s-t)\ud t
    =R_\star:=2\int_0^{1/2}K_\star(u)\ud u\in(0,\infty).
  \end{equation}
\end{lemma}

\begin{proof}
  Dropping the positive quadratic damping bounds the mixture by the Gamma
  integral $|u|^{-a}$. Positivity and monotonicity follow directly from the
  integrand, and $a < 1$ gives local integrability. Put $\Phi(t) =
  \int_0^tK_\star(u)\ud u$. It is concave because its integrand is decreasing on
  positive arguments. The row is $\Phi(s)+\Phi(1-s) \le 2\Phi(1/2)$, with
  equality at $s = 1/2$. The diagonal value changes no time integral. This
  proves Lemma~\ref{L:occupation-kernel-row}.
\end{proof}

Let $\mathcal{O}$ be the full class of strongly $H^1(\R^d;\R)$-measurable fields
$s\mapsto g_s$ on $(0,1)$ with $\|g_s\|_2 = 1$ a.e. and $T(g) =
\int_0^1b(g_s)\ud s < \infty$. No time smoothness, common compact support or
optimizer condition is imposed.

\LeanMarker{P:occupation-physical-class}
\begin{proposition}[The full physical occupation class]
  \label{P:occupation-physical-class}
  The pairings $(s,t)\mapsto B_\Gamma(g_s^2,g_t^2)$ are measurable,
  representative independent and equal to their physical integrals. With
  \[
    J(g)=\int_0^1\int_0^1
    K_\star(s-t)B_\Gamma(g_s^2,g_t^2)\ud s\ud t,
  \]
  every $g \in \mathcal{O}$ satisfies
  \begin{equation}\label{E:occupation-physical-bound}
    0\le J(g)\le R_\star R_{\mathrm{sp}}T(g)^{\alpha/2}<\infty.
  \end{equation}
  The energy over exactly this full class satisfies
  \begin{equation}\label{E:occupation-own-energy}
    0<\mathcal{E}_\star^{\mathcal{O}}
    :=\sup_{g\in\mathcal{O}}\big(J(g)-T(g)/2\big)
    \le R_\star^{2/(2-\alpha)}\mathcal{E}_{\mathrm{sp}}<\infty.
  \end{equation}
\end{proposition}

\begin{proof}
  The map $s\mapsto b(g_s)$ is measurable by continuity of the squared gradient
  norm. Also $\int_0^1\|g_s\|_{H^1}^2\ud s = 1+T(g)$. Use a separably valued
  strong representative and replace exceptional null times by one fixed real
  unit compact bump. Joint measurable spatial representatives exist: approximate
  this Bochner $L^2$ field by simple $H^1$-valued functions with summable
  $L^2(0,1;H^1)$ errors, and choose measurable spatial versions for their
  countably many values. A subsequence has a measurable spacetime $L^2$ limit,
  and for a.e. $s$ its spatial $L^2$ class is $g_s$.
  Lemma~\ref{L:spatial-physical-pair} makes every pairing independent of the
  spatial versions. The continuous pairing map on the separable space $H^1\times
  H^1$ from Proposition~\ref{P:spatial-four-linear} consequently gives a
  measurable function of $(s,t)$ agreeing a.e. with the nonnegative physical
  integral.

  Set $F(s) = b(g_s)^{\alpha/4}$. Equations~\eqref{E:spatial-homogeneous-bound}
  and~\eqref{E:spatial-square-CS} give $B_\Gamma(g_s^2,g_t^2) \le
  R_{\mathrm{sp}}F(s)F(t)$. Since $0 < \alpha/2 < 1$, concavity on the unit time
  interval yields $\int_0^1F^2 = \int_0^1b(g_s)^{\alpha/2}\ud s \le
  T(g)^{\alpha/2}$. Now use $2F(s)F(t) \le F(s)^2+F(t)^2$ and the exact row
  bound \eqref{E:occupation-kernel-row}, whose column bound is the same because
  $K_\star$ is even, to obtain \eqref{E:occupation-physical-bound}. Positive
  Tonelli is valid before finiteness is known. Null-time modifications affect
  only a product-null set, so they cannot change $J$.

  The scalar maximum of $R_\star R_{\mathrm{sp}}T^{\alpha/2}-T/2$ over $T \ge 0$
  is $R_\star^{2/(2-\alpha)}\mathcal{E}_{\mathrm{sp}}$, by the same
  differentiation as~\eqref{E:spatial-energy-map}. This bounds the occupation
  supremum from above. For positivity take a constant-in-time unit compact
  profile and its spatial dilation $U_\ell$. Its $J$ is $J_0\ell^\alpha
  A_{\mathrm{sp}}(g)$, where $J_0 = \int_0^1\int_0^1K_\star(s-t)\ud s\ud t \in
  (0,\infty)$, while $T = \ell^2b(g)$. For sufficiently small $\ell > 0$,
  $\alpha < 2$ makes $J-T/2 > 0$. Thus, the nonempty supremum is a positive
  finite real number. This is a comparison bound; no attaining profile or
  equality in it has been asserted. No spatial point trace or rough stochastic
  evolution was used. This proves Proposition~\ref{P:occupation-physical-class}.
\end{proof}

For zero covariance all spatial pairings vanish and set $R_{\mathrm{sp}} = 0$.
The spatial and occupation energies are zero suprema of $-b/2$ and $-T/2$: small
spatial dilations of one fixed unit compact profile make these energies tend to
zero, with the profile constant in time for the latter. No zero-gradient unit
state is used. The temporal identities remain valid, and the moment model is the
free case already treated.

\section{Physical spatial pairings and the full occupation class}
\label{S:spatial-occupation-foundation}

Retain a nonzero original covariance from Hypothesis~\ref{H:homog} with its
amplitude. Set $a = \alpha_0$, $k = 3-\alpha > 1$, $q = 4-\alpha = k+1$ and $D =
q-2a > 0$. The scalar $D$ is fixed. All profiles are real. The original positive
temporal exponent and threshold imply $\alpha < 2$; this is the strict
inequality used in the scalar powers and concavity. The finite variational value
over nonnegative controls $S_1^{\mathrm{pos}}$ is that of
Proposition~\ref{P:variational-positive-scores}.

\LeanMarker{L:spatial-physical-pair}
\begin{lemma}[The physical two-point integral]\label{L:spatial-physical-pair}
  The positive measure $\nu(\ud x,\ud y) = \ud x\,\Gamma(\ud y)$ is
  sigma-finite. For nonnegative functions define
  \begin{equation}\label{E:spatial-physical-pair}
    B_\Gamma(f,h)=\int f(x)h(x-y)\nu(\ud x,\ud y)
  \end{equation}
  initially as an extended nonnegative integral, using finite Borel
  representatives of Lebesgue classes. Its value is representative independent.
  Lebesgue-a.e. convergence transfers to $\nu$-a.e. convergence at both
  positions $x$ and $x-y$.
\end{lemma}

\begin{proof}
  Local finiteness of $\Gamma$ gives sigma-finiteness on exhausting balls, and
  Lebesgue measure is sigma-finite. If $N$ is Borel and Lebesgue null, Tonelli
  applied first in $x$ gives $\nu\{x\in N\} = 0$ and $\nu\{x-y\in N\} = 0$,
  because every fixed translate of $N$ is null. An arbitrary Lebesgue null set
  has a Borel null superset. Thus, arbitrary versions are measurable in the
  $\nu$ completion with the same value, and the same observation applies to the
  exceptional sets of a.e. convergence. Signed pairings will be used only after
  absolute convergence. This proves Lemma~\ref{L:spatial-physical-pair}.
\end{proof}

Put $A_{\mathrm{sp}}(g) = B_\Gamma(g^2,g^2)$ and $b(g) = \|\nabla g\|_2^2$ on
$H^1$; initially $A_{\mathrm{sp}}$ is used only for smooth compact profiles
before the extension below. For smooth compact $z$, write $U_z = \Gamma*z$ in
its usual pointwise sense.

\LeanMarker{L:spatial-static-controls}
\begin{lemma}[Smooth temporal controls]\label{L:spatial-static-controls}
  For every nonnegative nonzero $z \in C_c^\infty(\R^d)$, let
  \[
    \lambda_z(c)^2=
    \sup_{\substack{g\in H^1(\R^d;\R)\\\|g\|_2=1}}
    \left(c\int U_zg^2-b(g)\right),\qquad c>0.
  \]
  This is positive and finite, and $\lambda_z$ is Lipschitz on every compact
  positive $c$ interval. With the exact $M_a$
  in~\eqref{E:temporal-equilibrium-mass},
  \begin{equation}\label{E:spatial-static-score}
    S_1^{\mathrm{pos}}\ge
    \lambda_z(1)-\frac{M_a}{2}B_\Gamma(z,z).
  \end{equation}
\end{lemma}

\begin{proof}
  We first obtain the static Rayleigh rate from an admissible smooth control.
  We then approximate the exact temporal equilibrium by increasing compact
  smooth controls and pass their rewards and original covariance costs to the
  limit separately.


  \medskip
  \noindent\textit{Step 1.~}We obtain the static rate and its local Lipschitz
  bound.
  Choose nonnegative nonzero $f \in C_c^\infty(0,1)$ and $s_0 = 1/2$. Let
  $k_f(s_0) = \int_0^1|s_0-t|^{-a}f(t)\ud t > 0$. The smooth test $\Psi(t,x) =
  cf(t)z(x)/k_f(s_0)$ has mean $cU_z$ at $s_0$.
  Proposition~\ref{P:variational-unmasked-rate} gives boundedness of this
  potential, the displayed real Rayleigh formula, and a positive rate for every
  $c > 0$. Comparing identical unit tests makes $\lambda_z(c)^2$ Lipschitz with
  constant $\|U_z\|_\infty$. On a positive compact interval its rate has a
  positive floor by monotonicity, giving the square-root Lipschitz bound.

  For two positive coefficients, the difference of the squared Rayleigh rates
  is bounded by the coefficient difference times $\|U_z\|_\infty$. To pass to
  their square roots, divide by the sum of the positive rates. On a compact
  positive coefficient interval, monotonicity bounds that denominator away
  from zero by the rate at its left endpoint.

  \medskip
  \noindent\textit{Step 2.~}We approximate the equilibrium density by
  admissible controls.
  Use the nonincreasing smooth cutoff $\chi$ constructed in the proof of
  Proposition~\ref{P:variational-remove-mask}. For integers $n \ge 4$, set on
  $(0,1)$
  \[
    f_n(t)=f_a(t)\chi(1/(nt))\chi(1/(n(1-t))),\qquad
    k_n(s)=\int_0^1|s-t|^{-a}f_n(t)\ud t.
  \]
  Extend $f_n$ by zero. It is smooth compact, vanishes near both endpoints, is
  nonzero, and increases to $f_a$. Thus, $\Psi_n = f_n\otimes z$ is a positive
  smooth control. Monotone convergence
  and~\eqref{E:temporal-equilibrium-potential} give $k_n(s)\uparrow1$ for every
  $s \in [0,1]$. Since the kernel is at least one on the unit square off its
  null diagonal, $0 < c_0: = \int_0^1f_4 \le k_n(s) \le 1$. The mean is
  $k_n(s)U_z$; hence its rate is $\lambda_z(k_n(s))$, converging to
  $\lambda_z(1)$ and bounded by that finite number. Dominated convergence gives
  $I_{\Psi_n} \to \lambda_z(1)$.

  As $n$ increases, both arguments of the nonincreasing cutoff decrease, so
  both cutoff factors increase. Their product increases to one at every
  interior time. The temporal potentials are therefore increasing and remain
  between $c_0$ and one, a compact positive interval on which the rate
  comparison just proved applies.

  \medskip
  \noindent\textit{Step 3.~}We pass the covariance cost and the score to the
  limit.
  The Gaussian cost is exactly the nonnegative Tonelli pairing
  \[
    v_{\Psi_n}=B_\Gamma(z,z)\int_0^1f_n(s)k_n(s)\ud s.
  \]
  Its time integrand increases to $f_a$, so the cost tends to
  $M_aB_\Gamma(z,z)$. The spatial factor is finite and positive: its smooth
  compact autocorrelation is positive near zero, and homogeneity scales any
  origin-centered ball of positive $\Gamma$ mass to every other positive radius.
  Such a ball exists by exhaustion. Take the limit of the individual
  inequalities $S_1^{\mathrm{pos}} \ge I_{\Psi_n}-v_{\Psi_n}/2$. The variational
  expression differences need not increase. No control depends on original large
  time and no rough stochastic evolution has been used. This proves
  Lemma~\ref{L:spatial-static-controls}.
\end{proof}

The next estimate also allows zero covariance and does not use the finiteness
of $S_1^{\mathrm{pos}}$.

\LeanMarker{L:spatial-fourier-bound}
\begin{lemma}[Deterministic Fourier bound for the spatial energy]
  \label{L:spatial-fourier-bound}
  Under Hypothesis~\ref{H:homog}, allowing $\Gamma=0$, let
  $g\in C_c^\infty(\R^d;\R)$ satisfy $\|g\|_2=1$ and put
  $b=\|\nabla g\|_2^2$. Then $b>0$ and
  \begin{equation}\label{E:spatial-fourier-bound}
    0\le A_{\mathrm{sp}}(g)
    \le 2^{1+\alpha}R_\alpha b^{\alpha/2},
  \end{equation}
  where $R_\alpha$ is defined in~\eqref{E:intro-spectral-mass}.
\end{lemma}

\begin{proof}
  We identify the spatial energy spectrally without assuming a measure
  density. A mass bound and a derivative bound on the Fourier transform give a
  resolvent majorant, whose homogeneous integral supplies the desired
  constant.


  \medskip
  \noindent\textit{Step 1.~}We identify the energy as a finite spectral
  integral.
  If $b=0$, smoothness makes $\nabla g=0$ everywhere. Thus, $g$ is constant,
  and compact support forces $g=0$, contrary to $\|g\|_2=1$. Put $Q=g^2$ and
  use the unnormalized Fourier transform
  \[
    \mathcal{F}_0Q(\xi)=\int_{\R^d}e^{-i\xi\cdot x}Q(x)\ud x.
  \]
  In the unitary convention of Section~\ref{S:masked-covariance}, this is
  $c_d^{-1}\widehat Q$, with $c_d=(2\pi)^{-d/2}$.
  The autocorrelation
  \[
    c_Q(y)=\int_{\R^d}Q(x)Q(x-y)\ud x
  \]
  is smooth and compactly supported. Apply the distributional spectral
  identity~\eqref{E:masked-spectral-convention} to this Schwartz function.
  Its inner Fourier integral is $|\mathcal{F}_0Q(\xi)|^2$, since $Q$ is real.
  Hence
  \[
    A_{\mathrm{sp}}(g)=\int c_Q(y)\Gamma(\ud y)
    =\int|\mathcal{F}_0Q(\xi)|^2\mu(\ud\xi).
  \]
  Both integrals are finite: $c_Q$ is compactly supported and $\Gamma$ is
  Radon, while $|\mathcal{F}_0Q|^2$ is Schwartz and $\mu$ is tempered.
  No density of either measure is required.


  \medskip
  \noindent\textit{Step 2.~}We bound the Fourier transform by a resolvent
  weight.
  Since $\|Q\|_1=1$, we have $|\mathcal{F}_0Q|\le1$. Integration by parts
  and the Cauchy--Schwarz inequality give
  \[
    |\xi|\,|\mathcal{F}_0Q(\xi)|
    \le\|\nabla Q\|_1
    \le2\|g\|_2\|\nabla g\|_2=2\sqrt b.
  \]
  Splitting into $|\xi|^2\le4b$ and $|\xi|^2>4b$ therefore yields
  \[
    |\mathcal{F}_0Q(\xi)|^2\le\frac{8b}{4b+|\xi|^2}.
  \]
  In the region $|\xi|^2\le4b$, the right side of the preceding display is at
  least one, so the mass bound applies. In the other region, the derivative
  bound gives $4b/|\xi|^2$, which is at most $8b/(4b+|\xi|^2)$ because
  $4b\le|\xi|^2$. Thus, the two estimates give one majorant over the whole
  spectrum.

  \medskip
  \noindent\textit{Step 3.~}We integrate the majorant and keep the original
  spectral normalization.
  The deterministic resolvent calculation in the proof of
  Lemma~\ref{L:damped-homogeneous-norms}, together
  with~\eqref{E:intro-spectral-mass}, gives, for $q>0$,
  \[
    \mathcal{R}_\Gamma(q):=\int\frac{\mu(\ud\xi)}{q^2+|\xi|^2}
    =q^{\alpha-2}R_\alpha<\infty.
  \]
  Consequently,
  \[
    A_{\mathrm{sp}}(g)\le8b\mathcal{R}_\Gamma(2\sqrt b)
    =2^{1+\alpha}R_\alpha b^{\alpha/2}.
  \]
  Nonnegativity follows from the spectral identity. If $\Gamma=0$, then
  $\mu=0$ and both sides of~\eqref{E:spatial-fourier-bound} vanish. This proves
  Lemma~\ref{L:spatial-fourier-bound}.
\end{proof}

Lemma~\ref{L:spatial-static-controls} gives a lower bound for each nonzero,
nonnegative smooth spatial control of compact support. We now scale and
dilate a single profile and optimize. The resulting bound depends on the
profile only through the ratio of $A_{\mathrm{sp}}$ to a power of $b$.

\LeanMarker{P:spatial-smooth-ratio}
\begin{proposition}[The finite smooth spatial ratio]
  \label{P:spatial-smooth-ratio}
  For real unit $g \in C_c^\infty(\R^d)$ put $R(g) =
  A_{\mathrm{sp}}(g)/b(g)^{\alpha/2}$. Its supremum $R_0$ over these profiles is
  a positive finite real number, and
  \begin{equation}\label{E:spatial-smooth-score}
    S_1^{\mathrm{pos}}\ge
    k\left[
    \frac{R(g)\alpha^{\alpha/2}}{8M_aq^{q/2}}
    \right]^{1/k}.
  \end{equation}
\end{proposition}

\begin{proof}
  We first bound the smooth spatial ratio. We then vary the amplitude of a
  squared dilated profile so that its Rayleigh test has a prescribed positive
  value, and optimize the resulting one-variable score.


  \medskip
  \noindent\textit{Step 1.~}We prove positivity and finiteness of the ratio
  supremum.
  The gradient energy $b(g)$ is positive: a nonzero $L^2$ function with zero
  weak gradient would have Fourier transform supported only at zero, and hence
  would vanish. Its square is nonnegative nonzero smooth compact, so the
  autocorrelation argument in the proof of
  Lemma~\ref{L:spatial-static-controls} gives
  $0 < A:=A_{\mathrm{sp}}(g)<\infty$. Lemma~\ref{L:spatial-fourier-bound} gives
  $R(g)\le2^{1+\alpha}R_\alpha$ for every such profile. A normalized compact
  bump supplies a positive ratio. Thus, $0<R_0<\infty$, independently of the
  finiteness of $S_1^{\mathrm{pos}}$. It remains to prove the score inequality.
  With $U_\ell g(x)=\ell^{d/2}g(\ell x)$, the physical changes of variables give
  \begin{equation}\label{E:spatial-dilation}
    A_{\mathrm{sp}}(U_\ell g)=\ell^\alpha A,\qquad
    b(U_\ell g)=\ell^2b(g).
  \end{equation}
  The factors in the first identity are $\ell^{2d}$ from the two squares,
  $\ell^{-d}$ from $\ud x$, and $\ell^{\alpha-d}$ from the Radon pushforward.


  \medskip
  \noindent\textit{Step 2.~}We choose the amplitude and spatial scale of a
  single control.
  For $\theta,\ell > 0$ choose the smooth control $z = \theta(U_\ell g)^2$. The
  Rayleigh test $U_\ell g$ gives $\lambda_z(1)^2 \ge \theta\ell^\alpha
  A-\ell^2b$, while $B_\Gamma(z,z) = \theta^2\ell^\alpha A$. For any $L > 0$,
  put $u = \alpha/q > 0$ and choose
  \[
    \ell^2=uL^2/b,\qquad
    \theta=(1+u)L^2/(\ell^\alpha A).
  \]
  The chosen scale gives $\ell^2b=uL^2$, and the chosen amplitude gives
  $\theta\ell^\alpha A=(1+u)L^2$. Their difference is $L^2$. In the cost,
  substituting these same choices gives $\theta^2\ell^\alpha
  A=(1+u)^2L^4/(\ell^\alpha A)$; replacing $\ell^\alpha A$ by
  $u^{\alpha/2}L^\alpha R(g)$ yields the coefficient $C$ below.
  The radicand is $L^2$. Lemma~\ref{L:spatial-static-controls} consequently
  gives
  \[
    S_1^{\mathrm{pos}}\ge L-CL^q,\qquad
    C=\frac{M_a(1+u)^2}{2R(g)u^{\alpha/2}}>0.
  \]
  The derivative of $L-CL^q$ is $1-qCL^{q-1}$. It changes sign once from
  positive to negative since $q>1$. At its zero, $CL^q=L/q$, so the value is
  $(1-1/q)L=(k/q)L$. This gives the maximum stated below without assuming any
  optimizer for the spatial profiles.
  Its maximum occurs at $L^k = 1/(qC)$ and is $(k/q)(qC)^{-1/k}$. Since $1+u =
  4/q$, $1/(q^qC) = R(g)\alpha^{\alpha/2}/(8M_aq^{q/2})$,
  proving~\eqref{E:spatial-smooth-score}. This argument retains the finite
  real value $S_1^{\mathrm{pos}}$ fixed at the start of the section; only the
  proof of $R_0<\infty$ is independent of that input. This proves
  Proposition~\ref{P:spatial-smooth-ratio}.
\end{proof}

So far $A_{\mathrm{sp}}$ has been used only for smooth compact profiles.
Proposition~\ref{P:occupation-physical-class} needs it and the square pairings
for real $H^1$ profiles, so we extend them with the same constant $R_0$.

\LeanMarker{P:spatial-four-linear}
\begin{proposition}[The physical four-linear extension]
  \label{P:spatial-four-linear}
  For real $f \in H^1(\R^d)$ define $N(f) = \|f\|_2^{1-\alpha/4}\|\nabla
  f\|_2^{\alpha/4}$. Every physical square pairing is finite and
  \begin{align}
    A_{\mathrm{sp}}(f)&\le R_0N(f)^4,\label{E:spatial-homogeneous-bound}\\
    0\le B_\Gamma(f^2,h^2)
    &\le\sqrt{A_{\mathrm{sp}}(f)A_{\mathrm{sp}}(h)}.
    \label{E:spatial-square-CS}
  \end{align}
  The signed integral $C_4(f_1,f_2,f_3,f_4) = B_\Gamma(f_1f_2,f_3f_4)$ is
  absolutely convergent and jointly continuous on $(H^1)^4$, with
  \begin{equation}\label{E:spatial-four-linear}
    \int|f_1(x)f_2(x)f_3(x-y)f_4(x-y)|\ud\nu
    \le R_0\prod_{j=1}^4N(f_j).
  \end{equation}
  The ratio supremum $R_{\mathrm{sp}}$ over all real unit $H^1$ profiles equals
  $R_0$. Every nonzero $H^1$ profile has $A_{\mathrm{sp}}(f) > 0$.
\end{proposition}

\begin{proof}
  We first obtain the square-pair bound for smooth profiles and extend it by
  Fatou. We then use ordinary Cauchy–Schwarz under the physical measure to
  prove an absolutely convergent four-linear bound. That bound, rather than
  Fatou alone, gives continuity and the full-class ratio identity.


  \medskip
  \noindent\textit{Step 1.~}We establish the smooth square-pair estimates.
  For smooth compact $f$, normalization in $R_0$ proves
  \eqref{E:spatial-homogeneous-bound}, including $f = 0$. All exponents in $N$
  are positive and $N(f) \le \|f\|_{H^1}$. For smooth compact signed $Q,P$, the
  absolute correlation is compactly supported and bounded by $\|Q\|_2\|P\|_2$.
  Local finiteness of $\Gamma$ therefore makes its physical integral finite.
  Smooth signed covariance is positive: with the ordinary transform
  $\widehat{Q}_{\mathrm{ord}}(\xi) = \int e^{i\xi\cdot x}Q(x)\ud x$, the model
  convention gives
  \[
    B_\Gamma(Q,Q)=\int|\widehat{Q}_{\mathrm{ord}}(\xi)|^2\mu(\ud\xi)
    \ge0,\qquad Q\in C_c^\infty(\R^d;\R).
  \]
  Pair $\Gamma$ with the smooth compact correlation to obtain this identity;
  rapid Fourier decay and tempered $\mu$ give finiteness. Evenness of $\Gamma$
  and $(x,y)\mapsto(x-y,-y)$ give symmetry. The quadratic polynomial for $Q+tP$
  gives covariance Cauchy--Schwarz. In particular, for smooth real $f,h$,
  \[
    0\le B_\Gamma(f^2,h^2)
    \le R_0N(f)^2N(h)^2.
  \]
  The squares of signed smooth profiles are already smooth and nonnegative.


  \medskip
  \noindent\textit{Step 2.~}We approximate Sobolev profiles and obtain finite
  physical pairings.
  Smooth compact functions are dense in $H^1$. Cutoffs equal to one on the
  radius-$R$ ball, vanishing beyond $2R$ and with gradient at most $C/R$,
  converge by the function and gradient tails and the bound $C\|f\|_2/R$ for the
  extra gradient term. Mollification then converges in $H^1$ by $L^2$
  translation continuity of the function and its weak gradients. The space is
  separable through its closed isometric image $f\mapsto(f,\nabla f)$ in
  $L^2(\R^d)^{d+1}$.

  For arbitrary $f,h \in H^1$ take smooth approximants converging in $H^1$ and
  along a common subsequence Lebesgue-a.e. Lemma~\ref{L:spatial-physical-pair}
  transfers this convergence at both positions to $\nu$-a.e. convergence. Fatou
  and continuity of $N$ give $B_\Gamma(f^2,h^2) \le R_0N(f)^2N(h)^2 < \infty$,
  including the diagonal bound. At this stage Fatou supplies an inequality, not
  continuity.


  \medskip
  \noindent\textit{Step 3.~}We prove the four-linear estimate and continuity.
  Apply ordinary Cauchy--Schwarz under the positive measure $\nu$ to
  $f_1(x)f_3(x-y)$ and $f_2(x)f_4(x-y)$. The two squared integrals are finite by
  the last bound; their square root is at most $R_0\prod_jN(f_j)$. This
  proves~\eqref{E:spatial-four-linear}, absolute convergence and multilinearity.
  Telescoping a difference one argument at a time and using $N(f) \le
  \|f\|_{H^1}$ proves joint continuity. Thus, $A_{\mathrm{sp}}$ and the square
  pairings are continuous. PSD for smooth compact profiles passes to finite
  linear combinations of $H^1$ squares by this continuity, and its two-profile
  Cauchy--Schwarz gives \eqref{E:spatial-square-CS}. This completes the physical
  pairing argument beyond Fatou.

  In a telescoping difference of four-linear expressions, each term contains
  exactly one difference of arguments and three unchanged or approximating
  arguments. The latter stay bounded in $H^1$, while the norm of the
  difference tends to zero. The four-linear bound therefore makes each term
  tend to zero. This is the continuity needed to pass positive
  semidefiniteness from smooth squares to Sobolev squares.

  \medskip
  \noindent\textit{Step 4.~}We identify the full ratio supremum and strict
  positivity.
  Every real unit $H^1$ profile has positive gradient energy by the Fourier
  argument above. The homogeneous bound gives $R_{\mathrm{sp}} \le R_0$, and
  smooth inclusion gives the reverse. Normalized smooth approximations preserve
  the ratio by continuity. Finally, for nonzero $f$, choose a bounded set $E$ of
  finite positive measure where $|f| \ge c > 0$. The correlation of $1_E$ is
  continuous and positive near zero by $L^2$ translation continuity. An
  origin-centered ball there has positive $\Gamma$ mass, so $A_{\mathrm{sp}}(f)
  \ge c^4B_\Gamma(1_E,1_E) > 0$. This uses a nonnegative integral, not an $H^1$
  claim for $1_E$. Finiteness was already proved. This proves
  Proposition~\ref{P:spatial-four-linear}.
\end{proof}

By Proposition~\ref{P:spatial-four-linear}, the ratio supremum is the same
over smooth and over $H^1$ profiles. We now express it through an energy and
restate the score bound in those terms.

\LeanMarker{P:spatial-energy-map}
\begin{proposition}[The spatial energy and static lower variational expression]
  \label{P:spatial-energy-map}
  Over all real unit $H^1$ profiles define the energy
  \[
    \mathcal{E}_{\mathrm{sp}}
    =\sup_{\|g\|_2=1}
    \left(A_{\mathrm{sp}}(g)-\frac{b(g)}2\right).
  \]
  Then
  \begin{equation}\label{E:spatial-energy-map}
    \mathcal{E}_{\mathrm{sp}}
    =\frac{2-\alpha}{2}\alpha^{\alpha/(2-\alpha)}
    R_{\mathrm{sp}}^{2/(2-\alpha)}
    \in(0,\infty).
  \end{equation}
  In particular,
  \begin{equation}\label{E:spatial-static-energy-score}
    S_1^{\mathrm{pos}}\ge M_a^{-1/k}F_0(\mathcal{E}_{\mathrm{sp}}),
    \qquad
    F_0(z)=k\left[
    \frac{(2z/(2-\alpha))^{(2-\alpha)/2}}{8q^{q/2}}
    \right]^{1/k},\quad z>0.
  \end{equation}
  The subscript in $F_0$ labels this scalar map; the covariance retains its
  original temporal exponent $a$.
\end{proposition}

\begin{proof}
  We optimize spatial dilation for a fixed profile, then take the supremum
  over all profiles. The resulting identity expresses the ratio constant in
  terms of the energy and lets us rewrite the static score bound.

  Physical change of variables, or smooth approximation and
  Proposition~\ref{P:spatial-four-linear}, extends \eqref{E:spatial-dilation} to
  real $H^1$ profiles. For fixed unit $g$, with $A,b > 0$, the maximum of
  $\ell^\alpha A-\ell^2b/2$ occurs at $\ell^{2-\alpha} = \alpha A/b$ and equals
  \[
    \frac{2-\alpha}{2}\alpha^{\alpha/(2-\alpha)}
    R(g)^{2/(2-\alpha)}.
  \]
  Differentiating the scalar expression gives $\alpha A\ell^{\alpha-1}-b\ell$.
  Its positive zero satisfies the stated equation. At that point,
  $b\ell^2=\alpha A\ell^\alpha$, so the value is $(1-\alpha/2)A\ell^\alpha$.
  Substitution of the critical scale gives the displayed power of $R(g)$.
  Dilation preserves the whole unit class and includes $\ell = 1$. Taking both
  numerical suprema, then profiles with ratio tending to $R_{\mathrm{sp}}$,
  proves~\eqref{E:spatial-energy-map}. No optimizer is assumed. Smooth ratios
  also approach $R_{\mathrm{sp}}$ by Proposition~\ref{P:spatial-four-linear}.
  Pass them through~\eqref{E:spatial-smooth-score} and substitute
  \[
    R_{\mathrm{sp}}\alpha^{\alpha/2}
    =\left(\frac{2\mathcal{E}_{\mathrm{sp}}}{2-\alpha}
    \right)^{(2-\alpha)/2}.
  \]
  This gives~\eqref{E:spatial-static-energy-score}. This proves
  Proposition~\ref{P:spatial-energy-map}.
\end{proof}

\subsection{The kernel row and full occupation energy}

Define, for $u \ne 0$,
\begin{equation}\label{E:occupation-damped-kernel}
  K_\star(u)=\frac1{\Gamma_E(a)}
  \int_0^\infty
  e^{-(\lambda+\lambda^2/(2D))|u|}
  \lambda^{a-1}\ud\lambda,
\end{equation}
and set $K_\star(0) = 0$ for time integration. The spatial estimates above
concern individual $H^1$ profiles. To apply them to time-dependent families,
we first compute the exact row bound for this time kernel.

\LeanMarker{L:occupation-kernel-row}
\begin{lemma}[Exact row maximum]\label{L:occupation-kernel-row}
  The kernel is even, positive off zero, locally integrable, decreasing on
  $(0,\infty)$, and $K_\star(u) \le |u|^{-a}$ off zero. Its exact row maximum on
  $[0,1]$ is
  \begin{equation}\label{E:occupation-kernel-row}
    \max_{s\in[0,1]}\int_0^1K_\star(s-t)\ud t
    =R_\star:=2\int_0^{1/2}K_\star(u)\ud u\in(0,\infty).
  \end{equation}
\end{lemma}

\begin{proof}
  Dropping the positive quadratic damping bounds the mixture by the Gamma
  integral $|u|^{-a}$. Positivity and monotonicity follow directly from the
  integrand, and $a < 1$ gives local integrability. Put $\Phi(t) =
  \int_0^tK_\star(u)\ud u$. It is concave because its integrand is decreasing on
  positive arguments. The row is $\Phi(s)+\Phi(1-s) \le 2\Phi(1/2)$, with
  equality at $s = 1/2$. The diagonal value changes no time integral. This
  proves Lemma~\ref{L:occupation-kernel-row}.
\end{proof}

Let $\mathcal{O}$ be the full class of strongly $H^1(\R^d;\R)$-measurable fields
$s\mapsto g_s$ on $(0,1)$ with $\|g_s\|_2 = 1$ a.e. and $T(g) =
\int_0^1b(g_s)\ud s < \infty$. No time smoothness, common compact support or
optimizer condition is imposed.

\LeanMarker{P:occupation-physical-class}
\begin{proposition}[The full physical occupation class]
  \label{P:occupation-physical-class}
  The pairings $(s,t)\mapsto B_\Gamma(g_s^2,g_t^2)$ are measurable,
  representative independent and equal to their physical integrals. With
  \[
    J(g)=\int_0^1\int_0^1
    K_\star(s-t)B_\Gamma(g_s^2,g_t^2)\ud s\ud t,
  \]
  every $g \in \mathcal{O}$ satisfies
  \begin{equation}\label{E:occupation-physical-bound}
    0\le J(g)\le R_\star R_{\mathrm{sp}}T(g)^{\alpha/2}<\infty.
  \end{equation}
  The energy over exactly this full class satisfies
  \begin{equation}\label{E:occupation-own-energy}
    0<\mathcal{E}_\star^{\mathcal{O}}
    :=\sup_{g\in\mathcal{O}}\big(J(g)-T(g)/2\big)
    \le R_\star^{2/(2-\alpha)}\mathcal{E}_{\mathrm{sp}}<\infty.
  \end{equation}
\end{proposition}

\begin{proof}
  We first identify measurable physical pairings for the full occupation
  class. We bound the time integral by the kernel row estimate and then
  compare its energy with the spatial energy, retaining the full class
  throughout.


  \medskip
  \noindent\textit{Step 1.~}We identify measurable representatives and their
  pairings.
  The map $s\mapsto b(g_s)$ is measurable by continuity of the squared gradient
  norm. Also $\int_0^1\|g_s\|_{H^1}^2\ud s = 1+T(g)$. Use a separably valued
  strong representative and replace exceptional null times by one fixed real
  unit compact bump. Joint measurable spatial representatives exist: approximate
  this Bochner $L^2$ field by simple $H^1$-valued functions with summable
  $L^2(0,1;H^1)$ errors, and choose measurable spatial versions for their
  countably many values. A subsequence has a measurable spacetime $L^2$ limit,
  and for a.e. $s$ its spatial $L^2$ class is $g_s$.
  Lemma~\ref{L:spatial-physical-pair} makes every pairing independent of the
  spatial versions. The continuous pairing map on the separable space $H^1\times
  H^1$ from Proposition~\ref{P:spatial-four-linear} consequently gives a
  measurable function of $(s,t)$ agreeing a.e. with the nonnegative physical
  integral.


  \medskip
  \noindent\textit{Step 2.~}We apply the spatial bound and the temporal row
  estimate.
  Set $F(s) = b(g_s)^{\alpha/4}$. Equations~\eqref{E:spatial-homogeneous-bound}
  and~\eqref{E:spatial-square-CS} give $B_\Gamma(g_s^2,g_t^2) \le
  R_{\mathrm{sp}}F(s)F(t)$. Since $0 < \alpha/2 < 1$, concavity on the unit time
  interval yields $\int_0^1F^2 = \int_0^1b(g_s)^{\alpha/2}\ud s \le
  T(g)^{\alpha/2}$. Now use $2F(s)F(t) \le F(s)^2+F(t)^2$ and the exact row
  bound \eqref{E:occupation-kernel-row}, whose column bound is the same because
  $K_\star$ is even, to obtain \eqref{E:occupation-physical-bound}. Positive
  Tonelli is valid before finiteness is known. Null-time modifications affect
  only a product-null set, so they cannot change $J$.


  \medskip
  \noindent\textit{Step 3.~}We bound the energy and show that it is positive.
  The scalar maximum of $R_\star R_{\mathrm{sp}}T^{\alpha/2}-T/2$ over $T \ge 0$
  is $R_\star^{2/(2-\alpha)}\mathcal{E}_{\mathrm{sp}}$, by the same
  differentiation as~\eqref{E:spatial-energy-map}. This bounds the occupation
  supremum from above. For positivity take a constant-in-time unit compact
  profile and its spatial dilation $U_\ell$. Its $J$ is $J_0\ell^\alpha
  A_{\mathrm{sp}}(g)$, where $J_0 = \int_0^1\int_0^1K_\star(s-t)\ud s\ud t \in
  (0,\infty)$, while $T = \ell^2b(g)$. For sufficiently small $\ell > 0$,
  $\alpha < 2$ makes $J-T/2 > 0$. Thus, the nonempty supremum is a positive
  finite real number. This is a comparison bound; no attaining profile or
  equality in it has been asserted. No spatial point trace or rough stochastic
  evolution was used. This proves Proposition~\ref{P:occupation-physical-class}.
\end{proof}

For zero covariance all spatial pairings vanish and set $R_{\mathrm{sp}} = 0$.
The spatial and occupation energies are zero suprema of $-b/2$ and $-T/2$: small
spatial dilations of one fixed unit compact profile make these energies tend to
zero, with the profile constant in time for the latter. No zero-gradient unit
state is used. The temporal identities remain valid, and the moment model is the
free case already treated.

\section{Model and standing assumptions}\label{S:model}

We study the hyperbolic Anderson model
\cite{chen:author2026:time-dependency}*{(1.1)}
\begin{equation}\label{E:HAM}
  \frac{\partial^2 u}{\partial t^2}(t,x)=\Delta u(t,x)+\W(t,x)u(t,x),
  \qquad u(0,x)=1,\quad \frac{\partial u}{\partial t}(0,x)=0,
\end{equation}
for $(t,x) \in \R_+\times\R^d$, $d \in \{1,2,3\}$, interpreted in the
Stratonovich sense (limit of classical wave equations driven by smoothed noises
$\W_{\epsilon,\delta}$). The noise is centered Gaussian with
\begin{equation}\label{E:cov}
  \operatorname{Cov}\big(\W(s,x),\W(t,y)\big)=|s-t|^{-\alpha_0}\gamma(x-y),
  \qquad 0<\alpha_0<1,
\end{equation}
where $\gamma \ge 0$ is nonnegative definite with spectral measure $\mu$.

\begin{hypothesis}[Homogeneous covariance]\label{H:homog}
  Let $d \in \{1,2,3\}$ and $0 < \alpha_0 < 1$. The spatial covariance is a
  nonnegative, even, positive-definite tempered Radon measure $\Gamma$ on
  $\R^d$, with positive even tempered Radon spectral measure $\mu$ in the
  convention
  \[
    \Gamma=\int e^{i\xi\cdot x}\mu(\ud\xi)
    \quad\text{in the sense of tempered distributions}.
  \]
  The symbol $\gamma$ in~\eqref{E:cov} denotes this covariance as a function
  when one exists, and as a distribution otherwise. For every Borel set $A$ and
  $c > 0$, $\Gamma(cA) = c^{d-\alpha}\Gamma(A)$, and
  \begin{equation}\label{E:threshold}
    \alpha_0+\alpha<2,
  \end{equation}
  with $0 < \alpha < d$, or $\alpha = d = 1$. In particular, the last case
  includes spatial white noise $\Gamma = \delta_0$.
\end{hypothesis}

\paragraph{Zero covariance.}
Hypothesis~\ref{H:homog} allows $\Gamma = 0$ by design. In this case the
centered Gaussian noise is $\W = 0$, and the solution with initial data
$(1,0)$ is $u(t,x) = 1$. Thus $\E|u(t,x)|^p = 1$ for every $p > 0$, and
all normalized logarithmic moment rates are zero. Results asserting strict
positivity of covariance-dependent quantities impose $\Gamma \ne 0$ locally.

\section{Covariance bounds for physically masked wave fields}
\label{S:masked-covariance}

\input{notes/figures/mask-window-figure}

We establish the covariance estimates needed for the operator construction.
Throughout this section, Hypothesis~\ref{H:homog} holds, $a = \alpha_0$, and
$\Gamma \ne 0$. For zero spatial covariance, the spatial spectral measures and
noise variances vanish; the purely temporal estimates remain valid independently
of $\Gamma$. We use the unitary Fourier transform
\[
  \widehat{f}(\xi) = c_d\int_{\R^d}e^{-i\xi\cdot x}f(x)\ud x,
  \qquad c_d = (2\pi)^{-d/2}.
\]
The spectral convention in Hypothesis~\ref{H:homog} means that, for every
complex Schwartz test $\varphi$,
\begin{equation}\label{E:masked-spectral-convention}
  \int_{\R^d}\varphi(x)\Gamma(\ud x)
  = \int_{\R^d}\left(\int_{\R^d}e^{i\xi\cdot x}
  \varphi(x)\ud x\right)\mu(\ud\xi).
\end{equation}
This is an identity of distribution pairings. We will not interchange two
infinite measures in the associated oscillatory expression.

A positive tempered Radon measure has polynomial mass growth. Indeed, choose a
nonnegative compact smooth test that is at least one on the unit ball, and apply
a defining Schwartz seminorm bound for the distribution to its dilations. Its
mass on a ball of radius $R \ge 1$ is bounded by a power of $R$. Summation over
dyadic shells then gives integrability of $(1+|x|)^{-N}$ for some $N$, and hence
absolute integrability of every Schwartz test. These facts apply to both
$\Gamma$ and $\mu$. Smooth cutoffs converge in the Schwartz topology, so these
integrals agree with the tempered pairings.

\subsection{Translated spectral growth and physical masks}

\LeanMarker{P:masked-spectral-growth}
\begin{proposition}[Translated spectral mass]\label{P:masked-spectral-growth}
  Put
  \[
    M_\Gamma = \int_{\R^d}e^{-|x|^2/4}\Gamma(\ud x),
    \qquad C_\Gamma = \frac{eM_\Gamma}{(2\sqrt{\pi})^d}.
  \]
  Then $0 < M_\Gamma < \infty$, and for $y \in \R^d$ and $r > 0$,
  \begin{align}
    \int_{\R^d}e^{-|\xi-y|^2/r^2}\mu(\ud\xi)
    &= \left(\frac{r}{2\sqrt{\pi}}\right)^d
    \int_{\R^d}e^{-r^2|x|^2/4}e^{-iy\cdot x}\Gamma(\ud x),
    \label{E:masked-gaussian-transform}\\
    0 \le \int_{\R^d}e^{-|\xi-y|^2/r^2}\mu(\ud\xi)
    &\le \frac{r^\alpha M_\Gamma}{(2\sqrt{\pi})^d},
    \qquad \mu(B(y,r)) \le C_\Gamma r^\alpha.
    \label{E:masked-translated-ball}
  \end{align}
  The ball may be open or closed.
\end{proposition}

\begin{proof}
  Temperedness gives finiteness of $M_\Gamma$. Its positivity follows from
  $\Gamma \ne 0$. To check the Fourier constant, set $J(b) = \int_\R
  e^{-Ax^2}e^{ibx}\ud x$ for $A > 0$. Squaring the real Gaussian integral and
  using polar coordinates gives $J(0) = \sqrt{\pi/A}$. Differentiation and
  integration by parts give $J'(b) = -bJ(b)/(2A)$. Therefore, $J(b) =
  \sqrt{\pi/A}e^{-b^2/(4A)}$. Taking products over the coordinates gives
  \[
    \int_{\R^d}e^{-r^2|x|^2/4}e^{i(\xi-y)\cdot x}\ud x
    = \left(\frac{2\sqrt{\pi}}r\right)^d e^{-|\xi-y|^2/r^2}.
  \]
  Apply~\eqref{E:masked-spectral-convention} to the single Schwartz test
  $e^{-r^2|x|^2/4}e^{-iy\cdot x}$ to obtain \eqref{E:masked-gaussian-transform}.
  Its left side is nonnegative. Taking the absolute value on the right removes
  the complex phase because $\Gamma$ is nonnegative. The pushforward of $\Gamma$
  by $x \mapsto rx$ assigns mass $r^{\alpha-d}\Gamma(A)$ to a Borel set $A$.
  Thus,
  \[
    \int e^{-r^2|x|^2/4}\Gamma(\ud x) = r^{\alpha-d}M_\Gamma.
  \]
  This proves the Gaussian bound. On either ball $B(y,r)$ the integrand
  $e^{-|\xi-y|^2/r^2}$ is at least $e^{-1}$, proving the mass bound. This proves
  Proposition~\ref{P:masked-spectral-growth}.
\end{proof}

Choose and fix a real nonzero Schwartz function $\zeta$. For $R > 0$ put
$\zeta_R(x) = \zeta(x/R)$, $k_R = \widehat{\zeta_R}$ and $L =
\|\widehat{\zeta}\|_1$. Then $k_R(u) = R^d\widehat{\zeta}(Ru)$ and $\|k_R\|_1 =
L > 0$. Define
\begin{equation}\label{E:masked-auxiliary-measure}
  \nu_R(\ud\eta) = \rho_R(\eta)\ud\eta,
  \qquad \rho_R(\eta) = c_d^2L\int |k_R(\xi-\eta)|\mu(\ud\xi).
\end{equation}

\LeanMarker{P:masked-auxiliary-norm}
\begin{proposition}[Auxiliary mask norm]\label{P:masked-auxiliary-norm}
  The measure $\nu_R$ is even, tempered and Radon, with
  \begin{equation}\label{E:masked-auxiliary-growth}
    \nu_R(B(y,r)) \le c_d^2L^2C_\Gamma r^\alpha
  \end{equation}
  for all $y \in \R^d$ and $r > 0$, uniformly in $R > 0$. If $H$ is a separable
  complex Hilbert space and $F$ is an $H$-valued Schwartz function, then
  \begin{equation}\label{E:masked-hilbert-fourier}
    \int \|\widehat{\zeta_RF}(\xi)\|_H^2\mu(\ud\xi)
    \le \int \|\widehat{F}(\eta)\|_H^2\nu_R(\ud\eta).
  \end{equation}
  For a real $H$, the same assertion holds after complexification.
\end{proposition}

\begin{proof}
  Schwartz decay of $k_R$ and polynomial growth of $\mu$ show that
  $\rho_R(\eta)$ is finite at every $\eta$. On compact sets of $\eta$, the
  integrands have a common integrable bound. Dominated convergence therefore
  makes $\rho_R$ continuous, and $\nu_R$ is Radon. Since $\zeta$ is real,
  $k_R(-u) = \overline{k_R(u)}$. Evenness of $\mu$ gives evenness of $\rho_R$.
  Tonelli's theorem gives
  \[
    \nu_R(B(y,r)) = c_d^2L\int |k_R(u)|\mu(B(y+u,r))\ud u.
  \]
  Proposition~\ref{P:masked-spectral-growth} and $\|k_R\|_1 = L$ give
  \eqref{E:masked-auxiliary-growth}. Polynomial growth on centered balls proves
  temperedness by the same dyadic argument as above.

  Bochner Fourier integration gives
  \[
    \widehat{\zeta_RF}(\xi)
    = c_d\int k_R(\xi-\eta)\widehat{F}(\eta)\ud\eta.
  \]
  For scalar Schwartz functions this follows by Fourier inversion and absolute
  integrability. It follows next for finite-rank Hilbert-valued functions. For
  general $F$, take projections onto the successive spans of a countable
  orthonormal basis of its separable range closure. They converge pointwise in
  norm. The Schwartz bounds on $F$ dominate the Fourier integrals, proving the
  identity by dominated convergence. In a finite-dimensional space the full
  identity projection suffices; the zero space is immediate. The triangle
  inequality and weighted Cauchy--Schwarz yield
  \[
    \|\widehat{\zeta_RF}(\xi)\|_H^2
    \le c_d^2L\int |k_R(\xi-\eta)|\|\widehat{F}(\eta)\|_H^2\ud\eta.
  \]
  Integrating against $\mu$ and applying Tonelli's theorem proves
  \eqref{E:masked-hilbert-fourier}. Its right side is finite by Schwartz decay
  and temperedness of $\nu_R$. This proves
  Proposition~\ref{P:masked-auxiliary-norm}.
\end{proof}

\subsection{Temporal completion and the original noise}

The noise in~\eqref{E:cov} is also correlated in time, with kernel
$|t-s|^{-a}$. We now complete the temporal tests in that form.

\LeanMarker{L:masked-time-completion}
\begin{lemma}[Temporal covariance completion]\label{L:masked-time-completion}
  Let $T > 0$. The form
  \[
    \langle f,g\rangle_a
    = \int_0^T\int_0^T f(t)g(s)|t-s|^{-a}\ud t\ud s
  \]
  is positive semidefinite on real compact smooth tests. Its null-quotient
  completion $H_a$ is separable. The inclusion of tests extends continuously
  from $L^2(0,T)$, with
  \begin{equation}\label{E:masked-time-schur}
    \LeanEqMarker{E:masked-time-schur}
    \|f\|_a^2 \le \frac{2T^{1-a}}{1-a}\|f\|_2^2.
  \end{equation}
  In its complexification the norm is given by the same double integral with
  $f(t)\overline{f(s)}$. In particular, bounded measurable amplitudes on a
  subinterval, extended by zero, belong to this completion.
\end{lemma}

\begin{proof}
  Write $\Gamma_E$ for Euler's Gamma function. Away from the null diagonal,
  \[
    |t-s|^{-a} = \frac1{\Gamma_E(a)}\int_0^\infty
    \lambda^{a-1}e^{-\lambda|t-s|}\ud\lambda.
  \]
  This follows by substitution in the defining Gamma integral. Moreover,
  \[
    e^{-\lambda|t-s|}
    = \int_{-\infty}^{\min(t,s)}2\lambda
    e^{-\lambda(t-u)}e^{-\lambda(s-u)}\ud u.
  \]
  For each $\lambda > 0$ its quadratic form is an integral of a square. All
  exchanges for compact smooth tests are absolutely integrable because the
  kernel row integral is at most $2T^{1-a}/(1-a)$. The same bound and $2|xy| \le
  |x|^2+|y|^2$ prove~\eqref{E:masked-time-schur}. A countable $L^2$-dense family
  of compact smooth tests is consequently dense in $H_a$, proving separability.
  The complex quadratic form is the sum of those of the real and imaginary
  parts, since symmetry cancels the imaginary cross term. Thus, the same bound
  holds after complexification. Smooth density in $L^2$, together with the Schur
  bound on the absolute bilinear form, extends the inclusion and the integral
  identity to $L^2$. This includes the stated interval amplitudes. This proves
  Lemma~\ref{L:masked-time-completion}.
\end{proof}

Let $\mathcal{D}$ consist of finite sums $F(t,x) = \sum_j\tau_j(t)f_j(x)$ with
real $\tau_j \in C_c^\infty(0,T)$ and real Schwartz $f_j$. These give
$H_a$-valued Schwartz functions of $x$. Let $W$ denote the original real
isonormal noise in~\eqref{E:cov}.

\LeanMarker{P:masked-noise-completion}
\begin{proposition}[Completion on the same noise]\label{P:masked-noise-completion}
  The space $\mathcal{D}$ belongs to the original test-noise completion and
  \begin{equation}\label{E:masked-actual-noise-norm}
    \E W(F)^2 = (2\pi)^d\int
    \|\widehat{F}(\xi)\|_{H_a^{\mathbb{C}}}^2\mu(\ud\xi).
  \end{equation}
  Define the auxiliary seminorm
  \[
    N_R(F)^2 = (2\pi)^d\int
    \|\widehat{F}(\eta)\|_{H_a^{\mathbb{C}}}^2\nu_R(\ud\eta).
  \]
  Then $F \mapsto W(\zeta_RF)$ extends as a contraction from the null-quotient
  completion for $N_R$ into the original $L^2(\Omega)$. The extension is
  centered and jointly Gaussian. For finitely many real $F_1,\ldots,F_m \in
  \mathcal{D}$, their masked covariance matrix is bounded in positive
  semidefinite order by the matrix of auxiliary inner products. This gives
  covariance comparison for every finite linear Gaussian image and every convex
  norm power $p \ge 1$.
\end{proposition}

\begin{proof}
  For real spatial Schwartz $f,g$, put $h(z) = \int f(y+z)g(y)\ud y$. Fourier
  inversion and Parseval give
  \[
    h(z) = \int e^{i\xi\cdot z}\widehat{f}(\xi)
    \overline{\widehat{g}(\xi)}\ud\xi.
  \]
  Therefore,
  \[
    \int e^{i\xi\cdot z}h(z)\ud z
    = (2\pi)^d\widehat{f}(-\xi)\overline{\widehat{g}(-\xi)}.
  \]
  Apply~\eqref{E:masked-spectral-convention} to the single test $h$ and reflect
  $\xi$ using evenness of $\mu$. The spatial covariance is thus
  $(2\pi)^d\int\widehat{f}(\xi)\overline{\widehat{g}(\xi)}\mu(\ud\xi)$. Summing
  over the finitely many temporal factors gives
  \eqref{E:masked-actual-noise-norm}. Schwartz decay, temperedness and temporal
  integrability justify these pairings. If $W$ is initially given only on
  compact spatial tests, smooth expanding cutoffs converge in the Schwartz
  topology. The spectral form is continuous in that topology, admitting
  $\mathcal{D}$ in the same completion.

  Proposition~\ref{P:masked-auxiliary-norm}, applied with $H =
  H_a^{\mathbb{C}}$, gives $\E W(\zeta_RF)^2 \le N_R(F)^2$. It annihilates
  auxiliary-null tests. An auxiliary Cauchy sequence thus gives a Cauchy
  sequence of noise evaluations in $L^2(\Omega)$, proving the contraction
  extension and its uniqueness. Every finite linear combination of the limits is
  an $L^2$ limit of centered Gaussian variables. Their variances converge, so
  their characteristic functions prove that the limits remain jointly centered
  Gaussian.

  For the finite family let $A$ be the covariance matrix and $B$ the auxiliary
  Gram matrix. Applying the norm inequality to $\sum_i b_iF_i$ gives
  $b^{\mathsf{T}}Ab \le b^{\mathsf{T}}Bb$ for every real vector $b$.
  Diagonalizing $B-A$ constructs a centered Gaussian vector $Z$ of that
  covariance, independent of the evaluation vector $X$, on a product space. Then
  $X+Z$ has covariance $B$. Conditional Jensen gives $\E\|X\|^p \le \E\|X+Z\|^p$
  for every norm and $p \ge 1$. Finite Gaussian moments ensure finiteness of
  both sides. This proves Proposition~\ref{P:masked-noise-completion}.
\end{proof}

The auxiliary completion depends on $R$ and $\zeta$. Its construction does not
define multiplication by $\zeta_R$ on the original noise quotient. Nor does the
finite linear Gaussian comparison assert a covariance comparison for nonlinear
wave solutions. The auxiliary and original test norms need not dominate one
another. No spatial representative for every auxiliary class, nonnegative
physical inverse covariance for $\nu_R$, or uniform density bound in $R$ is
asserted.

\LeanMarker{P:masked-quotient-obstruction}
\begin{proposition}[A physical mask need not preserve noise-null tests]
  \label{P:masked-quotient-obstruction}
  In the admissible case $d = 2$, $\alpha = 1$, $0 < a < 1$ and $\Gamma =
  \delta_0(\ud x_1)\ud x_2$, there is a real compact smooth mask $\zeta$ such
  that no finite $C$ satisfies $\E W(\zeta F)^2 \le C\E W(F)^2$ for every
  compact smooth test $F$.
\end{proposition}

\begin{proof}
  Fourier inversion on Schwartz tests gives $\mu =
  (2\pi)^{-1}\ud\xi_1\delta_0(\ud\xi_2)$ in the stated spectral convention. The
  measure $\Gamma$ is nonnegative, even, positive definite and tempered, with
  $\Gamma(cA) = c\Gamma(A)$. It therefore satisfies Hypothesis~\ref{H:homog}
  with $\alpha = 1$. This is not spatial white noise in dimension two. For real
  compact smooth $g$, its spatial covariance is, by ordinary Fubini,
  \[
    V(g) = \int_\R\left(\int_\R g(x_1,x_2)\ud x_2\right)^2\ud x_1.
  \]
  Choose real even $\chi \in C_c^\infty(\R)$ equal to one on $[-1,1]$ and zero
  outside $[-2,2]$. For example, on $1 < r = |x| < 2$ use
  \[
    \chi(x) = \frac{e^{-1/(2-r)}}{e^{-1/(2-r)}+e^{-1/(r-1)}}.
  \]
  The endpoint exponentials vanish to every order, so the plateau pastes
  smoothly. Let $I_\chi = \int(\chi')^2 > 0$ and $J_\chi = \int\chi^4 > 0$.
  Positivity of $I_\chi$ follows because $\chi$ is not constant, and $J_\chi \ge
  2$. Put $g(x) = \chi(x_1)\chi''(x_2)$ and $\zeta(x) = \chi(x_1)\chi(x_2)$.
  Their unmasked marginal vanishes since $\int\chi'' = 0$, whereas integration
  by parts gives the masked marginal $-I_\chi\chi(x_1)^2$. Thus, $V(g) = 0$ and
  $V(\zeta g) = J_\chi I_\chi^2 > 0$. For a nonzero nonnegative $\tau \in
  C_c^\infty(0,T)$, set $F(t,x) = \tau(t)g(x)$. The temporal factor
  $\|\tau\|_a^2$ is finite and strictly positive because its nonnegative
  integrand is positive on a set of positive two-dimensional measure.
  Consequently, the full unmasked variance is zero and the masked variance is
  positive. This proves Proposition~\ref{P:masked-quotient-obstruction}.
\end{proof}

No-go~\ref{P:nogo-22} records the quotient interpretation of this witness:
the mask separates two representatives of the same noise-null class.
This is why we mask tests before constructing the auxiliary completion.


\subsection{Temporal modulation and continuous wave factors}

Write $\langle x\rangle = \sqrt{1+x^2}$. Let $I = [s,s+h] \subset [0,T]$, with
$0 < h \le 1$. Norms below are those of the complexified temporal completion in
Lemma~\ref{L:masked-time-completion}.

\LeanMarker{L:masked-temporal-modulation}
\begin{lemma}[Modulated amplitudes]\label{L:masked-temporal-modulation}
  Let $A \in C^1(I;\mathbb{C})$, $M = \sup_I|A|$ and $D = \sup_I|A'|$. For every
  real $\omega$,
  \begin{equation}\label{E:masked-temporal-modulation}
    \|1_IA(t)e^{i\omega t}\|_a^2
    \le C_0(a)h^{2-a}(M+hD)^2\langle h\omega\rangle^{a-1},
    \qquad C_0(a) = \frac{16}{1-a}.
  \end{equation}
  The bound is uniform in $T,s,h,A$ and $\omega$.
\end{lemma}

\begin{proof}
  Put $c(v) = \int_s^{s+h-v}A(u+v)\overline{A(u)}\ud u$ for $0 \le v \le h$.
  Differentiation at the moving endpoint gives
  \[
    c'(v) = -A(s+h)\overline{A(s+h-v)}
    +\int_s^{s+h-v}A'(u+v)\overline{A(u)}\ud u.
  \]
  Thus, $c(h) = 0$, $|c(v)| \le hM^2$ and $|c'(v)| \le M^2+hMD$. Splitting the
  covariance square into its two conjugate triangles gives
  \[
    \|1_IA(t)e^{i\omega t}\|_a^2
    = 2\operatorname{Re}\int_0^h v^{-a}e^{i\omega v}c(v)\ud v
    = 2h^{2-a}\operatorname{Re}\int_0^1 y^{-a}e^{i\omega hy}f(y)\ud y,
  \]
  where $f(y) = c(hy)/h$. Set $K = M+hD$. Then $f(1) = 0$ and $|f|,|f'| \le
  K^2$. If $K = 0$ the assertion is immediate. If $|\omega h| \le 1$, the last
  integral is at most $K^2/(1-a)$ in absolute value, and $\langle\omega
  h\rangle^{a-1} \ge 2^{(a-1)/2}$. The claimed constant is sufficient in this
  case.

  Now suppose $k = |\omega h| > 1$. On $[0,1/k]$ the integral is at most
  $K^2k^{a-1}/(1-a)$. On $[1/k,1]$, integrate the exponential by parts. The
  endpoint at one vanishes, while the other endpoint contributes at most
  $K^2k^{a-1}$. The derivative integral, divided by $k$, is bounded by
  \[
    \frac{K^2}k\left(a\int_{1/k}^1 y^{-a-1}\ud y
    +\int_{1/k}^1 y^{-a}\ud y\right)
    \le K^2 k^{a-1}\left(1+\frac1{1-a}\right).
  \]
  Here $k^{-1} \le k^{a-1}$ since $a > 0$. The total is at most
  $4K^2k^{a-1}/(1-a)$. Finally, $k^{a-1} \le \sqrt{2}\langle\omega
  h\rangle^{a-1}$ and $8\sqrt{2} \le 16$. Including the factor $2h^{2-a}$ proves
  the bound for either sign of $\omega$. This proves
  Lemma~\ref{L:masked-temporal-modulation}.
\end{proof}

Lemma~\ref{L:masked-temporal-modulation} treats one exponential mode. A
product of two sines or cosines is a sum of four such modes, which we bound
next.

\LeanMarker{L:masked-trig-products}
\begin{lemma}[Correlated trigonometric products]\label{L:masked-trig-products}
  Let $r,\rho \ge 0$. Let $P_r$ be either $\cos(rt)$ or $\sin(rt)$, and let
  $Q_\rho$ be either $\cos(\rho t)$ or $\sin(\rho t)$. With the amplitude and
  constant above,
  \begin{align}
    \|1_IAP_rQ_\rho\|_a^2
    &\le C_0(a)h^{2-a}(M+hD)^2\mathcal{Q}_h(r,\rho),
    \label{E:masked-trig-products}\\
    \mathcal{Q}_h(r,\rho)
    &:= \langle h(r-\rho)\rangle^{a-1}
    +\langle h(r+\rho)\rangle^{a-1}.
    \label{E:masked-frequency-pair}
  \end{align}
\end{lemma}

\begin{proof}
  Expanding the two trigonometric factors gives four complex exponential modes
  with frequencies $\sigma r+\tau\rho$, $\sigma,\tau \in \{-1,1\}$, and
  coefficients of absolute value $1/4$. In a Hilbert space,
  \[
    \left\|\sum_j c_jv_j\right\|^2
    \le \left(\sum_j|c_j|\right)\sum_j|c_j|\|v_j\|^2.
  \]
  The first coefficient sum is one. Apply
  Lemma~\ref{L:masked-temporal-modulation} to each mode. Two frequencies have
  absolute value $|r-\rho|$ and two have absolute value $r+\rho$. This gives one
  half of the asserted right side, and hence the bound. Repeated or zero
  frequencies are permitted; no orthogonality or independence of the modes is
  used. This proves Lemma~\ref{L:masked-trig-products}.
\end{proof}

\LeanMarker{L:masked-continuous-wave}
\begin{lemma}[Continuous sine factors]\label{L:masked-continuous-wave}
  Define $c_r(t) = \cos(rt)$ and
  \[
    b_r(t) = \langle r\rangle\frac{\sin(rt)}r\quad(r > 0),
    \qquad b_0(t) = t.
  \]
  For $F_r \in \{c_r,b_r\}$ and $G_\rho \in \{c_\rho,b_\rho\}$,
  \begin{equation}\label{E:masked-continuous-wave}
    \|1_IF_rG_\rho\|_a^2
    \le C_1(a)(1+T)^4h^{2-a}\mathcal{Q}_h(r,\rho),
    \qquad C_1(a) = 4\cdot3^{1-a}C_0(a).
  \end{equation}
\end{lemma}

\begin{proof}
  For $r \ge 1$, $b_r$ is a sine times a coefficient at most $\sqrt{2}$. For $0
  \le r \le 1$, the fundamental theorem of calculus gives
  \[
    b_r(t) = t\langle r\rangle\int_0^1\cos(vrt)\ud v.
  \]
  This includes $r = 0$. Hence, a product $F_rG_\rho$ is a scalar coefficient of
  absolute value at most two times a normalized integral of
  $t^jP_{r'}(t)Q_{\rho'}(t)$, where $j \in \{0,1,2\}$ and $|r'-r|,|\rho'-\rho|
  \le 1$. The integral is over zero, one or two copies of $[0,1]$; with no
  averages, use a one-point probability space. Its parameter dependence is
  continuous in $L^2(0,T)$ by dominated convergence.
  Lemma~\ref{L:masked-time-completion} makes it continuous in $H_a$ as well.
  Thus, the averages are Bochner integrals, and their $H_a$ images agree with
  the pointwise averages by continuity of the inclusion. Hilbert Jensen bounds
  the squared norm of an average by the average of the squared norms.

  Apply Lemma~\ref{L:masked-trig-products} to each integrand with $A(t) = t^j$.
  For $j = 0,1,2$, respectively, $M+hD$ is at most $1$, $T+1$ and $T^2+2T$. In
  every case $(M+hD)^2 \le (1+T)^4$. For real $x,e$, the triangle inequality and
  its reverse imply
  \[
    \frac{\langle x\rangle}{1+|e|}
    \le \langle x+e\rangle \le (1+|e|)\langle x\rangle.
  \]
  The changes in $h(r\pm\rho)$ have magnitude at most two, since $h \le 1$.
  Because $a-1 < 0$, each altered bracket to the power $a-1$ is at most
  $3^{1-a}$ times the original one. Apply these comparisons, Hilbert Jensen and
  the squared scalar coefficient bound four to obtain
  \eqref{E:masked-continuous-wave}. This proves
  Lemma~\ref{L:masked-continuous-wave}.
\end{proof}

These estimates concern covariance forms and temporal factors. Their application
to displacement and velocity is in Section~\ref{S:wave-variance}; the subsequent
operator, source and moment results require their separate proofs. They do not
follow from covariance domination alone.

\section{The unmasked Rayleigh rate and nonnegative controls}
\label{S:unmasked-variational}

Fix a nonzero covariance satisfying Hypothesis~\ref{H:homog}, including its
original amplitude, and set $a = \alpha_0$. Define the full class
\begin{equation}\label{E:variational-positive-class}
  \mathcal{C}_{\mathrm{pos}}
  =\{\Psi\in C_c^\infty((0,1)\times\R^d;\R):
  \Psi\ge0,\ \Psi\ne0\}.
\end{equation}

\input{notes/figures/nonseparable-controls}
It includes nonseparable tests. It is nonempty: if $b(z) = e^{-1/(1-z^2)}$ on
$|z| < 1$ and zero otherwise, then $b(4s-2)\prod_{j=1}^db(x_j)$ belongs to this
class. All boundary derivatives vanish because exponential decay dominates every
power of $(1-z^2)^{-1}$. For each $\Psi$ retain the original mean $h_{\Psi,s}$
and original variance $v_\Psi = \E W(\Psi)^2$ from~\eqref{E:compact-raw-mean}--%
\eqref{E:compact-raw-cost}. Neither the test nor its covariance is changed by a
physical mask. We write $h_s$ when $\Psi$ is fixed.

\subsection{Global bounds and the unmasked spectral bottom}

We first record a bound for the covariance smoothed by the heat kernel. The
proof of Lemma~\ref{L:variational-global-mean} uses it to bound
$\Gamma*|D\Psi(t,\cdot)|$ uniformly in $t,x$.

\LeanMarker{L:variational-heat-envelope}
\begin{lemma}[Gaussian envelope for the original covariance]
  \label{L:variational-heat-envelope}
  For the normalized heat kernel $p_\epsilon(x) =
  (2\pi\epsilon)^{-d/2}e^{-|x|^2/(2\epsilon)}$,
  \begin{equation}\label{E:variational-heat-envelope}
    (\Gamma*p_\epsilon)(x)
    =\int e^{i\xi\cdot x}e^{-\epsilon|\xi|^2/2}\mu(\ud\xi),
    \qquad
    0\le(\Gamma*p_\epsilon)(x)
    \le(\Gamma*p_\epsilon)(0)<\infty.
  \end{equation}
  The value at zero is strictly positive.
\end{lemma}

\begin{proof}
  A positive tempered Radon measure has polynomial ball growth. Indeed, evaluate
  its tempered bound on a nonnegative smooth cutoff equal to one on $B_1$,
  dilated by $R \ge 1$. Its finitely many Schwartz seminorms are bounded by
  $CR^N$, so positivity gives a ball-mass bound $CR^N$. Summing dyadic annuli
  shows that $(1+|x|)^{-L}$ is integrable for $L > N$. Apply this separately to
  $\Gamma$ and $\mu$. Gaussian weights are therefore integrable against both
  measures. The classical convolution on the left is finite and smooth locally
  in $x$: on compact $x$ sets its translated Gaussian derivatives are bounded by
  any fixed sufficiently fast polynomial weight in the integration variable.

  Pair the convolution with a Schwartz test $\varphi$. Absolute Fubini is
  justified by these weights. The resulting pairing is that of $\Gamma$ with
  $p_\epsilon*\varphi$. Applying the spectral convention in
  Hypothesis~\ref{H:homog} and the identity $\int p_\epsilon(y)e^{-i\xi\cdot
  y}\ud y = e^{-\epsilon|\xi|^2/2}$ gives exactly the right side
  of~\eqref{E:variational-heat-envelope} as a distribution. The heat kernel is
  even, so either Fourier sign gives that same Gaussian multiplier. Its spectral
  integral is absolutely finite and continuous in $x$ by dominated convergence.
  Thus, equality of the two continuous distributions gives equality at every
  $x$, without an extra normalization factor. The physical convolution is
  nonnegative. Evenness of $\mu$ makes the spectral integral real, and its
  absolute value is bounded by its total Gaussian mass, the value at zero.
  Finally, $\Gamma \ne 0$ and $p_\epsilon > 0$ give strict positivity there.
  This proves Lemma~\ref{L:variational-heat-envelope}.
\end{proof}

\LeanMarker{L:variational-global-mean}
\begin{lemma}[Global regularity of the original mean]
  \label{L:variational-global-mean}
  Every mixed time and space derivative of $h_s$ is globally bounded in space,
  uniformly on compact time intervals. In particular, for $I = [0,1]$ there are
  finite constants $M,C_\Psi$ such that
  \begin{equation}\label{E:variational-global-mean}
    0\le h_s(x)\le M,\qquad
    \|h_s-h_t\|_\infty\le C_\Psi|s-t|,
    \qquad s,t\in I.
  \end{equation}
\end{lemma}

\begin{proof}
  Extend $\Psi$ by zero in time; its compact support inside $(0,1)$ makes this a
  smooth extension. For any mixed derivative $D\Psi$, a fixed $\epsilon > 0$ and
  compactness of its common spatial support give $|D\Psi(t,x)| \le
  C_Dp_\epsilon(x)$ for all $t,x$. Here the heat kernel has a positive minimum
  on that support. Lemma~\ref{L:variational-heat-envelope} consequently bounds
  $\Gamma*|D\Psi(t,\cdot)|$ uniformly in $t,x$. Ordinary local differentiation
  against the Radon measure gives the corresponding derivatives of the
  convolution. After the change $r = s-t$,
  \[
    h_s(x)=\int_\R |r|^{-a}
    (\Gamma*\Psi(s-r,\cdot))(x)\ud r.
  \]
  For $s$ in a compact interval the relevant $r$ lie in a fixed bounded
  interval. The preceding global bound times $|r|^{-a}$ is integrable since $a <
  1$. Every derivative thus falls on the smooth test and passes through the
  integral by dominated convergence, with the claimed global spatial bounds.
  Nonnegativity follows from positive kernels. The bounded time derivative
  proves the Lipschitz estimate. No directional decay of the mean has been
  inferred. This proves Lemma~\ref{L:variational-global-mean}.
\end{proof}

Lemma~\ref{L:variational-global-mean} shows that $h_s$ is bounded by $M$ for $s
\in I$ and is Lipschitz in $s$ in the supremum norm. We now use these two bounds
to define the rate $\lambda_{\Psi,s}$ of the potential $h_s$ without a mask and
to show that it is Lipschitz on $I$.

\LeanMarker{P:variational-unmasked-rate}
\begin{proposition}[The unmasked Rayleigh rate]
  \label{P:variational-unmasked-rate}
  On complex $L^2(\R^d)$, the operator $H_s = -\Delta-h_s$ is self-adjoint on
  $H^2$, with form domain $H^1$ and form $q_s(f) = \|\nabla f\|_2^2-\int
  h_s|f|^2$. Its spectral bottom $E_s$ is uniformly negative on $I$. The
  positive rate $\lambda_{\Psi,s} = \sqrt{-E_s}$ satisfies
  \begin{equation}\label{E:variational-real-rayleigh}
    \lambda_{\Psi,s}^2
    =\sup_{\substack{f\in H^1(\R^d;\R)\\\|f\|_2=1}}
    \left(\int h_sf^2-\|\nabla f\|_2^2\right).
  \end{equation}
  It is Lipschitz on $I$, and $I_\Psi: = \int_0^1\lambda_{\Psi,s}\ud s$ is
  positive and finite. No eigenstate at the unmasked bottom is asserted.
\end{proposition}

\begin{proof}
  The Fourier Laplacian $A = -\Delta$ is self-adjoint on $H^2$, being
  multiplication by $|\xi|^2$ with its maximal domain. Multiplication $B_s =
  -h_s$ is bounded self-adjoint with norm at most $M$. Thus, $H_s = A+B_s$ is
  densely defined, symmetric and closed on $H^2$; the two graph norms are
  equivalent under a bounded perturbation. For real $b > M$,
  \[
    H_s-ib=(I+B_s(A-ib)^{-1})(A-ib).
  \]
  The first factor has a Neumann inverse because $\|B_s(A-ib)^{-1}\| \le M/b <
  1$, and the same holds with $+ib$. To check self-adjointness directly, for $u
  \in D(H_s^*)$ solve $(H_s-ib)v = (H_s^*-ib)u$ with $v \in D(H_s)$. Then $u-v
  \in \ker(H_s^*-ib) = \operatorname{Ran}(H_s+ib)^\perp = \{0\}$. Hence,
  $D(H_s^*) = D(H_s)$.

  Set $c_{\mathrm{shift}} = 1+M$ and $J_s = H_s+c_{\mathrm{shift}} \ge 1$.
  Integration by parts on $H^2$ gives
  \[
    \|J_s^{1/2}f\|_2^2=q_s(f)+c_{\mathrm{shift}}\|f\|_2^2,\qquad
    \|f\|_{H^1}^2\le\|J_s^{1/2}f\|_2^2
    \le c_{\mathrm{shift}}\|f\|_{H^1}^2.
  \]
  Spectral cutoffs $1_{[1,n]}(J_s)$ show that $H^2 = D(J_s)$ is a core for the
  half-power norm; Fourier cutoffs give its density in $H^1$. Completion in the
  equivalent norms identifies $D(J_s^{1/2})$ and $H^1$ as $L^2$ subspaces, since
  both norms dominate $L^2$. This proves the stated form domain.

  The spectral lower bound gives one direction of the Rayleigh identity. For the
  reverse, nonzero spectral projections in $[E_s,E_s+\epsilon]$ give unit $H^2$
  vectors with Rayleigh values at most $E_s+\epsilon$. The bottom is finite
  because the operator is bounded below and has finite Rayleigh tests. Let
  $\epsilon\downarrow0$. Real tests give the same infimum: $q_s(u+iv) =
  q_s(u)+q_s(v)$, so the quotient is the norm-weighted average of the nonzero
  real quotients. This proves the Rayleigh identity with $-E_s$ on the left. The
  following negative upper bound justifies writing that value as
  $\lambda_{\Psi,s}^2$ in~\eqref{E:variational-real-rayleigh}.

  Proposition~\ref{P:compact-original-common-trial} supplies, for this original
  $\Psi$, one unit compactly supported $w$ and a finite plateau mask with a
  common strictly negative upper bound $\epsilon_0 > 0$. Its trial has support
  in $B_{2L}$; choose its plateau radius $R_0 \ge 2L$. The unmasked form on this
  same $w$ is no larger. Consequently, $-M \le E_s \le -\epsilon_0 < 0$, and
  $\sqrt{\epsilon_0} \le \lambda_{\Psi,s} \le \sqrt M$. Comparing identical unit
  form tests gives $|E_s-E_t| \le \|h_s-h_t\|_\infty$ and therefore
  \[
    |\lambda_{\Psi,s}-\lambda_{\Psi,t}|
    \le \frac{C_\Psi}{2\sqrt{\epsilon_0}}|s-t|.
  \]
  This also proves positivity and finiteness of $I_\Psi$. No compact-potential
  spectral theorem was applied to $H_s$. This proves
  Proposition~\ref{P:variational-unmasked-rate}.
\end{proof}

\subsection{Removing masks from the numerical lower rate}

Proposition~\ref{P:raw-block-original-lower} gives the lower bound with the rate
of the masked potential $m_Rh_s$, for one fixed mask. We now let the mask radius
$R$ tend to infinity and replace this rate by the rate $\lambda_{\Psi,s}$ of
Proposition~\ref{P:variational-unmasked-rate}.

\LeanMarker{P:variational-remove-mask}
\begin{proposition}[Monotone rates and the original point moment]
  \label{P:variational-remove-mask}
  There is an increasing family of admissible radial masks $m_R$ tending to one,
  for which the compact ground rates satisfy
  $\lambda_{R,s}\uparrow\lambda_{\Psi,s}$ uniformly on $I$ as $R \to \infty$,
  with the compact ground statements used for $R \ge R_0(\Psi)$. For the
  original canonical solution and any fixed $x$,
  \begin{equation}\label{E:variational-unmasked-point-lower}
    \liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,x)|^2
    \ge2I_\Psi-\frac{v_\Psi}{2}.
  \end{equation}
\end{proposition}

\begin{proof}
  Put $F(t) = e^{-1/t}$ for $t > 0$ and zero otherwise, and define
  \[
    \chi(r)=\frac{F(2-r)}{F(2-r)+F(r-1)},\qquad
    m_R(x)=\chi(|x|/R).
  \]
  The denominator is positive. The numerator decreases and the second
  denominator term increases; thus $\chi$ is nonincreasing, equals one on $r \le
  1$, and zero on $r \ge 2$. Flatness of $F$ at zero gives smoothness. Since
  $\chi$ is constant near zero, $m_R$ is smooth also at the spatial origin.
  These even $[0,1]$ masks increase pointwise to one.

  Choose $R_0$ so the preceding common trial lies in the plateau. Every $R \ge
  R_0$ has the same negative margin. For each fixed $R$, the compact-family
  theorem applies to $m_Rh_s$. Its real Rayleigh formula follows from the same
  real/complex quotient argument as~\eqref{E:variational-real-rayleigh}. The
  rates are continuous, positive, at most $\sqrt M$ and monotone in $R$. For
  every fixed real unit $f \in H^1$, dominated convergence gives $\int m_Rh_sf^2
  \to \int h_sf^2$. Hence,
  \[
    \sup_{R\ge R_0}\lambda_{R,s}^2
    =\sup_{\|f\|_2=1,\ f\in H^1(\R^d;\R)}
    \sup_{R\ge R_0}
    \left(\int m_Rh_sf^2-\|\nabla f\|_2^2\right)
    =\lambda_{\Psi,s}^2.
  \]
  Only two numerical suprema were interchanged. Positive square roots give the
  pointwise rate convergence. For $R_n = R_0+n$ and $\epsilon > 0$, the
  increasing open sets $\{s:\lambda_{\Psi,s}-\lambda_{R_n,s}<\epsilon\}$ cover
  compact $I$. A finite subcover lies in the largest one, proving uniform
  convergence. Monotonicity extends it to all real $R \to \infty$. Therefore,
  $\int_I\lambda_{R,s}\ud s\uparrow I_\Psi$.

  At each fixed $R$ use this very same mask in the operator path and mean when
  applying Proposition~\ref{P:raw-block-original-lower}. It gives the lower
  bound with $2\int_I\lambda_{R,s}\ud s-v_\Psi/2$ for the same original point
  liminf. Taking the numerical supremum over these individual bounds proves
  \eqref{E:variational-unmasked-point-lower}. There is no $R(T)$, unmasked noisy
  wave, uniform-in-mask Gaussian estimate or exchange of source and time limits.
  This proves Proposition~\ref{P:variational-remove-mask}.
\end{proof}

\subsection{Full-class dilation and finite variational values for nonnegative controls}

The bound~\eqref{E:variational-unmasked-point-lower} holds for each $\Psi \in
\mathcal{C}_{\mathrm{pos}}$ separately. Before we take the supremum over $\Psi$,
we record how $h_{\Psi,s}$, $v_\Psi$, $\lambda_{\Psi,s}$ and $I_\Psi$ change
under a spatial dilation of the control. The proof of
Proposition~\ref{P:variational-positive-scores} uses this to relate the
variational values for different coefficients $c$.

\LeanMarker{L:variational-control-dilation}
\begin{lemma}[Radon and Rayleigh dilation]\label{L:variational-control-dilation}
  For $r > 0$ the map
  \begin{equation}\label{E:variational-control-dilation}
    (D_r\Psi)(s,x)=r^{d+2-\alpha}\Psi(s,rx)
  \end{equation}
  is a bijection of $\mathcal{C}_{\mathrm{pos}}$, with inverse $D_{1/r}$. With
  the same original covariance,
  \begin{equation}\label{E:variational-dilation-values}
    h_{D_r\Psi,s}(x)=r^2h_{\Psi,s}(rx),\qquad
    v_{D_r\Psi}=r^{4-\alpha}v_\Psi,\qquad
    \lambda_{D_r\Psi,s}=r\lambda_{\Psi,s},\quad
    I_{D_r\Psi}=rI_\Psi.
  \end{equation}
\end{lemma}

\begin{proof}
  Positive dilation preserves the full smooth positive nonzero class;
  multiplication of its amplitude powers gives the stated inverse. Homogeneity
  of the physical Radon measure gives, for every nonnegative measurable $G$,
  \[
    \int G(ry)\Gamma(\ud y)
    =r^{\alpha-d}\int G(z)\Gamma(\ud z).
  \]
  Indeed, its left pushforward measure on a Borel set $A$ is $\Gamma(A/r) =
  r^{\alpha-d}\Gamma(A)$. Positive Tonelli applies this identity inside the
  temporal convolution, giving $h_{D_r\Psi,s}(x) = r^2h_{\Psi,s}(rx)$. Now use
  the \emph{Lebesgue} substitution $z = rx$ in the original pairing:
  \[
    v_{D_r\Psi}=r^{d+2-\alpha}r^2r^{-d}v_\Psi
    =r^{4-\alpha}v_\Psi.
  \]
  This distinguishes the Radon pushforward from the spatial volume factor; it
  includes singular covariances and white noise.

  The map $U_rf(x) = r^{d/2}f(rx)$ is a unitary bijection on real $L^2$,
  preserving $H^1,H^2$ as sets under its inverse. Weak derivatives give
  $\|\nabla U_rf\|_2^2 = r^2\|\nabla f\|_2^2$ and $\int h_{D_r\Psi,s}(U_rf)^2 =
  r^2\int h_{\Psi,s}f^2$. The real Rayleigh bijection proves the squared rate
  identity; the rates are positive, so their square roots and integrals give
  \eqref{E:variational-dilation-values}. On $H^2$ this is also $H_{D_r\Psi,s}U_r
  = r^2U_rH_{\Psi,s}$. Time has not been rescaled. For compact masks the
  corresponding identity uses $m_R(rx) = m_{R/r}(x)$; a fixed mask radius is not
  invariant. This proves Lemma~\ref{L:variational-control-dilation}.
\end{proof}

\LeanMarker{P:variational-positive-scores}
\begin{proposition}[Intrinsic variational lower bounds]
  \label{P:variational-positive-scores}
  For $c > 0$, the set of variational values
  $\{cI_\Psi-v_\Psi/2:\Psi\in\mathcal{C}_{\mathrm{pos}}\}$ is nonempty and
  bounded above. Its finite real supremum $S_c^{\mathrm{pos}}$ is positive and
  satisfies
  \begin{equation}\label{E:variational-score-homogeneity}
    S_c^{\mathrm{pos}}=c^{(4-\alpha)/(3-\alpha)}S_1^{\mathrm{pos}},
    \qquad
    Q_2\ge S_2^{\mathrm{pos}},
  \end{equation}
  where $Q_2$ is the original point liminf. The coefficient-one value is an
  auxiliary variational value; no first-moment or source-constant identification
  is asserted.
\end{proposition}

\begin{proof}
  Write $k = 3-\alpha > 0$, $q = 4-\alpha = k+1$ and $m = q/k > 1$. Fix one
  $\Psi_0$ from the nonempty class and put $I_0 = I_{\Psi_0} > 0$, $v_0 =
  v_{\Psi_0} > 0$. Positivity of the latter is
  Lemma~\ref{L:compact-raw-profile}. For $r_0 = (I_0/v_0)^{1/k} > 0$, which need
  not be less than one, the exact real-power identity gives
  \[
    I_{D_{r_0}\Psi_0}-v_{D_{r_0}\Psi_0}/2
    =r_0I_0-r_0^qv_0/2=r_0I_0/2>0.
  \]
  This positive value is obtained from a smooth nonnegative control.

  For $c > 0$ put $r_c = c^{1/k}$ and $K_c = c^m > 0$.
  Lemma~\ref{L:variational-control-dilation} and the exact scalar powers give,
  for every $\Psi$,
  \begin{equation}\label{E:variational-score-set-scaling}
    cI_{D_{r_c}\Psi}-v_{D_{r_c}\Psi}/2
    =K_c(I_\Psi-v_\Psi/2).
  \end{equation}
  Since $D_{r_c}$ is a bijection of the entire class, the whole set of
  variational values with coefficient $c$ is $K_c$ times the whole
  coefficient-one set of variational values.

  We prove boundedness before taking finite-real suprema. The independent
  Theorem~\ref{T:full-upper}, for the \emph{same canonical solution}, gives
  \[
    \E|u(T,0)|^2\le e(T+2)^d e^{C_{\mathrm{up}}T^\kappa},
    \qquad T\ge1,\quad C_{\mathrm{up}}<\infty.
  \]
  Its prefactor is negligible: choose $\epsilon = \kappa/2 > 0$, use $\log T \le
  (T^\epsilon-1)/\epsilon$ and $\log(T+2) \le \log T+\log3$ for $T \ge 1$, and
  divide by $T^\kappa$. Thus, the original point limsup is at most
  $C_{\mathrm{up}}$. One fixed test/mask lower bound gives a finite lower bound
  for $Q_2$, so $Q_2$ is finite. By~\eqref{E:variational-unmasked-point-lower}
  every coefficient-two variational expression is at most this same $Q_2 \le
  C_{\mathrm{up}}$, uniformly in $\Psi$.
  Equation~\eqref{E:variational-score-set-scaling} at $c = 2$ consequently
  bounds all coefficient-one variational values by $C_{\mathrm{up}}/K_2$. The
  nonempty set has a finite real supremum $S_1^{\mathrm{pos}} > 0$ by the
  positive test above.

  For general $c$, set equality and $K_c > 0$ show that $K_cS_1^{\mathrm{pos}}$
  is the supremum: it is an upper bound, and every smaller number is exceeded by
  $K_c$ times a coefficient-one variational expression within the corresponding
  positive distance of its supremum. This proves finiteness, positivity and the
  first identity in~\eqref{E:variational-score-homogeneity}. Taking the
  numerical supremum of the coefficient-two lower bounds proves the second.
  Although $2^m > 2$, the resulting $S_2^{\mathrm{pos}} > 2S_1^{\mathrm{pos}}$
  is only a comparison of auxiliary variational values. It is not a Jensen
  improvement without a first-moment identification. This proves
  Proposition~\ref{P:variational-positive-scores}.
\end{proof}

For $\Gamma = 0$, the mean, original variance and Rayleigh rate are zero. Set
all these variational values to zero; the free canonical solution $u = 1$ has
the corresponding zero moment rate. Strict positivity above is restricted to
nonzero covariance. Matching an upper variational rate and identifying an exact
pointwise limit remain later steps.

\section{The temporal equilibrium density}
\label{S:temporal-equilibrium}

The density in Lemma~\ref{L:temporal-equilibrium} is the classical interval
Riesz equilibrium density. Its probability-normalized form on $[-1,1]$ is
recalled by Calef and Hardin~\cite{calef.hardin:source2008:riesz}*{p.~1}.
We prove the potential identity with the normalization used here.

We use Euler's function $\Gamma_E(x) = \int_0^\infty u^{x-1}e^{-u}\ud u$, $x >
0$, and $\operatorname{Beta}(x,y) = \int_0^1t^{x-1}(1-t)^{y-1}\ud t$, $x,y > 0$.
Both integrals are positive and finite.

\LeanMarker{L:equilibrium-gamma-beta}
\begin{lemma}[Gamma and Beta identities]\label{L:equilibrium-gamma-beta}
  For positive arguments,
  \begin{align*}
    \Gamma_E(x+1)&=x\Gamma_E(x),\\
    \operatorname{Beta}(x,y)
    &=\frac{\Gamma_E(x)\Gamma_E(y)}{\Gamma_E(x+y)},\\
    \Gamma_E(2x)
    &=\frac{2^{2x-1}\Gamma_E(x)\Gamma_E(x+1/2)}{\sqrt\pi}.
  \end{align*}
\end{lemma}

\begin{proof}
  Integration by parts gives the recurrence, with vanishing boundary terms. In
  the positive product Gamma integral, use $(u,v) = (Rt,R(1-t))$, with Jacobian
  $R$; Tonelli gives the Beta--Gamma identity. For the last identity, symmetry,
  then $t = (1+v)/2$ and $v^2 = w$, give $\operatorname{Beta}(x,x) =
  2^{1-2x}\operatorname{Beta}(1/2,x)$. Also $\Gamma_E(1/2) = \int_\R e^{-z^2}\ud
  z = \sqrt\pi$: the substitution $u = z^2$ gives the first equality, and
  squaring the positive Gaussian integral and using polar coordinates gives
  $2\pi\int_0^\infty re^{-r^2}\ud r = \pi$. Substituting these into the
  Beta--Gamma identity proves duplication. This proves
  Lemma~\ref{L:equilibrium-gamma-beta}.
\end{proof}

\LeanMarker{L:temporal-equilibrium}
\begin{lemma}[Exact potential and mass]\label{L:temporal-equilibrium}
  For $0 < a < 1$ put $r = (1-a)/2$ and, for $0 < t < 1$,
  \begin{equation}\label{E:temporal-equilibrium-density}
    f_a(t)=\frac{[t(1-t)]^{-r}}{\Gamma_E(r)\Gamma_E(1-r)}.
  \end{equation}
  This positive function belongs to $L^1(0,1)\cap L^2(0,1)$, and
  \begin{align}
    \int_0^1|s-t|^{-a}f_a(t)\ud t&=1
    &&\text{for every }s\in[0,1],\label{E:temporal-equilibrium-potential}\\
    M_a:=\int_0^1f_a(t)\ud t
    &=\frac{\Gamma_E((1+a)/2)}
    {\Gamma_E(1+a)\Gamma_E((1-a)/2)}>0.
    \label{E:temporal-equilibrium-mass}
  \end{align}
\end{lemma}

\begin{proof}
  The positive binomial series $(1-z)^{-p} = \sum_{n\ge0}(p)_nz^n/n!$, $p > 0$,
  $0 \le z < 1$, follows directly as follows. Its coefficient ratio gives
  uniform convergence on compact subintervals, including for its derivative.
  Termwise differentiation gives $(1-z)y' = py$, $y(0) = 1$; hence $(1-z)^py$ is
  constant one. Here $(p)_0 = 1$, $(p)_n = p(p+1)\cdots(p+n-1)$. All terms used
  below are nonnegative, so Tonelli applies.

  For $0 < s < 1$ split the integral of $|s-t|^{2r-1}[t(1-t)]^{-r}$ into left
  and right parts. In the left part, $t = su$ gives
  \begin{align*}
    L(s)
    &=s^r\int_0^1(1-u)^{2r-1}u^{-r}(1-su)^{-r}\ud u\\
    &=\operatorname{Beta}(1-r,2r)s^r
    \sum_{n\ge0}
    \frac{(r)_n(1-r)_ns^n}{(1+r)_n n!}.
  \end{align*}
  The Beta ratio follows from the Gamma recurrence. Since $(r)_n/(1+r)_n =
  r/(r+n)$, expanding the incomplete Beta integral gives
  \[
    L(s)=r\operatorname{Beta}(1-r,2r)B_s(r,r),\qquad
    B_s(r,r)=\int_0^s u^{r-1}(1-u)^{r-1}\ud u.
  \]
  Reflection makes the right part $L(1-s)$, while symmetry gives
  $B_s(r,r)+B_{1-s}(r,r) = \operatorname{Beta}(r,r)$. The full integral is
  therefore $r\operatorname{Beta}(1-r,2r)\operatorname{Beta}(r,r) =
  \Gamma_E(r)\Gamma_E(1-r)$. All integrals are finite because $-r > -1$ and
  $2r-1 > -1$. At $s = 0$ the integral is directly $\operatorname{Beta}(r,1-r) =
  \Gamma_E(r)\Gamma_E(1-r)$; $s = 1$ follows by reflection. This proves
  \eqref{E:temporal-equilibrium-potential}, including the endpoints without a
  limiting argument there.

  Integrability and square integrability follow from $-r > -1$ and $-2r = a-1 >
  -1$. Finally,
  \[
    M_a=
    \frac{\operatorname{Beta}(1-r,1-r)}
    {\Gamma_E(r)\Gamma_E(1-r)}
    =\frac{\Gamma_E(1-r)}{\Gamma_E(2-2r)\Gamma_E(r)},
  \]
  which is~\eqref{E:temporal-equilibrium-mass}. Its time mass has not been
  renormalized. This proves Lemma~\ref{L:temporal-equilibrium}.
\end{proof}
