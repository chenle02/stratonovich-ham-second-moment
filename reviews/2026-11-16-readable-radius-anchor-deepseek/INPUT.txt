\section{Bounds for the operator path with explicit radius dependence}
\label{S:radius-stochastic-anchor}

Retain every original covariance and amplitude in Hypothesis~\ref{H:homog}, with
$a = \alpha_0$, and the real phase $E = H^{1/2}\times H^{-1/2}$. Use the fixed
cutoff profile $\eta$ of Section~\ref{S:uniform-mean-regulation} and physical
mask $m_R = \eta(\cdot/R)^2$, $R \ge 1$. In the variance construction the base
mask is $\zeta = \eta^2$. The balanced unitary is $J_0(u,v) =
(\Lambda^{1/2}u,\Lambda^{-1/2}v)$. Fix a finite horizon $B \ge 1$ and parameters
\begin{equation}\label{E:radius-anchor-parameters}
  0<\gamma<\min\left(\frac12,\frac{2-\alpha}{2}\right),\qquad
  \frac12<\theta<\eta_t<\beta_t<
  \min\left(1-\frac a2,\frac{3-a-\alpha}{2}\right).
\end{equation}
The strict subcritical assumptions make these intervals nonempty. Fix one even
integer $Q \ge 2$ such that
\begin{equation}\label{E:radius-anchor-Q}
  Q\gamma>d,\qquad
  \beta_t-\eta_t-\frac1Q>0,\qquad
  Q>\frac{2d}{1-a}.
\end{equation}
Such a $Q$ exists because all three lower thresholds are finite. It stays fixed
as $\delta$ tends to zero. Constants below may depend on the model, $B,Q$, these
exponents and the fixed auxiliary cutoff profiles. They are independent of the
spatial radius, noise-basis truncation, moment index and deterministic residual
projection unless explicitly stated otherwise.

\subsection{Uniform outer factors and the exact whole-space envelope}

Fix real $\chi_0 \in C_c^\infty(\R^d)$, equal to one on $B_1$ and supported in
the closed ball of radius two. Choose real $\psi_0 \in C_c^\infty(\R^d)$, equal
to one on a neighbourhood of $\operatorname{supp}\chi_0$ and supported inside
$B_3$. For $\rho \ge 1$ put $\chi_\rho(x) = \chi_0(x/\rho)$, $\psi_\rho(x) =
\psi_0(x/\rho)$ and $L = 2\pi\rho$. Since $3 < \pi$, both cutoffs are supported
strictly inside the cube of side $L$. The auxiliary torus has unnormalized
$L^2(Q_L,\ud x)$, orthonormal modes $L^{-d/2}e^{ik\cdot x/\rho}$ and Sobolev
weights $\langle k/\rho\rangle^r$.

\begin{lemma}[Uniform rescaled outer maps]\label{L:radius-outer-maps}
  The cutoff periodization and recovery maps from
  Lemma~\ref{L:cutoff-factorization} satisfy, for every real $r$,
  \[
    \|\mathcal{P}_{\psi_\rho}:H^r(\R^d)\to H^r(\mathbb{T}_L)\|
    \le C_{\psi_0,r},\qquad
    \|\mathcal{E}_{\chi_\rho}:H^r(\mathbb{T}_L)\to H^r(\R^d)\|
    \le C_{\chi_0,r},
  \]
  independently of $\rho \ge 1$.
\end{lemma}

\begin{proof}
  The exact weighted Fourier kernels are
  \begin{align*}
    &L^{-d/2}
    \left(\frac{\langle k/\rho\rangle}{\langle\xi\rangle}\right)^r
    \widehat{\psi_\rho}(k/\rho-\xi),\\
    &L^{-d/2}
    \left(\frac{\langle\xi\rangle}{\langle k/\rho\rangle}\right)^r
    \widehat{\chi_\rho}(\xi-k/\rho).
  \end{align*}
  There is no additional $(2\pi)^{-d/2}$: it cancels between Fourier inversion
  and the ordinary integral in the first kernel, while the torus normalization
  gives the second. For either base profile $\phi_0$, $\widehat{\phi_\rho}(t) =
  \rho^d\widehat{\phi_0}(\rho t)$. Peetre's inequality~\eqref{E:cutoff-peetre},
  also for negative $r$, and $\langle(\rho\xi-k)/\rho\rangle \le
  \langle\rho\xi-k\rangle$ bound the relevant kernel by
  \[
    C_r\rho^{d/2}F_r(k-\rho\xi),\qquad
    F_r(t)=|\widehat{\phi_0}(\pm t)|\langle t\rangle^{|r|}.
  \]
  The nonnegative $F_r$ is integrable and bounded by arbitrarily high inverse
  powers of $\langle t\rangle$. At fixed $k$, integration in $\xi$ gives a bound
  $C\rho^{-d/2}$. At fixed $\xi$, summation in $k$ gives $C\rho^{d/2}$:
  translate $\rho\xi$ into a unit lattice cell and sum a power tail of order
  greater than $d$, as in~\eqref{E:cutoff-lattice-sum}. This lattice spacing is
  one and has no remaining $\rho$ dependence. The mixed Schur estimate bounds
  the operator norm by the square root of the product of those two quantities.
  Their radius powers cancel. For recovery the row and column roles are
  reversed.

  Bounded kernel extension from Schwartz functions and torus polynomials gives
  the full Sobolev maps by density. Their distributional meanings are the same
  cutoff maps as in Lemma~\ref{L:cutoff-factorization}; no point sampling of a
  rough Fourier function is used. This includes negative indices. This proves
  Lemma~\ref{L:radius-outer-maps}.
\end{proof}

Lemma~\ref{L:radius-outer-maps} bounds the two outer maps independently of
$\rho \ge 1$. The proof below places a diagonal inclusion on the torus between
them. We now sum the $q$th powers of the singular values of this inclusion and
take the $q$th root, which produces the factor $\rho^{d/q}$.

\begin{proposition}[Radius-explicit compact envelopes]
  \label{P:radius-compact-envelopes}
  For real $\sigma$, $\gamma > 0$ and $q \ge 1$ with $q\gamma > d$,
  \begin{equation}\label{E:radius-cutoff-Schatten}
    \|\Lambda^\sigma\chi_\rho\Lambda^{-\sigma-\gamma}\|_{S_q}
    \le C_{\sigma,\gamma,q,d,\chi_0,\psi_0}\rho^{d/q}.
  \end{equation}
  In the balanced coordinates of~\eqref{E:compact-two-factors},
  \begin{equation}\label{E:radius-two-envelopes}
    \|K_{\rm in}\|_{S_q}=\|K_{\rm out}\|_{S_q}
    \le C_q\rho^{d/q}.
  \end{equation}
\end{proposition}

\begin{proof}
  The exact recovery identity, with $\chi_\rho\psi_\rho = \chi_\rho$, gives
  \[
    \Lambda^\sigma\chi_\rho\Lambda^{-\sigma-\gamma}
    =\Lambda^\sigma\mathcal{E}_{\chi_\rho}\mathcal{I}_\gamma
    \mathcal{P}_{\psi_\rho}\Lambda^{-\sigma-\gamma}.
  \]
  Here $\mathcal{I}_\gamma:H^{\sigma+\gamma}(\mathbb{T}_L) \to
  H^\sigma(\mathbb{T}_L)$ has diagonal singular values $\langle
  k/\rho\rangle^{-\gamma}$. The outer Bessel maps are unitary on their indicated
  Sobolev domains. The preceding lemma bounds both other outer maps uniformly in
  $\rho$. This is an equality on the full whole-space domain by the recovery
  proof and density, including negative $\sigma$.

  Put $u = q\gamma > d$. If $x \in k+[-1/2,1/2]^d$, then
  \[
    \langle x/\rho\rangle
    \le\sqrt2\langle k/\rho\rangle\langle(x-k)/\rho\rangle
    \le c_d'\langle k/\rho\rangle,\qquad
    c_d'=\sqrt{2(1+d/4)}.
  \]
  Integrate over the unit cubes to obtain
  \begin{equation}\label{E:radius-lattice-volume}
    \sum_{k\in\mathbb{Z}^d}\langle k/\rho\rangle^{-u}
    \le(c_d')^u\rho^d\int_{\R^d}\langle y\rangle^{-u}\ud y
    \le C_{d,u}\rho^d .
  \end{equation}
  The integral is finite: dyadic annuli outside the unit ball give a geometric
  series with terms bounded by $C_d2^{j(d-u)}$.

  Arrange the inclusion's diagonal values in decreasing order $(\lambda_j)$.
  They tend to zero. Retaining its first $j-1$ coordinates has rank less than
  $j$ and error $\lambda_j$. Compose this approximation with the bounded outer
  factors. The singular-value approximation argument in
  Proposition~\ref{P:cutoff-singular-values} gives $s_j \le
  C_{\chi_0,\sigma}C_{\psi_0,\sigma+\gamma}\lambda_j$. Finite-rank approximation
  also proves compactness. Summing $q$th powers and
  using~\eqref{E:radius-lattice-volume} proves~\eqref{E:radius-cutoff-Schatten}.
  No separate Schatten ideal theorem is needed.

  The two blocks of $K_{\rm out} = P_{\chi_\rho}D_\gamma$ are the whole-space
  operators with $\sigma = 1/2,-1/2$. Their $q$th power norms add, proving its
  bound. Proposition~\ref{P:compact-two-envelopes} gives $K_{\rm in} = V K_{\rm
  out}V$ for the unitary coordinate swap $V(z_1,z_2) = (z_2,z_1)$, so their
  norms agree. In~\eqref{E:radius-anchor-parameters}, $\gamma < 1/2$ and
  $q\gamma > d$ force $q > 2d \ge 2$; there is no Hilbert--Schmidt conclusion.
  Neither $P_{\chi_\rho}$ nor its adjoint is assumed selfadjoint or idempotent.
  This proves Proposition~\ref{P:radius-compact-envelopes}.
\end{proof}

\subsection{The operator path and covariance contractions}

Write $M_R(W) = A_{m_R}(W)$ for the $E$ interaction operator path starting at
zero on $[0,B]$ from Proposition~\ref{P:actual-operator-driver}. The phase
remains the whole-space $E$ throughout.

\begin{proposition}[Radius-explicit path moments]
  \label{P:radius-actual-moments}
  Under~\eqref{E:radius-anchor-parameters}--\eqref{E:radius-anchor-Q}, for every
  finite $v \ge 1$,
  \begin{equation}\label{E:radius-actual-path-moments}
    \|M_R(W)\|_{L^v(C^\theta([0,B];\mathcal{L}(E)))}
    \le C_{Q,B}\sqrt v\,R^{d/Q},\qquad R\ge1.
  \end{equation}
  The constant does not depend on $v$ or $R$. This is an $E$ operator-path
  estimate; the two constant columns on $F$ require their own estimates. Any
  later projection used in a Cameron comparison must also satisfy its
  orthogonality and covariance hypotheses.
\end{proposition}

\begin{proof}
  Use the given smooth-test basis from Lemma~\ref{L:driver-noise-basis}, $g_j =
  W(f_j)$, and its finite paths $M_N = \sum_{j\le N}g_jA_j$. These coefficient
  paths use the same given noise as in Lemma~\ref{L:driver-holder-limit}, not
  source approximations. Choose
  \[
    \rho=2R+B+2\le(B+4)R,\qquad \chi=\chi_0(\cdot/\rho).
  \]
  The plateau covers $B_{2R+B+1}$. Thus, both explicit localization identities
  \eqref{E:driver-actual-localization} hold for this envelope. The physical
  noise mask remains $m_R$. Both envelopes $K_{\rm in},K_{\rm out}$ here use
  this same $\chi_\rho$.

  For an interval $I \subset [0,B]$ of length $0 < h \le 1$,
  Proposition~\ref{P:wave-two-variances} and
  Lemma~\ref{L:driver-actual-variances} give both finite weighted variances with
  bound $V_{B,\mathrm{var}}h^{2-a}B_\alpha(h)$, uniformly in $R,N,I$. In
  particular, the base mask is the same fixed $\eta^2$ required by that variance
  theorem. Corollary~\ref{C:finite-gaussian-operators}, applied in the balanced
  coordinates, gives for every even $q \ge Q$
  \[
    \|M_N(I)\|_{L^q(S_q)}
    \le\sqrt{q-1}\sqrt{V_{B,\mathrm{var}}}\,
    h^{1-a/2}B_\alpha(h)^{1/2}\|K_{\rm out}\|_{S_q},
  \]
  where $B_\alpha(h)$ is $1$, $1+\log(1/h)$ or $h^{1-\alpha}$ according as
  $\alpha < 1$, $\alpha = 1$ or $\alpha > 1$. Both variances and both
  localizations have been checked; individual basis-dependent coefficient bounds
  are not summed.

  For $q \ge Q$, the unnormalized Schatten norms obey $\|K_{\rm out}\|_{S_q} \le
  \|K_{\rm out}\|_{S_Q}$. Indeed, normalize a nonzero singular-value sequence by
  its $\ell^Q$ norm; each entry is at most one, so its $q$th power sum is at
  most its $Q$th power sum. The zero case is immediate.
  Apply~\eqref{E:radius-two-envelopes} only at the fixed index $Q$. Use $\|A\|
  \le \|A\|_{S_q}$ and $\sqrt{q-1} \le \sqrt q$. The balanced Hilbert
  complexification has the same operator norm as its real restriction, and
  $\rho^{d/Q} \le (B+4)^{d/Q}R^{d/Q}$. Since $\beta_t$ is strictly below the
  corresponding short-time power, it follows that
  \begin{equation}\label{E:radius-finite-increment}
    \|M_N(t)-M_N(u)\|_{L^q(\mathcal{L}(E))}
    \le C_{Q,B}\sqrt q\,R^{d/Q}|t-u|^{\beta_t},
    \qquad |t-u|\le1.
  \end{equation}
  At $\alpha = 1$, the factor $\sqrt{1+\log(1/h)}$ is absorbed by the strict gap
  $1-a/2-\beta_t > 0$. For $\alpha > 1$, the power is $(3-a-\alpha)/2$. These
  power/log constants are independent of $R,N,q$. Longer increments are finite
  sums of increments of length at most one, with constants depending only on the
  fixed $B$.

  Let $D_n(F)$ be the largest adjacent increment on the grid $kB2^{-n}$ of
  $[0,B]$. For fine levels $B2^{-n} \le 1$, the finite-maximum bound gives
  \[
    \|D_n(M_N)\|_{L^q}
    \le2^{n/q}C_{Q,B}\sqrt q\,R^{d/Q}(B2^{-n})^{\beta_t}.
  \]
  No independence of increments is used. The sum after multiplication by
  $2^{n\eta_t}$ converges uniformly for even $q \ge Q$, because its geometric
  ratio is at most $2^{-(\beta_t-\eta_t-1/Q)} < 1$. The finitely many coarse
  levels are controlled by unit partitions. The deterministic dyadic chaining
  argument in Lemma~\ref{L:driver-holder-limit} therefore bounds the
  $L^qC^{\eta_t}$ norms by $C_{Q,B}\sqrt q\,R^{d/Q}$. Each finite path is
  continuous and starts at zero, as required there. The fixed-horizon embedding
  $C^{\eta_t}\hookrightarrow C^\theta$ gives the same estimate in $C^\theta$.

  For each fixed $R$, Lemma~\ref{L:driver-holder-limit} identifies $M_R(W)$ as
  the all-finite-moment $C^\theta$ limit of these very $M_N$. Pass the uniform
  finite-$N$ estimate to this limit. Uniform convergence in $R$ is unnecessary:
  the limiting bound has its explicit radius dependence at every $R$. For
  arbitrary $v \ge 1$, choose an even $q \ge \max(v,Q)$ with $q \le v+Q+2 \le
  (Q+3)v$. Lyapunov's inequality gives~\eqref{E:radius-actual-path-moments},
  absorbing $\sqrt{Q+3}$ into the fixed constant. This proves
  Proposition~\ref{P:radius-actual-moments}.
\end{proof}

Proposition~\ref{P:radius-actual-moments} concerns the path $M_R(W)$ of the
noise $W$ itself. Proposition~\ref{P:radius-mean-anchor} works instead with
residual noises $Y_m$, obtained from $W$ through a deterministic orthogonal
projection. We now transfer the moment bound to such noises.

\begin{corollary}[Deterministic covariance residuals]
  \label{C:radius-covariance-residuals}
  Let $Y$ be a centered Gaussian linear noise on the coupled space, with
  $\operatorname{Var}Y(f) \le C\|f\|_{H_W}^2$, $C \ge 0$. Its transferred
  operator path satisfies
  \[
    \|M_R(Y)\|_{L^v(C^\theta)}
    \le\sqrt C\,C_{Q,B}\sqrt v\,R^{d/Q}.
  \]
  In particular, this holds with $C = 1$ for $Y_m(f) = W((I-\Pi)T_{\delta,m}f)$,
  for any deterministic orthogonal projection $\Pi$ on the slow $H_W$. Its rank
  may be infinite and it may depend on $\delta,R$.
\end{corollary}

\begin{proof}
  Proposition~\ref{P:source-gaussian-transfer} is a linear Gaussian covariance
  contraction theorem for this same physical mask and these same deterministic
  coefficient maps. It bounds the path norm by $\sqrt C$ times the reference
  isonormal operator-path norm. The reference has the canonical operator-path
  law used in Proposition~\ref{P:radius-actual-moments}; the $Y$ is not replaced
  in this comparison. The zero case gives zero directly. This proves the first
  assertion. For the second, the original isometry \eqref{E:raw-block-pullback},
  proved in Lemma~\ref{L:raw-block-inputs}, and $\|I-\Pi\| \le 1$ give
  $\operatorname{Var}Y_m(f) \le \|f\|_{H_W}^2$, where the right side is the
  original norm in fast coordinates. The joint residuals retain the given slow
  $W$ and may be correlated. They need not lie in the span of an individual fast
  noise, which is not a hypothesis of Gaussian transfer. No random
  noise-dependent projection is used. This proves
  Corollary~\ref{C:radius-covariance-residuals}.
\end{proof}

\subsection{The growing-radius expected-norm anchor}

Corollary~\ref{C:radius-covariance-residuals} bounds each residual path with
the growing factor $R^{d/Q}$. We now average over the time blocks with the
amplitude $g_\delta$. The last condition in~\eqref{E:radius-anchor-Q} makes
$g_\delta R^{d/Q}$ tend to zero at the radius $R_\delta = \delta^{-1}+2$.

\LeanMarker{P:radius-mean-anchor}
\begin{proposition}[Averaged block mean tends to zero]
  \label{P:radius-mean-anchor}
  Fix integer $B \ge 1$ and use the true block partition $B,\delta,N,\ell_m$ of
  Section~\ref{S:actual-cameron-mean}, with $0 < \delta < \min(1,1/B)$ and $N =
  \lfloor1/(\delta B)\rfloor$. For $Y_m$ as in
  Corollary~\ref{C:radius-covariance-residuals}, put
  \[
    X_m(u)=g_\delta M_R(Y_m)(\min(u,\ell_m)),\qquad
    Z_\delta=\left(\delta\sum_{m=0}^N
    \|X_m\|_{C^\theta}^p\right)^{1/p},\qquad
    g_\delta=\delta^{(1-a)/2},
  \]
  for any fixed $p \ge 1$. Then
  \begin{equation}\label{E:radius-mean-anchor}
    \E Z_\delta\le(2/B)^{1/p}C_{Q,B}\sqrt p\,
    g_\delta R^{d/Q}.
  \end{equation}
  In particular, for $R_\delta = \delta^{-1}+2$,
  \begin{equation}\label{E:radius-mean-anchor-decay}
    2\E Z_\delta
    \le2(2/B)^{1/p}C_{Q,B}\sqrt p\,3^{d/Q}
    \delta^{e_{\rm anchor}}\longrightarrow0,\qquad
    e_{\rm anchor}=\frac{1-a}{2}-\frac dQ>0 .
  \end{equation}
  Both bounds are uniform in the deterministic orthogonal projection.
\end{proposition}

\begin{proof}
  Stopping an interaction primitive starting at zero and extending it constantly
  to $B$ does not increase its supremum or H\"older seminorm. An empty last
  interval gives the zero path. Probability normalization and Lyapunov give
  \begin{align*}
    \E Z_\delta
    &\le(\E Z_\delta^p)^{1/p}
    =\left(\delta\sum_{m=0}^N
    \E\|X_m\|_{C^\theta}^p\right)^{1/p}\\
    &\le[\delta(N+1)]^{1/p}g_\delta
    C_{Q,B}\sqrt p\,R^{d/Q}.
  \end{align*}
  This is linearity for a finite sum; no time-block independence or maximum over
  blocks is used. Since $\delta B < 1$, we have $N \ge 1$ and $N+1 \le 2/(\delta
  B)$, proving~\eqref{E:radius-mean-anchor}.

  Now $R_\delta \le 3/\delta$ because $\delta < 1$. The last condition
  in~\eqref{E:radius-anchor-Q} is equivalent to $Q(1-a) > 2d$. Dividing by $2Q >
  0$ gives $e_{\rm anchor} > 0$. Substitute this radius
  into~\eqref{E:radius-mean-anchor} to
  obtain~\eqref{E:radius-mean-anchor-decay}. The estimate holds at each fixed
  $B,p,Q$ and model, uniformly in every deterministic residual projection. This
  proves Proposition~\ref{P:radius-mean-anchor}.
\end{proof}

These bounds concern the expected operator-path norm on $E$. They do not give an
$F$ estimate with constant initial-data columns. The radius substitution is
numerical; cone coverage and source-to-point transfer require their separate
arguments. No concentration, integration over an infinite sequence of Gaussian
coordinates or nonlinear moment upper bound follows from this section alone. The
constants need not be uniform as $Q,B$ tend to infinity or as model parameters
approach an endpoint. For zero covariance, the operator paths and stochastic
bounds vanish directly.

\section{Bounds for the operator path with explicit radius dependence}
\label{S:radius-stochastic-anchor}

Retain every original covariance and amplitude in Hypothesis~\ref{H:homog}, with
$a = \alpha_0$, and the real phase $E = H^{1/2}\times H^{-1/2}$. Use the fixed
cutoff profile $\eta$ of Section~\ref{S:uniform-mean-regulation} and physical
mask $m_R = \eta(\cdot/R)^2$, $R \ge 1$. In the variance construction the base
mask is $\zeta = \eta^2$. The balanced unitary is $J_0(u,v) =
(\Lambda^{1/2}u,\Lambda^{-1/2}v)$. Fix a finite horizon $B \ge 1$ and parameters
\begin{equation}\label{E:radius-anchor-parameters}
  0<\gamma<\min\left(\frac12,\frac{2-\alpha}{2}\right),\qquad
  \frac12<\theta<\eta_t<\beta_t<
  \min\left(1-\frac a2,\frac{3-a-\alpha}{2}\right).
\end{equation}
The strict subcritical assumptions make these intervals nonempty. Fix one even
integer $Q \ge 2$ such that
\begin{equation}\label{E:radius-anchor-Q}
  Q\gamma>d,\qquad
  \beta_t-\eta_t-\frac1Q>0,\qquad
  Q>\frac{2d}{1-a}.
\end{equation}
Such a $Q$ exists because all three lower thresholds are finite. It stays fixed
as $\delta$ tends to zero. Constants below may depend on the model, $B,Q$, these
exponents and the fixed auxiliary cutoff profiles. They are independent of the
spatial radius, noise-basis truncation, moment index and deterministic residual
projection unless explicitly stated otherwise.

\subsection{Uniform outer factors and the exact whole-space envelope}

Fix real $\chi_0 \in C_c^\infty(\R^d)$, equal to one on $B_1$ and supported in
the closed ball of radius two. Choose real $\psi_0 \in C_c^\infty(\R^d)$, equal
to one on a neighbourhood of $\operatorname{supp}\chi_0$ and supported inside
$B_3$. For $\rho \ge 1$ put $\chi_\rho(x) = \chi_0(x/\rho)$, $\psi_\rho(x) =
\psi_0(x/\rho)$ and $L = 2\pi\rho$. Since $3 < \pi$, both cutoffs are supported
strictly inside the cube of side $L$. The auxiliary torus has unnormalized
$L^2(Q_L,\ud x)$, orthonormal modes $L^{-d/2}e^{ik\cdot x/\rho}$ and Sobolev
weights $\langle k/\rho\rangle^r$.

\begin{lemma}[Uniform rescaled outer maps]\label{L:radius-outer-maps}
  The cutoff periodization and recovery maps from
  Lemma~\ref{L:cutoff-factorization} satisfy, for every real $r$,
  \[
    \|\mathcal{P}_{\psi_\rho}:H^r(\R^d)\to H^r(\mathbb{T}_L)\|
    \le C_{\psi_0,r},\qquad
    \|\mathcal{E}_{\chi_\rho}:H^r(\mathbb{T}_L)\to H^r(\R^d)\|
    \le C_{\chi_0,r},
  \]
  independently of $\rho \ge 1$.
\end{lemma}

\begin{proof}
  We rescale the exact Fourier kernels and bound their integral and
  lattice-sum norms separately. The radius powers cancel in the mixed Schur
  estimate, after which density gives the full Sobolev maps.

  The exact weighted Fourier kernels are
  \begin{align*}
    &L^{-d/2}
    \left(\frac{\langle k/\rho\rangle}{\langle\xi\rangle}\right)^r
    \widehat{\psi_\rho}(k/\rho-\xi),\\
    &L^{-d/2}
    \left(\frac{\langle\xi\rangle}{\langle k/\rho\rangle}\right)^r
    \widehat{\chi_\rho}(\xi-k/\rho).
  \end{align*}
  There is no additional $(2\pi)^{-d/2}$: it cancels between Fourier inversion
  and the ordinary integral in the first kernel, while the torus normalization
  gives the second. For either base profile $\phi_0$, $\widehat{\phi_\rho}(t) =
  \rho^d\widehat{\phi_0}(\rho t)$. Peetre's inequality~\eqref{E:cutoff-peetre},
  also for negative $r$, and $\langle(\rho\xi-k)/\rho\rangle \le
  \langle\rho\xi-k\rangle$ bound the relevant kernel by
  \[
    C_r\rho^{d/2}F_r(k-\rho\xi),\qquad
    F_r(t)=|\widehat{\phi_0}(\pm t)|\langle t\rangle^{|r|}.
  \]
  The nonnegative $F_r$ is integrable and bounded by arbitrarily high inverse
  powers of $\langle t\rangle$. At fixed $k$, integration in $\xi$ gives a bound
  $C\rho^{-d/2}$. At fixed $\xi$, summation in $k$ gives $C\rho^{d/2}$:
  translate $\rho\xi$ into a unit lattice cell and sum a power tail of order
  greater than $d$, as in~\eqref{E:cutoff-lattice-sum}. This lattice spacing is
  one and has no remaining $\rho$ dependence. The mixed Schur estimate bounds
  the operator norm by the square root of the product of those two quantities.
  Their radius powers cancel. For recovery the row and column roles are
  reversed.

  In the integral bound, the substitution $\rho\xi-k$ contributes $\rho^{-d}$.
  Multiplying by the kernel's factor $\rho^{d/2}$ gives $\rho^{-d/2}$. In the
  sum bound, the
  same variable has unit lattice spacing, so there is no density-of-lattice
  factor depending on $\rho$. Their product is therefore independent of radius
  before its square root is taken.
  Bounded kernel extension from Schwartz functions and torus polynomials gives
  the full Sobolev maps by density. Their distributional meanings are the same
  cutoff maps as in Lemma~\ref{L:cutoff-factorization}; no point sampling of a
  rough Fourier function is used. This includes negative indices. This proves
  Lemma~\ref{L:radius-outer-maps}.
\end{proof}

Lemma~\ref{L:radius-outer-maps} bounds the two outer maps independently of
$\rho \ge 1$. The proof below places a diagonal inclusion on the torus between
them. We now sum the $q$th powers of the singular values of this inclusion and
take the $q$th root, which produces the factor $\rho^{d/q}$.

\begin{proposition}[Radius-explicit compact envelopes]
  \label{P:radius-compact-envelopes}
  For real $\sigma$, $\gamma > 0$ and $q \ge 1$ with $q\gamma > d$,
  \begin{equation}\label{E:radius-cutoff-Schatten}
    \|\Lambda^\sigma\chi_\rho\Lambda^{-\sigma-\gamma}\|_{S_q}
    \le C_{\sigma,\gamma,q,d,\chi_0,\psi_0}\rho^{d/q}.
  \end{equation}
  In the balanced coordinates of~\eqref{E:compact-two-factors},
  \begin{equation}\label{E:radius-two-envelopes}
    \|K_{\rm in}\|_{S_q}=\|K_{\rm out}\|_{S_q}
    \le C_q\rho^{d/q}.
  \end{equation}
\end{proposition}

\begin{proof}
  We factor the cutoff through the auxiliary torus. We compare the diagonal
  inclusion sum with a Euclidean integral and then use finite-rank
  approximations to transfer that sum to the whole-space operator.


  \medskip
  \noindent\textit{Step 1.~}We use the full-domain recovery factorization.
  The exact recovery identity, with $\chi_\rho\psi_\rho = \chi_\rho$, gives
  \[
    \Lambda^\sigma\chi_\rho\Lambda^{-\sigma-\gamma}
    =\Lambda^\sigma\mathcal{E}_{\chi_\rho}\mathcal{I}_\gamma
    \mathcal{P}_{\psi_\rho}\Lambda^{-\sigma-\gamma}.
  \]
  Here $\mathcal{I}_\gamma:H^{\sigma+\gamma}(\mathbb{T}_L) \to
  H^\sigma(\mathbb{T}_L)$ has diagonal singular values $\langle
  k/\rho\rangle^{-\gamma}$. The outer Bessel maps are unitary on their indicated
  Sobolev domains. The preceding lemma bounds both other outer maps uniformly in
  $\rho$. This is an equality on the full whole-space domain by the recovery
  proof and density, including negative $\sigma$.


  \medskip
  \noindent\textit{Step 2.~}We bound the diagonal sum by a volume integral.
  Put $u = q\gamma > d$. If $x \in k+[-1/2,1/2]^d$, then
  \[
    \langle x/\rho\rangle
    \le\sqrt2\langle k/\rho\rangle\langle(x-k)/\rho\rangle
    \le c_d'\langle k/\rho\rangle,\qquad
    c_d'=\sqrt{2(1+d/4)}.
  \]
  Integrate over the unit cubes to obtain
  \begin{equation}\label{E:radius-lattice-volume}
    \sum_{k\in\mathbb{Z}^d}\langle k/\rho\rangle^{-u}
    \le(c_d')^u\rho^d\int_{\R^d}\langle y\rangle^{-u}\ud y
    \le C_{d,u}\rho^d .
  \end{equation}
  The integral is finite: dyadic annuli outside the unit ball give a geometric
  series with terms bounded by $C_d2^{j(d-u)}$.

  The unit cubes partition Euclidean space and each has volume one. The
  preceding weight comparison bounds the constant cube value $\langle
  k/\rho\rangle^{-u}$ by $(c_d')^u\langle x/\rho\rangle^{-u}$. Summing the
  cube integrals and substituting $y=x/\rho$ gives the factor $\rho^d$. Taking
  a $q$th root later gives $\rho^{d/q}$.

  \medskip
  \noindent\textit{Step 3.~}We transfer the singular-value bound to the two
  whole-space envelopes.
  Arrange the inclusion's diagonal values in decreasing order $(\lambda_j)$.
  They tend to zero. Retaining its first $j-1$ coordinates has rank less than
  $j$ and error $\lambda_j$. Compose this approximation with the bounded outer
  factors. The singular-value approximation argument in
  Proposition~\ref{P:cutoff-singular-values} gives $s_j \le
  C_{\chi_0,\sigma}C_{\psi_0,\sigma+\gamma}\lambda_j$. Finite-rank approximation
  also proves compactness. Summing $q$th powers and
  using~\eqref{E:radius-lattice-volume} proves~\eqref{E:radius-cutoff-Schatten}.
  No separate Schatten ideal theorem is needed.

  The two blocks of $K_{\rm out} = P_{\chi_\rho}D_\gamma$ are the whole-space
  operators with $\sigma = 1/2,-1/2$. Their $q$th power norms add, proving its
  bound. Proposition~\ref{P:compact-two-envelopes} gives $K_{\rm in} = V K_{\rm
  out}V$ for the unitary coordinate swap $V(z_1,z_2) = (z_2,z_1)$, so their
  norms agree. In~\eqref{E:radius-anchor-parameters}, $\gamma < 1/2$ and
  $q\gamma > d$ force $q > 2d \ge 2$; there is no Hilbert--Schmidt conclusion.
  Neither $P_{\chi_\rho}$ nor its adjoint is assumed selfadjoint or idempotent.
  This proves Proposition~\ref{P:radius-compact-envelopes}.
\end{proof}

\subsection{The operator path and covariance contractions}

Write $M_R(W) = A_{m_R}(W)$ for the $E$ interaction operator path starting at
zero on $[0,B]$ from Proposition~\ref{P:actual-operator-driver}. The phase
remains the whole-space $E$ throughout.

\begin{proposition}[Radius-explicit path moments]
  \label{P:radius-actual-moments}
  Under~\eqref{E:radius-anchor-parameters}--\eqref{E:radius-anchor-Q}, for every
  finite $v \ge 1$,
  \begin{equation}\label{E:radius-actual-path-moments}
    \|M_R(W)\|_{L^v(C^\theta([0,B];\mathcal{L}(E)))}
    \le C_{Q,B}\sqrt v\,R^{d/Q},\qquad R\ge1.
  \end{equation}
  The constant does not depend on $v$ or $R$. This is an $E$ operator-path
  estimate; the two constant columns on $F$ require their own estimates. Any
  later projection used in a Cameron comparison must also satisfy its
  orthogonality and covariance hypotheses.
\end{proposition}

\begin{proof}
  We keep the physical noise mask fixed while choosing a larger localization
  envelope. We use only one fixed Schatten index to extract the radius factor,
  then apply the dyadic path estimate uniformly over all higher moment
  indices.


  \medskip
  \noindent\textit{Step 1.~}We choose a localization envelope covering the
  propagated support.
  Use the given smooth-test basis from Lemma~\ref{L:driver-noise-basis}, $g_j =
  W(f_j)$, and its finite paths $M_N = \sum_{j\le N}g_jA_j$. These coefficient
  paths use the same given noise as in Lemma~\ref{L:driver-holder-limit}, not
  source approximations. Choose
  \[
    \rho=2R+B+2\le(B+4)R,\qquad \chi=\chi_0(\cdot/\rho).
  \]
  The plateau covers $B_{2R+B+1}$. Thus, both explicit localization identities
  \eqref{E:driver-actual-localization} hold for this envelope. The physical
  noise mask remains $m_R$. Both envelopes $K_{\rm in},K_{\rm out}$ here use
  this same $\chi_\rho$.


  \medskip
  \noindent\textit{Step 2.~}We estimate increments using both variance bounds.
  For an interval $I \subset [0,B]$ of length $0 < h \le 1$,
  Proposition~\ref{P:wave-two-variances} and
  Lemma~\ref{L:driver-actual-variances} give both finite weighted variances with
  bound $V_{B,\mathrm{var}}h^{2-a}B_\alpha(h)$, uniformly in $R,N,I$. In
  particular, the base mask is the same fixed $\eta^2$ required by that variance
  theorem. Corollary~\ref{C:finite-gaussian-operators}, applied in the balanced
  coordinates, gives for every even $q \ge Q$
  \[
    \|M_N(I)\|_{L^q(S_q)}
    \le\sqrt{q-1}\sqrt{V_{B,\mathrm{var}}}\,
    h^{1-a/2}B_\alpha(h)^{1/2}\|K_{\rm out}\|_{S_q},
  \]
  where $B_\alpha(h)$ is $1$, $1+\log(1/h)$ or $h^{1-\alpha}$ according as
  $\alpha < 1$, $\alpha = 1$ or $\alpha > 1$. Both variances and both
  localizations have been checked; individual basis-dependent coefficient bounds
  are not summed.


  \medskip
  \noindent\textit{Step 3.~}We extract the radius factor at the fixed index.
  For $q \ge Q$, the unnormalized Schatten norms obey $\|K_{\rm out}\|_{S_q} \le
  \|K_{\rm out}\|_{S_Q}$. Indeed, normalize a nonzero singular-value sequence by
  its $\ell^Q$ norm; each entry is at most one, so its $q$th power sum is at
  most its $Q$th power sum. The zero case is immediate.
  Apply~\eqref{E:radius-two-envelopes} only at the fixed index $Q$. Use $\|A\|
  \le \|A\|_{S_q}$ and $\sqrt{q-1} \le \sqrt q$. The balanced Hilbert
  complexification has the same operator norm as its real restriction, and
  $\rho^{d/Q} \le (B+4)^{d/Q}R^{d/Q}$. Since $\beta_t$ is strictly below the
  corresponding short-time power, it follows that
  \begin{equation}\label{E:radius-finite-increment}
    \|M_N(t)-M_N(u)\|_{L^q(\mathcal{L}(E))}
    \le C_{Q,B}\sqrt q\,R^{d/Q}|t-u|^{\beta_t},
    \qquad |t-u|\le1.
  \end{equation}
  At $\alpha = 1$, the factor $\sqrt{1+\log(1/h)}$ is absorbed by the strict gap
  $1-a/2-\beta_t > 0$. For $\alpha > 1$, the power is $(3-a-\alpha)/2$. These
  power/log constants are independent of $R,N,q$. Longer increments are finite
  sums of increments of length at most one, with constants depending only on the
  fixed $B$.

  Using the envelope estimate only at $Q$ is what prevents its constant from
  varying with the moment index $q$. The inequality between the unnormalized
  Schatten norms retains the fixed factor $R^{d/Q}$ for every $q\ge Q$. The
  only remaining moment dependence in the Gaussian estimate is the explicit
  $\sqrt q$.

  \medskip
  \noindent\textit{Step 4.~}We sum dyadic increments and pass to the path
  limit.
  Let $D_n(F)$ be the largest adjacent increment on the grid $kB2^{-n}$ of
  $[0,B]$. For fine levels $B2^{-n} \le 1$, the finite-maximum bound gives
  \[
    \|D_n(M_N)\|_{L^q}
    \le2^{n/q}C_{Q,B}\sqrt q\,R^{d/Q}(B2^{-n})^{\beta_t}.
  \]
  No independence of increments is used. The sum after multiplication by
  $2^{n\eta_t}$ converges uniformly for even $q \ge Q$, because its geometric
  ratio is at most $2^{-(\beta_t-\eta_t-1/Q)} < 1$. The finitely many coarse
  levels are controlled by unit partitions. The deterministic dyadic chaining
  argument in Lemma~\ref{L:driver-holder-limit} therefore bounds the
  $L^qC^{\eta_t}$ norms by $C_{Q,B}\sqrt q\,R^{d/Q}$. Each finite path is
  continuous and starts at zero, as required there. The fixed-horizon embedding
  $C^{\eta_t}\hookrightarrow C^\theta$ gives the same estimate in $C^\theta$.

  For each fixed $R$, Lemma~\ref{L:driver-holder-limit} identifies $M_R(W)$ as
  the all-finite-moment $C^\theta$ limit of these very $M_N$. Pass the uniform
  finite-$N$ estimate to this limit. Uniform convergence in $R$ is unnecessary:
  the limiting bound has its explicit radius dependence at every $R$. For
  arbitrary $v \ge 1$, choose an even $q \ge \max(v,Q)$ with $q \le v+Q+2 \le
  (Q+3)v$. Lyapunov's inequality gives~\eqref{E:radius-actual-path-moments},
  absorbing $\sqrt{Q+3}$ into the fixed constant. This proves
  Proposition~\ref{P:radius-actual-moments}.
\end{proof}

Proposition~\ref{P:radius-actual-moments} concerns the path $M_R(W)$ of the
noise $W$ itself. Proposition~\ref{P:radius-mean-anchor} works instead with
residual noises $Y_m$, obtained from $W$ through a deterministic orthogonal
projection. We now transfer the moment bound to such noises.

\begin{corollary}[Deterministic covariance residuals]
  \label{C:radius-covariance-residuals}
  Let $Y$ be a centered Gaussian linear noise on the coupled space, with
  $\operatorname{Var}Y(f) \le C\|f\|_{H_W}^2$, $C \ge 0$. Its transferred
  operator path satisfies
  \[
    \|M_R(Y)\|_{L^v(C^\theta)}
    \le\sqrt C\,C_{Q,B}\sqrt v\,R^{d/Q}.
  \]
  In particular, this holds with $C = 1$ for $Y_m(f) = W((I-\Pi)T_{\delta,m}f)$,
  for any deterministic orthogonal projection $\Pi$ on the slow $H_W$. Its rank
  may be infinite and it may depend on $\delta,R$.
\end{corollary}

\begin{proof}
  Proposition~\ref{P:source-gaussian-transfer} is a linear Gaussian covariance
  contraction theorem for this same physical mask and these same deterministic
  coefficient maps. It bounds the path norm by $\sqrt C$ times the reference
  isonormal operator-path norm. The reference has the canonical operator-path
  law used in Proposition~\ref{P:radius-actual-moments}; the $Y$ is not replaced
  in this comparison. The zero case gives zero directly. This proves the first
  assertion. For the second, the original isometry \eqref{E:raw-block-pullback},
  proved in Lemma~\ref{L:raw-block-inputs}, and $\|I-\Pi\| \le 1$ give
  $\operatorname{Var}Y_m(f) \le \|f\|_{H_W}^2$, where the right side is the
  original norm in fast coordinates. The joint residuals retain the given slow
  $W$ and may be correlated. They need not lie in the span of an individual fast
  noise, which is not a hypothesis of Gaussian transfer. No random
  noise-dependent projection is used. This proves
  Corollary~\ref{C:radius-covariance-residuals}.
\end{proof}

\subsection{The growing-radius expected-norm anchor}

Corollary~\ref{C:radius-covariance-residuals} bounds each residual path with
the growing factor $R^{d/Q}$. We now average over the time blocks with the
amplitude $g_\delta$. The last condition in~\eqref{E:radius-anchor-Q} makes
$g_\delta R^{d/Q}$ tend to zero at the radius $R_\delta = \delta^{-1}+2$.

\LeanMarker{P:radius-mean-anchor}
\begin{proposition}[Averaged block mean tends to zero]
  \label{P:radius-mean-anchor}
  Fix integer $B \ge 1$ and use the true block partition $B,\delta,N,\ell_m$ of
  Section~\ref{S:actual-cameron-mean}, with $0 < \delta < \min(1,1/B)$ and $N =
  \lfloor1/(\delta B)\rfloor$. For $Y_m$ as in
  Corollary~\ref{C:radius-covariance-residuals}, put
  \[
    X_m(u)=g_\delta M_R(Y_m)(\min(u,\ell_m)),\qquad
    Z_\delta=\left(\delta\sum_{m=0}^N
    \|X_m\|_{C^\theta}^p\right)^{1/p},\qquad
    g_\delta=\delta^{(1-a)/2},
  \]
  for any fixed $p \ge 1$. Then
  \begin{equation}\label{E:radius-mean-anchor}
    \E Z_\delta\le(2/B)^{1/p}C_{Q,B}\sqrt p\,
    g_\delta R^{d/Q}.
  \end{equation}
  In particular, for $R_\delta = \delta^{-1}+2$,
  \begin{equation}\label{E:radius-mean-anchor-decay}
    2\E Z_\delta
    \le2(2/B)^{1/p}C_{Q,B}\sqrt p\,3^{d/Q}
    \delta^{e_{\rm anchor}}\longrightarrow0,\qquad
    e_{\rm anchor}=\frac{1-a}{2}-\frac dQ>0 .
  \end{equation}
  Both bounds are uniform in the deterministic orthogonal projection.
\end{proposition}

\begin{proof}
  Stopping an interaction primitive starting at zero and extending it constantly
  to $B$ does not increase its supremum or H\"older seminorm. An empty last
  interval gives the zero path. Probability normalization and Lyapunov give
  \begin{align*}
    \E Z_\delta
    &\le(\E Z_\delta^p)^{1/p}
    =\left(\delta\sum_{m=0}^N
    \E\|X_m\|_{C^\theta}^p\right)^{1/p}\\
    &\le[\delta(N+1)]^{1/p}g_\delta
    C_{Q,B}\sqrt p\,R^{d/Q}.
  \end{align*}
  This is linearity for a finite sum; no time-block independence or maximum over
  blocks is used. Since $\delta B < 1$, we have $N \ge 1$ and $N+1 \le 2/(\delta
  B)$, proving~\eqref{E:radius-mean-anchor}.

  The finite expectation sum carries $\delta(N+1)$, rather than a factor from
  a maximum. The inequality $N+1\le2/(\delta B)$ reduces it to $2/B$ before
  the radius is substituted. Thus, no independence assumption is needed to
  offset the increasing number of blocks.
  Now $R_\delta \le 3/\delta$ because $\delta < 1$. The last condition
  in~\eqref{E:radius-anchor-Q} is equivalent to $Q(1-a) > 2d$. Dividing by $2Q >
  0$ gives $e_{\rm anchor} > 0$. Substitute this radius
  into~\eqref{E:radius-mean-anchor} to
  obtain~\eqref{E:radius-mean-anchor-decay}. The estimate holds at each fixed
  $B,p,Q$ and model, uniformly in every deterministic residual projection. This
  proves Proposition~\ref{P:radius-mean-anchor}.
\end{proof}

These bounds concern the expected operator-path norm on $E$. They do not give an
$F$ estimate with constant initial-data columns. The radius substitution is
numerical; cone coverage and source-to-point transfer require their separate
arguments. No concentration, integration over an infinite sequence of Gaussian
coordinates or nonlinear moment upper bound follows from this section alone. The
constants need not be uniform as $Q,B$ tend to infinity or as model parameters
approach an endpoint. For zero covariance, the operator paths and stochastic
bounds vanish directly.

\section{Model and standing assumptions}\label{S:model}

We study the hyperbolic Anderson model
\cite{chen:author2026:time-dependency}*{(1.1)}
\begin{equation}\label{E:HAM}
  \frac{\partial^2 u}{\partial t^2}(t,x)=\Delta u(t,x)+\W(t,x)u(t,x),
  \qquad u(0,x)=1,\quad \frac{\partial u}{\partial t}(0,x)=0,
\end{equation}
for $(t,x) \in \R_+\times\R^d$, $d \in \{1,2,3\}$, interpreted in the
Stratonovich sense (limit of classical wave equations driven by smoothed noises
$\W_{\epsilon,\delta}$). The noise is centered Gaussian with
\begin{equation}\label{E:cov}
  \operatorname{Cov}\big(\W(s,x),\W(t,y)\big)=|s-t|^{-\alpha_0}\gamma(x-y),
  \qquad 0<\alpha_0<1,
\end{equation}
where $\gamma \ge 0$ is nonnegative definite with spectral measure $\mu$.

\begin{hypothesis}[Homogeneous covariance]\label{H:homog}
  Let $d \in \{1,2,3\}$ and $0 < \alpha_0 < 1$. The spatial covariance is a
  nonnegative, even, positive-definite tempered Radon measure $\Gamma$ on
  $\R^d$, with positive even tempered Radon spectral measure $\mu$ in the
  convention
  \[
    \Gamma=\int e^{i\xi\cdot x}\mu(\ud\xi)
    \quad\text{in the sense of tempered distributions}.
  \]
  The symbol $\gamma$ in~\eqref{E:cov} denotes this covariance as a function
  when one exists, and as a distribution otherwise. For every Borel set $A$ and
  $c > 0$, $\Gamma(cA) = c^{d-\alpha}\Gamma(A)$, and
  \begin{equation}\label{E:threshold}
    \alpha_0+\alpha<2,
  \end{equation}
  with $0 < \alpha < d$, or $\alpha = d = 1$. In particular, the last case
  includes spatial white noise $\Gamma = \delta_0$.
\end{hypothesis}

\paragraph{Zero covariance.}
Hypothesis~\ref{H:homog} allows $\Gamma = 0$ by design. In this case the
centered Gaussian noise is $\W = 0$, and the solution with initial data
$(1,0)$ is $u(t,x) = 1$. Thus $\E|u(t,x)|^p = 1$ for every $p > 0$, and
all normalized logarithmic moment rates are zero. Results asserting strict
positivity of covariance-dependent quantities impose $\Gamma \ne 0$ locally.

\section{Finite Gaussian moments and whole-space Sobolev compactness}
\label{S:compact-gaussian}

We prove the finite Gaussian and compact-cutoff estimates needed to construct
the wave operator path. All traces and Schatten norms in this section are
unnormalized. The spatial states remain on $\R^d$, $d \in \{1,2,3\}$. Here $p$
is an auxiliary Schatten and Gaussian-moment index. The auxiliary periodic space
below is used only to factor a compactly supported multiplication operator. No
infinite Gaussian series or time path is constructed in this section.

\subsection{Finite Gaussian matrices with both variance powers}

\LeanMarker{L:gaussian-scalar-powers}
\begin{lemma}[Scalar power estimates]\label{L:gaussian-scalar-powers}
  For $0 < \theta < 1$ and $u,v \ge 0$,
  \begin{equation}\label{E:gaussian-weighted-power}
    u^\theta v^{1-\theta} \le \theta u+(1-\theta)v.
  \end{equation}
  Consequently, if $L > 0$ and $0 \le q \le L$ are integers, then for real
  $x,y$,
  \begin{equation}\label{E:gaussian-integer-power}
    |x|^q|y|^{L-q}
    \le \frac qL|x|^L+\frac{L-q}L|y|^L.
  \end{equation}
  For nonnegative finite sequences, the usual H\"older inequality holds with
  exponents $m$ and $m/(m-1)$, $m > 1$. If $Z \ge 0$ has finite expectation $F$
  and $0 < r < 1$, then $\E Z^r \le F^r$.
\end{lemma}

\begin{proof}
  If $v > 0$, divide~\eqref{E:gaussian-weighted-power} by $v$ and put $t = u/v$.
  The function $\theta t+1-\theta-t^\theta$ decreases to zero at one and then
  increases, by its derivative $\theta(1-t^{\theta-1})$. Continuity covers $t =
  0$; if $v = 0$ the inequality is immediate. For $0 < q < L$, apply this with
  $u = |x|^L$, $v = |y|^L$ and $\theta = q/L$. The endpoint cases $q = 0,L$ are
  identities, with the ordinary integer convention $a^0 = 1$.

  To prove finite H\"older, divide the two sequences by their respective
  $\ell^m$ and $\ell^{m/(m-1)}$ norms. If either norm vanishes, the result is
  immediate. Apply~\eqref{E:gaussian-weighted-power} with $\theta = 1/m$ to the
  corresponding normalized powers and sum; the sum of products is at most one.
  Undo the normalization. Finally, $F = 0$ gives $Z = 0$ almost surely. If $F >
  0$, apply \eqref{E:gaussian-weighted-power} to $u = Z/F$, $v = 1$ and $\theta
  = r$, and integrate. The right side has expectation one. This proves
  Lemma~\ref{L:gaussian-scalar-powers}.
\end{proof}

\LeanMarker{L:gaussian-trace-bounds}
\begin{lemma}[Two Hermitian trace bounds]\label{L:gaussian-trace-bounds}
  Let $B,Y$ be finite Hermitian matrices, $m \ge 2$ an integer, and $L = 2m-2$.
  For integers $0 \le q \le L$,
  \begin{equation}\label{E:gaussian-trace-word}
    \operatorname{Re}\operatorname{Tr}(BY^qBY^{L-q})
    \le \operatorname{Tr}(B^2|Y|^L).
  \end{equation}
  If $D \ge 0$ is Hermitian, then
  \begin{equation}\label{E:gaussian-trace-holder}
    \operatorname{Tr}(D|Y|^{2m-2})
    \le (\operatorname{Tr}D^m)^{1/m}
    (\operatorname{Tr}|Y|^{2m})^{(m-1)/m}.
  \end{equation}
\end{lemma}

\begin{proof}
  Diagonalize $Y$, with real eigenvalues $\lambda_i$, and write $b_{ij}$ for the
  entries of $B$ in that basis. Hermitian symmetry gives $b_{ji} =
  \overline{b_{ij}}$. The trace word is the real sum
  \[
    \sum_{i,j}|b_{ij}|^2\lambda_j^q\lambda_i^{L-q}.
  \]
  Bound each product by its absolute value and use
  \eqref{E:gaussian-integer-power}. After interchanging $i,j$, the two resulting
  sums agree because $|b_{ij}| = |b_{ji}|$. Each equals
  $\sum_{i,j}|b_{ij}|^2|\lambda_i|^L = \operatorname{Tr}(B^2|Y|^L)$. The two
  weights sum to one, proving \eqref{E:gaussian-trace-word}. No commutation of
  $B$ and $Y$ is used.

  In the same basis put $d_i = D_{ii} \ge 0$. If $\theta_j$ are the nonnegative
  eigenvalues of $D$, a unitary change of basis gives $d_i =
  \sum_j|U_{ij}|^2\theta_j$. Both row and column sums of $|U_{ij}|^2$ equal one.
  Finite convexity therefore yields
  \[
    \sum_i d_i^m \le \sum_{i,j}|U_{ij}|^2\theta_j^m
    = \operatorname{Tr}D^m.
  \]
  For completeness, the convexity step follows from $t^m \ge s^m+ms^{m-1}(t-s)$
  for $s,t \ge 0$, obtained by differentiating the difference. Average this
  inequality with $s$ equal to the weighted mean. Now apply finite H\"older from
  Lemma~\ref{L:gaussian-scalar-powers} to $\sum_i d_i|\lambda_i|^{2m-2}$. Since
  $(2m-2)m/(m-1) = 2m$, the last display proves \eqref{E:gaussian-trace-holder},
  including zero sums. This proves Lemma~\ref{L:gaussian-trace-bounds}.
\end{proof}

The estimate below is a finite-dimensional Gaussian noncommutative Khintchine
inequality. For its origins in the work of Lust-Piquard and a Gaussian
integration-by-parts proof, see Tropp~\cite{tropp:source2018:second-order}*{arXiv:1504.05919v2,
Proposition~2.3 and Section~7}. We include the proof and Hermitian dilation
to fix the normalization and retain both variance operators.

\LeanMarker{P:finite-gaussian-matrix}
\begin{proposition}[Finite Gaussian matrix moment]\label{P:finite-gaussian-matrix}
  Let $A_1,\ldots,A_N$ be complex $o$-by-$n$ matrices, with $o,n,N \ge 1$, and
  let $g_k$ be independent real standard Gaussians. Put
  \[
    X=\sum_k g_kA_k,\qquad V_{\rm out}=\sum_k A_kA_k^*,
    \qquad V_{\rm in}=\sum_k A_k^*A_k.
  \]
  For even $p = 2m \ge 2$, with $\|A\|_{S_p}^p = \operatorname{Tr}((A^*A)^m)$,
  \begin{align}
    (\E\|X\|_{S_p}^p)^{1/p}
    &\le \sqrt{p-1}\left(
    \frac{\operatorname{Tr}V_{\rm out}^m+
    \operatorname{Tr}V_{\rm in}^m}{2}\right)^{1/p}
    \label{E:finite-gaussian-average}\\
    &\le \sqrt{p-1}\max\left(
    \|V_{\rm out}^{1/2}\|_{S_p},\|V_{\rm in}^{1/2}\|_{S_p}\right).
    \label{E:finite-gaussian-maximum}
  \end{align}
  The constants are independent of $o,n,N$. The first inequality is an equality
  at $p = 2$. Empty or zero families give the zero identity.
\end{proposition}

\begin{proof}
  First suppose $B_k$ are Hermitian and put $Y = \sum_k g_kB_k$ and $D = \sum_k
  B_k^2$. All polynomial moments are finite because Gaussian decay dominates
  every fixed power. Componentwise integration by parts against the product
  Gaussian density gives $\E[g_kP(g)] = \E[\partial_kP(g)]$, with zero boundary
  terms. It applies to complex polynomials through their real and imaginary
  parts; absolute polynomial integrability justifies Fubini.

  For $m \ge 2$, put $F = \E\operatorname{Tr}Y^{2m}$ and $L = 2m-2$. Write
  $\operatorname{Tr}Y^{2m} = \sum_k g_k \operatorname{Tr}(B_kY^{2m-1})$ and
  differentiate the $2m-1$ factors in their order. This gives exactly
  \[
    F=\sum_k\sum_{q=0}^{L}
    \E\operatorname{Tr}(B_kY^qB_kY^{L-q}).
  \]
  The eigenbasis calculation above shows these traces are real. The
  differentiated polynomial is real by conjugation and trace cyclicity. No
  derivative of a spectral absolute value or measurable choice of an eigenbasis
  is needed. The two trace bounds give
  \begin{align*}
    F&\le (2m-1)\E\operatorname{Tr}(D|Y|^{2m-2})\\
    &\le (2m-1)(\operatorname{Tr}D^m)^{1/m}
    \E(\operatorname{Tr}|Y|^{2m})^{(m-1)/m}\\
    &\le (2m-1)(\operatorname{Tr}D^m)^{1/m}F^{(m-1)/m}.
  \end{align*}
  Here $\operatorname{Tr}|Y|^{2m} = \operatorname{Tr}Y^{2m}$, and the last
  inequality is the expectation bound in Lemma~\ref{L:gaussian-scalar-powers}.
  If $F = 0$ the conclusion is immediate. Otherwise divide by its positive
  fractional power and raise to $m$ to obtain
  \begin{equation}\label{E:finite-hermitian-gaussian}
    \E\operatorname{Tr}Y^{2m} \le (2m-1)^m\operatorname{Tr}D^m.
  \end{equation}
  For $m = 1$, independence gives directly $\E\operatorname{Tr}Y^2 =
  \sum_k\operatorname{Tr}B_k^2 = \operatorname{Tr}D$, proving the same formula
  as equality.

  For rectangular $A_k$, use the Hermitian dilation
  \[
    B_k=\begin{pmatrix}0&A_k\\A_k^*&0\end{pmatrix},\qquad
    Y=\begin{pmatrix}0&X\\X^*&0\end{pmatrix},\qquad
    D=\operatorname{diag}(V_{\rm out},V_{\rm in}).
  \]
  Finite entrywise summation proves $\operatorname{Tr}(UV) =
  \operatorname{Tr}(VU)$ for compatible rectangular matrices. Since $(XX^*)^m =
  X(X^*X)^{m-1}X^*$, it follows that
  \[
    \operatorname{Tr}Y^{2m}=2\operatorname{Tr}((X^*X)^m),\qquad
    \operatorname{Tr}D^m=\operatorname{Tr}V_{\rm out}^m+
    \operatorname{Tr}V_{\rm in}^m.
  \]
  Substitute in~\eqref{E:finite-hermitian-gaussian}, divide by two and take the
  $p$th root. This proves~\eqref{E:finite-gaussian-average}. For positive $V$,
  spectral diagonalization gives $\|V^{1/2}\|_{S_p}^p = \operatorname{Tr}V^m$,
  and the average of two nonnegative numbers is bounded by their maximum. At $p
  = 2$, $\E\operatorname{Tr}X^*X = \operatorname{Tr}V_{\rm in} =
  \operatorname{Tr}V_{\rm out}$, proving the claimed equality. This proves
  Proposition~\ref{P:finite-gaussian-matrix}.
\end{proof}

Both variance directions are needed: the row-matrix example in
No-go~\ref{P:nogo-23} rules out a dimension-independent estimate that
keeps only one of them.


\LeanMarker{R:gaussian-one-variance-test}
\begin{remark}[Both variance directions are necessary]
  \label{R:gaussian-one-variance-test}
  Let $n = 16$ and $A_k = e_k^{\mathsf{T}}$ be the coordinate row matrices. The
  only nonzero singular value of $X$ is $(\sum_k g_k^2)^{1/2}$. Gaussian
  integration by parts gives $\E g^2 = 1$, $\E g^4 = 3$, and hence
  \[
    \E\|X\|_{S_4}^4=3n+n(n-1)=n(n+2)=288.
  \]
  Here $V_{\rm out} = n$ and $V_{\rm in} = I_n$. Keeping only the input variance
  in~\eqref{E:finite-gaussian-maximum} would give fourth power
  $9\operatorname{Tr}I_n^2 = 144$, which is false. Transposing the family
  refutes the corresponding output-only bound. The average and maximum bounds
  have fourth powers $1224$ and $2304$, respectively. This finite example
  asserts nothing about a specialized wave estimate.
\end{remark}

\subsection{Whole-space cutoffs through an auxiliary periodic space}

Use the unitary Fourier transform with $c_d = (2\pi)^{-d/2}$ and $\Lambda =
(1-\Delta)^{1/2}$. For real $r$, the norm on $H^r(\R^d)$ is
$\|\langle\xi\rangle^r\widehat{u}\|_2$. The deterministic cutoff arguments allow
any smoothing order $\gamma > 0$. Fix real $\sigma$ and a real compact smooth
$\chi$. Singular values $s_j$, $j \ge 1$, are counted with multiplicity, with
zeros appended when necessary. For compact $K$, put $\|K\|_{S_p}^p = \sum_j
s_j(K)^p$.

\begin{lemma}[Cutoff factorization]\label{L:cutoff-factorization}
  Choose a cube $Q_L = (-L/2,L/2)^d$ containing $\operatorname{supp}\chi$ in its
  interior, and a real $\psi \in C_c^\infty(Q_L)$ equal to one near
  $\operatorname{supp}\chi$. On the auxiliary torus $\mathbb{T}_L$ use Lebesgue
  measure and the orthonormal basis
  \[
    e_k(x)=L^{-d/2}e^{i\kappa_k\cdot x},\qquad
    \kappa_k=2\pi k/L,\qquad k \in \mathbb{Z}^d,
  \]
  with $\|v\|_{H^r(\mathbb{T}_L)}^2 = \sum_k\langle\kappa_k\rangle^{2r}|v_k|^2$.
  Periodization of $\psi u$ and multiplication of a periodic function by $\chi$
  extend to bounded maps
  \begin{equation}\label{E:cutoff-periodic-maps}
    \mathcal{P}_\psi:H^r(\R^d)\to H^r(\mathbb{T}_L),\qquad
    \mathcal{E}_\chi:H^r(\mathbb{T}_L)\to H^r(\R^d)
  \end{equation}
  for every real $r$. If $\mathcal{I}_\gamma$ denotes the identity inclusion
  from $H^{\sigma+\gamma}(\mathbb{T}_L)$ to $H^\sigma(\mathbb{T}_L)$, then
  \begin{equation}\label{E:cutoff-whole-space-recovery}
    \mathcal{E}_\chi\mathcal{I}_\gamma\mathcal{P}_\psi u=\chi u,
    \qquad u \in H^{\sigma+\gamma}(\R^d).
  \end{equation}
\end{lemma}

\begin{proof}
  For $x,y \in \R^d$ and real $r$, the elementary inequality $\langle x\rangle
  \le \sqrt{2}\langle y\rangle\langle x-y\rangle$ gives
  \begin{equation}\label{E:cutoff-peetre}
    \left(\frac{\langle x\rangle}{\langle y\rangle}\right)^r
    \le 2^{|r|/2}\langle x-y\rangle^{|r|}.
  \end{equation}
  For negative $r$, interchange $x,y$ before raising to $|r|$. Integration by
  parts gives rapid decay of the Fourier transform of every compact smooth
  cutoff. For any lattice spacing $a > 0$ and $N > d$,
  \begin{equation}\label{E:cutoff-lattice-sum}
    \sup_y\sum_{k\in\mathbb{Z}^d}\langle y-ak\rangle^{-N}<\infty.
  \end{equation}
  Indeed, translate $y$ into a centered lattice cell, where $|y| \le a\sqrt
  d/2$. For $|k| \ge \sqrt d$ we have $|y-ak| \ge a|k|/2$. The remaining terms
  are finite; the tail converges by dyadic lattice counting, since there are at
  most $C_d2^{jd}$ lattice points in the $j$th shell in $k$. More precisely, for
  $|k| \ge \sqrt d$, the preceding inequality gives
  \[
    \langle y-ak\rangle^{-N}
    \le \max(1,2/a)^N\langle k\rangle^{-N}.
  \]
  The comparison-series terms are bounded by $2^{-jN}$ on each dyadic $k$-shell.
  Thus, its convergent geometric sum carries constants depending on $a,d,N$, as
  does~\eqref{E:cutoff-lattice-sum}.

  We use the mixed Schur bound for a kernel whose absolute row and column
  integrals are bounded by $M_1,M_2$. Weighted Cauchy--Schwarz at each output
  point, followed by Tonelli, bounds its squared operator norm by $M_1M_2$. The
  same proof works with counting measure on one side and Lebesgue measure on the
  other, first on dense finite or compact inputs.

  For Schwartz $u$, the $k$th coefficient of its cutoff periodization is
  \[
    (\mathcal{P}_\psi u)_k
    = L^{-d/2}\int\psi(x)u(x)e^{-i\kappa_k\cdot x}\ud x
    = L^{-d/2}\int\widehat{\psi}(\kappa_k-\xi)\widehat{u}(\xi)\ud\xi.
  \]
  The $c_d$ from Fourier inversion cancels the $c_d^{-1}$ in the ordinary
  Fourier integral of $\psi$. After weighting, its exact kernel is
  \[
    L^{-d/2}\left(\frac{\langle\kappa_k\rangle}{\langle\xi\rangle}\right)^r
    \widehat{\psi}(\kappa_k-\xi).
  \]
  For a finite torus polynomial $v$, the normalized Fourier kernel of
  $\mathcal{E}_\chi v$ is
  \[
    L^{-d/2}\left(\frac{\langle\xi\rangle}{\langle\kappa_k\rangle}\right)^r
    \widehat{\chi}(\xi-\kappa_k).
  \]
  There is no additional $c_d$ in this kernel. By~\eqref{E:cutoff-peetre}, each
  is bounded in absolute value by a rapidly decreasing function of its frequency
  difference, times a fixed constant. One Schur quantity is its $L^1$ integral
  and the other a uniformly translated lattice sum, bounded
  by~\eqref{E:cutoff-lattice-sum}. This proves~\eqref{E:cutoff-periodic-maps},
  including negative $r$.

  Smooth compact approximation in weighted Fourier $L^2$ proves Schwartz density
  on $\R^d$; weighted sequence truncation proves polynomial density on the
  torus. Hence, these maps extend uniquely. Their distributional meanings follow
  by testing against compact smooth functions and density: each periodic
  coefficient pairs the distribution with a compact smooth test. No point
  sampling of a rough Fourier function is used. The same full-Lebesgue Schur
  argument proves smooth compact multiplication bounded on every whole-space
  $H^r$.

  For Schwartz $u$, the periodic Fourier series of $\psi u$ converges smoothly
  on the cube. Since $\chi\psi = \chi$ and $\chi$ is supported strictly inside
  the cube, multiplying this series by $\chi$ recovers $\chi u$ with no alias
  term. Both sides are bounded from $H^{\sigma+\gamma}(\R^d)$ to
  $H^\sigma(\R^d)$, using the just-proved maps, the inclusion and smooth
  multiplication. Density proves \eqref{E:cutoff-whole-space-recovery} on that
  full domain. This proves Lemma~\ref{L:cutoff-factorization}.
\end{proof}

\begin{proposition}[Whole-space cutoff singular values]\label{P:cutoff-singular-values}
  For real $\sigma$ and $\gamma > 0$, the initially Schwartz-defined operator
  $K_{\chi,\sigma,\gamma} = \Lambda^\sigma\chi\Lambda^{-\sigma-\gamma}$ extends
  uniquely to a compact operator on $L^2(\R^d)$, and
  \begin{equation}\label{E:cutoff-singular-values}
    s_j(K_{\chi,\sigma,\gamma})
    \le C_{\chi,\sigma,\gamma,d}j^{-\gamma/d},\qquad j \ge 1.
  \end{equation}
  Consequently, $K_{\chi,\sigma,\gamma} \in S_p$ whenever $p \ge 1$ and $p\gamma
  > d$.
\end{proposition}

\begin{proof}
  The zero cutoff is immediate. Fix one cube and cutoff pair $L,\psi$ as in
  Lemma~\ref{L:cutoff-factorization}. Its recovery identity gives
  \[
    K_{\chi,\sigma,\gamma}
    =\Lambda^\sigma\mathcal{E}_\chi\mathcal{I}_\gamma
    \mathcal{P}_\psi\Lambda^{-\sigma-\gamma}.
  \]
  In the weighted torus coordinates, $\mathcal{I}_\gamma$ is diagonal with
  entries $\langle\kappa_k\rangle^{-\gamma}$. Let $Q_n$ retain $|k|_\infty \le
  n$. Its rank is $(2n+1)^d$, and
  \[
    \|(1-Q_n)\mathcal{I}_\gamma\|
    \le [1+(2\pi/L)^2(n+1)^2]^{-\gamma/2}\longrightarrow0.
  \]
  Replacing the inclusion by $Q_n\mathcal{I}_\gamma$ gives finite-rank $K_n$
  converging in norm to $K_{\chi,\sigma,\gamma}$. This proves compactness:
  finite-dimensional images have finite nets of each positive radius, and norm
  approximation transfers those nets to the limiting image of the unit ball.

  Put
  \[
    M=\|\mathcal{E}_\chi:H^\sigma(\mathbb{T}_L)\to H^\sigma(\R^d)\|
    \|\mathcal{P}_\psi:H^{\sigma+\gamma}(\R^d)
    \to H^{\sigma+\gamma}(\mathbb{T}_L)\|.
  \]
  The outer Bessel maps are unitary. For $j \ge 3^d$, choose $n = \lfloor
  j^{1/d}/4\rfloor$. Then
  \[
    (2n+1)^d\le(j^{1/d}/2+1)^d<j,\qquad
    \|K-K_n\|\le M(2L/\pi)^\gamma j^{-\gamma/d}.
  \]
  The second estimate follows from $n+1 > j^{1/d}/4$. For a compact operator,
  every rank less than $j$ approximation $F$ satisfies $s_j(K) \le \|K-F\|$: the
  span of the first $j$ right singular vectors contains a unit vector in $\ker
  F$, on which $\|K\cdot\| \ge s_j(K)$. If $s_j(K) = 0$ the inequality is
  immediate. For $j < 3^d$ use $s_j(K) \le \|K\| \le M$. Thus, one may take
  \[
    C_{\chi,\sigma,\gamma,d}
    =M\max(3^\gamma,(2L/\pi)^\gamma).
  \]
  Finally, $\sum_j j^{-p\gamma/d}$ converges for $p\gamma > d$. This proves
  Proposition~\ref{P:cutoff-singular-values}.
\end{proof}

No uniform support-radius constant or endpoint necessity is asserted in
Proposition~\ref{P:cutoff-singular-values}. The periodic inclusion was only a
factor of the recovered whole-space operator.

\subsection{Balanced adjoints and localized variance envelopes}

For $E = H^{1/2}(\R^d)\times H^{-1/2}(\R^d)$ use the physical unitary
coordinates
\[
  J_0(u,v)=(\Lambda^{1/2}u,\Lambda^{-1/2}v),\qquad
  H=L^2(\R^d;\mathbb{C}^2).
\]
The frequency map $J$ in Section~\ref{S:wave-variance} is
$(\mathcal{F}\oplus\mathcal{F})J_0$. Let $D_\chi$ multiply both phase entries
and define
\begin{align}
  P_\chi&=J_0D_\chi J_0^{-1},\qquad
  D_\gamma=\operatorname{diag}(\Lambda^{-\gamma},\Lambda^{-\gamma}),
  \label{E:compact-balanced-cutoff}\\
  K_{\rm out}&=P_\chi D_\gamma,\qquad
  K_{\rm in}=P_\chi^*D_\gamma.
  \label{E:compact-two-factors}
\end{align}

\begin{proposition}[Both compact envelopes]\label{P:compact-two-envelopes}
  The operator $P_\chi$ is bounded. For the unitary swap $U(v,w) = (w,v)$,
  \[
    P_\chi^*=UP_\chi U,\qquad K_{\rm in}=UK_{\rm out}U.
  \]
  Both factors are compact and have equal $S_p$ norms for $p \ge 1$, $p\gamma >
  d$. For $p \ge 2$, $p\gamma > d$, the positive compact operators
  \[
    B_{\rm out}=P_\chi D_\gamma^2P_\chi^*=K_{\rm out}K_{\rm out}^*,
    \qquad B_{\rm in}=P_\chi^*D_\gamma^2P_\chi=K_{\rm in}K_{\rm in}^*
  \]
  satisfy
  \begin{equation}\label{E:compact-envelope-traces}
    \operatorname{Tr}B_{\rm out}^{p/2}
    =\operatorname{Tr}B_{\rm in}^{p/2}=\|K_{\rm out}\|_{S_p}^p.
  \end{equation}
\end{proposition}

\begin{proof}
  Put $A_\sigma = \Lambda^\sigma\chi\Lambda^{-\sigma}$. The full-Lebesgue Schur
  argument in Lemma~\ref{L:cutoff-factorization} shows boundedness for every
  real $\sigma$. On Schwartz functions, self-adjointness of real multiplication
  and of the Bessel powers gives $A_\sigma^* = A_{-\sigma}$. Density and
  boundedness extend this to the Hilbert adjoints. No evenness of $\chi$ is
  needed. Thus,
  \[
    P_\chi=\operatorname{diag}(A_{1/2},A_{-1/2}),\qquad
    P_\chi^*=\operatorname{diag}(A_{-1/2},A_{1/2})=UP_\chi U.
  \]
  The swap commutes with $D_\gamma$, giving the claimed factor identity. The
  blocks of $K_{\rm out}$ are $K_{\chi,1/2,\gamma}$ and $K_{\chi,-1/2,\gamma}$.
  Proposition~\ref{P:cutoff-singular-values} gives compactness and $S_p$
  membership. The singular values of their direct sum form the multiset union,
  so
  \[
    \|K_{\rm out}\|_{S_p}^p
    =\|K_{\chi,1/2,\gamma}\|_{S_p}^p+
    \|K_{\chi,-1/2,\gamma}\|_{S_p}^p.
  \]
  Unitary conjugation preserves them. Finally, the positive eigenvalues of
  $KK^*$ and $K^*K$ agree with multiplicity: on an eigenspace with eigenvalue
  $\lambda > 0$, the maps $K/\sqrt\lambda$ and $K^*/\sqrt\lambda$ are inverse
  isometries. Zero eigenvalues contribute nothing to the trace. Summing these
  scalar powers proves~\eqref{E:compact-envelope-traces}. This proves
  Proposition~\ref{P:compact-two-envelopes}.
\end{proof}

We use the usual positive square root of a bounded positive operator and the
spectral decomposition of compact positive operators. The needed order argument
is recorded explicitly.

\begin{lemma}[Compact positive order and trace]\label{L:compact-positive-order}
  If $A,B$ are bounded self-adjoint operators with $0 \le A \le B$ and $B$
  compact positive, then $A$ is compact and its ordered eigenvalues satisfy
  $\lambda_j(A) \le \lambda_j(B)$, including zero padding. Consequently, for $t
  \ge 1$ with finite $\operatorname{Tr}B^t$,
  \begin{equation}\label{E:compact-positive-trace}
    \operatorname{Tr}A^t\le\operatorname{Tr}B^t.
  \end{equation}
\end{lemma}

\begin{proof}
  Let $Q_n$ be finite spectral projections of $B$ whose tails tend to zero in
  norm. Then
  \[
    \|(1-Q_n)A^{1/2}\|^2
    =\|(1-Q_n)A(1-Q_n)\|
    \le\|(1-Q_n)B(1-Q_n)\|\longrightarrow0.
  \]
  Thus, $A^{1/2}$ is a norm limit of finite-rank operators and is compact, as is
  $A$. For compact positive operators, the max-min formula for $\lambda_j$
  follows in an eigenbasis: the span of the first $j$ eigenvectors gives the
  lower bound for the supremum over $j$-dimensional subspaces of their minimum
  Rayleigh quotient, while every such subspace meets the orthogonal complement
  of the first $j-1$ eigenvectors, giving the upper bound. Compare the Rayleigh
  quotients of $A$ and $B$ to obtain the eigenvalue inequality. Sum the ordered
  scalar powers to prove \eqref{E:compact-positive-trace}. No inequality $A^t
  \le B^t$ is asserted. This proves Lemma~\ref{L:compact-positive-order}.
\end{proof}

\begin{proposition}[Localized finite variance bounds]\label{P:localized-finite-variance}
  Suppose finitely many bounded operators $Z_\ell$ on $H$ satisfy
  \begin{equation}\label{E:localized-finite-hypotheses}
    P_\chi Z_\ell=Z_\ell=Z_\ell P_\chi,\qquad
    V_{\rm out}:=\sum_\ell Z_\ell Z_\ell^*\le CD_\gamma^2,\qquad
    V_{\rm in}:=\sum_\ell Z_\ell^*Z_\ell\le CD_\gamma^2,
  \end{equation}
  with $C \ge 0$. If $p \ge 2$ and $p\gamma > d$, every $Z_\ell$ is compact in
  $S_p$, the two variances are compact in $S_{p/2}$, and
  \begin{align}
    V_{\rm out}&\le CB_{\rm out},\qquad V_{\rm in}\le CB_{\rm in},
    \label{E:localized-two-envelopes}\\
    \operatorname{Tr}V_{\rm out}^{p/2},\quad
    \operatorname{Tr}V_{\rm in}^{p/2}
    &\le C^{p/2}\|K_{\rm out}\|_{S_p}^p.
    \label{E:localized-trace-bounds}
  \end{align}
\end{proposition}

\begin{proof}
  Left localization gives $V_{\rm out} = P_\chi V_{\rm out}P_\chi^*$. Apply
  positive congruence to the output bound
  in~\eqref{E:localized-finite-hypotheses} to get $V_{\rm out} \le CP_\chi
  D_\gamma^2P_\chi^* = CB_{\rm out}$. Right localization similarly gives $V_{\rm
  in} = P_\chi^*V_{\rm in}P_\chi \le CB_{\rm in}$.
  Proposition~\ref{P:compact-two-envelopes} and
  Lemma~\ref{L:compact-positive-order} prove compactness and the two trace
  bounds with $t = p/2$. If $C = 0$, both variances and every $Z_\ell$ vanish,
  so all assertions are immediate.

  For each $\ell$, $Z_\ell Z_\ell^* \le V_{\rm out}$ is compact by the same
  lemma. Its finite spectral projections $Q_n$ give $\|(1-Q_n)Z_\ell\|^2 =
  \|(1-Q_n)Z_\ell Z_\ell^*(1-Q_n)\| \to 0$. Hence, $Z_\ell$ is compact. Its
  singular values squared are the nonzero eigenvalues of $Z_\ell Z_\ell^*$.
  Comparing with $V_{\rm out}$ and summing proves $Z_\ell \in S_p$ with
  $\|Z_\ell\|_{S_p}^p \le \operatorname{Tr}V_{\rm out}^{p/2}$. This proves
  Proposition~\ref{P:localized-finite-variance}.
\end{proof}

The two localization identities are hypotheses here; a noise coefficient family
has not yet been shown to satisfy them. In particular, $P_\chi$ is not assumed
self-adjoint or a projection.

\begin{corollary}[Finite Gaussian operators]\label{C:finite-gaussian-operators}
  Under~\eqref{E:localized-finite-hypotheses}, let $g_\ell$ be independent real
  standard Gaussians. For even $p \ge 2$ with $p\gamma > d$, the finite sum $X =
  \sum_\ell g_\ell Z_\ell$ satisfies
  \begin{equation}\label{E:finite-gaussian-operator-bound}
    (\E\|X\|_{S_p}^p)^{1/p}
    \le\sqrt{p-1}\sqrt C\|K_{\rm out}\|_{S_p}.
  \end{equation}
  Its extended norm is measurable, with finite expectation as displayed.
\end{corollary}

\begin{proof}
  Choose finite-rank orthogonal projections $\Pi_n$ from a countable orthonormal
  basis of $H$, with $\Pi_n \to I$ strongly. Put $Z_{\ell,n} = \Pi_nZ_\ell\Pi_n$
  and $X_n = \sum_\ell g_\ell Z_{\ell,n}$, viewed as matrices on
  $\operatorname{ran}\Pi_n$. Their output variance satisfies
  \[
    \sum_\ell\Pi_nZ_\ell\Pi_nZ_\ell^*\Pi_n
    \le\Pi_nV_{\rm out}\Pi_n,
  \]
  since the difference is $\sum_\ell\Pi_nZ_\ell(1-\Pi_n)Z_\ell^*\Pi_n \ge 0$.
  The input inequality is identical with adjoints interchanged. Compact max-min
  also gives $\lambda_j(\Pi_nV\Pi_n) \le \lambda_j(V)$ for positive compact $V$,
  since Rayleigh subspaces inside $\operatorname{ran}\Pi_n$ form a subset of all
  subspaces. Thus, both variance trace powers of these matrices are bounded by
  $C^{p/2}\|K_{\rm out}\|_{S_p}^p$.

  Proposition~\ref{P:finite-gaussian-matrix} applies to these finite complex
  matrices, the same real independent Gaussians and the even $p$. Taking the
  average of the two already bounded trace powers gives
  \begin{equation}\label{E:finite-gaussian-compression-bound}
    \E\|X_n\|_{S_p}^p
    \le(p-1)^{p/2}C^{p/2}\|K_{\rm out}\|_{S_p}^p.
  \end{equation}
  No factor two is added by this average.

  For each Gaussian coordinate vector, $X$ is a finite sum of compact operators.
  Strong convergence of $\Pi_n$ implies $\Pi_nX\Pi_n \to X$ in operator norm:
  approximate $X$ by a finite-rank $F$, use uniform strong convergence on the
  finite-dimensional ranges of $F$ and $F^*$, and control the remainder using
  $\|\Pi_n\| \le 1$. Singular values are norm-Lipschitz. Indeed, their formula
  \[
    s_j(A)=\inf_{\dim L<j}
    \sup_{v\in L^\perp,\ \|v\|=1}\|Av\|
  \]
  follows from compact singular vectors and gives $s_j(A) \le s_j(B)+\|A-B\|$,
  and the reverse bound by interchanging $A,B$. Hence, $s_j(X_n) \to s_j(X)$ for
  each $j$. Fatou for the nonnegative singular-value sum gives $\|X\|_{S_p}^p
  \le \liminf_n\|X_n\|_{S_p}^p$, allowing extended infinity. These variables are
  measurable: the finite Gaussian coordinate map into operators is
  norm-continuous, each singular value is continuous, and the norm power is
  their nonnegative measurable sum. Fatou in probability
  and~\eqref{E:finite-gaussian-compression-bound} prove the bound and
  almost-sure finiteness. No $S_p$-Bochner convergence is assumed. This proves
  Corollary~\ref{C:finite-gaussian-operators}.
\end{proof}

\subsection{A nonzero witness and the need for a compact cutoff}

Proposition~\ref{P:localized-finite-variance} and
Corollary~\ref{C:finite-gaussian-operators} assume
\eqref{E:localized-finite-hypotheses}. We check that these hypotheses admit a
nonzero operator, and then test what is lost without the compact cutoff.

\begin{proposition}[Whole-space cutoff tests]\label{P:cutoff-boundary-tests}
  If $\chi$ equals one on a nonempty open set, there is a nonzero rank-one
  operator satisfying~\eqref{E:localized-finite-hypotheses} with a finite common
  $C$. On the other hand, $\Lambda^{-\gamma}$ on $L^2(\R^d)$ is not compact for
  any $\gamma > 0$.
\end{proposition}

\begin{proof}
  Choose nonzero real $v,f \in C_c^\infty$ supported inside the plateau, and put
  $\widetilde{v} = (\Lambda^{1/2}v,0)$ and $\widetilde{w} =
  (\Lambda^{-1/2}f,0)$. Both are Schwartz. The block formulas give
  $P_\chi\widetilde{v} = \widetilde{v}$ and $P_\chi^*\widetilde{w} =
  \widetilde{w}$. Therefore, $Zx = \langle x,\widetilde{w}\rangle\widetilde{v}$
  satisfies $P_\chi Z = Z = ZP_\chi$. Its two variances are
  $\|\widetilde{w}\|^2\widetilde{v}\otimes\widetilde{v}^*$ and
  $\|\widetilde{v}\|^2\widetilde{w}\otimes\widetilde{w}^*$. For any Schwartz $a$
  and any $x \in H$,
  \[
    |\langle x,a\rangle|
    =|\langle D_\gamma x,D_\gamma^{-1}a\rangle|
    \le\|D_\gamma x\|\|D_\gamma^{-1}a\|.
  \]
  The inverse multiplier is applied only to Schwartz $a$, so it belongs to $H$.
  Apply the inequality with $a = \widetilde{v},\widetilde{w}$ to obtain the two
  variance bounds with
  \[
    C=\max\left(
    \|\widetilde{w}\|^2\|D_\gamma^{-1}\widetilde{v}\|^2,
    \|\widetilde{v}\|^2\|D_\gamma^{-1}\widetilde{w}\|^2\right)<\infty.
  \]
  This witness uses a plateau; no nonzero family is inferred for every compact
  cutoff or identified with a source family.

  For noncompactness, take $f$ with nonzero Gaussian Fourier transform. Then $g
  = \Lambda^{-\gamma}f$ is a nonzero Schwartz function. The images of translates
  of $f$ are the same translates of $g$. With $T_hg(x) = g(x-h)$, Plancherel
  gives
  \[
    \langle T_hg,g\rangle
    =\int e^{-i\xi\cdot h}|\widehat{g}(\xi)|^2\ud\xi
    =c_d^{-1}\widehat{|\widehat{g}|^2}(h).
  \]
  This tends to zero because $|\widehat{g}|^2$ is Schwartz. Choose a large
  spacing so that any two distinct translates in the resulting sequence have
  image distance at least $\|g\|$. The input norms are identical, but the images
  have no Cauchy subsequence. This disproves compactness. This proves
  Proposition~\ref{P:cutoff-boundary-tests}.
\end{proof}

The translation obstruction is restated in No-go~\ref{P:nogo-24}:
frequency decay alone does not replace compact spatial localization.


Thus, the choice $\chi = 1$ cannot be substituted into the compact-cutoff
argument. No necessity at the Schatten endpoint $p\gamma = d$ is asserted. The
results concern finite matrices and conditional finite operator sums.
Coefficient localization, compact variance tails, an infinite Gaussian series
and time regularity are proved in Section~\ref{S:actual-operator-driver}; source
and Young identification are proved in
Sections~\ref{S:source-driver-convergence}--\ref{S:constant-phase-young}. They
are not consequences of the finite comparison alone.

\section{The Gaussian operator path}
\label{S:actual-operator-driver}

We now construct the integrated operator path in the interaction representation
on the same probability space as the original noise. Retain
Hypothesis~\ref{H:homog}, set $a = \alpha_0$, $s = 1/2$ and $E_r =
H^r(\R^d)\times H^{r-1}(\R^d)$, and initially use real phases $E = E_s$. Let
$H_W$ be the real completion of $C_c^\infty((0,\infty)\times\R^d)$ modulo its
covariance-null tests. The given isonormal process on it is $W$. If $\Gamma =
0$, the operator path and all covariance forms are zero, so the conclusions
below hold directly. Assume $\Gamma \ne 0$ in the construction.

Choose and fix $R,T > 0$ and a real even nonnegative $\zeta \in
C_c^\infty(\R^d)$ equal to one on $B_1$ and zero off $B_2$. The physical mask is
$\zeta_R(x) = \zeta(x/R)$. The free group $S(t)$ and forms $\ell_R$ are those of
Section~\ref{S:wave-variance}; both phase entries and the continuous
zero-frequency sine quotient are retained. Fix parameters
\begin{equation}\label{E:driver-parameters}
  \LeanEqMarker{E:driver-parameters}
  0<\gamma<\min\left(\frac12,\frac{2-\alpha}{2}\right),\qquad
  \frac12<\theta<\beta<\min\left(1-\frac a2,
  \frac{3-a-\alpha}{2}\right).
\end{equation}
The strict intervals are nonempty. The $C^\theta$ norm is the supremum norm plus
the $\theta$-H\"older seminorm.

\begin{proposition}[Compact operator path on the original noise space]\label{P:actual-operator-driver}
  There is an orthonormal basis $\{f_j\}$ of $H_W$ with real smooth compact test
  representatives, independent standard real Gaussians $g_j = W(f_j)$, and
  deterministic operators $A_j(0,t)$ on $E$ such that
  \begin{equation}\label{E:driver-path-convergence}
    X_N(t):=\sum_{j\le N}g_jA_j(0,t)\longrightarrow M_R(t)
    \quad\text{in }L^p(\Omega;C^\theta([0,T];\mathcal{B}(E)))
  \end{equation}
  for every finite $p \ge 1$. The limit is a centered Gaussian path on the
  original probability space, starts at zero, and has compact values at every
  time on one probability-one path set. For each fixed real Schwartz pair $z,w$
  and $t \in [0,T]$,
  \begin{equation}\label{E:driver-raw-row}
    \langle M_R(t)z,w\rangle_E=\ell_R((0,t);z,w)
    \quad\text{in }L^2(\Omega).
  \end{equation}
  For long intervals, use the completed covariance test, equivalently the
  additive unit-interval construction justified below. Different compact smooth
  test bases give indistinguishable paths on the same $W$. For constants $C,c >
  0$ depending on the fixed data but not the basis or $p$,
  \begin{equation}\label{E:driver-gaussian-norm}
    (\E\|M_R\|_{C^\theta}^p)^{1/p}\le C\sqrt p,\qquad
    \E\exp(c\|M_R\|_{C^\theta}^2)<\infty.
  \end{equation}
  Its complexification has the corresponding sesquilinear covariance identity.
  Fixed-interval Schatten moments hold at the admissible even indices
  in~\eqref{E:driver-fixed-schatten}. Arbitrary $p$ above refers to
  $\mathcal{B}(E)$, not arbitrary Schatten indices or a Schatten-valued path.
\end{proposition}

\subsection{A test basis and explicit covariance coefficients}

\begin{lemma}[Basis in the prescribed quotient]\label{L:driver-noise-basis}
  The nonzero space $H_W$ is separable and infinite-dimensional. It has a real
  orthonormal basis with compact smooth representatives, whose evaluations by
  $W$ are independent standard real Gaussians.
\end{lemma}

\begin{proof}
  For tests supported in $[0,L]\times K$, with $K$ compact, the physical
  covariance and the temporal Schur bound give the absolute estimate
  \begin{equation}\label{E:driver-test-continuity}
    |\langle F,G\rangle_{H_W}|
    \le |K|\Gamma(K-K)\frac{2L^{1-a}}{1-a}
    \|\|F(t)\|_\infty\|_{L^2(0,L)}
    \|\|G(t)\|_\infty\|_{L^2(0,L)}.
  \end{equation}
  Only spatial differences in $K-K$ contribute. On each rational compact box
  inside positive time, rational step approximations followed by compact smooth
  mollifiers at rational scales give a countable uniformly dense test
  collection, supported in a fixed slightly larger box. The estimate proves
  noise-norm approximation. The union over increasing boxes is dense in the test
  completion. This proves separability without assuming the covariance form is
  injective.

  Choose $L \ge 1$ with $0 < \Gamma(B_L) < \infty$ and nonnegative compact
  smooth $q \ge 1$ on $B_{2L+1}$. Then
  \[
    Q_{\rm sp}:=\int\Gamma(\ud y)\int q(x)q(x-y)\ud x
    \ge |B_{L+1}|\Gamma(B_L)>0,
  \]
  and the compact-support estimate makes it finite. For every real compact
  smooth positive-time $\tau$, the tensor identity is
  \[
    \|\tau\otimes q\|_{H_W}^2=Q_{\rm sp}\|\tau\|_a^2.
  \]
  Here the temporal norm is computed on any finite interval containing the
  support of $\tau$. It is strictly positive for signed nonzero $\tau$. Indeed,
  zero-extend this deterministic test and use the positive Laplace factorization
  from Lemma~\ref{L:masked-time-completion} to write
  \begin{align*}
    \|\tau\|_a^2
    &=\frac1{\Gamma_E(a)}\int_0^\infty\lambda^{a-1}
    E_\lambda(\tau)\ud\lambda,\\
    E_\lambda(\tau)
    &=\int_\R 2\lambda\left|\int_r^\infty
    e^{-\lambda(t-r)}\tau(t)\ud t\right|^2\ud r.
  \end{align*}
  Signed exchanges are absolutely integrable on the test support, by the
  temporal Schur estimate. If the norm vanished, some $\lambda > 0$ would have
  $E_\lambda = 0$. Its inner integral $v(r)$ is continuous and satisfies $v' =
  \lambda v-\tau$. Vanishing almost everywhere makes it identically zero and
  forces $\tau = 0$. Thus, infinitely many linearly independent compact time
  bumps inject into $H_W$ after multiplication by $q$.

  Apply Gram--Schmidt to a countable dense real test collection, discarding zero
  residuals in the given null quotient. Infinitude gives an infinite basis; each
  representative is a finite linear combination of compact smooth tests. The
  isonormal identity gives $\E g_jg_k = \delta_{jk}$. Joint Gaussian
  characteristic functions then factor, giving independence and unit variances.
  No physical null direction is removed beyond the prescribed quotient, and no
  $H_W$-valued white Gaussian vector is used. This proves
  Lemma~\ref{L:driver-noise-basis}.
\end{proof}

The operator-norm topology is essential here: in dimensions two and three,
No-go~\ref{P:nogo-21} rules out a compatible square-integrable
Hilbert--Schmidt realization on a same-index Sobolev phase pair,
for suitable coefficient masks and a sufficiently large observation cutoff.


\begin{lemma}[Covariance coefficients]\label{L:driver-covariance-coefficients}
  For each basis representative put
  \begin{equation}\label{E:driver-covariance-coefficient}
    H_{j,R}(t,x)=\zeta_R(x)\int_{u>0}|t-u|^{-a}
    \int\Gamma(\ud y)f_j(u,x-y)\ud u.
  \end{equation}
  These functions are real smooth on finite time-space compacts, supported
  spatially in $B_{2R}$, with bounded derivatives there. For smooth compact real
  positive-time $F$, and also for the admitted wave tests
  of~\eqref{E:variance-actual-form},
  \begin{equation}\label{E:driver-literal-covariance}
    \E[W_{\mathbb{C}}(\zeta_RF)g_j]
    =\int H_{j,R}(t,x)F(t,x)\ud t\ud x.
  \end{equation}
  The coefficient on $t \ge 0$ depends only on the $H_W$ class of $f_j$.
\end{lemma}

\begin{proof}
  Extend $f_j$ smoothly by zero in time and substitute $v = t-u$. The unmasked
  expression is $\int_\R |v|^{-a}\int\Gamma(\ud y)f_j(t-v,x-y)\ud v$. On each
  compact $(t,x)$ set, the integrand involves only bounded $v$ and a common
  compact set of $y$. Local Radon mass and local integrability of $|v|^{-a}$
  allow every mixed derivative to move onto the smooth test. Dominated
  integration proves smoothness and the derivative bounds. Multiplication by
  $\zeta_R$ gives the stated support. Zero extension is only of a deterministic
  test; no negative-time noise is introduced.

  The physical covariance formula and absolute Fubini give
  \eqref{E:driver-literal-covariance} first for real compact tests. The compact
  difference set in $\Gamma$ and integrable temporal singularity justify the
  exchange. For wave tests use the common-support original approximations of
  Proposition~\ref{P:actual-wave-form}, converging in $L^2$ time in the spatial
  supremum norm after masking. The left side converges by noise Cauchy--Schwarz
  and the right side by the coefficient's boundedness on that support. Real and
  imaginary parts give the complex version. If two representatives give the same
  class, their Gaussian difference is zero. Testing the covariance identity
  against every compact smooth $F$ makes the difference of their masked
  coefficients zero as a distribution for $t > 0$. Smoothness, then continuity
  at zero, proves the representative assertion on $t \ge 0$. This proves
  Lemma~\ref{L:driver-covariance-coefficients}.
\end{proof}

\subsection{Gain, adjoints and localization}

\begin{lemma}[Deterministic coefficient operators]\label{L:driver-gain-localization}
  For $I \subset [0,T]$, the statewise integral
  \begin{equation}\label{E:driver-coefficient-operator}
    A_j(I)z=\int_I S(-t)B(H_{j,R}(t))S(t)z\ud t
  \end{equation}
  satisfies, for every real $r$,
  \begin{equation}\label{E:driver-gain-one}
    \|A_j(I)\|_{\mathcal{B}(E_r,E_{r+1})}
    \le C_{j,r,R,T}|I|.
  \end{equation}
  Its $E_s$ adjoint has the gain $E_{2s-r-1} \to E_{2s-r}$, so both operators
  act from $E_{s-\gamma}$ to $E$. Choose real compact smooth $\chi = 1$ on
  $B_{2R+T+1}$. On every $E_r$,
  \begin{equation}\label{E:driver-actual-localization}
    D_\chi A_j(I)=A_j(I)=A_j(I)D_\chi.
  \end{equation}
  For each fixed Schwartz pair,
  \begin{equation}\label{E:driver-covariance-row}
    \E[g_j\ell_R(I;z,w)]=\langle A_j(I)z,w\rangle_E.
  \end{equation}
\end{lemma}

\begin{proof}
  The full massless wave is strongly continuous and bounded on every $E_r$ on
  finite time intervals. Its normalized off-diagonal Fourier entries have sizes
  at most $C_T$ and one, respectively; continuity and domination prove the
  strong assertion. Multiplication by $H_{j,R}(t)$ is bounded on every $H^r$,
  continuously in $t$, by weighted Fourier convolution and its compact smooth
  derivative bounds. Hence, $B(H_{j,R}(t)):E_r \to E_{r+1}$ is bounded: its
  second entry is a multiplier of the displacement in $H^r$. The integrand
  in~\eqref{E:driver-coefficient-operator} is strongly continuous with values in
  $E_{r+1}$ for each input in $E_r$, and uniformly bounded in that operator
  norm. Statewise Bochner integration proves~\eqref{E:driver-gain-one},
  including the Lipschitz operator-norm primitive $A_j(0,t)$. Norm continuity of
  $S(t)$ is not needed.

  Under the $E_s$ pairing, the dual of $E_r$ is isometrically $E_{2s-r}$,
  directly by the two Bessel weights. Dualizing the bound with a gain of one
  derivative therefore gives the asserted adjoint gain. It agrees on Schwartz
  phases with the bounded $E_s$ adjoint. Since $\gamma < 1$, gain one at every
  index implies the two stated maps from $E_{s-\gamma}$ to $E$.

  For smooth data the free-wave energy on a shrinking ball has nonpositive
  boundary flux: $v\partial_nu-(v^2+|\nabla u|^2)/2 \le 0$. Vanishing initial
  displacement and velocity give zero velocity and gradient in the cone;
  integration of velocity from the zero displacement gives zero displacement
  there as well. Time reversal gives the backward assertion. Thus, both entries
  of $S(\pm t)$ enlarge supports by at most $|t|$. Density and testing against
  compact smooth functions extend this support assertion to Sobolev phases. In
  particular, for input $(1-D_\chi)z$, approximate $z$ by Schwartz phases and
  retain the factor $1-D_\chi$; the approximants vanish on $B_{2R+T+1}$. Their
  free displacement vanishes on $B_{2R}$ throughout $[0,T]$, and the same is
  true in the distributional limit. They contribute nothing to the integral. Its
  output is supported in $B_{2R+T}$, by the support of $H_{j,R}$ and backward
  finite propagation. Since $\chi = 1$ on a neighborhood of this ball, both
  identities in~\eqref{E:driver-actual-localization} follow. This uses both
  entries, not a scalar restart or a positivity theorem.

  Finally, apply the physical Sobolev pairing
  \eqref{E:variance-physical-adjoint} and~\eqref{E:driver-literal-covariance} to
  the state integral. They give \eqref{E:driver-covariance-row}.
  In~\eqref{E:driver-covariance-coefficient}, the mask is after convolution;
  replacing it by a mask inside $\Gamma*f_j$ would change this identity. This
  proves Lemma~\ref{L:driver-gain-localization}.
\end{proof}

\subsection{Finite variances and first Wiener chaos tails}

Put $J_0(u,v) = (\Lambda^{1/2}u,\Lambda^{-1/2}v)$, $H_0 = L^2(\R^d;\R^2)$, $a_j
= J_0A_j(I)J_0^{-1}$ and $P_\chi = J_0D_\chi J_0^{-1}$. The operator $D_\gamma =
\operatorname{diag}(\Lambda^{-\gamma},\Lambda^{-\gamma})$ acts on these physical
coordinates. Its two compact envelopes $B_{\rm out},B_{\rm in}$ are those of
Proposition~\ref{P:compact-two-envelopes}, after complexification when Schatten
norms are used. For $I \subset [0,T]$ with $0 < h = |I| \le 1$, set
\begin{equation}\label{E:driver-short-variance-budget}
  b_I=C_T h^{2-a}B_\alpha(h),
\end{equation}
where $C_T$ is the uniform-in-$R$ constant in
Proposition~\ref{P:wave-two-variances}.

\begin{lemma}[Both variances and their tails]\label{L:driver-actual-variances}
  For every $N$,
  \begin{equation}\label{E:driver-two-variances}
    V_{{\rm out},N}:=\sum_{j\le N}a_ja_j^*\le b_ID_\gamma^2,
    \qquad V_{{\rm in},N}:=\sum_{j\le N}a_j^*a_j\le b_ID_\gamma^2.
  \end{equation}
  The finite families satisfy both localizations by $P_\chi$. There are bounded
  positive limits $V_{{\rm out},\infty},V_{{\rm in},\infty}$ with positive tails
  $T_{{\rm out},N},T_{{\rm in},N} \to 0$ strongly, and
  \begin{equation}\label{E:driver-compact-tail-envelopes}
    T_{{\rm out},N}\le b_IB_{\rm out},\qquad
    T_{{\rm in},N}\le b_IB_{\rm in}.
  \end{equation}
  For each fixed $z \in E_{s-\gamma}$, the real column maps satisfy
  \begin{equation}\label{E:driver-column-expansions}
    m_{R,I}(z)=\sum_jg_jA_j(I)z,\qquad
    n_{R,I}(z)=\sum_jg_jA_j(I)^{*,E}z
    \quad\text{in }L^2(\Omega;E).
  \end{equation}
\end{lemma}

\begin{proof}
  By~\eqref{E:driver-covariance-row}, for real Schwartz $z,w$,
  \[
    \langle X_N(I)z,w\rangle_E
    =\sum_{j\le N}g_j\E[g_j\ell_R(I;z,w)]
    =\mathcal{P}_N\ell_R(I;z,w),
  \]
  where $\mathcal{P}_N$ is the orthogonal projection in $L^2(\Omega)$ onto the
  span of $g_1,\ldots,g_N$. In particular,
  \[
    \sum_{j\le N}|\langle A_j(I)z,w\rangle_E|^2
    =\E|\langle X_N(I)z,w\rangle_E|^2
    \le\E|\ell_R(I;z,w)|^2.
  \]
  These are projections onto the noise-test basis, not source regularizations.
  The two variance bounds of Proposition~\ref{P:wave-two-variances}, with the
  mean-square columns of Proposition~\ref{P:wave-square-columns}, apply to these
  same covariance forms and the bound~\eqref{E:driver-short-variance-budget}.

  Choose a real Schwartz orthonormal basis $\{e_k\}$ of $E$. Fix real Schwartz
  $w$. For every finite phase truncation $K$,
  \[
    \begin{split}
      \sum_{k\le K}\sum_{j\le N}
      |\langle A_j(I)e_k,w\rangle_E|^2
      &=\sum_{k\le K}\E|\langle X_N(I)e_k,w\rangle_E|^2\\
      &\le\sum_{k\le K}\E|\ell_R(I;e_k,w)|^2
      \le b_I\|w\|_{E_{s-\gamma}}^2,
    \end{split}
  \]
  by~\eqref{E:wave-input-variance}. Let $K \to \infty$. Parseval in the phase
  space and nonnegative monotone convergence, with $N$ still finite, give
  \[
    \sum_{j\le N}\|A_j(I)^{*,E}w\|_E^2
    =\E\|X_N(I)^{*,E}w\|_E^2
    \le b_I\|w\|_{E_{s-\gamma}}^2.
  \]
  Conversely, for fixed real Schwartz $z$, use \eqref{E:wave-output-variance}
  with the output basis varying:
  \[
    \begin{split}
      \sum_{k\le K}\sum_{j\le N}
      |\langle A_j(I)z,e_k\rangle_E|^2
      &=\sum_{k\le K}\E|\langle X_N(I)z,e_k\rangle_E|^2\\
      &\le\sum_{k\le K}\E|\ell_R(I;z,e_k)|^2
      \le b_I\|z\|_{E_{s-\gamma}}^2.
    \end{split}
  \]
  The same phase Parseval limit gives
  \[
    \sum_{j\le N}\|A_j(I)z\|_E^2
    =\E\|X_N(I)z\|_E^2
    \le b_I\|z\|_{E_{s-\gamma}}^2.
  \]
  Thus, both directions use their respective Hilbert-Bessel bounds; neither
  follows from retaining only one variance.

  These inequalities extend at each finite $N$ to the full $E_{s-\gamma}$ input
  space. Indeed, Lemma~\ref{L:driver-gain-localization} gives $A_j:E_{s-\gamma}
  \to E_{s+1-\gamma} \subset E$. Its adjoint gain, at $r = s+\gamma-1$, gives
  the same map for $A_j^{*,E}$. The inclusion holds since $\gamma < 1$.
  Approximate the input by Schwartz phases and pass each of the finitely many
  operator terms to the limit. The resulting constant is still $b_I$,
  independent of $N$; individual $j$-dependent gain constants were not summed.
  Restricting to $E$ and using $\|z\|_{E_{s-\gamma}} = \|D_\gamma J_0z\|_{H_0}$
  gives \eqref{E:driver-two-variances} as two operator quadratic inequalities.
  Both localizations follow from~\eqref{E:driver-actual-localization}. No
  self-adjointness or projection property of $P_\chi$ is assumed.

  The coordinates of $m_{R,I}(z)$, for Schwartz $z$, are the original covariance
  forms $\ell_R(I;z,e_k)$ and belong to the closed first Wiener chaos $W(H_W)$.
  Their coefficients against $g_j$ are $\langle A_jz,e_k\rangle_E$. Parseval in
  that chaos, followed by Parseval in $E$ and positive Tonelli, gives
  \[
    \sum_j\|A_jz\|_E^2=\E\|m_{R,I}(z)\|_E^2,
    \qquad m_{R,I}(z)=\sum_jg_jA_jz\quad\text{in }L^2(\Omega;E).
  \]
  The same argument for the adjoint column gives the other expansion. Membership
  in this first chaos comes from linear noise tests, not merely Gaussianity. To
  extend to a fixed $E_{s-\gamma}$ input, approximate by Schwartz phases. The
  finite inequalities extend to that space by the gain bounds and have a
  constant uniform in $N$. Apply them to the input difference, then use the
  Schwartz expansion and continuity of $m,n$. This
  proves~\eqref{E:driver-column-expansions} on the full claimed input domain.

  The convergent positive quadratic sums, restricted to $E$ and transported by
  $J_0$, define the bounded positive operators $V_{\cdot,\infty}$ by
  polarization and Riesz. They retain the bounds $b_ID_\gamma^2$. Their tails
  have quadratic forms tending to zero. Since they are uniformly bounded and
  positive, $\|T_Nv\|^2 \le \|T_N\|\langle T_Nv,v\rangle \to 0$. Passing to the
  limit in the finite localization identities gives
  \[
    V_{{\rm out},\infty}=P_\chi V_{{\rm out},\infty}P_\chi^*,\qquad
    V_{{\rm in},\infty}=P_\chi^*V_{{\rm in},\infty}P_\chi,
  \]
  and the same identities for their tails. Positive congruence applied to $T_N
  \le b_ID_\gamma^2$ proves~\eqref{E:driver-compact-tail-envelopes}.

  Complexification preserves these statements and constants. The Hilbert norms
  split into real and imaginary squared norms; real operators extend
  isometrically. A real orthonormal basis complexifies to a complex orthonormal
  basis, preserving singular values and multiplicities. The complex column maps
  have the same bounds and sesquilinear row identities. This proves
  Lemma~\ref{L:driver-actual-variances}.
\end{proof}

Lemma~\ref{L:driver-actual-variances} gives tails
$T_{{\rm out},N},T_{{\rm in},N}$ that tend to zero only strongly. They lie
below the compact envelopes in~\eqref{E:driver-compact-tail-envelopes}. We
now use this compact domination to improve the mode of convergence.

\begin{lemma}[Compact domination of positive tails]\label{L:driver-dominated-tails}
  If $0 \le T_N \le B$, with positive compact $B \in S_m$, $m \ge 1$, and $T_N
  \to 0$ weakly, then
  \begin{equation}\label{E:driver-trace-tails}
    \|T_N\|\longrightarrow0,\qquad \operatorname{Tr}T_N^m\longrightarrow0.
  \end{equation}
\end{lemma}

\begin{proof}
  Choose a finite spectral projection $Q$ of $B$ with $\|(1-Q)B(1-Q)\| <
  \epsilon$. Positive quadratic Cauchy--Schwarz gives, for unit $x$,
  \[
    \langle T_Nx,x\rangle
    \le2\langle T_NQx,Qx\rangle
    +2\langle T_N(1-Q)x,(1-Q)x\rangle
    \le2\|QT_NQ\|+2\epsilon.
  \]
  Weak convergence makes the finite-dimensional matrix $QT_NQ$ converge in norm
  to zero. Letting $\epsilon\downarrow0$ proves norm convergence.
  Lemma~\ref{L:compact-positive-order} gives $\lambda_j(T_N) \le \lambda_j(B)$,
  while each $\lambda_j(T_N) \to 0$ by norm convergence. Dominated summation of
  $\lambda_j(B)^m$ proves the trace limit. No order comparison of the operator
  powers is used. This proves Lemma~\ref{L:driver-dominated-tails}.
\end{proof}

Trace powers of variances must still be turned into moments of Gaussian sums.
Proposition~\ref{P:finite-gaussian-matrix} keeps the average of the two trace
powers, but only for matrices. Corollary~\ref{C:finite-gaussian-operators}
bounds compact sums only through the envelope norm $\|K_{\rm out}\|_{S_p}$.

\begin{lemma}[Retained finite Gaussian variance powers]\label{L:driver-gaussian-tail-average}
  Let $Z_\ell$ be finitely many compact operators whose two variances $V_{\rm
  out},V_{\rm in}$ belong to $S_{q/2}$, for an even $q \ge 2$. With independent
  standard real $g_\ell$,
  \begin{equation}\label{E:driver-retained-gaussian-average}
    \E\left\|\sum_\ell g_\ell Z_\ell\right\|_{S_q}^q
    \le(q-1)^{q/2}\frac{\operatorname{Tr}V_{\rm out}^{q/2}
    +\operatorname{Tr}V_{\rm in}^{q/2}}2.
  \end{equation}
\end{lemma}

\begin{proof}
  Compress by finite-rank orthogonal $Q_k \to I$ strongly. The compressed output
  variance is at most $Q_kV_{\rm out}Q_k$, since their difference is $\sum_\ell
  Q_kZ_\ell(1-Q_k)Z_\ell^*Q_k \ge 0$; the input inequality is the same with
  adjoints interchanged. Positive-order and compression max-min compare their
  trace powers with the uncompressed ones. Apply
  Proposition~\ref{P:finite-gaussian-matrix} on the finite range, retaining the
  average. For each finite Gaussian coordinate vector, compactness gives
  operator-norm convergence of the compressed sum to the original sum. The
  singular-value norm-Lipschitz argument in the proof of
  Corollary~\ref{C:finite-gaussian-operators}, then Fatou over singular values
  and probability, proves the displayed bound. The extended norm is measurable
  by the same finite-coordinate continuity argument. No infinite Gaussian series
  or $S_q$-Bochner convergence is assumed. This proves
  Lemma~\ref{L:driver-gaussian-tail-average}.
\end{proof}

\subsection{Interval limits on the given probability space}

The two previous lemmas control tails of the series $\sum_jg_jA_j(I)$ through
its two variances. We now fix one interval and use them to pass to the limit in
$N$, without leaving the original probability space.

\LeanMarker{L:driver-interval-limit}
\begin{lemma}[Fixed-interval operator construction]\label{L:driver-interval-limit}
  For each fixed short interval $I$, $X_N(I)$ converges in every finite
  $L^p(\Omega;\mathcal{B}(E))$ to a compact-valued centered Gaussian operator
  $X_R(I)$ on the original probability space. For every admissible even $q \ge
  2$, $q\gamma > d$,
  \begin{equation}\label{E:driver-fixed-schatten}
    (\E\|X_R(I)\|_{S_q}^q)^{1/q}
    \le\sqrt{q-1}\sqrt{C_T}\,h^{1-a/2}B_\alpha(h)^{1/2}
    \|K_\chi\|_{S_q},\qquad K_\chi=P_\chi D_\gamma.
  \end{equation}
  Its fixed Schwartz rows equal the covariance forms in $L^2$.
\end{lemma}

\begin{proof}
  The finite families $a_j$, $j \le N$, satisfy every hypothesis of
  Proposition~\ref{P:localized-finite-variance} and
  Corollary~\ref{C:finite-gaussian-operators}: both localizations, both weighted
  variances and the independent real $g_j$ have been proved in
  Lemma~\ref{L:driver-actual-variances}. Thus, the coefficients are compact, and
  their finite Gaussian sums satisfy~\eqref{E:driver-fixed-schatten}, uniformly
  in $N$. The fixed cutoff norm may depend on $R,T$.

  Fix an admissible even $q \ge 2$, $q\gamma > d$. Both positive compact
  envelopes in~\eqref{E:driver-compact-tail-envelopes} belong to $S_{q/2}$.
  Lemma~\ref{L:driver-dominated-tails} therefore gives
  \[
    \operatorname{Tr}T_{{\rm out},N}^{q/2}\longrightarrow0,
    \qquad
    \operatorname{Tr}T_{{\rm in},N}^{q/2}\longrightarrow0.
  \]
  For $M > N$, put $Y_{N,M} = \sum_{j=N+1}^Mg_jA_j(I)$. After the unitary $J_0$
  transport, its two finite variances are
  \[
    V_{{\rm out},M}-V_{{\rm out},N}\le T_{{\rm out},N},\qquad
    V_{{\rm in},M}-V_{{\rm in},N}\le T_{{\rm in},N}.
  \]
  Lemma~\ref{L:driver-gaussian-tail-average} and positive eigenvalue max-min
  comparison now give the uniform tail estimate
  \[
    \sup_{M>N}\E\|Y_{N,M}\|_{S_q}^q
    \le(q-1)^{q/2}
    \frac{\operatorname{Tr}T_{{\rm out},N}^{q/2}
    +\operatorname{Tr}T_{{\rm in},N}^{q/2}}2
    \longrightarrow0.
  \]
  The average of the two trace powers is retained. The comparison uses
  eigenvalues, not operator monotonicity of the $q/2$ power. Since $\|Y_{N,M}\|
  \le \|Y_{N,M}\|_{S_q}$, this proves $L^q(\Omega;\mathcal{B}(E))$ Cauchy
  convergence; a uniform moment bound alone would not suffice.

  All finite sums lie in the separable operator-norm closed real span
  $\mathcal{F}_I$ of the countably many deterministic $A_j(I)$. Choose $N_k$
  increasing so that $\|X_{N_{k+1}}(I)-X_{N_k}(I)\|_{L^q\mathcal{B}(E)} \le
  2^{-k}$. Their $L^1$ norms are summable. Tonelli therefore shows that the
  pointwise operator-norm differences sum absolutely almost surely. Completeness
  gives a limit in $\mathcal{F}_I$ on the original probability space; assign it
  zero on the exceptional null set. It is strongly measurable as a limit of
  separably valued finite Gaussian sums. Fatou and the Cauchy property give
  convergence of the whole sequence in $L^q$ to this limit $X_R(I)$. No new
  noise or probability space has been introduced. Limits constructed at
  different admissible even indices agree by uniqueness of convergence in
  probability. For each requested finite $p \ge 1$, choose an admissible even $q
  \ge p$ and use Lyapunov to obtain $L^p$ operator-norm convergence to that same
  limit. One fixed Schatten index is not used to infer all moment orders.

  The compact operators are norm closed, so an almost surely norm-convergent
  subsequence makes $X_R(I)$ compact valued. Singular-value Fatou applied to the
  finite uniform bound gives \eqref{E:driver-fixed-schatten} at each admissible
  even $q$. Every continuous real linear functional of $X_R(I)$ is an $L^2$
  limit of centered finite Gaussian sums; convergence of means, variances and
  characteristic functions makes it centered Gaussian. The same applies to
  finite linear combinations of these functionals, proving joint Gaussianity on
  the given space.

  For each fixed real Schwartz $z,w$, the scalar rows $\langle
  X_N(I)z,w\rangle_E$ are the orthogonal projections of the completed covariance
  form $\ell_R(I;z,w)$ onto the complete noise basis. That form lies in the
  original closed first Wiener chaos, so these projections converge to it in
  $L^2(\Omega)$. Operator-norm $L^2$ convergence also gives convergence to
  $\langle X_R(I)z,w\rangle_E$. Uniqueness of the $L^2$ limit identifies these
  two variables on the same space, not merely their laws.

  For a fixed long interval, choose a finite partition into intervals of length
  at most one. Their indicator tests add to the long-interval test except at
  finitely many endpoints. The masked wave products have common compact spatial
  support, and the temporal estimate~\eqref{E:driver-test-continuity} makes
  those endpoint changes have zero original covariance norm. Covariance
  completion and linearity give exact additivity of the forms. The deterministic
  statewise coefficient integrals, hence every $X_N$, add on the same partition.
  Passing their finite-sum $L^p$ limits gives the corresponding operator
  additivity and covariance identity for matrix elements, independently of the
  partition by common refinement. These are statements at each fixed interval
  and partition; no uncountable intersection of independently selected null sets
  is used. No long-interval version of the short-$h$ bound is asserted. The
  Schatten conclusion is a fixed-interval moment at admissible even indices, not
  a Schatten-valued path or arbitrary-index claim. This proves
  Lemma~\ref{L:driver-interval-limit}.
\end{proof}

\subsection{Time-path convergence and Gaussian norm bounds}

Lemma~\ref{L:driver-interval-limit} works one interval at a time.
Proposition~\ref{P:actual-operator-driver} asks for more: convergence in the
$C^\theta$ path norm and compact values at every time on one path event.

\LeanMarker{L:driver-holder-limit}
\begin{lemma}[H\"older path limit]\label{L:driver-holder-limit}
  The finite paths $X_N(t) = X_N((0,t))$ converge in
  $L^p(\Omega;C^\theta([0,T];\mathcal{B}(E)))$ for every finite $p \ge 1$, in a
  separable closed path span on the same probability space. The limit starts at
  zero and has compact values for all times on one path event. For sufficiently
  large even $q_0$, its $C^\theta$ norm has $L^q$ bound $C\sqrt q$, uniformly
  over even $q \ge q_0$ and the basis.
\end{lemma}

\begin{proof}
  Choose $\theta < \eta < \beta$ and an even $q_0 \ge 2$ with $q_0\gamma > d$
  and $\beta-\eta-1/q_0 > 0$. For $q \ge q_0$, the singular-value sequence gives
  $\|K_\chi\|_{S_q} \le \|K_\chi\|_{S_{q_0}}$: if the latter is nonzero,
  normalize by it, so every singular value is at most one, and sum its $q$th
  powers. The zero case is immediate. Apply the finite Gaussian operator
  estimate with the time-refined variances from
  Proposition~\ref{P:wave-two-variances}. This gives, uniformly in $N$ and even
  $q \ge q_0$,
  \begin{equation}\label{E:driver-increment-moments}
    \|X_N(t)-X_N(u)\|_{L^q(\mathcal{B}(E))}
    \le C\sqrt q\,|t-u|^\beta,\qquad |t-u|\le1.
  \end{equation}
  The constant includes the fixed $S_{q_0}$ cutoff norm and does not depend on
  $q$. Longer increments are finite sums of unit increments, with constants
  depending on $T$.

  Let $D_n(F)$ be the maximum of adjacent increments on the grid $kT2^{-n}$, $0
  \le k \le 2^n$. At fine levels $T2^{-n} \le 1$, the elementary bound for a
  finite maximum yields
  \[
    \|D_n(X_N)\|_{L^q}
    \le 2^{n/q}C\sqrt q\,(T2^{-n})^\beta.
  \]
  No independence of these increments is used. The finitely many coarse levels
  are controlled by unit partitions. Minkowski and a geometric sum therefore
  give
  \begin{equation}\label{E:driver-dyadic-moment-sum}
    \sup_N\left\|\sum_{n\ge0}2^{n\eta}D_n(X_N)\right\|_{L^q}
    \le C_\eta\sqrt q,
  \end{equation}
  since the fine-level ratio is at most $2^{-(\beta-\eta-1/q_0)} < 1$.

  For a continuous Banach-valued $F$ starting at zero, this weighted dyadic sum
  controls its $C^\eta$ norm. Approximate two times by dyadic points at a scale
  comparable to their separation. The two grid points differ by at most two
  adjacent increments, and each endpoint's finer binary corrections are bounded
  by the tail sum of $D_n$. Multiplying by the separation to the power $-\eta$
  bounds that total by the weighted sum, up to a constant depending on $T,\eta$.
  Continuity identifies the endpoint limits and also controls the supremum norm
  from $F(0) = 0$. The finite paths are Lipschitz by~\eqref{E:driver-gain-one}.
  Thus, \eqref{E:driver-dyadic-moment-sum} gives uniform $L^qC^\eta$ bounds, and
  the triangle inequality gives the same bound for their differences.

  On a fixed finite time grid, Lemma~\ref{L:driver-interval-limit} gives $L^q$
  convergence of every point value. For continuous $F$,
  \[
    \|F\|_\infty\le\max_k\|F(t_k)\|
    +C(\operatorname{mesh})^\eta\|F\|_{C^\eta}.
  \]
  Apply this to $X_N-X_M$, send $N,M \to \infty$ at a fixed grid, and then send
  the mesh to zero. The uniform $C^\eta$ bound proves sup-norm $L^q$ Cauchy
  convergence. The deterministic interpolation inequality
  \begin{equation}\label{E:driver-holder-interpolation}
    \|F\|_{C^\theta}
    \le C\|F\|_\infty^{1-\theta/\eta}
    \|F\|_{C^\eta}^{\theta/\eta}
  \end{equation}
  follows by bounding each increment by both $2\|F\|_\infty$ and its $C^\eta$
  bound, and splitting at their balancing scale. Its supremum term is controlled
  too. Scalar H\"older in probability proves the $L^qC^\theta$ Cauchy property.
  For any requested finite $p$, choose an even $q \ge \max(p,q_0)$ and use
  Lyapunov.

  The full H\"older/operator space need not be separable. Nevertheless, every
  finite path lies in the span of the countably many deterministic Lipschitz
  paths $A_j(0,\cdot)$. Their $C^\theta$-closed real span is a separable Banach
  space. In it the same elementary Cauchy/subsequence construction as in
  Lemma~\ref{L:driver-interval-limit} gives a strongly measurable limit $M_R$ on
  the original space. Different $p$ limits agree by uniqueness of convergence in
  probability in the path norm. An almost surely $C^\theta$-convergent
  subsequence converges uniformly in operator norm. Finite paths have compact
  values on short intervals, and on long ones by finite additivity. Therefore,
  the limit has compact values at every time on that single path set. It starts
  at zero. Passing the uniform bounds from~\eqref{E:driver-dyadic-moment-sum}
  through the path convergence proves the asserted $C\sqrt q$ bound. This proves
  Lemma~\ref{L:driver-holder-limit}.
\end{proof}

\begin{proof}[Proof of Proposition~\ref{P:actual-operator-driver}]
  Lemmas~\ref{L:driver-noise-basis}--\ref{L:driver-holder-limit} construct the
  stated finite sums and their path limit, with no new probability space.
  Evaluation at a fixed time agrees with the interval limit by uniqueness in
  probability. Its covariance identity for matrix elements is
  \eqref{E:driver-raw-row}, by Lemma~\ref{L:driver-interval-limit}, including
  the long-interval interpretation. The same additivity identifies each fixed
  increment with $X_R(I)$, proving its stated Schatten moment bound.

  The finite paths are Gaussian linear combinations of deterministic paths.
  Every continuous linear functional on their closed path span converges in
  $L^2$ to a centered Gaussian; finite linear combinations do as well. Thus,
  $M_R$ is a centered Gaussian path. The four real pair identities give its
  complexified sesquilinear row identity.

  The construction used only compact smooth representatives, orthonormality and
  completeness, so it applies to any such noise basis. Compare two constructions
  using a fixed countable dense Schwartz phase collection and rational times.
  Both rows equal the same covariance forms. A countable null-set union,
  boundedness of the operators, phase density and path continuity give equality
  at all times on one probability-one set. This proves indistinguishability
  without an assumed common event for uncountably many original evaluations.

  For $p \ge 1$, choose even $q \ge \max(p,q_0)$ with $q \le p+q_0+2 \le
  (q_0+3)p$. Lemma~\ref{L:driver-holder-limit} and Lyapunov give the first bound
  in~\eqref{E:driver-gaussian-norm}. Write its resulting constant as $C_1$ and
  let $Y = \|M_R\|_{C^\theta}$. If $C_1 = 0$, then $Y = 0$ almost surely and the
  exponential bound is immediate. Otherwise, $\E Y^{2m} \le
  (C_1\sqrt{2m})^{2m}$. Since $m! \ge (m/e)^m$, as follows by integrating $\log
  x$ from one to $m$, the $m$th term in the exponential series is at most
  $(2ecC_1^2)^m$. For $0 < c < 1/(2eC_1^2)$, positive Tonelli and the geometric
  series prove the second bound. This proves
  Proposition~\ref{P:actual-operator-driver}.
\end{proof}

All constants here may depend on $R,T$ and the fixed admissible parameters. No
uniform-in-$R$ path-norm estimate or $C^\theta S_q$-valued path is asserted.
This constructs the local masked integrated operator on the original noise.
Source-cutoff convergence, Young evolution, canonical initial data and
solution-moment asymptotics remain separate parts of the argument.

\section{Frequency estimates and both wave variance directions}
\label{S:wave-variance}

We now apply the covariance and temporal estimates of
Section~\ref{S:masked-covariance} to the full wave phase. We retain
Hypothesis~\ref{H:homog}, write $a = \alpha_0$, and choose and fix
\begin{equation}\label{E:variance-gamma-range}
  0 < \gamma < \min\left(\frac12,\frac{2-\alpha}{2}\right).
\end{equation}
The interval is nonempty because $\alpha+a < 2$. We use $\langle\xi\rangle =
\sqrt{1+|\xi|^2}$ and set
\begin{equation}\label{E:variance-scale-profile}
  B_\alpha(h) =
  \begin{cases}
    1, & \alpha < 1,\\
    1+\log(1/h), & \alpha = 1,\\
    h^{1-\alpha}, & \alpha > 1,
  \end{cases}
  \qquad 0 < h \le 1.
\end{equation}

\subsection{A scalar row estimate for a measure}

Let $\nu$ be a positive Radon measure satisfying
\begin{equation}\label{E:frequency-translated-growth}
  \nu(B(y,r)) \le C r^\alpha,
  \qquad y \in \R^d,\quad r > 0,
\end{equation}
where $C \ge 0$. If the estimate is initially known for open balls, apply it at
radius $r+\epsilon$ and let $\epsilon \downarrow 0$ to obtain the closed-ball
estimate. Neither evenness nor a density is assumed here.
Using~\eqref{E:masked-frequency-pair}, define
\begin{align}
  \Phi_h(\xi,\eta)
  &:= \frac{\mathcal{Q}_h(|\xi|,|\eta|)}
  {\langle\xi\rangle\langle\eta\rangle},
  \label{E:frequency-row-kernel}\\
  J_h(\xi)
  &:= \langle\xi\rangle^{2\gamma}
  \int_{\R^d}\Phi_h(\xi,\xi-\lambda)\nu(\ud\lambda).
  \label{E:frequency-row-integral}
\end{align}
These are initially nonnegative extended integrals. In particular, the notation
denotes a pushforward of $\nu$, not integration against an assumed spectral
density.

\LeanMarker{P:frequency-row}
\begin{proposition}[Uniform scalar row]\label{P:frequency-row}
  There is a finite $K = K(\alpha,a,\gamma)$, independent of $d,\nu,C,h$ and
  $\xi$, such that
  \begin{equation}\label{E:frequency-row-bound}
    \sup_{\xi\in\R^d}J_h(\xi) \le K C B_\alpha(h),
    \qquad 0 < h \le 1.
  \end{equation}
\end{proposition}

\begin{proof}
  If $C = 0$, a countable covering by finite balls gives $\nu = 0$, and the
  result is immediate. Suppose $C > 0$. For each $\xi$, let $\nu_\xi$ be the
  pushforward under $\lambda \mapsto \eta = \xi-\lambda$. Then $\nu_\xi(B(0,R))
  \le CR^\alpha$. For $q \ge 1$, put
  \[
    A_\xi(q) = \int_{|\eta|\le5q}\langle\eta\rangle^{-1}
    \nu_\xi(\ud\eta).
  \]
  Choose $N = \lceil\log_2(5q)\rceil$, so $5q \le 2^N < 10q$. Splitting into the
  unit ball and the annuli $2^j < |\eta| \le 2^{j+1}$ gives
  \[
    A_\xi(q) \le C\left(1+2^\alpha
    \sum_{j=0}^{N-1}2^{j(\alpha-1)}\right).
  \]
  The geometric sum is bounded when $\alpha < 1$, is $N$ when $\alpha = 1$, and
  is at most $2^{N(\alpha-1)}/(2^{\alpha-1}-1)$ when $\alpha > 1$. Hence,
  \begin{equation}\label{E:frequency-inverse-bessel}
    A_\xi(q) \le K_\alpha C
    \begin{cases}
      1, & \alpha < 1,\\
      1+\log q, & \alpha = 1,\\
      q^{\alpha-1}, & \alpha > 1,
    \end{cases}
  \end{equation}
  with a constant depending only on $\alpha$.

  Set $q = \max(1,|\xi|)$ and $\epsilon = 1-2\gamma > 0$. On $|\lambda| \le 4q$
  we have $|\eta| \le 5q$. Both bracket powers in $\mathcal{Q}_h$ are at most
  one, since $a-1 < 0$. The contribution of this region to $J_h(\xi)$ is at most
  \[
    2\langle\xi\rangle^{-\epsilon}A_\xi(q)
    \le 2q^{-\epsilon}A_\xi(q).
  \]
  If $\alpha < 1$, this is uniformly bounded
  by~\eqref{E:frequency-inverse-bessel}. If $\alpha = 1$, the bound follows from
  \[
    e^{-\epsilon u}(1+u) \le 1+\frac1{e\epsilon},
    \qquad u = \log q \ge 0,
  \]
  since $u e^{-\epsilon u}$ has maximum $1/(e\epsilon)$. If $\alpha > 1$, the
  remaining power is $q^{\alpha+2\gamma-2}$, whose exponent is negative
  by~\eqref{E:variance-gamma-range}. Thus, the near contribution is at most a
  constant times $C$ in all three cases.

  On the far region write $r = |\lambda| > 4q$. Then
  \[
    |\eta| \ge \frac{3r}4,\qquad
    \big||\eta|-|\xi|\big| \ge \frac r2,\qquad
    |\eta|+|\xi| \ge \frac r2.
  \]
  Each oscillatory bracket is at least $\langle hr\rangle/2$, whereas
  $\langle\eta\rangle^{-1} \le 4/(3r)$ and $\langle\xi\rangle^{-\epsilon} \le
  1$. Consequently, the far contribution is at most
  \[
    \frac43\,2^{2-a}\int_{|\lambda|>4q}
    |\lambda|^{-1}\langle h|\lambda|\rangle^{a-1}\nu(\ud\lambda).
  \]
  Enlarge the domain to $|\lambda| > 1$ and split it into dyadic annuli.
  Applying~\eqref{E:frequency-translated-growth} to their masses bounds this
  expression by a constant depending only on $\alpha,a$ times $C S(h)$, where
  \[
    S(h) = \sum_{j=0}^\infty
    2^{j(\alpha-1)}\langle h2^j\rangle^{a-1}.
  \]
  Choose $L = \lfloor\log_2(1/h)\rfloor \ge 0$. For $j \le L$ the bracket power
  is at most one. For $j \ge L+1$ it is at most $(h2^j)^{a-1}$. Since
  $\alpha+a-2 < 0$, the tail is
  \[
    \frac{h^{a-1}2^{(L+1)(\alpha+a-2)}}{1-2^{\alpha+a-2}}
    \le \frac{h^{1-\alpha}}{1-2^{\alpha+a-2}}.
  \]
  The inequality uses $2^{L+1} > 1/h$ and the negative exponent. The finite sum
  is bounded by $1/(1-2^{\alpha-1})$ for $\alpha < 1$, by $1+\log_2(1/h)$ for
  $\alpha = 1$, and by
  \[
    \frac{2^{\alpha-1}h^{1-\alpha}}{2^{\alpha-1}-1}
  \]
  for $\alpha > 1$. Since $h^{1-\alpha} \le 1$ in the first case, we obtain
  $S(h) \le K_{\alpha,a}B_\alpha(h)$. Combining the near and far bounds, and
  using $B_\alpha(h) \ge 1$, proves the assertion and finiteness of the original
  integral. All constants arose from numerical shell estimates and are
  independent of $d$. This proves Proposition~\ref{P:frequency-row}.
\end{proof}

For $\alpha = 1$ the profile $B_\alpha(h)$ in
Proposition~\ref{P:frequency-row} carries a logarithm. After multiplication
by $h^{2-a}$, we absorb it into a power of $h$.

\LeanMarker{C:frequency-time}
\begin{corollary}[A time exponent above one half]\label{C:frequency-time}
  For every
  \begin{equation}\label{E:variance-beta-range}
    \frac12 < \beta < \min\left(1-\frac a2,
    \frac{3-a-\alpha}{2}\right),
  \end{equation}
  there is a finite $K_\beta = K_\beta(\alpha,a,\gamma,\beta)$ such that
  \begin{equation}\label{E:frequency-time-bound}
    h^{2-a}\sup_\xi J_h(\xi) \le K_\beta C h^{2\beta},
    \qquad 0 < h \le 1.
  \end{equation}
\end{corollary}

\begin{proof}
  Both upper endpoints in~\eqref{E:variance-beta-range} exceed one half, by $a <
  1$ and $\alpha+a < 2$. If $\alpha < 1$, use $2-a > 2\beta$ in
  Proposition~\ref{P:frequency-row}. If $\alpha > 1$, use $3-a-\alpha > 2\beta$.
  For $\alpha = 1$, set $\delta = 2-a-2\beta > 0$ and $u = \log(1/h)$. Then
  \[
    h^\delta(1+\log(1/h))
    = e^{-\delta u}(1+u) \le 1+\frac1{e\delta}.
  \]
  This absorbs the critical logarithm. No endpoint value of $\beta$ is used.
  This proves Corollary~\ref{C:frequency-time}.
\end{proof}

\subsection{Forms on the full phase}

Choose and fix $T > 0$. Set $\Lambda = (1-\Delta)^{1/2}$, $s = 1/2$ and $E_r =
H^r(\R^d)\times H^{r-1}(\R^d)$. On the complexification of $E = E_s$, use inner
products linear in the first slot. The map
\[
  Jz(\xi) = \left(\langle\xi\rangle^{1/2}\widehat{z}_1(\xi),
  \langle\xi\rangle^{-1/2}\widehat{z}_2(\xi)\right)
\]
is unitary onto $L^2(\R^d;\mathbb{C}^2)$. The full massless wave group has
Fourier matrix
\begin{equation}\label{E:variance-free-matrix}
  S(t,\xi) =
  \begin{pmatrix}
    \cos(t|\xi|) & \sin(t|\xi|)/|\xi|\\
    -|\xi|\sin(t|\xi|) & \cos(t|\xi|)
  \end{pmatrix},
\end{equation}
where the sine quotient is $t$ at zero. Let $B(b)(u,v) = (0,bu)$. Choose and fix
a real even nonnegative $\zeta \in C_c^\infty(\R^d)$ equal to one on $B_1$ and
zero off $B_2$, and put $\zeta_R(x) = \zeta(x/R)$ for $R > 0$. For $I =
[t_0,t_0+h] \subset [0,T]$, $0 < h \le 1$, and Schwartz phases $z,w$, put
\begin{align}
  v_z(t) &= [S(t)z]_1,
  \label{E:variance-displacement}\\
  G_w(t) &= -\frac{\sin(t|D|)}{|D|}\Lambda w_1
  +\cos(t|D|)\Lambda^{-1}w_2,
  \label{E:variance-dual-wave}\\
  F_{z,w}(t,x) &= 1_I(t)v_z(t,x)\overline{G_w(t,x)},
  \qquad \ell_R(I;z,w) = W_{\mathbb{C}}(\zeta_RF_{z,w}),
  \label{E:variance-actual-form}
\end{align}
where $W_{\mathbb{C}}(F) = W(\operatorname{Re}F)+iW(\operatorname{Im}F)$. There
is no observation cutoff in this definition.

We first assume $\Gamma \ne 0$. If $\Gamma = 0$, all forms and the column maps
below are zero, so their bounds hold directly. For the nonzero case, let $\nu_R$
be~\eqref{E:masked-auxiliary-measure} and set
\begin{equation}\label{E:variance-actual-growth-constant}
  L_\zeta = \|\widehat{\zeta}\|_1,\qquad
  C_\nu = c_d^2L_\zeta^2C_\Gamma.
\end{equation}
Proposition~\ref{P:masked-auxiliary-norm} gives a positive even tempered Radon
measure with $\nu_R(B(y,r)) \le C_\nu r^\alpha$, uniformly in $R$.

\LeanMarker{P:actual-wave-form}
\begin{proposition}[Form and phase kernel]\label{P:actual-wave-form}
  The tests $\zeta_RF_{z,w}$ in~\eqref{E:variance-actual-form} belong to the
  original noise completion. The form $\ell_R$ is linear in $z$ and conjugate
  linear in $w$, and uses the same $W$ as the model. For a smooth deterministic
  coefficient $b$ with common compact spatial support and bounded spatial
  derivatives on $[0,T]$,
  \begin{equation}\label{E:variance-physical-adjoint}
    \langle S(-t)B(b(t))S(t)z,w\rangle_E
    = \int b(t,x)v_z(t,x)\overline{G_w(t,x)}\ud x.
  \end{equation}
  For a real compact smooth $\chi$, multiplication $D_\chi$ on both phase
  entries has adjoint
  \begin{equation}\label{E:variance-cutoff-adjoint}
    D_\chi^{*,E}w
    = \left(\Lambda^{-1}\chi\Lambda w_1,
    \Lambda\chi\Lambda^{-1}w_2\right).
  \end{equation}
  With $U = Jz$, $V = Jw$, $r = |\xi|$, $\rho = |\eta|$ and the continuous
  functions $c_r,b_r$ of Lemma~\ref{L:masked-continuous-wave}, set
  \begin{equation}\label{E:variance-normalized-kernel}
    \mathcal{K}(t;\xi,\eta)
    = \frac1{\sqrt{\langle\xi\rangle\langle\eta\rangle}}
    \begin{pmatrix}
      -b_rc_\rho & -b_rb_\rho\\
      c_rc_\rho & c_rb_\rho
    \end{pmatrix}(t).
  \end{equation}
  Then
  \begin{equation}\label{E:variance-normalized-covariance}
    \E|\ell_R(I;z,w)|^2
    \le \int\left\|\int\sum_{i,j=1}^2
    1_I\mathcal{K}_{ij}(\cdot;\eta+\lambda,\eta)
    U_j(\eta)\overline{V_i(\eta+\lambda)}\ud\eta
    \right\|_{H_a^{\mathbb{C}}}^2\nu_R(\ud\lambda).
  \end{equation}
\end{proposition}

\begin{proof}
  The wave multipliers are smooth at zero through the continuous sine quotient,
  with polynomially bounded frequency derivatives on finite time intervals.
  Bessel multipliers preserve Schwartz space. Thus, $v_z,G_w$ and their product
  are continuous in all Schwartz seminorms on $[0,T]$. For tests supported in a
  fixed compact $K$, the absolute spatial pairing is bounded by
  \[
    |K|\Gamma(K-K)\|F(t)\|_\infty\|H(u)\|_\infty.
  \]
  Indeed, only $y \in K-K$ contributes to $\int\Gamma(\ud y)\int
  F(t,x)\overline{H(u,x-y)}\ud x$. Combine this with the temporal Schur bound
  and approximate the interval indicator and the continuous time dependence by
  smooth interior temporal tests, keeping the spatial support of $\zeta_R$. This
  admits the masked product in the original completion in every allowed
  dimension.

  We also identify its auxiliary and spectral limits. Approximate a
  Schwartz-valued function on $I$ by time partitions with frozen spatial values,
  and then replace the finitely many indicators by compact smooth temporal
  functions. A diagonal sequence $F_n$ of finite temporal tensors converges in
  $L^2$ time in every Schwartz seminorm. Fourier integration by parts and the
  spatial weighted supremum bounds give, for sufficiently large fixed $N$,
  \[
    \|\widehat{F_n-F}(\xi)\|_{L^2(0,T)}
    \le \epsilon_n\langle\xi\rangle^{-N},\qquad \epsilon_n \to 0.
  \]
  Choose $N$ so that the square weight is integrable against both tempered
  measures $\mu$ and $\nu_R$. Lemma~\ref{L:masked-time-completion} gives
  convergence in their spectral $H_a$ norms. Multiplication by the fixed
  $\zeta_R$ preserves these $L^2$ Schwartz-seminorm limits, so the masked
  differences converge in the $\mu$ spectral norm too. The preceding
  compact-support estimate also gives convergence in the original completion.
  Hence, the auxiliary extension and original-noise limit agree on these
  products. Constants for this admission argument may depend on $R$; they are
  not the uniform variance constants below.

  For real tensor tests $f,g$, the Hermitian symmetry of their spatial
  transforms gives
  \[
    A(-\xi) = \overline{A(\xi)},\qquad
    A(\xi) = \langle\widehat{f}(\xi),\widehat{g}(\xi)\rangle_{H_a^{\mathbb{C}}}.
  \]
  Integration against either even $\mu$ or even $\nu_R$ cancels the imaginary
  mixed term. Expanding the squared norm of $\widehat{f}+i\widehat{g}$, then
  using the preceding approximation, extends
  Proposition~\ref{P:masked-noise-completion} to the present complex tests:
  \begin{equation}\label{E:variance-complex-noise}
    \E|W_{\mathbb{C}}(\zeta_RF)|^2
    = (2\pi)^d\int\|\widehat{\zeta_RF}(\xi)\|_{H_a^{\mathbb{C}}}^2\mu(\ud\xi)
    \le (2\pi)^d\int\|\widehat{F}(\lambda)\|_{H_a^{\mathbb{C}}}^2\nu_R(\ud\lambda).
  \end{equation}
  This is a bound for the masked noise, not a declaration that its law is
  stationary. Complex linearity of $W_{\mathbb{C}}$ and linearity of $v_z,G_w$
  prove the stated sesquilinearity.

  Smooth multiplication is bounded on each Sobolev space: Fourier convolution,
  the inequality $\langle\xi\rangle^r\langle\eta\rangle^{-r} \le
  2^{|r|/2}\langle\xi-\eta\rangle^{|r|}$ and weighted Schwartz decay reduce this
  to the $L^1*L^2$ bound. The free matrix is strongly continuous and bounded on
  $E$ on compact time intervals, directly by its weighted Fourier entries. Thus,
  the ordinary first-response integral exists on states. Since $S(-t)(0,f) =
  (-\sin(t|D|)f/|D|,\cos(t|D|)f)$, moving the real self-adjoint multipliers and
  the two Sobolev weights onto $w$ proves \eqref{E:variance-physical-adjoint}.
  The same weighted duality gives \eqref{E:variance-cutoff-adjoint}. Inserting
  this adjoint in $G_w$ is the correctly observed form; $D_\chi$ need not be
  self-adjoint on $E$. This deterministic identity does not identify a
  positive-width source with the original noise or assert a source limit.

  The definitions give exactly
  \begin{align*}
    \widehat{v}_z(t,\eta)
    &= \langle\eta\rangle^{-1/2}
    [c_\rho(t)U_1(\eta)+b_\rho(t)U_2(\eta)],\\
    \widehat{G}_w(t,\xi)
    &= \langle\xi\rangle^{-1/2}
    [-b_r(t)V_1(\xi)+c_r(t)V_2(\xi)].
  \end{align*}
  Their product gives~\eqref{E:variance-normalized-kernel}. At zero frequencies
  one uses the continuous sine quotient, not division by zero; at $\xi = \eta =
  0$ the matrix is
  $\left(\begin{smallmatrix}-t&-t^2\\1&t\end{smallmatrix}\right)$. Fourier
  multiplication with the conjugate yields
  \[
    \widehat{F}_{z,w}(t,-\lambda)
    = c_d1_I(t)\int\sum_{i,j=1}^2\mathcal{K}_{ij}(t;\eta+\lambda,\eta)
    U_j(\eta)\overline{V_i(\eta+\lambda)}\ud\eta.
  \]
  The spatial integral is absolutely convergent uniformly on the finite time
  interval by Schwartz decay. The Schur bound also makes it a Bochner
  $H_a^{\mathbb{C}}$ integral. Reflect $\lambda$ using evenness of $\nu_R$
  in~\eqref{E:variance-complex-noise}. Since $(2\pi)^dc_d^2 = 1$, this
  proves~\eqref{E:variance-normalized-covariance} with no remaining Fourier
  factor. This proves Proposition~\ref{P:actual-wave-form}.
\end{proof}

\subsection{Bessel's inequality and the two variances}

Proposition~\ref{P:actual-wave-form} bounds the variance of one form
$\ell_R(I;z,w)$. To sum it over an orthonormal family of phases, we need a
Bessel inequality with values in $H_a^{\mathbb{C}}$.
Proposition~\ref{P:wave-two-variances} uses it in both directions.

\LeanMarker{L:phase-bessel}
\begin{lemma}[Hilbert-valued finite Bessel inequality]\label{L:phase-bessel}
  Let $H$ be a separable complex Hilbert space, let $\{u_k\}$ be a finite
  orthonormal family in $L^2(\R^d;\mathbb{C}^2)$, and let $g = (g_1,g_2) \in
  L^2(\R^d;H^2)$. Then
  \begin{equation}\label{E:phase-bessel}
    \sum_k\left\|\int\sum_{j=1}^2u_{k,j}(\eta)g_j(\eta)\ud\eta\right\|_H^2
    \le \int\sum_{j=1}^2\|g_j(\eta)\|_H^2\ud\eta.
  \end{equation}
  The same bound holds with $\overline{u_{k,j}}$ in place of $u_{k,j}$.
\end{lemma}

\begin{proof}
  Expand $g$ in a countable orthonormal basis of $H$. On each scalar coefficient
  apply scalar Bessel to the orthonormal family $\{\overline{u_k}\}$, or to
  $\{u_k\}$ for the conjugated version. Scalar finite Bessel follows by
  subtracting an orthogonal finite projection and expanding its nonnegative
  squared norm. Sum over the $H$ basis and use Parseval and nonnegative Tonelli.
  The Bochner integrals exist by $L^2$ Cauchy--Schwarz. A finite-dimensional $H$
  uses its finite basis, and the zero space is immediate. No independence is
  involved. This proves Lemma~\ref{L:phase-bessel}.
\end{proof}

\LeanMarker{P:wave-two-variances}
\begin{proposition}[Both variance directions]\label{P:wave-two-variances}
  There is a finite $C_T$, depending on the fixed model, mask,
  $a,\alpha,\gamma,T$, but not on $R,I$ or the orthonormal family, such that for
  every finite complex $E$-orthonormal Schwartz family $\{e_k\}$ and every
  Schwartz phase $z$,
  \begin{align}
    \sum_k\E|\ell_R(I;e_k,z)|^2
    &\le C_T h^{2-a}B_\alpha(h)\|z\|_{E_{s-\gamma}}^2,
    \label{E:wave-input-variance}\\
    \sum_k\E|\ell_R(I;z,e_k)|^2
    &\le C_T h^{2-a}B_\alpha(h)\|z\|_{E_{s-\gamma}}^2.
    \label{E:wave-output-variance}
  \end{align}
  For every $\beta$ in~\eqref{E:variance-beta-range}, either right side may be
  replaced by $C_{T,\beta}h^{2\beta}\|z\|_{E_{s-\gamma}}^2$.
\end{proposition}

\begin{proof}
  Each entry in~\eqref{E:variance-normalized-kernel} is a product of two
  continuous wave factors times $(\langle\xi\rangle\langle\eta\rangle)^{-1/2}$,
  with a possible minus sign. Lemma~\ref{L:masked-continuous-wave} therefore
  gives
  \begin{equation}\label{E:variance-kernel-time-bound}
    \|1_I\mathcal{K}_{ij}(\cdot;\xi,\eta)\|_{H_a^{\mathbb{C}}}^2
    \le A_T h^{2-a}\Phi_h(\xi,\eta),
    \qquad A_T = C_1(a)(1+T)^4.
  \end{equation}
  The positive even Radon measure $\nu_R$ satisfies the growth bound
  \eqref{E:variance-actual-growth-constant}. Thus, all hypotheses of
  Proposition~\ref{P:frequency-row} hold, and
  \begin{equation}\label{E:variance-actual-row-bound}
    \int\Phi_h(\xi,\xi-\lambda)\nu_R(\ud\lambda)
    \le K C_\nu B_\alpha(h)\langle\xi\rangle^{-2\gamma}.
  \end{equation}
  No nonnegative physical inverse covariance of $\nu_R$ is needed.

  The kernels used in Lemma~\ref{L:phase-bessel} belong to its stated $L^2$
  space for each fixed $\lambda$. Indeed, $|c_r(t)| \le 1$ and $|b_r(t)| \le
  \sqrt{2}\max(1,T)$ on $[0,T]$, by the high- and low-frequency representations.
  The kernels are continuous into $L^2(0,T)$ in their frequency variables, hence
  into $H_a^{\mathbb{C}}$ by Schur. Separability gives strong measurability.
  Multiplication by shifted Schwartz components of $U$ or $V$ then gives the
  required $L^2$ integrability. Later $\lambda$ integrals are of nonnegative
  functions; their finiteness follows from the estimates below.

  First fix $V = Jz$ and let $U_k = Je_k$ be the input family.
  In~\eqref{E:variance-normalized-covariance} set
  \[
    g_j^\lambda(\eta) = \sum_{i=1}^2
    1_I\mathcal{K}_{ij}(\cdot;\eta+\lambda,\eta)
    \overline{V_i(\eta+\lambda)}.
  \]
  Lemma~\ref{L:phase-bessel} and Cauchy--Schwarz over the two output components
  give
  \begin{align*}
    \sum_j\|g_j^\lambda(\eta)\|_{H_a^{\mathbb{C}}}^2
    &\le 2\sum_{i,j}|V_i(\eta+\lambda)|^2
    \|1_I\mathcal{K}_{ij}(\cdot;\eta+\lambda,\eta)\|_{H_a^{\mathbb{C}}}^2\\
    &\le 4A_T h^{2-a}\Phi_h(\eta+\lambda,\eta)
    \sum_i|V_i(\eta+\lambda)|^2.
  \end{align*}
  Change variables $\xi = \eta+\lambda$ in the Lebesgue integral for each
  $\lambda$, apply Tonelli and~\eqref{E:variance-actual-row-bound}, and use
  $\|z\|_{E_{s-\gamma}} = \|\langle\xi\rangle^{-\gamma}Jz\|_2$. This
  proves~\eqref{E:wave-input-variance} with $C_T = 4A_TKC_\nu$.

  Next fix $U = Jz$ and let $V_k = Je_k$ be the output family. After $\eta =
  \xi-\lambda$ in~\eqref{E:variance-normalized-covariance}, put
  \[
    f_i^\lambda(\xi) = \sum_{j=1}^2
    1_I\mathcal{K}_{ij}(\cdot;\xi,\xi-\lambda)U_j(\xi-\lambda).
  \]
  Apply the conjugated version of Lemma~\ref{L:phase-bessel}. Cauchy--Schwarz
  over the two input components bounds its right side by
  \[
    4A_T h^{2-a}\int\Phi_h(\xi,\xi-\lambda)
    \sum_j|U_j(\xi-\lambda)|^2\ud\xi.
  \]
  After the change of variables and Tonelli, the row at fixed $\eta$ is
  $\int\Phi_h(\eta+\lambda,\eta)\nu_R(\ud\lambda)$. Reflect $\lambda$ using the
  evenness of $\nu_R$ and then use symmetry of $\Phi_h$. This row becomes
  $\int\Phi_h(\eta,\eta-\lambda)\nu_R(\ud\lambda)$, which is bounded
  by~\eqref{E:variance-actual-row-bound} with $\xi = \eta$. This proves
  \eqref{E:wave-output-variance} with the same $C_T$.

  Use Corollary~\ref{C:frequency-time} for $h^{2-a}$ times the row integral at
  the two applications above to obtain the asserted $h^{2\beta}$ bounds,
  including strict absorption of the critical logarithm. The constants are
  uniform in $R,I$ and family size. The $R$-dependent admission estimate was not
  used as this uniform variance bound. This proves
  Proposition~\ref{P:wave-two-variances}.
\end{proof}

\subsection{Mean-square columns on the original probability space}

Write $E_r^{\mathrm{real}}$ for the real phase space in dimension $d$.

\begin{proposition}[Column and adjoint-column maps]\label{P:wave-square-columns}
  On the same probability space there are deterministic bounded real linear maps
  \[
    m_{R,I},n_{R,I}:E_{s-\gamma}^{\mathrm{real}}\longrightarrow
    L^2(\Omega;E_s^{\mathrm{real}}),
  \]
  with squared bounds $C_T h^{2-a}B_\alpha(h)$, such that, for every fixed real
  Schwartz pair $z,w$,
  \begin{equation}\label{E:variance-square-columns}
    \langle m_{R,I}z,w\rangle_E
    = \ell_R(I;z,w) = \langle z,n_{R,I}w\rangle_E
    \quad\text{in }L^2(\Omega).
  \end{equation}
  Fixed-vector values and finite families of values are centered Gaussian.
\end{proposition}

\begin{proof}
  Choose a countable real Schwartz orthonormal basis $\{e_k\}$ of $E_s$. Such a
  basis follows by Gram--Schmidt from a countable dense Schwartz family,
  discarding zero residuals. Schwartz density follows by smooth Fourier cutoff
  approximation in the weighted $L^2$ norms: approximate by compact smooth
  Fourier coordinates, replace $g(\xi)$ by $(g(\xi)+\overline{g(-\xi)})/2$, and
  apply the inverse Bessel/Fourier maps. The resulting phases are real Schwartz,
  and Gram--Schmidt preserves that class. The same density holds in
  $E_{s-\gamma}$.

  For fixed real Schwartz $z$, define
  \[
    m_N(z) = \sum_{k\le N}\ell_R(I;z,e_k)e_k,
    \qquad n_N(z) = \sum_{k\le N}\ell_R(I;e_k,z)e_k.
  \]
  Both are finite Gaussian sums on the original $W$. By
  Proposition~\ref{P:wave-two-variances}, their coefficient expected squares
  have finite sum. Orthogonality gives the squared tail norms as the
  corresponding sums of expected squares. Thus, both sequences converge in
  $L^2(\Omega;E_s)$ to $m(z),n(z)$ with the asserted bounds, without
  independence of the coefficients. Finite-sum measurability and separable $E_s$
  give strong measurability of the limits. The bounded real linear maps extend
  uniquely from Schwartz phases to $E_{s-\gamma}$.

  If $w_N$ is the $E_s$ projection of a Schwartz $w$ onto the first $N$ basis
  vectors, then $\langle m_N(z),w\rangle_E = \ell_R(I;z,w_N)$. The singleton
  form of~\eqref{E:wave-output-variance}, after normalizing $w-w_N$ in $E_s$,
  gives
  \[
    \E|\ell_R(I;z,w-w_N)|^2
    \le C_T h^{2-a}B_\alpha(h)\|z\|_{E_{s-\gamma}}^2\|w-w_N\|_{E_s}^2.
  \]
  The zero residual is immediate. The right side tends to zero, giving the first
  identity in~\eqref{E:variance-square-columns}. For the second, project $z$
  instead and use the singleton input bound \eqref{E:wave-input-variance}. A
  fixed countable Schwartz orthonormal basis and Parseval show basis
  independence of the constructed limits.

  Continuous linear functionals of the limits, and finite linear combinations of
  such functionals, are $L^2$ limits of centered Gaussian variables with
  convergent variances. Their characteristic functions therefore converge to
  those of centered Gaussians. The same reasoning after the bounded input
  extension proves joint Gaussianity for arbitrary fixed inputs and finite
  families of the two maps. This proves Proposition~\ref{P:wave-square-columns}.
\end{proof}

These are maps into $L^2(\Omega;E_s)$ for fixed intervals. No common uncountable
almost-sure version, random bounded operator, Hilbert--Schmidt realization, time
path, source convergence or new solution-moment bound is inferred by this
section alone. Both variance directions are retained in
Section~\ref{S:actual-operator-driver}.

\subsection{Gaussian transfer in the full path norm}

\LeanMarker{L:source-finite-comparison}
\begin{lemma}[Finite Gaussian linear images]\label{L:source-finite-comparison}
  Let $v_1,\ldots,v_N$ lie in a real Banach space and let $\xi$ be a centered
  Gaussian vector with covariance bounded by $CI$, $C \ge 0$. For independent
  standard real $g_j$ and $p \ge 1$,
  \begin{equation}\label{E:source-finite-comparison}
    \left(\E\left\|\sum_j\xi_jv_j\right\|^p\right)^{1/p}
    \le\sqrt C\left(\E\left\|\sum_jg_jv_j\right\|^p\right)^{1/p}.
  \end{equation}
\end{lemma}

\begin{proof}
  The function $\varphi(x) = \|\sum_jx_jv_j\|^p$ is continuous and convex, even
  when the linear image gives only a seminorm on $\R^N$. Its Gaussian
  expectations are finite by the finite-sum bound. On an auxiliary space take an
  independent centered Gaussian $Z$ of covariance $CI-\operatorname{Cov}\xi$.
  Conditional Jensen gives $\E\varphi(\xi) \le \E\varphi(\xi+Z)$. The latter
  vector has the law of $\sqrt C\,g$, proving the claim. The case $C = 0$ is
  immediate. This auxiliary comparison changes neither the coupling of $\xi$ nor
  any nonlinear wave solution. This proves
  Lemma~\ref{L:source-finite-comparison}.
\end{proof}

Lemma~\ref{L:source-finite-comparison} compares two finite Gaussian sums with
the same vectors. We apply it to the coefficient paths $A_j$, keeping the
original basis $f_j$ and replacing $W(f_j)$ by the evaluations of another
Gaussian noise.

\begin{proposition}[Gaussian transfer and stability]
  \label{P:source-gaussian-transfer}
  Let $Y$ be a centered Gaussian linear noise on smooth compact positive-time
  tests, on a common coupled space, with
  \begin{equation}\label{E:source-raw-variance-bound}
    \operatorname{Var}Y(F)\le C\|F\|_{H_W}^2.
  \end{equation}
  It extends to $H_W$. The finite paths $M_{Y,N} = \sum_{j\le N}Y(f_j)A_j$
  converge in every finite $L^p(\mathscr{B}_\theta)$ to a centered Gaussian path
  $M_Y$ with compact values at all times on one path event, and
  \begin{align}
    \|M_Y-M_{Y,N}\|_{L^p\mathscr{B}_\theta}
    &\le\sqrt C\|M_R-M_N\|_{L^p\mathscr{B}_\theta},\label{E:source-transfer-tail}\\
    \|M_Y\|_{L^p\mathscr{B}_\theta}
    &\le\sqrt C\|M_R\|_{L^p\mathscr{B}_\theta}.\label{E:source-transfer-norm}
  \end{align}
  For a fixed real Schwartz pair and time,
  \begin{equation}\label{E:source-transferred-row}
    \langle M_Y(t)z,w\rangle_E=Y(h_{R,t;z,w})\quad\text{in }L^2,
  \end{equation}
  where $h_{R,t;z,w}$ is the explicit original test class
  $\zeta_R1_{(0,t)}v_zG_w$ with the adjoint from~\eqref{E:variance-dual-wave}.
  The path is basis-independent up to indistinguishability, is linear in jointly
  Gaussia noise families, and complexifies sesquilinearly.

  If $(W,Y_n)$ are jointly centered Gaussian, $Y_n$ and $D_n = W-Y_n$ have
  uniform bounds of the form~\eqref{E:source-raw-variance-bound}, and
  $\operatorname{Var}D_n(F) \to 0$ on every smooth compact test, then $M_{Y_n}
  \to M_R$ in every finite $L^p(\mathscr{B}_\theta)$.
\end{proposition}

\begin{proof}
  The variance bound annihilates covariance-null tests and extends $Y$ as a
  bounded linear map into its $L^2$ space. Completion preserves Gaussian
  finite-dimensional laws. For each finite original orthonormal family, the
  covariance of its $Y$ evaluations is bounded by $CI$, by applying the variance
  bound to every real linear combination. Independence of $Y(f_j)$ is not
  assumed.

  Apply Lemma~\ref{L:source-finite-comparison} with $v_j = A_j(\cdot)$ in
  $\mathscr{B}_\theta$, also to finite tails. The standard comparison vector has
  the law of the original $W(f_j)$. The already-proved original path convergence
  makes $M_{Y,N}$ Cauchy in every $L^p$. It lies in the same separable closed
  span of the countably many deterministic paths. The elementary
  Cauchy/subsequence construction of Lemma~\ref{L:driver-holder-limit} gives a
  strongly measurable limit on the space of $Y$. Different $p$ limits agree by
  convergence in probability. Passing to limits in the finite inequalities gives
  both displayed bounds. An almost surely uniformly convergent subsequence gives
  compact values for all times. Functional $L^2$ limits of Gaussian linear
  combinations prove that the path is Gaussian.

  For real Schwartz $z,w$, the coefficient identity is
  \[
    \langle A_j(t)z,w\rangle_E
    =\E[W(h_{R,t;z,w})W(f_j)]
    =\langle h_{R,t;z,w},f_j\rangle_{H_W}.
  \]
  Therefore, the finite transferred row is $Y(P_Nh)$, with $P_N$ the original
  $H_W$ orthogonal projection. Since $P_Nh \to h$, the noise bound gives
  $Y(P_Nh) \to Y(h)$ in $L^2$. Operator-path $L^2$ convergence
  proves~\eqref{E:source-transferred-row}. Long intervals use the same
  covariance completion and endpoint nullity as in
  Lemma~\ref{L:driver-interval-limit}. No filter is commuted through a physical
  mask or wave group. Four real pair identities give the complex assertion.
  Countably many dense Schwartz phases and rational times, followed by phase
  density and path continuity, prove basis indistinguishability.

  For jointly Gaussian $Y,Z$, their real linear combinations have an original
  variance bound by the $L^2$ triangle inequality. The finite sums are linear in
  the noise, hence so are the path limits. In the stability assertion $D_n$ is
  centered Gaussian by the joint hypothesis. Its uniform bound gives
  \[
    \|M_{D_n}-M_{D_n,N}\|_{L^p\mathscr{B}_\theta}
    \le\sqrt{C_D}\|M_R-M_N\|_{L^p\mathscr{B}_\theta}.
  \]
  For fixed $N$, the covariance matrix of $(D_n(f_j))_{j\le N}$ is positive and
  bounded by $s_N(n)I$, where $s_N(n) = \sum_{j\le N}\operatorname{Var}D_n(f_j)
  \to 0$. The finite comparison gives $\|M_{D_n,N}\|_{L^p\mathscr{B}_\theta} \le
  \sqrt{s_N(n)}\|M_N\|_{L^p\mathscr{B}_\theta}$. Choose $N$ large for the
  uniform tail, then $n$ large for the finite head. Linearity identifies
  $M_{D_n} = M_R-M_{Y_n}$. The same argument proves a multiparameter limit
  whenever each fixed test error tends to zero, with no relation between
  parameter rates. This proves Proposition~\ref{P:source-gaussian-transfer}.
\end{proof}

\subsection{The spatial covariance space and an extension preserving the given noise}

Proposition~\ref{P:source-gaussian-transfer} reduces source-path convergence
to uniform variance domination and vanishing residual variance on each test,
for noises jointly Gaussian with $W$. We now construct such regularized
sources, beginning with the spatial covariance space and heat regularization.

\LeanMarker{L:source-spatial-heat}
\begin{lemma}[Original spatial space and heat]\label{L:source-spatial-heat}
  Let $H_\Gamma$ be the real test-null completion with
  \[
    B_\Gamma(f,g)=\int\Gamma(\ud y)\int f(x)g(x-y)\ud x
    =(2\pi)^d\int\widehat{f}(\xi)\overline{\widehat{g}(\xi)}\mu(\ud\xi).
  \]
  It is separable and includes Schwartz tests. With
  \begin{equation}\label{E:source-heat-definition}
    p_\varepsilon(x)=(2\pi\varepsilon)^{-d/2}
    e^{-|x|^2/(2\varepsilon)},\qquad C_\varepsilon f=p_\varepsilon*f,
  \end{equation}
  heat is a positive self-adjoint contraction on $H_\Gamma$, $I-C_\varepsilon$
  is a contraction, and $C_\varepsilon \to I$ strongly.
\end{lemma}

\begin{proof}
  The spectral normalization is the spatial computation in
  Proposition~\ref{P:masked-noise-completion}; its integral is real on real
  tests by evenness of $\mu$. The local bound $|B_\Gamma(f,g)| \le
  |K|\Gamma(K-K)\|f\|_\infty\|g\|_\infty$ on a common compact support gives
  separability by countable smooth approximation. Expanding smooth cutoffs of a
  Schwartz test converge in Schwartz seminorms, and its covariance correlation
  pairing converges against tempered $\Gamma$. Thus, Schwartz tests enter the
  same completion. The Fourier multiplier of $C_\varepsilon$ is
  $m_\varepsilon(\xi) = e^{-\varepsilon|\xi|^2/2}$. On dense Schwartz tests, the
  displayed norm gives positivity, self-adjointness and both contraction bounds.
  Dominated convergence gives strong convergence there; uniform boundedness
  extends it to the completion. This uses no physical mask operator on a noise
  quotient. This proves Lemma~\ref{L:source-spatial-heat}.
\end{proof}

Set $\nu_a(\ud\lambda) = \lambda^{a-1}\ud\lambda/\Gamma_E(a)$ and
\[
  \mathcal{G}=L^2((0,\infty)_\lambda\times\R_r,
  \nu_a(\ud\lambda)\ud r;H_\Gamma).
\]
This is a real separable Hilbert space: countable simple functions on rational
rectangles with a countable dense spatial collection suffice.

\begin{lemma}[Feature isometry preserving the given noise]\label{L:source-retaining-isometry}
  On finite smooth temporal-spatial tensors define
  \begin{equation}\label{E:source-feature-isometry}
    \Phi F(\lambda,r)=\sqrt{2\lambda}
    \int_{\substack{t>0\\t\ge r}}e^{-\lambda(t-r)}F(t,\cdot)\ud t.
  \end{equation}
  It extends to an isometry $\Phi:H_W \to \mathcal{G}$ with closed range, and
  the formula represents smooth compact tests almost everywhere. On a product
  extension retaining the original $W$, there is an isonormal $\mathcal{I}$ on
  $\mathcal{G}$ with
  \begin{equation}\label{E:source-given-W-retention}
    \mathcal{I}(\Phi h)=W(h)\quad\text{in }L^2
    \quad\text{for each }h\in H_W.
  \end{equation}
\end{lemma}

\begin{proof}
  The identity
  \[
    \int_{-\infty}^{\min(t,u)}
    2\lambda e^{-\lambda(t+u-2r)}\ud r
    =e^{-\lambda|t-u|}
  \]
  and the original Laplace formula give $\|\Phi F\|_{\mathcal{G}}^2 =
  \|F\|_{H_W}^2$. All signed exchanges for finite tensors are absolutely
  integrable: compact temporal factors give integrals of
  $|\tau(t)\sigma(u)||t-u|^{-a}$, finite for $a < 1$, times fixed finite spatial
  inner products. No oscillatory infinite-measure Fubini is used. Smooth time
  partitions on one compact support approximate a general smooth spacetime test
  uniformly by finite tensors. The local covariance
  bound~\eqref{E:driver-test-continuity} proves their density in $H_W$. Hence,
  the isometry extends with closed range.

  For a general smooth compact $F$, those approximants also converge uniformly
  in spatial $H_\Gamma$ norm in time, by local Radon mass. For fixed
  $\lambda,r$, their integrals in~\eqref{E:source-feature-isometry} converge in
  $H_\Gamma$. Meanwhile the isometry gives convergence in $\mathcal{G}$, with an
  almost everywhere convergent subsequence. Its two limits agree, proving the
  stated formula. The coordinate $r$ is auxiliary; no value of original
  negative-time noise has been assigned.

  Let $P$ project orthogonally onto $\Phi H_W$. On a product extension take an
  independent isonormal $J$ on $(\Phi H_W)^\perp$, built from independent real
  standard Gaussians on a finite or countable basis. Define
  \[
    \mathcal{I}(g)=W(\Phi^{-1}Pg)+J((I-P)g).
  \]
  Orthogonality and independence make it isonormal on $\mathcal{G}$ and
  prove~\eqref{E:source-given-W-retention} with the original random variable,
  not only its law. For the countable fixed original basis, these equalities
  hold on one common probability-one set. Thus, the original finite paths and
  their $L^p$ limits are precisely pullbacks of the already constructed original
  operator path. Cutoff fields may use $J$; they are not asserted to be
  functions of $W$ alone. This proves Lemma~\ref{L:source-retaining-isometry}.
\end{proof}

The auxiliary component can be nonzero: No-go~\ref{P:nogo-32}
constructs an orthogonal witness for every positive finite cutoff point
feature at positive time. Proposition~\ref{P:constant-raw-average} later
proves that the tested fixed-mask limit is $W$-measurable, using the
operator-path and Young-flow convergence.

\subsection{Exact joint source law and convergence}

Let $P_\eta$ multiply $\mathcal{G}$ by $1_{\{\lambda\le\eta^{-1}\}}$. Let
$C_\varepsilon$ act fiberwise on $H_\Gamma$. These commute. For $\rho =
(\varepsilon,\eta) > 0$, define
\begin{equation}\label{E:source-cutoff-noises}
  Q_\rho=C_\varepsilon P_\eta,\qquad
  Y_\rho(F)=\mathcal{I}(Q_\rho\Phi F),\qquad
  D_\rho(F)=\mathcal{I}((I-Q_\rho)\Phi F)=W(F)-Y_\rho(F).
\end{equation}
The last equality uses the coupling preserving the given noise. Define
\begin{equation}\label{E:source-covariance-functions}
  \Gamma_e(x)=(p_e*\Gamma)(x),\qquad
  K_\eta(v)=\frac1{\Gamma_E(a)}
  \int_0^{\eta^{-1}}e^{-\lambda|v|}\lambda^{a-1}\ud\lambda.
\end{equation}
The temporal operation is a Laplace prefix, not temporal convolution.

\begin{proposition}[Canonical sources on the extended space containing the given noise]
  \label{P:source-canonical-convergence}
  The noises in~\eqref{E:source-cutoff-noises} have random field realizations
  with joint covariance
  \begin{equation}\label{E:source-joint-covariance}
    \operatorname{Cov}(Y_{\varepsilon,\eta}(t,x),
    Y_{\varepsilon',\eta'}(s,y))
    =K_{\max(\eta,\eta')}(t-s)\Gamma_{\varepsilon+\varepsilon'}(x-y).
  \end{equation}
  Mean-square random field integration agrees with $Y_\rho(F)$ on tests. Both
  source and full residual obey
  \begin{equation}\label{E:source-two-dominations}
    \operatorname{Var}Y_\rho(F),\quad\operatorname{Var}D_\rho(F)
    \le\|F\|_{H_W}^2,
  \end{equation}
  and $\operatorname{Var}D_\rho(F) \to 0$ for every $F \in H_W$ as
  $\varepsilon,\eta\downarrow0$ at arbitrary relative rates. Consequently, for
  every finite $p \ge 1$,
  \begin{equation}\label{E:source-driver-limit}
    M_R^\rho:=M_{Y_\rho}\longrightarrow M_R
    \quad\text{in }L^p(\mathscr{B}_\theta),\qquad
    M_{D_\rho}=M_R-M_R^\rho.
  \end{equation}
  The source and residual have uniform local Gaussian operator-path moments and
  square-exponential norms, with constants for fixed $R,T$ and exponents.
\end{proposition}

\begin{proof}
  Apply the spatial distribution identity to a single shifted heat test:
  \[
    \Gamma_e(x)=\int e^{i\xi\cdot x}e^{-e|\xi|^2/2}\mu(\ud\xi).
  \]
  Gaussian spectral moments of every order are finite for tempered $\mu$. Thus,
  this is smooth with bounded derivatives. Since
  $\widehat{p_\varepsilon(x-\cdot)} = c_de^{-i\xi\cdot
  x}e^{-\varepsilon|\xi|^2/2}$, the two heat factors in the spatial inner
  product cancel $(2\pi)^d$ exactly, giving
  \begin{equation}\label{E:source-heat-inner-product}
    B_\Gamma(p_\varepsilon(x-\cdot),p_{\varepsilon'}(y-\cdot))
    =\Gamma_{\varepsilon+\varepsilon'}(x-y).
  \end{equation}
  Evenness of $\mu$ removes the reflected argument.

  For each positive $\rho$ and point $(t,x)$ put
  \begin{equation}\label{E:source-point-feature}
    v_{\rho;t,x}(\lambda,r)
    =1_{\{\lambda\le\eta^{-1}\}}\sqrt{2\lambda}
    e^{-\lambda(t-r)}1_{\{r\le t\}}p_\varepsilon(x-\cdot).
  \end{equation}
  Its squared $\mathcal{G}$ norm is $K_\eta(0)\Gamma_{2\varepsilon}(0) <
  \infty$. Define $Y_\rho(t,x) = \mathcal{I}(v_{\rho;t,x})$. The common
  $\lambda$ range for two such vectors is $\lambda \le
  \min(\eta^{-1},\eta'^{-1}) = \max(\eta,\eta')^{-1}$. Integrating $r$ gives
  $e^{-\lambda|t-s|}$, and~\eqref{E:source-heat-inner-product} gives the spatial
  factor. This proves~\eqref{E:source-joint-covariance} with the prescribed
  widths and original amplitudes, also for singular Radon and white covariances.

  The representing vector is continuous in $\mathcal{G}$ norm, directly from
  this continuous covariance. Its norm is bounded on compact parameter sets.
  Hence, for a smooth compact spacetime $F$, its Bochner integral against
  $v_{\rho;t,x}$ exists. Integrate the heat translates in $x$, then the time
  feature, to obtain exactly $Q_\rho\Phi F$. Applying the bounded map
  $\mathcal{I}:\mathcal{G} \to L^2$ proves the random field-test identity on the
  same space. Common-support indicator tests follow by covariance completion and
  $L^1$ approximation; bounded point-feature norms and the domination below give
  the same $L^2$ limit.

  Orthogonality of the two $\lambda$ ranges gives, for $g \in \mathcal{G}$,
  \[
    \|(I-Q_\rho)g\|_{\mathcal{G}}^2
    =\|(I-P_\eta)g\|_{\mathcal{G}}^2
    +\|P_\eta(I-C_\varepsilon)g\|_{\mathcal{G}}^2
    \le\|g\|_{\mathcal{G}}^2.
  \]
  Also $\|Q_\rho g\|_{\mathcal{G}} \le \|g\|_{\mathcal{G}}$. Together with the
  isometry these prove both variance bounds. Disjoint $\lambda$ ranges are
  independent; the full heat residual need not be independent of the source.

  The $L^2$ tail integral gives $P_\eta \to I$ strongly. Fiberwise heat
  converges strongly on simple $\mathcal{G}$ functions by
  Lemma~\ref{L:source-spatial-heat}, then on all of $\mathcal{G}$ by the
  contraction bound. The estimate
  \[
    \|(I-Q_\rho)g\|_{\mathcal{G}}
    \le\|(I-C_\varepsilon)g\|_{\mathcal{G}}
    +\|(I-P_\eta)g\|_{\mathcal{G}}\longrightarrow0
  \]
  holds at arbitrary relative rates. Thus, the residual variance tends to zero
  for every original $F$. Proposition~\ref{P:source-gaussian-transfer}
  gives~\eqref{E:source-driver-limit} without commuting a filter through
  $\zeta_R$ or a wave group.

  The same comparison gives $\|M_R^\rho\|_{L^p\mathscr{B}_\theta},
  \|M_{D_\rho}\|_{L^p\mathscr{B}_\theta} \le \|M_R\|_{L^p\mathscr{B}_\theta}$.
  The original $\sqrt p$ estimate and its even-moment exponential-series
  argument give uniform Gaussian operator path bounds. They are statements about
  $L^p$ equivalence classes on the extension preserving the given noise, not a
  simultaneous path event for all uncountably many cutoffs. This proves
  Proposition~\ref{P:source-canonical-convergence}.
\end{proof}

\subsection{Regularized random fields and the ordinary first response}

\begin{lemma}[Regularity for each fixed cutoff]\label{L:source-pointfield-regularity}
  For each fixed $\rho > 0$, the random field has a version continuous in time
  and smooth in space, with every spatial derivative jointly continuous on
  compact time-space rectangles.
\end{lemma}

\begin{proof}
  For each spatial multiindex $b$, differentiate \eqref{E:source-point-feature}
  in $x$ as a $\mathcal{G}$ vector. All derivatives exist because their spatial
  squared norms are the finite heat moments
  \[
    \int \xi^{2b}e^{-\varepsilon|\xi|^2}\mu(\ud\xi),
  \]
  and the next moments dominate difference quotients. Set $Z_b(t,x) =
  \mathcal{I}(\partial_x^bv_{\rho;t,x})$. At fixed $x$, use $1-e^{-\lambda h}
  \le \lambda h$ in the finite $\lambda$ integral; at fixed $t$, use
  $|e^{i\xi\cdot x}-e^{i\xi\cdot y}| \le |\xi||x-y|$ and the next heat moment.
  The squared triangle inequality gives
  \begin{equation}\label{E:source-derivative-increments}
    \E|Z_b(t,x)-Z_b(s,y)|^2
    \le C_{\rho,b}(|t-s|+|x-y|^2).
  \end{equation}
  This computation uses~\eqref{E:source-joint-covariance}, including its spatial
  width, and applies to singular spectral measures without a density assumption.

  Here is a direct continuity construction. In $m = d+1$ parameter dimensions on
  a compact rectangle, even Gaussian $q$th increment moments have root bounded
  by $C\sqrt q$ times the parameter distance to the power $1/2$ at small
  distances. The dyadic grid has at most $C_m2^{nm}$ adjacent edges at level
  $n$. Their maximum has $L^q$ norm at most $C\sqrt q\,2^{-n(1/2-m/q)}$, with
  constants for this rectangle and fixed $\rho,b$. Choose $q$ large and $0 < h_0
  < 1/2-m/q$. The sum of these maxima weighted by $2^{nh_0}$ has finite $L^q$
  norm. Binary refinement, using a fixed number of adjacent edges per coordinate
  at each scale, gives a continuous extension from dyadic points. At each fixed
  parameter the extension agrees with the given Gaussian variable in $L^2$,
  by~\eqref{E:source-derivative-increments}; it is a modification. Use countably
  many nested compact rectangles and all multiindices. The versions agree on
  overlaps on countable dense sets, hence everywhere there by continuity, so
  they patch.

  They are derivatives. The $\mathcal{G}$-valued fundamental theorem gives,
  along a coordinate segment,
  \[
    \partial_x^bv(t,x+he_i)-\partial_x^bv(t,x)
    =\int_0^h\partial_x^{b+e_i}v(t,x+ue_i)\ud u.
  \]
  Apply bounded linear $\mathcal{I}$ and $L^2$ integration. The continuous
  versions give the same ordinary integral by their Riemann sums, which converge
  both pathwise and in $L^2$. Enforce these identities first on countably many
  rational $t,x,h$ and multiindices, then use joint continuity to extend them to
  every segment in the compact rectangles. The ordinary fundamental theorem
  yields $\partial_iZ_b = Z_{b+e_i}$. Thus, all spatial derivatives exist and
  are jointly continuous with time. The versions are for this fixed $\rho$, not
  a common path event over uncountably many source parameters. This proves
  Lemma~\ref{L:source-pointfield-regularity}.
\end{proof}

\begin{proposition}[Ordinary source response]\label{P:source-ordinary-response}
  For each fixed $\rho > 0$, on its regularity event, put $b_\rho(t,x) =
  \zeta_R(x)Y_\rho(t,x)$. The statewise integral
  \begin{equation}\label{E:source-ordinary-response}
    L_\rho(t)z=\int_0^tS(-u)B(b_\rho(u))S(u)z\ud u
  \end{equation}
  defines an operator-norm Lipschitz path on $E$, and $L_\rho = M_R^\rho$ for
  all $t \in [0,T]$ on one probability-one set. In particular, it has the
  strongly measurable path version supplied by the Gaussian construction.
\end{proposition}

\begin{proof}
  The coefficient is compact smooth in space and continuous in time in every
  spatial derivative norm. Multiplication estimates and strong continuity of the
  full free group show that the statewise integral exists on every $E_r$. Its
  operator increments $E_r \to E_{r+1}$ are bounded by a finite sample constant
  times $|t-s|$, exactly by the statewise gain argument of
  Lemma~\ref{L:driver-gain-localization}. This proves Lipschitz continuity in
  $\mathcal{B}(E)$, without requiring norm continuity or an operator-norm
  derivative of $S(t)$.

  At each fixed $u$, the Sobolev duality calculation uses only spatial
  smoothness of $b_\rho(u,\cdot)$. From $S(-u)(0,f)$ and the two Bessel weights,
  for real Schwartz phases,
  \[
    \langle S(-u)B(b_\rho(u))S(u)z,w\rangle_E
    =\int b_\rho(u,x)[S(u)z]_1(x)G_w(u,x)\ud x.
  \]
  Thus, no prior hypothesis of smoothness in time is invoked for this
  instantaneous identity. For complex pairs conjugate the second factor.
  Integrating gives the ordinary source-field integral against the specified
  wave test for the original covariance $h_{R,t;z,w}$. For smooth tests, the
  point-feature integral has already been identified with $Y_\rho(F)$. The
  continuous random field version has the same integral: compact Riemann sums
  converge pathwise and in $L^2$ to the ordinary and feature integrals,
  respectively. Approximate the indicator wave test in the covariance completion
  with common compact spatial support and $L^2$-time spatial supremum error
  tending to zero. Its $L^1$ error also tends to zero. The point-feature norms
  are bounded on that support, so their integrals converge in $\mathcal{G}$; the
  original variance domination gives the same $L^2$ limit on the test side.
  Consequently,
  \[
    \langle L_\rho(t)z,w\rangle_E
    =Y_\rho(h_{R,t;z,w})
    =\langle M_R^\rho(t)z,w\rangle_E
    \quad\text{in }L^2.
  \]
  Compare a countable dense Schwartz phase collection and rational times.
  Boundedness and the path continuity of both sides extend this equality to all
  states and all times on one probability-one set. The equality is on the
  coupled space, not only in law. The set is for fixed $\rho$; the $L^p$ limit
  in~\eqref{E:source-driver-limit} needs no simultaneous version for every
  source parameter. This proves Proposition~\ref{P:source-ordinary-response}.
\end{proof}

\subsection{The common original noise and its exact Gaussian shift}

Choose an integer $B \ge 1$, later fixed by the cone estimate, and take $0 <
\delta < \min(1,1/B)$. Define
\begin{equation}\label{E:raw-block-clocks}
  N=\left\lfloor\frac1{\delta B}\right\rfloor\ge1,\quad
  s_m=m\delta B\ (0\le m\le N),\quad
  \tau=\delta^{-1}-NB\in[0,B),\quad
  g_\delta=\delta^{(1-a)/2}.
\end{equation}
For a smooth compact positive-fast-time test $f$, extended by zero, put
\begin{equation}\label{E:raw-block-pullback}
  T_{\delta,m}f(s,x)
  =\delta^{a/2-1}f((s-s_m)/\delta,x),\qquad
  W_{\delta,m}(f)=W(T_{\delta,m}f).
\end{equation}
Write $W_{\mathrm{fast}} = W_{\delta,0}$. Use the same admissible
$\gamma,\theta,\beta$ as~\eqref{E:driver-parameters}. Write $A_R(W_{\delta,m})$
for the $F$ integrated operator obtained by the bounded extension
\eqref{E:constant-phase-driver}, on the local horizon $[0,B]$, and set
\begin{equation}\label{E:raw-block-drivers}
  X_m=g_\delta A_R(W_{\delta,m}),\qquad
  H_m(u)=\int_0^uS_F(-r)iB_F(V_{s_m+\delta r})S_F(r)\ud r.
\end{equation}
The convolution formula for $h$ extends smoothly to $s \in [0,2]$, covering
these means. Ground templates will only use $s_m \in [0,1]$.

\LeanMarker{L:raw-block-inputs}
\begin{lemma}[Coupled Gaussian block inputs]\label{L:raw-block-inputs}
  Each $T_{\delta,m}$ is an isometry of the original test completion, respecting
  covariance-null tests. All $W_{\delta,m}$ and $Z$ are on the same given noise
  space, without an independence assertion. Let $\mathcal{K}(F)$ have the
  operator norm and let $\mathscr{B}_0 = h_0^\theta([0,B];\mathcal{K}(F))$ be
  the $C^\theta$ closure of zero-initial polygonal compact-operator paths. It is
  a separable Banach space. In $\mathscr{B}_\delta = \mathscr{B}_0^{N+1}$ with
  maximum norm, $(X,Z)$, $X = (X_m)_{m=0}^N$, is jointly centered Gaussian and
  \begin{equation}\label{E:raw-block-covariance}
    b:=\E[XZ]=\sqrt\delta\,H,\qquad H=(H_m)_{m=0}^N.
  \end{equation}
  Moreover,
  \begin{equation}\label{E:raw-block-moments}
    \|X_m\|_{L^p\mathscr{B}_0}
    \le C_{R,B}\sqrt p\,g_\delta,\qquad p\ge1.
  \end{equation}
\end{lemma}

\begin{proof}
  Substituting $s = s_m+\delta u$, $t = s_m+\delta r$ in the original covariance
  gives the factor $\delta^{a-2}\delta^2\delta^{-a} = 1$. Thus, the maps extend
  isometrically through the null quotient. Direct test substitution gives
  $W_{\delta,m}(f) = W_{\mathrm{fast}}(f(\cdot-mB,\cdot))$. These are explicit
  common-space identities.

  Finite-rank rational matrices in a countable real Hilbert basis are dense in
  $\mathcal{K}(F)$: for compact $T$, $P_nTP_n \to T$ in norm by compactness of
  $T,T^*$, and finite entries admit rational approximation. Polygonal paths with
  rational breakpoints and such matrices give separability of $\mathscr{B}_0$.
  To approximate breakpoints, use uniform convergence, uniform Lipschitz bounds
  and $[f]_\theta \le (2\|f\|_\infty)^{1-\theta}\operatorname{Lip}(f)^\theta$.
  The same inequality puts every zero-initial Lipschitz compact-valued path in
  $\mathscr{B}_0$ by polygonal approximation. The finite original coefficient
  paths $iA_jL_\psi$ have these properties. Their Gaussian sums converge in
  $L^pC^\theta$ and hence in the closed space $\mathscr{B}_0$. Their laws and
  the parent Gaussian norm bound give~\eqref{E:raw-block-moments}. Every finite
  joint coefficient list with $Z$ is Gaussian on the same $W$. Scalar continuous
  functionals pass through the $L^2$ limits, preserving Gaussianity by
  convergence of variances and characteristic functions. Thus, $(X,Z)$ is the
  asserted Banach Gaussian variable. Cauchy--Schwarz gives $\E\|XZ\| \le
  \|X\|_2\sqrt v < \infty$, so its Bochner covariance exists.

  The original covariance and the test pullback give
  \begin{equation}\label{E:raw-block-mean-jacobian}
    \E[W_{\delta,m}(f)Z]
    =\delta^{a/2}\int f(u,x)h_{s_m+\delta u}(x)\ud x\ud u.
  \end{equation}
  Absolute Fubini uses compact supports, local Radon mass and $a < 1$. This
  identity extends to the explicit matrix-element tests for the operator path on
  $E$: their time indicator is approximated by smooth bounded tests on one
  compact support, converging in the original covariance norm by dominated
  convergence against $|u-r|^{-a}\Gamma$; their mean pairings converge as well.
  The Sobolev adjoint is the one in~\eqref{E:driver-covariance-row}. Hence, the
  path covariance is $g_\delta\delta^{a/2}H_m = \sqrt\delta\,H_m$ on real
  Schwartz $E$ pairs. For constant inputs the two columns are the same $L_\psi$
  inputs $(\psi,0)$ and $(0,\psi)$, and total finite propagation identifies
  their deterministic response. The span of $E$ and both constant coordinates is
  dense in $F$; $E$ alone is not. Boundedness, rational times and path
  continuity prove~\eqref{E:raw-block-covariance} on all $F$. In particular, $H
  = b/\sqrt\delta \in \mathscr{B}_\delta$. This proves
  Lemma~\ref{L:raw-block-inputs}.
\end{proof}

The next identity is the covariance-direction Cameron--Martin formula;
see Janson~\cite{janson:97:gaussian}*{Theorem~14.1(v), pp.~216--217}.
We prove it on the norm-Borel space used here to verify the measurable
realization and its deterministic translate.

\begin{proposition}[Change of law on the whole norm-Borel space]
  \label{P:raw-block-gaussian-law}
  For every nonnegative norm-Borel $j$ on all of $\mathscr{B}_\delta$,
  \begin{equation}\label{E:raw-block-gaussian-law}
    \E j(X)=
    \E\left[e^{-Z/\sqrt\delta-v/(2\delta)}j(X+H)\right].
  \end{equation}
  The equality is valid also as an extended nonnegative integral.
\end{proposition}

\begin{proof}
  For real $s$, the density $e^{-sZ-s^2v/2}$ defines an equivalent probability
  measure $Q_s$ by completing the square in the $N(0,v)$ density. For any finite
  list $\ell$ of continuous real coordinates on $\mathscr{B}_\delta$, the
  Gaussian vector $(\ell(X),Z)$ is equal in law to $(TG,z\cdot G)$ for a finite
  standard Gaussian vector $G$. A positive covariance square root gives this
  representation even when singular; no inverse is used. Here $Tz = \ell(b)$ and
  $|z|^2 = v$. Substitution $G' = G+sz$ in its density gives
  $\E_{Q_s}f(\ell(X+sb)) = \E f(\ell(X))$ for every bounded Borel $f$.

  The matrix entries of each block in a real Hilbert basis at rational times,
  together with $0,B$, generate the norm-Borel sigma-field. To see the
  operator-norm point, for a countable dense set $\mathcal{D}$ of unit vectors
  obtained from normalized rational finite basis combinations, $\|T\| =
  \sup_{u,w\in\mathcal{D}}|\langle Tu,w\rangle|$. These pairings are finite
  linear combinations of matrix entries. Supremum and H\"older norms of
  continuous paths are suprema over rational times and pairs. Thus, distance to
  every fixed element of $\mathscr{B}_\delta$ is coordinate-measurable. Open
  balls are measurable by rational closed-ball approximation, and separability
  makes every open set a countable union of balls from dense centers. Conversely
  the coordinates are continuous. The cylinder sets form a generating pi-system;
  uniqueness of probability measures extends the finite-coordinate equality to
  all norm-Borel sets. Simple functions and monotone convergence give the
  identity for nonnegative $j$. Finally, take $s = \delta^{-1/2}$ and use $sb =
  H$ from~\eqref{E:raw-block-covariance}. The map and its deterministic
  translate are defined on the entire input space; there is no translated
  exceptional-domain assumption. This proves
  Proposition~\ref{P:raw-block-gaussian-law}.
\end{proof}

The identity retains the random shifted input $X+H$. Replacing its
expected contribution by evaluation at the deterministic mean can lose
the exponential constant, even in a scalar Gaussian calculation; see
No-go~\ref{P:nogo-16}. Positive semidefinite residual covariance does not
justify discarding that residual: No-go~\ref{P:nogo-19} gives an admissible
parent model whose rank-one projection has a larger small-coupling wave mean.


\subsection{Small noisy blocks and a cone in the displacement-velocity space}

For any $x \in \mathscr{B}_0$ define $D_x(u) = S_F(u)U_x(u,0)$ by the
everywhere-defined linear Young map of
Proposition~\ref{P:constant-young-linear}. Write $D_m^0 = D_{X_m}$ and $D_m^+ =
D_{X_m+H_m}$. The latter retains both the mean and the residual noise.

The shifted blocks $D_m^+$ are random. We now show that their largest distance
from the deterministic $P_{s_m}$ of~\eqref{E:compact-F-block}, over all blocks
$m \le N$, tends to zero in every finite $L^q$.

\begin{lemma}[Uniform block error]\label{L:raw-block-uniform-error}
  Let $M_\delta = \max_{m\le N}\|X_m\|_{\mathscr{B}_0}$, including its factor
  $g_\delta$. Then $M_\delta \to 0$ in every finite $L^q$, and, for some $c >
  0$,
  \begin{equation}\label{E:raw-block-maximum-exponential}
    \sup_\delta\E e^{cM_\delta^2}<\infty.
  \end{equation}
  With the deterministic $P_s$ in~\eqref{E:compact-F-block},
  \begin{equation}\label{E:raw-block-operator-error}
    E_{\mathrm{op},\delta}:=
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m^+(u)-P_{s_m}(u)\|_{\mathcal{B}(F)}
    \longrightarrow0
    \quad\hbox{in every finite }L^q.
  \end{equation}
\end{lemma}

\begin{proof}
  The maximum and~\eqref{E:raw-block-moments}, with $N+1 \le 2/\delta$, give
  \[
    \|M_\delta\|_p
    \le (N+1)^{1/p}C\sqrt p\,g_\delta
    \le C'\sqrt p\,\delta^{(1-a)/2-1/p}.
  \]
  For fixed $q$, choose $p > \max(q,2/(1-a))$ to prove the first limit. No
  independence is needed. Choose an integer $n_0$ with $n_0(1-a) \ge 1$. For $n
  \ge n_0$,
  \[
    \E M_\delta^{2n}
    \le2C^{2n}(2n)^n\delta^{n(1-a)-1}
    \le2C^{2n}(2n)^n.
  \]
  For $1 \le n < n_0$, Lyapunov bounds the moment by the appropriate power of
  the $2n_0$ moment. These finitely many terms are bounded. For the remaining
  exponential series, $n! \ge (n/e)^n$ gives a geometric tail $2\sum_{n\ge
  n_0}(2ecC^2)^n$ for small $c$. This
  proves~\eqref{E:raw-block-maximum-exponential}; no such assertion is made for
  the unscaled maximum.

  Let $H_m^{\mathrm{stat}}$ be the primitive for constant potential $V_{s_m}$.
  Compact mean smoothness on $[0,2]$ and the statewise primitive bound give
  \[
    \sup_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \sup_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C\delta.
  \]
  Indeed, the potential difference is at most $C\delta B$ in $L^\infty$ and
  $L^2$; integration of the bounded free groups gives its Lipschitz primitive
  bound on this fixed horizon. On $\{M_\delta\le1\}$, local Young stability on
  the fixed ball of radius $H_0+1$ gives $E_{\mathrm{op},\delta} \le
  C(M_\delta+\delta)$. This proves convergence in probability. The global Young
  bound also gives
  \[
    \max_m\sup_{u\le B}\|D_m^+(u)\|
    \le C_B\exp\big(C_B(1+H_0+M_\delta)^{1/\theta}\big).
  \]
  Since $1/\theta < 2$, every finite power of this bound is dominated by a
  constant times the exponential in~\eqref{E:raw-block-maximum-exponential}.
  Uniform higher moments yield uniform integrability, proving
  \eqref{E:raw-block-operator-error}. Suprema are measurable: on a countable
  dense set of unit states each difference is continuous in $u$, so rational
  times and those states suffice. Operator-norm continuity of the physical free
  group is not used. Only $s_m \le 1$ supplies the ground templates. This proves
  Lemma~\ref{L:raw-block-uniform-error}.
\end{proof}

Lemma~\ref{L:raw-block-uniform-error} bounds the distance of each shifted block
from $P_{s_m}$. It says nothing about the product of the blocks. The next lemma
is deterministic and holds for every $N \ge 1$. If each factor $D_m$ is within
$\epsilon_{\mathrm{op}}$ of $P_{\sigma_m}$, and adjacent indices $\sigma_m$ are
within $\omega$, then the product satisfies the lower bound
\eqref{E:raw-block-positive-product}.

\LeanMarker{L:raw-block-indexed-cone}
\begin{lemma}[Indexed cone and terminal displacement]
  \label{L:raw-block-indexed-cone}
  Fix $0 < \varrho < 1/2$. There are an integer $B \ge 1$ and positive
  $\omega,\epsilon_{\mathrm{op}},c_*$ such that the following holds for every $N
  \ge 1$. If $\sigma_0,\ldots,\sigma_N \in I$ have adjacent distances at most
  $\omega$ and real bounded operator families $D_m(u)$, $0 \le u \le B$, satisfy
  \[
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m(u)-P_{\sigma_m}(u)\|\le\epsilon_{\mathrm{op}},
  \]
  then, for $0 \le u \le B$,
  \begin{equation}\label{E:raw-block-positive-product}
    O_\phi D_N(u)D_{N-1}(B)\cdots D_0(B)e_0
    \ge c_*(1-\varrho)^{N-1}
    \exp\left(B\sum_{m<N}\lambda_{\sigma_m}
    +u\lambda_{\sigma_N}\right)>0.
  \end{equation}
  No positivity, composition or independence assumption on the $D_m$ is
  required.
\end{lemma}

\begin{proof}
  From~\eqref{E:compact-normalized-complement} fix $C_{\mathrm{gap}} \ge 1$,
  $\eta_{\mathrm{gap}} > 0$ so the normalized complementary norm is at most
  $C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}t}$. Define positive margins and
  finite bounds
  \[
    a_*=\inf_I\ell_s(e_0),\quad j_*=\inf_I O_\phi(e_s),\quad
    M_e=\max(1,\sup_I\|e_s\|),\quad
    M_\ell=\max(1,\sup_I\|\ell_s\|).
  \]
  Also put $M_P = \max(1,\sup_I\|\Pi_s\|)$, $M_Q = 1+M_P$, $M_z = M_e+1$, $E_0 =
  \|e_0\| = 1$ and $H_{\mathrm{obs}} = \max(1,\|O_\phi\|)$. Choose the aperture
  \emph{first}: $\vartheta = \min(1,j_*/(16H_{\mathrm{obs}}C_{\mathrm{gap}})) >
  0$. Next choose integer $B \ge 1$ with
  \[
    q_B:=C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}B}
    \le\min\left(\frac18,\frac{\vartheta a_*}{8M_QE_0}\right).
  \]
  Choose $\nu > 0$ such that $\nu M_z \le \min(\varrho/2,\vartheta,1/4)$.
  Uniform continuity of $\ell_s,\Pi_s$ supplies $\omega > 0$ for which adjacent
  distances at most $\omega$ give both norm differences at most $\nu$. Finally,
  choose $\epsilon_{\mathrm{op}} > 0$ satisfying all five smallness conditions
  \begin{align*}
    M_\ell\epsilon_{\mathrm{op}}M_z&\le\varrho/2,&
    M_Q\epsilon_{\mathrm{op}}M_z&\le\vartheta/4,\\
    M_\ell\epsilon_{\mathrm{op}}E_0&\le a_*/2,&
    M_Q\epsilon_{\mathrm{op}}E_0&\le\vartheta a_*/8,&
    H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z&\le j_*/8.
  \end{align*}
  All denominators are positive. This order of choices uses the normalized
  complement rather than a growing terminal norm.

  The cone at $s$ consists of $z = Ae_s+r$, with $A = \ell_s(z) > 0$, $\ell_s(r)
  = 0$ and $\|r\|_F \le \vartheta A$. In particular, $r = Q_sz$ and $\|z\| \le
  M_zA$. For entrance, $z_1 = D_0(B)e_0$, with $b = \sigma_0$, has normalized
  amplitude at least $a_*-M_\ell\epsilon_{\mathrm{op}}E_0 \ge a_*/2$. Its
  normalized residual is at most $q_BM_QE_0+M_Q\epsilon_{\mathrm{op}}E_0 \le
  \vartheta a_*/4$. Thus, it enters the cone with residual ratio at most
  $\vartheta/2$ and amplitude at least $e^{\lambda_bB}a_*/2$.

  For an old-cone state $z = Ae_b+r$ and an adjacent index $c$, rebase before
  using the complementary estimate:
  \[
    \ell_c(z)\ge(1-\nu M_z)A,\qquad
    \|Q_cz\|\le\|Q_bz\|+\|(\Pi_b-\Pi_c)z\|
    \le(\vartheta+\nu M_z)A.
  \]
  For the next full block $D$ close to $P_c(B)$, put $x =
  e^{-\lambda_cB}\ell_c(Dz)$ and $y = e^{-\lambda_cB}\|Q_cDz\|$. Since
  $\lambda_c > 0$, the perturbation and smallness conditions give
  \[
    x\ge(1-\nu M_z-M_\ell\epsilon_{\mathrm{op}}M_z)A
    \ge(1-\varrho)A>A/2,
  \]
  and
  \[
    y\le
    [q_B(\vartheta+\nu M_z)+M_Q\epsilon_{\mathrm{op}}M_z]A
    \le\vartheta A/2.
  \]
  The last inequality uses $q_B \le 1/8$, $0 \le \nu M_z \le \vartheta$ and
  $M_Q\epsilon_{\mathrm{op}}M_z \le \vartheta/4$. Together with $\varrho < 1/2$,
  these give $y \le \vartheta x$. These are precisely the finite residual and
  cone-conversion margins; the old residual was not assumed to lie in the new
  kernel. Each full block therefore preserves the new cone and multiplies its
  amplitude by at least $e^{\lambda_cB}(1-\varrho)$.

  For the terminal $0 \le u \le B$, rebase once more to $c = \sigma_N$. The same
  bounds give $\ell_c(z) \ge 3A/4$ and $\|Q_cz\| \le 2\vartheta A$.
  Consequently,
  \[
    e^{-\lambda_cu}O_\phi(P_c(u)z)/A
    \ge3j_*/4-2H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta.
  \]
  The perturbation changes this by at most
  $H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z \le j_*/8$. Since
  $16H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta \le j_*$, the result is at least
  $j_*/2$. Combine entrance, all $N-1$ intermediate full blocks and this
  terminal estimate to obtain~\eqref{E:raw-block-positive-product}, with $c_* =
  a_*j_*/4$. The induction is empty for $N = 1$ and the estimate includes $u =
  0$. This proves Lemma~\ref{L:raw-block-indexed-cone}.
\end{proof}

\subsection{Source restart and comparison with the original point}

So far the blocks $D_m^0$ and $D_m^+$ are defined one at a time, from the local
operator paths $X_m$ and $X_m+H_m$. We now show that each family forms the
successive pieces of its corresponding global flow, and that its finite
products are source limits in $L^2$ at fixed $\delta,N$. The proof of
Proposition~\ref{P:raw-block-point-comparison} uses this to verify the
source-limit premise of the masked comparison.

\LeanMarker{L:raw-block-restart}
\begin{lemma}[Composition over time blocks and regularized products]
  \label{L:raw-block-restart}
  Fix $\delta,B,N$ and let $L = \delta^{-1}$, $L_{\mathrm{ext}} = (N+1)B \ge L$.
  On this global fast horizon define $A_{\mathrm{glob}} = g_\delta
  A_R(W_{\mathrm{fast}})$. Its physical flow obeys
  \begin{equation}\label{E:raw-block-restart}
    P_{\mathrm{glob}}(mB+u,mB)=D_m^0(u),\qquad 0\le u\le B.
  \end{equation}
  The analogous identity holds with the deterministic mean $V_{\delta t}$ added
  globally and local operator paths $X_m+H_m$. Their finite block products,
  including the last remainder $\tau$, are source limits in $L^2$, at this fixed
  $\delta,N$.
\end{lemma}

\begin{proof}
  Take one regularized slow noise preserving the given noise $Y_{\epsilon,\eta}$
  on the extension of the given $W$ constructed in
  Section~\ref{S:source-driver-convergence}. Define the fast and local noises by
  \[
    Y_{\mathrm{fast}}(f)=Y_{\epsilon,\eta}(T_{\delta,0}f),\qquad
    Y_m(f)=Y_{\epsilon,\eta}(T_{\delta,m}f).
  \]
  The pullback isometry and~\eqref{E:source-two-dominations} give source and
  residual variance bounds by $\|f\|_{H_W}^2$. The residual tends to zero on
  each transformed smooth compact test at fixed $\delta,m$. Joint Gaussianity is
  retained. Proposition~\ref{P:source-gaussian-transfer} therefore gives the
  $L^pC^\theta$ operator-path limits on $[0,L_{\mathrm{ext}}]$ and on $[0,B]$,
  including both constant columns.

  Their random fields are $Y_{\mathrm{fast}}(u,x) =
  \delta^{a/2}Y_{\epsilon,\eta}(\delta u,x)$ and $Y_m(u,x) =
  Y_{\mathrm{fast}}(mB+u,x)$. Indeed, these follow by change of variables in
  their test integrals. The temporal law is $K_{\eta/\delta}$ because
  $\delta^aK_\eta(\delta r) = K_{\eta/\delta}(r)$; the spatial heat law and all
  joint width rules are unchanged. They are continuous in time and smooth in
  space. Their ordinary primitives equal their Gaussian operator paths: their
  matrix entries are $Y_m(h_R)$, by the indicator approximation used
  in~\eqref{E:raw-block-mean-jacobian} and the point-test integral identity.
  Total finite propagation extends this identity to both constant columns. This
  does not assume a separately chosen extension preserving the given noise
  automatically commutes with time dilation.

  Changing the ordinary source integration variable and composing the free
  groups gives the exact identity on $F$
  \[
    S_F(mB)\big[A_{\mathrm{glob}}^{\mathrm{src}}(mB+u)
    -A_{\mathrm{glob}}^{\mathrm{src}}(mB)\big]
    S_F(-mB)=X_m^{\mathrm{src}}(u).
  \]
  No physical mask is commuted through a wave group. Restriction, difference and
  conjugation are bounded maps on these fixed-horizon $C^\theta$ spaces. Passing
  the source limits gives the same identity for $A_{\mathrm{glob}},X_m$. For the
  finitely many $m$, a convergent subsequence or rational times and path
  continuity gives simultaneous identities in $u$. Conjugation norms need only
  be finite at this fixed $\delta$.

  Conjugate the global interaction Young equation on $[mB,mB+u]$ by $S_F(mB)$.
  Bounded linearity of its Riemann sums and that original identity give the
  local equation driven by $X_m$. Uniqueness identifies it with $U_{X_m}$,
  proving \eqref{E:raw-block-restart}. The ordinary deterministic primitive for
  $V_{\delta t}$ has the identical cocycle with local $H_m$. Composition now
  identifies the full product, with terminal $\tau$.

  For completeness, source convergence of these finite products requires no
  growing-product stability assertion. At fixed $\delta,N$ let $C \ge 1$ be the
  maximum norm of their source and original factors, and let $e_{\mathrm{src}}$
  be the maximum factor difference. The parent Young convergence gives every
  finite moment of $C$ uniformly in source widths and $e_{\mathrm{src}} \to 0$
  in every finite $L^p$. If $d(n)$ is the difference of the first $n$ ordered
  products, then
  \[
    d(n+1)\le C d(n)+C^n e_{\mathrm{src}},\qquad d(0)=0.
  \]
  Extend both finite sequences by identities so the recurrence holds for all
  natural $n$. The finite error-accumulation estimate gives $d(N+1) \le
  (N+1)C^{N+1}e_{\mathrm{src}}$. H\"older, using $C$ in $L^{4(N+1)}$ and the
  error in $L^4$, proves $L^2$ convergence. This is only a fixed-$\delta$ source
  limit. Shifted source products may use the fixed original mean plus the
  regularized noise; no finite-source Cameron mean is substituted. This proves
  Lemma~\ref{L:raw-block-restart}.
\end{proof}

\begin{proposition}[The square observable and canonical comparison]
  \label{P:raw-block-point-comparison}
  Define the signed averages without and with the deterministic mean
  \[
    U_{\delta,B}
    =O_\phi D_N^0(\tau)D_{N-1}^0(B)\cdots D_0^0(B)e_0,
    \qquad
    Y_{\delta,B}
    =O_\phi D_N^+(\tau)D_{N-1}^+(B)\cdots D_0^+(B)e_0.
  \]
  They have all finite unweighted moments, and
  \begin{align}
    \E U_{\delta,B}^2
    &=\E\left[e^{-Z/\sqrt\delta-v/(2\delta)}
    Y_{\delta,B}^2\right],
    \label{E:raw-block-square-identity}\\
    \E U_{\delta,B}^2
    &\le\E|V_\delta(1,0)|^2
    =\E|u(T,0)|^2,\qquad \delta=T^{-\kappa}.
    \label{E:raw-block-original-comparison}
  \end{align}
  The variance $v$ in the density is the original slow original variance.
\end{proposition}

\begin{proof}
  On all of $\mathscr{B}_\delta$ put
  \[
    j_\delta(x)=
    |O_\phi D_{x_N}(\tau)D_{x_{N-1}}(B)\cdots D_{x_0}(B)e_0|^2.
  \]
  The everywhere-defined continuous Young map, fixed-time bounded free groups,
  finite products and bounded $O_\phi$ make $j_\delta$ continuous and
  nonnegative on the whole input space. Apply
  Proposition~\ref{P:raw-block-gaussian-law} directly to these original paths.
  Since $X+H$ is its exact translate, this proves
  \eqref{E:raw-block-square-identity} with the single original $Z = W(\Psi)$ and
  its variance $v$. Finite products of the local Young flows have every finite
  unweighted moment by their Gaussian path bounds and H\"older, at fixed
  $\delta,N$. The Gaussian identity itself did not require prior integrability
  of $j_\delta$. No source convergence under its exponential density is used.

  To apply the masked comparison at the amplitude, put $\Gamma_\delta =
  g_\delta^2\Gamma$ and $W' = g_\delta W_{\mathrm{fast}}$. This is a nonzero
  member of the same admissible covariance class, with unchanged degree $\alpha$
  and temporal parameter $a$. Its integrated operator is $g_\delta$ times the
  original one: the original matrix forms are linear in noise, and
  basis-independent Gaussian transfer and the bounded $F$ extension retain both
  constant columns. The $n$th source chain is multiplied by $g_\delta^n$. Thus,
  this is exactly the model without the deterministic mean at that amplitude.

  We verify the source-limit premise of the masked comparison. Keep $\delta,B,N$
  fixed and use the single regularized noise preserving the given noise of
  Lemma~\ref{L:raw-block-restart}. For $m < N$ let $F_m = D_m^0(B)$ and put $F_N
  = D_N^0(\tau)$; write $F_{m,\rho}$ for the corresponding ordinary source
  factors, where both source widths in $\rho$ tend to zero.
  Lemma~\ref{L:raw-block-restart} and Proposition~\ref{P:constant-gaussian-flow}
  give factor convergence in every finite $L^q$ and uniform bounds for every
  finite moment of their norms. These statements are on the fixed finite
  horizons $[0,(N+1)B]$ and $[0,B]$, and include both constant-response columns.
  Set $n = N+1$ and
  \[
    C_\rho=\max\left(1,\max_{m\le N}\|F_{m,\rho}\|,
    \max_{m\le N}\|F_m\|\right),\qquad
    e_\rho=\max_{m\le N}\|F_{m,\rho}-F_m\|.
  \]
  Then $\sup_\rho\|C_\rho\|_q < \infty$ for every finite $q$, and $\|e_\rho\|_q
  \to 0$. Telescoping the ordered products gives
  \[
    \|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\|
    \le n C_\rho^n e_\rho.
  \]
  Therefore, H\"older with exponents $4n$ and $4$ gives
  \[
    \big\|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\big\|_{L^2}
    \le n\|C_\rho\|_{L^{4n}}^n\|e_\rho\|_{L^4}
    \longrightarrow0.
  \]
  No uniform estimate in the growing number $N$ is needed: this source limit is
  completed at each fixed $\delta,N$. The physical source identification in
  Lemma~\ref{L:constant-source-solution}, with the same $m_R$ and initial state
  $e_0$, identifies the source product with the tested ordinary masked solution
  at fast time $L = 1/\delta$. Boundedness of $O_\phi$ and $\|e_0\| = 1$ thus
  identify $U_{\delta,B}$ as its $L^2$ limit. The sources $g_\delta
  Y_{\mathrm{fast}}$ have the correct law, contraction and original-test limit
  by Lemma~\ref{L:raw-block-restart}.

  At positive source widths, $0 \le m_R \le 1$ multiplies each nonnegative
  displacement Wick integrand by the product of its vertex masks. Positive Green
  measures and covariance retain this comparison for unequal, self and empty
  chains as well. Proposition~\ref{P:masked-source-moments} gives the tested
  comparison, and the just-proved $L^2$ convergence discharges the premise of
  Proposition~\ref{P:masked-canonical-transfer} at $\Gamma_\delta$. Its
  canonical unmasked moment limit uses the fixed-order joint source limits and
  the absolutely summable order bounds. Since $\phi \ge 0$ and $\int\phi = 1$,
  these facts give
  \[
    \E U_{\delta,B}^2
    \le\E|u_{\Gamma_\delta}(L,0)|^2.
  \]
  Only the source products without the deterministic mean were needed for this
  passage. The shifted original product is already the everywhere-defined
  $j_\delta(X+H)$ in the Gaussian identity; no tilted source limit or change of
  the original slow $\Psi,Z,v$ has been made.

  The ordinary change from fast $u$ to slow $t = \delta u$, with phase
  $(V,\delta V_t)$, transforms the unmasked source equation into the regularized
  version of~\eqref{E:exact-fast-equation}, with slow terminal time one. Indeed,
  $g_\delta Y_{\mathrm{fast}}(u) = \sqrt\delta\,Y_{\epsilon,\eta}(\delta u)$,
  and its fast cutoff is $\eta/\delta$. The canonical fixed-order and absolutely
  convergent series construction passes this law identity at fixed $\delta$ to
  the original moment, preserving every mixed, unequal, self and empty chain.
  Lemma~\ref{L:exact-canonical-scaling} gives the final equality
  in~\eqref{E:raw-block-original-comparison}. The physical scale $r_T = T\delta$
  is distinct from the fixed mask radius $R$. The mask need not contain the
  growing backward cone: the comparison is a moment inequality, not equality of
  masked and full fields. All source limits preceded this large-time comparison.
  This proves Proposition~\ref{P:raw-block-point-comparison}.
\end{proof}

Proposition~\ref{P:raw-block-point-comparison} bounds $\E|u(T,0)|^2$ from below
by the weighted second moment of $Y_{\delta,B}$
in~\eqref{E:raw-block-square-identity}. We still need a lower bound for
$Y_{\delta,B}$ itself. The proof below obtains one on an event whose probability
tends to one. It uses the estimates of Lemma~\ref{L:raw-block-uniform-error} and
the cone bound of Lemma~\ref{L:raw-block-indexed-cone}.

\LeanMarker{P:raw-block-original-lower}
\begin{proposition}[Original point lower rate at the Gaussian quadratic cost]
  \label{P:raw-block-original-lower}
  For every original positive $\Psi$ and eligible fixed mask $R$ chosen above,
  and every fixed $x \in \R^d$,
  \begin{equation}\label{E:raw-block-original-lower}
    \liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,x)|^2
    \ge 2\int_0^1\lambda_s\ud s-\frac{v_\Psi}{2}.
  \end{equation}
\end{proposition}

\begin{proof}
  Fix $e_{\mathrm{rate}} > 0$. Choose $0 < \varrho < 1/2$ with $\log(1-\varrho)
  > -e_{\mathrm{rate}}/3$, then choose the cone constants, including an integer
  $B$ fixed independently of $\delta$, from
  Lemma~\ref{L:raw-block-indexed-cone}. The control $\Psi$ and eligible mask $R$
  remain fixed throughout.

  Choose a finite $p > 2/(1-a)$ and put $\gamma_p = p(1-a)/2-1 > 0$.
  From~\eqref{E:raw-block-moments} and $N+1 \le 2/\delta$, Markov and the union
  bound give, for $\eta > 0$,
  \[
    \begin{split}
      \mathbb{P}(M_\delta>\eta)
      &\le\eta^{-p}\sum_{m=0}^N
      \E\|X_m\|_{C^\theta}^p\\
      &\le A_p\eta^{-p}\delta^{\gamma_p},\qquad
      A_p=2(C_{R,B}\sqrt p)^p<\infty.
    \end{split}
  \]
  This estimate concerns the coupled blocks; no independence is needed. At these
  fixed data $A_p$ is independent of $\delta$.

  We also specify the deterministic tolerance used with this bound. The
  primitive estimates in Lemma~\ref{L:raw-block-uniform-error} give, for the
  constant-potential primitives $H_m^{\mathrm{stat}}$,
  \[
    \max_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \max_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C_B\delta.
  \]
  Apply Proposition~\ref{P:constant-young-linear} on the fixed ball of operator
  paths of radius $H_0+1$ and use the bounded free groups on $[0,B]$. There is
  $L_B \ge 1$, depending on these fixed $R,B,\Psi$ and the admissible exponents,
  such that on $\{M_\delta\le1\}$,
  \[
    E_{\mathrm{op},\delta}\le L_B(M_\delta+\delta).
  \]
  Set $\eta = \min(1,\epsilon_{\mathrm{op}}/(2L_B)) > 0$. If $\delta \le
  \epsilon_{\mathrm{op}}/(2L_B)$, then $M_\delta \le \eta$ implies
  $E_{\mathrm{op},\delta} \le \epsilon_{\mathrm{op}}$. Thus, for the measurable
  event on the original common space
  \[
    E_\delta=\{E_{\mathrm{op},\delta}\le\epsilon_{\mathrm{op}}\}
    \cap\{|Z|\le\delta^{-1/4}\},\qquad
    p_\delta=\mathbb{P}(E_\delta),
  \]
  Chebyshev for the same single $Z = W(\Psi)$, with $\E Z^2 = v$, gives
  \[
    p_\delta\ge1-A_p\eta^{-p}\delta^{\gamma_p}
    -v\delta^{1/2}\longrightarrow1.
  \]
  This is one union bound, with no independence between blocks or from $Z$ and
  no product of block probabilities.

  Let $C_\lambda$ be a Lipschitz bound for $s\mapsto\lambda_s$. Take $\delta$
  small enough, in addition to $0 < \delta < \min(1,1/B)$, that
  \[
    \begin{gathered}
      \delta B\le\omega,\qquad
      \delta\le\epsilon_{\mathrm{op}}/(2L_B),\qquad
      C_\lambda\delta B\le e_{\mathrm{rate}}/3,\\
      \delta|\log c_*|\le e_{\mathrm{rate}}/3,\qquad
      A_p\eta^{-p}\delta^{\gamma_p}+v\delta^{1/2}\le1/2.
    \end{gathered}
  \]
  Every constant has been fixed before this choice. The condition involving
  $C_\lambda$ is vacuous if that constant is zero. All source limits in
  Proposition~\ref{P:raw-block-point-comparison} have already been completed at
  each such fixed $\delta$.

  The indices $s_m = m\delta B$ lie in $I$ with adjacent gaps at most $\omega$.
  On $E_\delta$ the shifted full-$F$ blocks therefore satisfy
  Lemma~\ref{L:raw-block-indexed-cone}, retaining both phase entries and the
  final remainder $\tau$. In particular, $Y_{\delta,B} > 0$, and
  \[
    \delta\log Y_{\delta,B}
    \ge\delta\log c_*+\delta(N-1)\log(1-\varrho)
    +\delta B\sum_{m<N}\lambda_{s_m}
    +\delta\tau\lambda_{s_N}.
  \]
  Since $s_N+\delta\tau = 1$, the last two terms are the left Riemann sum for
  $I_R$ on the entire interval, including its last interval of length
  $\delta\tau$, with mesh at most $\delta B$. Its error is at most
  $C_\lambda\delta B$. Also $0 \le \delta(N-1) \le 1/B \le 1$, so the
  logarithmic loss is at least $\log(1-\varrho) > -e_{\mathrm{rate}}/3$. The two
  other losses are each at most $e_{\mathrm{rate}}/3$ by the choices above.
  Hence,
  \[
    Y_{\delta,B}\ge
    \exp((I_R-e_{\mathrm{rate}})/\delta)\quad\hbox{on }E_\delta,
  \]
  including the $N = 1$ and $\tau = 0$ cases of the cone estimate.

  The integrand in~\eqref{E:raw-block-square-identity} is nonnegative. On the
  same $E_\delta$ its linear Gaussian term satisfies $-Z/\sqrt\delta \ge
  -\delta^{-3/4}$. Restriction to this event
  and~\eqref{E:raw-block-original-comparison} therefore give the explicit bound
  for the original point moment
  \[
    \E|u(T,0)|^2\ge\E U_{\delta,B}^2
    \ge p_\delta\exp\left(
    \frac{2I_R-2e_{\mathrm{rate}}-v/2}{\delta}
    -\delta^{-3/4}\right),\qquad \delta=T^{-\kappa}.
  \]
  Here $p_\delta \ge 1/2$ for small $\delta$, so $-\delta\log2 \le \delta\log
  p_\delta \le 0$. Multiplying the logarithm of the preceding inequality by
  $\delta$ yields
  \[
    \delta\log\E|u(T,0)|^2
    \ge2I_R-2e_{\mathrm{rate}}-v/2
    -\delta^{1/4}+\delta\log p_\delta.
  \]
  Thus, the normalized linear penalty is $\delta\delta^{-3/4} = \delta^{1/4} \to
  0$, and $\delta\log p_\delta \to 0$. The quadratic Gaussian cost is precisely
  half the original variance $v$. Since $\kappa > 0$, $\delta = T^{-\kappa}$ is
  a bijection between all sufficiently large $T$ and all sufficiently small
  $\delta$. This gives the full continuous-time liminf $Q_2 \ge
  2I_R-2e_{\mathrm{rate}}-v/2$. For each $e_{\mathrm{rate}}$ the argument first
  fixed its own $B = B(e_{\mathrm{rate}})$ and then took $\delta\downarrow0$.
  The original point rate $Q_2$ is independent of $B$; only now let
  $e_{\mathrm{rate}}\downarrow0$. No choice $B = B(\delta)$ is made.

  Finally, spatial translation of all positive-width vertices preserves the
  original noise law, Green kernels and constant initial data. The fixed-order
  $L^2$ and absolute-series limits preserve equality of point second moments.
  Hence, zero can be replaced by every fixed $x$
  in~\eqref{E:raw-block-original-lower}. Every control and mask were fixed
  before asymptotics; a supremum over them may follow their individual
  inequalities. This proves Proposition~\ref{P:raw-block-original-lower}.
\end{proof}

\section{The compact mean and Cameron map on time blocks}
\label{S:actual-cameron-mean}

Retain the nonzero original covariance in Hypothesis~\ref{H:homog}, its
amplitude, $a = \alpha_0$, and the real phases $E = H^{1/2}\times H^{-1/2}$ and
$F = E\oplus\R^2$. Write $\Lambda = (1-\Delta)^{1/2}$. The form $B_\Gamma$ and
constant $R_{\mathrm{sp}}$ are those of
Section~\ref{S:spatial-occupation-foundation}. Let $H_{\mathrm{sp}}$ be the real
covariance-null completion of $C_c^\infty(\R^d)$ with inner product $B_\Gamma$,
and $H_t$ the analogous completion of $C_c^\infty(I)$, $I = (0,1)$, with time
kernel $|s-t|^{-a}$. Let $H = H_I$ be the completion of smooth tests supported
inside $I\times\R^d$. It embeds isometrically as a closed subspace of the given
$H_W$. We retain the given $W$ restricted to it.

Fix $R > 0$ and real even $\eta_R \in C_c^\infty(\R^d)$ with $0 \le \eta_R \le
1$, equal to one on the radius-$R$ ball and supported in the closed radius-$2R$
ball. The physical mask is $m_R = \eta_R^2$. Every operator path below takes its
mask argument equal to $m_R$: at this fixed radius the base profile $\zeta(y) =
m_R(Ry)$ meets the earlier hypotheses. The root $\eta_R$ factors state tests; it
does not multiply an original noise class. All compactness conclusions here fix
$R$.

\subsection{Fractional products and the compact spatial map}

The four-linear estimate~\eqref{E:spatial-four-linear} needs $H^1$ factors.
The first component of a state in $E$ lies only in $H^{1/2}$, so we first
extend that estimate to fractional Sobolev factors.

\begin{lemma}[Fractional covariance products]
  \label{L:cameron-fractional-products}
  For $\alpha/4 < s < 1/2$, products of real $H^s$ functions define elements of
  $H_{\mathrm{sp}}$, with
  \begin{equation}\label{E:cameron-fractional-product}
    \|fg\|_{H_{\mathrm{sp}}}
    \le C_s^2\sqrt{R_{\mathrm{sp}}}\|f\|_{H^s}\|g\|_{H^s}.
  \end{equation}
  Their Hilbert pairings equal their physical integrals. In particular,
  \begin{equation}\label{E:cameron-fractional-absolute}
    \int|f_1(x)f_2(x)f_3(x-y)f_4(x-y)|\ud x\,\Gamma(\ud y)
    \le C_s^4R_{\mathrm{sp}}\prod_{j=1}^4\|f_j\|_{H^s}.
  \end{equation}
\end{lemma}

\begin{proof}
  First approximate real $H^1$ factors by compact smooth ones. The absolute
  four-linear bound~\eqref{E:spatial-four-linear} and covariance PSD make their
  products Cauchy in $H_{\mathrm{sp}}$, with norm at most
  $\sqrt{R_{\mathrm{sp}}}N(f)N(g)$. Expanding product differences proves
  independence of approximation; physical continuity identifies all pairings.
  PSD for products also follows from $fg = ((f+g)^2-(f-g)^2)/4$.

  Use real radial Fourier bands $D_0 = 1_{\{|\xi|<1\}}$ and $D_j =
  1_{\{2^{j-1}\le|\xi|<2^j\}}$, $j \ge 1$. For $f \in H^s$ put $f_j = D_jf$.
  Each block is real and in $H^1$. Frequency support, Parseval and weighted
  Cauchy--Schwarz give
  \begin{equation}\label{E:cameron-dyadic-product}
    N(f_j)\le2^{j\alpha/4}\|f_j\|_2,\qquad
    \sum_{j\ge0}N(f_j)\le C_s\|f\|_{H^s}.
  \end{equation}
  Indeed, $\sum_j2^{-2j(s-\alpha/4)} < \infty$, while $\sum_j2^{2js}\|f_j\|_2^2
  \le 2^{2s}\|f\|_{H^s}^2$. The low block has factor one. No $L^p$ multiplier
  result is used. Also $\sum_j\|f_j\|_2 < \infty$ since $s > 0$. Thus,
  $\sum_j|f_j|$ lies in $L^2$ by Minkowski and monotone convergence, and the
  series is absolutely convergent a.e. with sum $f$.

  Apply the absolute $H^1$ four-linear estimate to each band quadruple. The sum
  of its absolute integrals is at most
  $R_{\mathrm{sp}}\prod_{\ell=1}^4\sum_jN((f_\ell)_j) < \infty$. Tonelli and the
  null-set transfer at both spatial positions in
  Lemma~\ref{L:spatial-physical-pair} identify the physical product series,
  proving~\eqref{E:cameron-fractional-absolute}. The double series
  $\sum_{j,k}f_jg_k$ is also absolutely convergent in $H_{\mathrm{sp}}$ by the
  $H^1$ product bound. Its partial sums are products of $H^1$ functions. The
  absolute quadruple calculation identifies their limiting physical pairings and
  gives~\eqref{E:cameron-fractional-product}. Compact smooth functions are dense
  in $H^s$: approximate in Fourier space by real Schwartz functions, then cut
  each off compactly in $H^1$ and hence in $H^s$. This also proves uniqueness of
  the bilinear extension. This proves Lemma~\ref{L:cameron-fractional-products}.
\end{proof}

With these products in $H_{\mathrm{sp}}$, we can extend the spatial mean map
from smooth tests to all of $H_{\mathrm{sp}}$.

\begin{proposition}[Compact spatial mean map]\label{P:cameron-spatial-map}
  The smooth-test map
  \[
    T_{\mathrm{sp}}q=iB_F(m_R(\Gamma*q))
  \]
  extends to a compact linear map $T_{\mathrm{sp}}:H_{\mathrm{sp}} \to
  \mathcal{K}(F)$, where $\mathcal{K}(F)$ has the operator norm. For $z =
  (z_1,z_2,c_1,c_2)$ and $w = (w_1,w_2,d_1,d_2)$ set $w' = \Lambda^{-1}w_2$. Its
  exact row identity is
  \begin{equation}\label{E:cameron-spatial-row}
    \langle T_{\mathrm{sp}}qz,w\rangle_F
    =\langle q,k(z,w)\rangle_{H_{\mathrm{sp}}},
    \qquad
    k(z,w)=(\eta_Rz_1+c_1\eta_R)(\eta_Rw').
  \end{equation}
\end{proposition}

\begin{proof}
  Fix $s$ with $\alpha/4 < s < 1/2$; this interval is nonempty because the
  original threshold implies $\alpha < 2$. For smooth compact $q$, $\Gamma*q$ is
  locally smooth by differentiation under the compactly restricted Radon
  integral. The masked potential is smooth compact. The velocity pairing in $E$
  has weight $\Lambda^{-1}$; hence physical Fubini
  gives~\eqref{E:cameron-spatial-row} on smooth states. Evenness of $\Gamma$
  changes $B_\Gamma(k,q)$ to $B_\Gamma(q,k)$. For general states the localized
  products are $L^1$, and $\Gamma*q$ is bounded on their compact support. $L^2$
  state approximation passes the physical integral to the limit, while
  Lemma~\ref{L:cameron-fractional-products} passes the $H_{\mathrm{sp}}$
  pairing. This proves the row.

  Multiplication by $\eta_R$ maps bounded $H^{1/2}$ sets to precompact $H^s$
  sets by Proposition~\ref{P:cutoff-singular-values}, with gain $1/2-s > 0$.
  This applies to $z_1$ and $w'$. The $c_1\eta_R$ family is finite dimensional
  and compact. The closures of both factor families for unit $z,w$ are compact,
  so their products have compact closure $\mathcal{L}$ in $H_{\mathrm{sp}}$ by
  the fractional product bound. The row identity gives $\|T_{\mathrm{sp}}q\| \le
  \sup_{k\in\mathcal{L}}\|k\|\|q\|$. Null covariance classes therefore map to
  zero.

  For smooth $q$, multiplication by its potential is compact $H^{1/2} \to
  H^{-1/2}$ by the same whole-space cutoff result with gain one. Its extra $c_1$
  column is finite rank, and the instantaneous $c_2$ column is zero. Thus, the
  smooth images lie in $\mathcal{K}(F)$, and so does the bounded extension by
  operator-norm closure.

  Choose a complete smooth-test orthonormal basis of $H_{\mathrm{sp}}$ by
  countable compact-support approximation and Gram--Schmidt, omitting null
  classes. Let $Q_n$ project onto its first $n$ terms. The row identity yields
  \[
    \|T_{\mathrm{sp}}(I-Q_n)\|
    \le\sup_{k\in\mathcal{L}}\|(I-Q_n)k\|_{H_{\mathrm{sp}}}
    \longrightarrow0.
  \]
  Strong projection convergence is uniform on compact $\mathcal{L}$ by a
  finite-net argument and $\|I-Q_n\| \le 1$. Each $T_{\mathrm{sp}}Q_n$ has
  finite-dimensional range; their operator-norm limit is a compact map. This is
  compactness of the map, in addition to compact values. This proves
  Proposition~\ref{P:cameron-spatial-map}.
\end{proof}

\subsection{The supported tensor and temporal mean}

Proposition~\ref{P:cameron-spatial-map} treats the spatial variable alone. A
control in $H$ depends on time and space, so we first split $H$ into a time
factor and a spatial factor.

\begin{lemma}[Supported original tensor space]\label{L:cameron-supported-tensor}
  Canonically $H = H_t\otimes H_{\mathrm{sp}}$ as real Hilbert spaces. It is
  separable and admits a complete orthonormal basis $e_j$ with smooth test
  representatives supported inside $I\times\R^d$. Projection $P_J$ onto its
  first $J$ elements is also an orthogonal projection on $H_W$ by the same
  finite formula.
\end{lemma}

\begin{proof}
  On separated smooth tests, absolute Fubini makes the original inner product
  exactly the tensor product after both null quotients. Finite sums are dense:
  approximate a smooth compact spacetime test uniformly by smooth time
  partitions times its spatial slices, on one common compact support inside
  $I\times\R^d$. A uniform error $\epsilon$ on spatial compact $A$ has squared
  original covariance norm at most
  \[
    \epsilon^2|A|\Gamma(A-A)
    \int_I\int_I|s-t|^{-a}\ud s\ud t<\infty.
  \]
  This also gives countable smooth-test density on exhausting compact sets, both
  here and in $H_{\mathrm{sp}}$. Gram--Schmidt skipping zeros preserves smooth
  compact support. The isometric embedding into $H_W$ has closed image by
  completeness; its finite orthonormal projection formula remains orthogonal
  there. It is not multiplication by a time indicator and gives no independence
  of disjoint time intervals. This proves
  Lemma~\ref{L:cameron-supported-tensor}.
\end{proof}

It remains to treat the time factor $H_t$, whose kernel is $|s-t|^{-a}$.
Proposition~\ref{P:cameron-full-mean} then combines the two maps.

\begin{proposition}[The compact temporal mean, without trace class]
  \label{P:cameron-temporal-map}
  Let $K$ on $L^2(I)$ have kernel $|s-t|^{-a}$. It is compact, self-adjoint,
  strictly positive and injective. The map $\iota:f\mapsto K^{1/2}f$ extends
  isometrically from $H_t$ onto $L^2(I)$, and the temporal mean
  \begin{equation}\label{E:cameron-temporal-map}
    C_t=K^{1/2}\iota:H_t\longrightarrow L^r(I)
  \end{equation}
  is bounded and compact for every $2 < r < 2/a$. The displayed formula first
  has its $L^2(I)$ meaning; the same completed map is then shown to take values
  compactly in $L^r(I)$. The operator $K$ is not trace class and $C_t:H_t \to
  L^2(I)$ is not Hilbert--Schmidt.
\end{proposition}

\begin{proof}
  Schur bounds $K$ by its finite row and column integrals. The Gamma
  representation of $|s-t|^{-a}$ and zero extension of $f \in L^2(I)$ give
  strict positivity directly: the kernel $e^{-\lambda|v|}$ has quadratic form
  \[
    \int_\R\frac{2\lambda}{\lambda^2+\xi^2}
    |\widehat{f}(\xi)|^2\ud\xi>0
    \quad(f\ne0,\ \lambda>0)
  \]
  in the unitary Fourier convention. Absolute Fubini is justified by Schur
  applied to $|f(s)f(t)|$. Integrating with positive weight
  $\lambda^{a-1}/\Gamma_E(a)$ proves positivity and injectivity. Symmetry gives
  self-adjointness. The deterministic truncated kernel $K_\epsilon =
  K1_{\{|s-t|\ge\epsilon\}}$ is square integrable, so finite separated-kernel
  approximations make it compact. The Schur norm of $K-K_\epsilon$ is at most
  $2\epsilon^{1-a}/(1-a)$, proving compactness of $K$. These truncations are not
  used as Gaussian covariances. Its positive square root is compact by the
  compact spectral theorem.

  On tests $\|K^{1/2}f\|_2 = \|f\|_{H_t}$. Smooth-test $L^2$ density and $\ker K
  = 0$ imply that its range is dense, so the isometry extends onto $L^2(I)$. On
  tests $C_tf = Kf$, and~\eqref{E:cameron-temporal-map} first gives compactness
  into $L^2$.

  Extend $K_a(v) = |v|^{-a}1_{\{|v|\le1\}}$ to the line. It lies in $L^{r/2}$.
  Scalar Young convolution gives
  \[
    \|g\|_{H_t}^2\le\|K_a\|_{r/2}\|g\|_{r'}^2,\qquad
    r'=r/(r-1).
  \]
  To verify the needed Young inequality, take $u = r/2$, $v = r'$, so $u,v < r$
  and $1/u+1/v = 1+1/r$. Split the absolute convolution integrand into
  $(|f|^u|h|^v)^{1/r}$, $|f|^{1-u/r}$ and $|h|^{1-v/r}$. H\"older with exponents
  $r$, $(1/u-1/r)^{-1}$ and $(1/v-1/r)^{-1}$, followed by integration and
  Tonelli, gives $\|f*h\|_r \le \|f\|_u\|h\|_v$. Covariance Cauchy--Schwarz and
  $L^r$ duality now imply $\|Kf\|_r \le C_{a,r}\|f\|_{H_t}$ on tests. Their $Kf$
  are bounded by the integrable-kernel row, making duality legitimate before
  completion. Density extends the estimate, with $L^2$ value equal
  to~\eqref{E:cameron-temporal-map}. Choose $r < r_2 < 2/a$. Compactness into
  $L^2$ and boundedness into $L^{r_2}$ give compactness into $L^r$ by $\|u\|_r
  \le \|u\|_2^\tau\|u\|_{r_2}^{1-\tau}$ with $1/r = \tau/2+(1-\tau)/r_2$.

  Finally, the orthonormal cell indicators $N^{1/2}1_{(j/N,(j+1)/N)}$, $0 \le j
  < N$, have $K$ quadratic forms summing to $2N^a/((1-a)(2-a)) \to \infty$.
  Thus, positive $K$ has infinite trace, and $K^{1/2}\iota$ is not
  Hilbert--Schmidt into $L^2$. No instantaneous $L^2$-valued Gaussian has been
  introduced. This proves Proposition~\ref{P:cameron-temporal-map}.
\end{proof}

No-go~\ref{P:nogo-27} explains the upper endpoint of this compactness
range: temporal concentration makes the mean map unbounded above
$2/\alpha_0$ and noncompact at $2/\alpha_0$. The endpoint witness does not
refute boundedness there.


\begin{proposition}[The compact time-space mean]\label{P:cameron-full-mean}
  Fix one $2 < r < 2/a$ throughout this proposition. For finite tensors with
  orthonormal spatial labels define
  \[
    \mathcal{U}\left(\sum_j f_j\otimes q_j\right)(s)
    =\sum_j(Kf_j)(s)q_j.
  \]
  It extends to a bounded map $H \to L^r(I;H_{\mathrm{sp}})$ with dimension-free
  bound. The map
  \begin{equation}\label{E:cameron-full-mean}
    T\Phi(s)=T_{\mathrm{sp}}(\mathcal{U}\Phi(s))
    :H\longrightarrow L^r(I;\mathcal{K}(F))
  \end{equation}
  is compact. For smooth supported $\Phi$, it is the multiplier
  $iB_F(m_Rh_\Phi(s))$, where $h_\Phi = |\cdot|^{-a}*(\Gamma*\Phi)$. For the
  supported projections of Lemma~\ref{L:cameron-supported-tensor},
  \begin{equation}\label{E:cameron-compact-tail}
    \epsilon_J:=\|T(I-P_J)\|_{H\to L^r(\mathcal{K}(F))}
    \longrightarrow0.
  \end{equation}
\end{proposition}

\begin{proof}
  For a finite-dimensional spatial field $g$, positivity of the time kernel and
  scalar Young give
  \[
    \|g\|_{H_t\otimes H_{\mathrm{sp}}}^2
    \le\int_I\int_I|s-t|^{-a}\|g_s\|\|g_t\|\ud s\ud t
    \le C_{a,r}^2\|g\|_{L^{r'}(H_{\mathrm{sp}})}^2.
  \]
  Hilbert Cauchy--Schwarz and finite-dimensional $L^r$ duality give
  $\|\mathcal{U}\Phi\|_{L^r(H_{\mathrm{sp}})} \le C_{a,r}\|\Phi\|_H$. The
  duality can be checked by the pointwise direction
  $\|\mathcal{U}\Phi\|^{r-2}\mathcal{U}\Phi$ and smooth coefficient
  approximation in $L^{r'}$, controlled by the displayed estimate. No dimension
  factor occurs. Tensor density and completeness extend this to a strongly
  measurable Bochner field. This time-only map still contains spatial vectors in
  the covariance space. Spatial covariance is applied once, by
  $T_{\mathrm{sp}}$.

  The physical mean identity on separated tests follows
  from~\eqref{E:cameron-spatial-row}. For a smooth spacetime test use the
  separated compact-support approximations in the tensor proof. Their uniform
  errors give uniformly small masked mean errors, using the finite time row and
  locally finite spatial covariance on the relevant compact differences. The
  multiplier bound by $L^\infty+L^2$ then gives uniform operator convergence,
  identifying the $L^r$ mean in~\eqref{E:cameron-full-mean}. General $H$
  controls denote completed operator fields, without a pointwise-potential
  assertion.

  Let $Q_n$ be the spatial projections in the compact-map proof. The maps
  $T_n\Phi = T_{\mathrm{sp}}Q_n\mathcal{U}\Phi$ approximate $T$ in operator norm
  by the dimension-free bound and $\|T_{\mathrm{sp}}(I-Q_n)\| \to 0$. For fixed
  $n$, each spatial coordinate extraction $H \to H_t$ has norm at most one and
  its image under $C_t$ is precompact in $L^r$. Finite combination with the
  compact operators $T_{\mathrm{sp}}q_j$ makes $T_n$ compact. Hence, $T$ is
  compact. Scalar time compactness alone tensored with an infinite identity
  would not prove this.

  To prove~\eqref{E:cameron-compact-tail}, if it failed choose unit $v_J$ on a
  subsequence with $\|T(I-P_J)v_J\|$ bounded below. The vectors $(I-P_J)v_J$
  tend weakly to zero, since their pairings with fixed $h$ are bounded by
  $\|(I-P_J)h\|$. Compactness gives a norm-convergent image subsequence with
  weak limit zero, a contradiction. This proves
  Proposition~\ref{P:cameron-full-mean}.
\end{proof}

The radius remains fixed in this approximation. No-go~\ref{P:nogo-25}
shows that translated white-noise controls prevent these fixed smooth
projections from giving tails uniform over growing radii.


\subsection{The noise residual and its Cameron operator}

Choose integer $B \ge 1$ and $0 < \delta < \min(1,1/B)$. Put $N =
\lfloor1/(\delta B)\rfloor$, $s_m = m\delta B$, $\ell_m = B$ for $m < N$ and
$\ell_N = \delta^{-1}-NB \in [0,B)$. The intervals $I_m =
[s_m,s_m+\delta\ell_m]$ partition $[0,1]$ up to endpoints. With $Q_J = I-P_J$ on
$H_W$ and the explicit pullback~\eqref{E:raw-block-pullback}, define
\[
  Y_{m,J}(f)=W(Q_JT_{\delta,m}f),\qquad
  g_\delta=\delta^{(1-a)/2}.
\]
Fix parameters
\begin{equation}\label{E:cameron-block-parameters}
  \frac12<\theta<\min\left(1-\frac a2,\frac{3-a-\alpha}{2}\right),
  \qquad \frac1{1-\theta}<r<\frac2a,\qquad
  \frac1\theta\le p<2<r .
\end{equation}
These intervals are nonempty.

\begin{lemma}[Supported residual paths]\label{L:cameron-actual-residual}
  Gaussian transfer with mask $m_R$ and its $F$ extension constructs the
  centered Gaussian paths
  \[
    X_{m,J}(u)=g_\delta A_{m_R}(Y_{m,J})(\min(u,\ell_m)),
    \qquad 0\le u\le B.
  \]
  Their finite tuple lies in the separable space $\mathscr{B}_0^{N+1}$, where
  $\mathscr{B}_0 = h_0^\theta([0,B];\mathcal{K}(F))$. Every continuous linear
  statistic $\ell(X)$ is $W(k_\ell)$ for some $k_\ell \in H_I$. An empty last
  interval gives zero.
\end{lemma}

\begin{proof}
  Isometry of the pullback and orthogonality of $Q_J$ give
  $\operatorname{Var}Y_{m,J}(f) \le \|f\|_{H_W}^2$.
  Proposition~\ref{P:source-gaussian-transfer} therefore applies. Its finite
  sums retain both deterministic localizations, and their limits retain the true
  $F$ extension and constant columns. Choose $\theta < \beta < \beta_2$ below
  the upper threshold in~\eqref{E:cameron-block-parameters}; read the
  operator-path construction at exponent $\beta$ with auxiliary $\beta_2$.
  Constant stopping preserves compact values and $C^\beta$ bounds, hence gives
  paths in the little $C^\theta$ closure by polygonal approximation. Its
  separability is Lemma~\ref{L:raw-block-inputs}.

  As in Lemma~\ref{L:raw-block-inputs}, every observed row's slow pullback
  belongs to $H_I$: its time support is in $[0,1]$, and approximating its
  indicator inside $I$ converges in original covariance norm by dominated
  convergence against the integrable time kernel and locally finite $\Gamma$ on
  the common compact support. This includes endpoints. Schwartz $E$ states and
  both true constant columns supply the identities, then boundedness extends
  them to $F$. Let $P_H$ be the orthogonal projection onto closed $H_I$.
  Transfer for the noise $f\mapsto W(P_HQ_JT_{\delta,m}f)$ leaves each observed
  row unchanged, since its residual slow test already lies in $H_I$. Countable
  states and rational times identify the same operator paths. This
  projected-transfer series has coefficient tests in $H_I$. Applying $\ell$
  gives an $L^2$ limit in the closed isonormal image $W(H_I)$, proving the
  assertion. Original global-basis tests were not assumed individually
  supported. No time-support independence or Hilbert-valued noise is used. This
  proves Lemma~\ref{L:cameron-actual-residual}.
\end{proof}

\begin{proposition}[Cameron covariance and averaged bound]
  \label{P:cameron-actual-embedding}
  The covariance map $L_{\delta,J}\Phi = \E[XW(\Phi)]$ is
  \begin{equation}\label{E:cameron-actual-covariance}
    (L_{\delta,J}\Phi)_m(u)=\sqrt\delta
    \int_0^{\min(u,\ell_m)}
    S_F(-v)T(Q_J\Phi)(s_m+\delta v)S_F(v)\ud v ,
    \qquad \Phi\in H.
  \end{equation}
  This is a compact-operator Bochner integral. Define the exact norm and
  constants
  \begin{equation}\label{E:cameron-averaged-norm}
    \|x\|_{\delta,p}
    =\left(\delta\sum_{m=0}^N\|x_m\|_{C^\theta}^p\right)^{1/p},
    \quad G_B=\sup_{|v|\le B}\|S_F(v)\|,
    \quad C_B=G_B^2[B^{1-1/r}+B^{1-1/r-\theta}].
  \end{equation}
  Then the Gaussian Cameron embedding satisfies
  \begin{equation}\label{E:cameron-averaged-bound}
    \|L_{\delta,J}\Phi\|_{\delta,p}
    \le\sqrt\delta\,C_B(2/B)^{1/p-1/r}
    \epsilon_J\|\Phi\|_H .
  \end{equation}
  Convergence $\epsilon_J \to 0$ fixes $R$ and is independent of $\delta$.
\end{proposition}

\begin{proof}
  For smooth supported $\Phi$, $Q_J\Phi$ is another such test. The explicit
  pullback covariance~\eqref{E:raw-block-mean-jacobian} gives
  \[
    \E[Y_{m,J}(f)W(\Phi)]
    =\delta^{a/2}\int f(v,x)
    h_{Q_J\Phi}(s_m+\delta v,x)\ud x\ud v .
  \]
  Absolute Fubini also holds for signed compact tests. Multiplying by $g_\delta$
  gives $\sqrt\delta$. Explicit rows and~\eqref{E:cameron-spatial-row} give
  \eqref{E:cameron-actual-covariance}. The constant-velocity input acquires
  displacement $v$ under $S_F(v)$; the constant displacement remains one. Both
  columns are included.

  Both $S_F$ and its adjoint are strongly continuous on Hilbert $F$. In
  Bessel-unitary $E$ coordinates the Fourier matrix is
  \[
    \begin{pmatrix}
      \cos(v|\xi|)&\langle\xi\rangle\sin(v|\xi|)/|\xi|\\
      -|\xi|\sin(v|\xi|)/\langle\xi\rangle&\cos(v|\xi|)
    \end{pmatrix}.
  \]
  Its continuous zero quotient and common bound $C(1+B)$ apply also to its
  adjoint; dominated convergence proves the assertion. The constant block is
  finite dimensional. For compact $A$, conjugation $S_F(-v)AS_F(v)$ is norm
  continuous: strong convergence is uniform on compact images, and the right
  factor follows by adjoints. Joint norm continuity in $(v,A)$ and uniform
  bounds give Bochner measurability for the conjugated strongly measurable
  compact-operator field. No norm continuity of the free group itself is
  assumed.

  Write $M = TQ_J\Phi$. H\"older and the primitive increment estimate give
  \begin{equation}\label{E:cameron-local-primitive-bound}
    \|(L_{\delta,J}\Phi)_m\|_{C^\theta}
    \le\sqrt\delta\,C_B\delta^{-1/r}\|M\|_{L^r(I_m)}.
  \end{equation}
  Its primitive exponent is $1-1/r > \theta$. Simple compact-operator integrand
  approximation in $L^r$ puts the primitive in $\mathscr{B}_0$; constant
  stopping does not increase the H\"older norm. Affine pullback preserves null
  sets. For fixed $\delta,J$ the covariance map is bounded by Bochner
  Cauchy--Schwarz and finite tuple $L^2$ moments. The integral is bounded by the
  displayed estimate, so equality on smooth controls extends by density to all
  $H$. Dense states and rational times make it a full operator-path identity.

  Let $H_{\mathrm{seen}}$ be the closed span of the $k_\ell$ from
  Lemma~\ref{L:cameron-actual-residual}. Then $\ell(L_{\delta,J}\Phi) = \langle
  k_\ell,\Phi\rangle_H$ and $\ker L_{\delta,J} = H_{\mathrm{seen}}^\perp$. The
  first Wiener chaos definition of the Cameron space is therefore
  $L_{\delta,J}(H_{\mathrm{seen}})$ with the minimum $H$ preimage norm. Extra
  directions are removed by orthogonal projection, without an inverse
  covariance. A bound in the original $H$ norm is thus the embedding bound.

  Put $t_m = \|M\|_{L^r(I_m)} \ge 0$ and $n = N+1 \ge 2$. Finite power mean at
  exponent $r/p > 1$, applied to $t_m^p$, gives
  \[
    \left(\sum_m t_m^p\right)^{1/p}
    \le n^{1/p-1/r}\left(\sum_m t_m^r\right)^{1/r}.
  \]
  The true partition gives $\sum_m t_m^r = \|M\|_{L^r(I)}^r$, including an empty
  last interval, and $n \le 2/(\delta B)$. Hence, the exact $\delta$-weighted
  norm gives
  \[
    \|L_{\delta,J}\Phi\|_{\delta,p}
    \le\sqrt\delta\,C_B(\delta n)^{1/p-1/r}\|M\|_{L^r}
    \le\sqrt\delta\,C_B(2/B)^{1/p-1/r}
    \epsilon_J\|\Phi\|_H.
  \]
  The $B$ factor remains explicit; covariance and mask occur once. This proves
  Proposition~\ref{P:cameron-actual-embedding}.
\end{proof}

These conclusions fix the radius and make no uniform-radius projection tail
assertion. No concentration estimate or original moment upper bound follows from
this section alone.

\section{Mean and Cameron bounds uniform in the spatial radius}
\label{S:uniform-mean-regulation}

Retain the original nonzero covariance, amplitude and real spaces of
Section~\ref{S:actual-cameron-mean}. Fix $\alpha/4 < s < 1/2$ and put $\beta =
1/2-s > 0$. Write $A_s = C_s^2\sqrt{R_{\mathrm{sp}}}$ for the constant
in~\eqref{E:cameron-fractional-product}. Choose a real even smooth compact
$\eta$, equal to one on the unit ball, with $0 \le \eta \le 1$ and support in
the closed radius-two ball. For $R \ge 1$ set $\eta_R(x) = \eta(x/R)$ and $m_R =
\eta_R^2$. In this section the mean acts on $E = H^{1/2}\times H^{-1/2}$. The
constant data columns belong to $F$, outside $E$. All uniform-radius claims
below concern $E$. The spatial map $T_R:H_{\mathrm{sp}} \to \mathcal{K}(E)$ is
the $E$ restriction of Proposition~\ref{P:cameron-spatial-map}. Its explicit row
is
\begin{equation}\label{E:uniform-mean-row}
  \langle T_Rqz,w\rangle_E
  =\langle q,k_R(z,w)\rangle_{H_{\mathrm{sp}}},
  \qquad
  k_R(z,w)=(\eta_Rz_1)(\eta_R\Lambda^{-1}w_2).
\end{equation}
Thus, the physical mask occurs once, after covariance convolution.

\subsection{Uniform multiplier and spatial tails}

We first bound the row $k_R$ in~\eqref{E:uniform-mean-row} uniformly in
$R \ge 1$. Both factors of this row carry the cutoff $\eta_R$, so we start with
its multiplier norm on $H^\sigma$.

\begin{lemma}[Uniform cutoff and physical mean]\label{L:uniform-mean-cutoff}
  For every real $\sigma$, multiplication by $\eta_R$ on $H^\sigma$ has norm at
  most
  \[
    M_\sigma=2^{|\sigma|/2}(2\pi)^{-d/2}
    \int_{\R^d}|\widehat{\eta}(p)|\langle p\rangle^{|\sigma|}\ud p,
  \]
  independently of $R \ge 1$. With $M = M_{1/2}$ and $M_0 = A_sM^2$,
  \begin{equation}\label{E:uniform-spatial-mean}
    \|k_R(z,w)\|_{H_{\mathrm{sp}}}
    \le M_0\|z\|_E\|w\|_E,\qquad
    \|T_R\|_{H_{\mathrm{sp}}\to\mathcal{L}(E)}\le M_0 .
  \end{equation}
\end{lemma}

\begin{proof}
  Peetre's inequality bounds the weighted Fourier kernel for multiplication by
  $\eta_R$ by convolution with
  \[
    2^{|\sigma|/2}(2\pi)^{-d/2}
    |\widehat{\eta_R}(q)|\langle q\rangle^{|\sigma|}.
  \]
  Here $\widehat{\eta_R}(q) = R^d\widehat{\eta}(Rq)$. Substitution $p = Rq$ and
  $\langle p/R\rangle \le \langle p\rangle$ bound its $L^1$ norm by $M_\sigma$.
  The $L^1*L^2$ inequality proves the assertion on Schwartz functions and
  density extends it to $H^\sigma$. The map $w_2\mapsto\Lambda^{-1}w_2$ is an
  isometry from $H^{-1/2}$ to $H^{1/2}$, and $H^{1/2}\hookrightarrow H^s$ has
  norm one. Apply the cutoff bound at $\sigma = 1/2$ and
  Lemma~\ref{L:cameron-fractional-products} to the two factors
  in~\eqref{E:uniform-mean-row}. Hilbert Cauchy--Schwarz then gives the operator
  bound. This is the existing completed physical map: the row agrees on smooth
  controls and hence on their completion. Its compact values at each fixed $R$
  are retained. This proves Lemma~\ref{L:uniform-mean-cutoff}.
\end{proof}

Lemma~\ref{L:uniform-mean-cutoff} bounds $T_R$ on all of $H_{\mathrm{sp}}$.
The approximation in Proposition~\ref{P:uniform-mean-regulator} also needs
$T_R$ to be small on high frequencies, uniformly in $R \ge 1$. We obtain this
from a bound on $T_R(I-C_\varepsilon)$ for the heat contraction
$C_\varepsilon$.

\begin{lemma}[Uniform heat and frequency tails]
  \label{L:uniform-mean-spatial-tails}
  Let $C_\varepsilon$ be the covariance-Hilbert heat contraction with Fourier
  multiplier $e^{-\varepsilon|\xi|^2/2}$. There is a real orthogonal projection
  $P_K$ on $H_{\mathrm{sp}}$ corresponding to the closed frequency ball
  $\{|\xi|\le K\}$. It commutes with heat. For $K \ge 1$, $\varepsilon > 0$ and
  $R \ge 1$,
  \begin{align}
    \|T_R(I-C_\varepsilon)\|
    &\le M_h\varepsilon^{\beta/2},
    &M_h&=4A_sM^2\E|G_d|^\beta,\label{E:uniform-mean-heat}\\
    \|T_R(I-P_K)\|
    &\le M_fK^{-\beta},
    &M_f&=\frac{M_h}{1-e^{-1/2}},\label{E:uniform-mean-frequency}
  \end{align}
  where $G_d$ is a standard real $d$-dimensional normal vector. Operator norms
  are from $H_{\mathrm{sp}}$ to $\mathcal{L}(E)$. The projection $P_K$ is
  allowed to have infinite rank.
\end{lemma}

\begin{proof}
  Translations $\tau_hf(x) = f(x-h)$ extend to strongly continuous isometries of
  the $H_{\mathrm{sp}}$. Indeed, in its covariance Fourier isometry they
  multiply by $e^{-i\xi\cdot h}$; dominated convergence gives continuity on
  tests, then density and isometry give it on the completion. The real subspace
  is preserved. The inequalities $|e^{-it}-1| \le 2\min(1,|t|^\beta)$ and
  $|\xi|^\beta\langle\xi\rangle^s \le \langle\xi\rangle^{1/2}$ show that
  \[
    \|\tau_hf-f\|_{H^s}
    \le2\min(1,|h|^\beta)\|f\|_{H^{1/2}} .
  \]
  Use this for $f = \eta_Rz_1$ and $g = \eta_R\Lambda^{-1}w_2$. The product
  identity
  \[
    \tau_h(fg)-fg=(\tau_hf-f)\tau_hg+f(\tau_hg-g)
  \]
  holds in $H_{\mathrm{sp}}$, first for smooth factors and then by the
  continuous fractional product extension. Lemma~\ref{L:uniform-mean-cutoff}
  gives
  \begin{equation}\label{E:uniform-mean-row-translation}
    \|\tau_hk_R(z,w)-k_R(z,w)\|_{H_{\mathrm{sp}}}
    \le4A_sM^2\min(1,|h|^\beta)\|z\|_E\|w\|_E .
  \end{equation}

  The Bochner integral $\int p_\varepsilon(h)\tau_hk\ud h$ exists, since the
  translation orbit is strongly continuous and bounded in the separable
  covariance Hilbert space. Its Fourier image is
  $e^{-\varepsilon|\xi|^2/2}\widehat{k}$: ordinary Gaussian integration proves
  this on tests and contraction extends it to all $k$. It is therefore the
  $C_\varepsilon$ of the source construction. Heat has variance $\varepsilon$
  here; there is no additional Fourier normalization factor. Integrating
  \eqref{E:uniform-mean-row-translation} with $h = \sqrt\varepsilon G_d$ bounds
  $\|(I-C_\varepsilon)k_R(z,w)\|$ by $M_h\varepsilon^{\beta/2}\|z\|_E\|w\|_E$.
  Selfadjointness in~\eqref{E:uniform-mean-row} proves
  \eqref{E:uniform-mean-heat}.

  To construct $P_K$, let $V$ be the closed image of the covariance Fourier
  isometry into $L^2(\mu)$, including its fixed $(2\pi)^{d/2}$ factor. No
  surjectivity is assumed. Its formula extends to Schwartz functions: compact
  cutoff approximation converges in Schwartz seminorms, and temperedness of
  $\mu$ controls the Fourier error in $L^2(\mu)$. Choose real even radial smooth
  $\chi_n$, equal to one on the closed ball of radius $K$, zero outside radius
  $K+1/n$, and between zero and one. For real Schwartz $f$, the function
  $\chi_n\widehat{f}$ is the transform of another real Schwartz function and so
  lies in $V$. Dominated convergence against $|\widehat{f}|^2\mu$ gives
  $1_{\{|\xi|\le K\}}\widehat{f} \in V$. The pointwise limit includes the
  boundary sphere, even if $\mu$ charges it. Test density and contraction extend
  this assertion to all of $V$. Indicator multiplication is selfadjoint and
  idempotent on $V$; pulling it back defines the asserted real orthogonal $P_K$.
  All Fourier multipliers here commute.

  On $|\xi| > K$ one has $1-e^{-|\xi|^2/(2K^2)} \ge 1-e^{-1/2}$, whence
  \[
    \|(I-P_K)k\|_{H_{\mathrm{sp}}}
    \le(1-e^{-1/2})^{-1}\|(I-C_{K^{-2}})k\|_{H_{\mathrm{sp}}}.
  \]
  Apply the row heat bound and selfadjointness of $P_K$
  in~\eqref{E:uniform-mean-row} to obtain \eqref{E:uniform-mean-frequency}. No
  density of $\mu$, periodic state space or finite rank was used. This proves
  Lemma~\ref{L:uniform-mean-spatial-tails}.
\end{proof}

\subsection{Tensor products and mean approximation}

We first record a tensor extension bound for maps into $L^r(I)$. The proof
of Proposition~\ref{P:uniform-mean-regulator} uses it to pass from a scalar
time map to the Hilbert-valued time map without a dimension factor.

\begin{lemma}[Dimension-free Hilbert tensor extension]
  \label{L:uniform-mean-tensor}
  If $H_0,G$ are real Hilbert spaces, $2 \le r < \infty$ and $A:H_0 \to L^r(I)$
  is bounded, then
  \[
    \|A\otimes I_G\|_{H_0\otimes G\to L^r(I;G)}\le\|A\|.
  \]
  The tensor extension need not be compact when $A$ is compact.
\end{lemma}

\begin{proof}
  For a finite tensor write $v = \sum_{j=1}^n f_j\otimes e_j$ with $(e_j)$
  orthonormal. Its squared norm is $\sum_j\|f_j\|^2$. Let $\gamma_j$ be
  independent standard real normals and $m_r = \E|\gamma_1|^r$. Finite
  covariance diagonalization gives
  \[
    \E\left\|\sum_j\gamma_j f_j\right\|_{H_0}^r
    \le m_r\left(\sum_j\|f_j\|_{H_0}^2\right)^{r/2}.
  \]
  In fact, if $\lambda_l$ are the covariance eigenvalues and $L =
  \sum_l\lambda_l$, the squared norm has law $\sum_l\lambda_l\zeta_l^2$ for
  independent standard $\zeta_l$. For $L > 0$, convexity at exponent $r/2 \ge 1$
  bounds $(\sum_l(\lambda_l/L)\zeta_l^2)^{r/2}$ by
  $\sum_l(\lambda_l/L)|\zeta_l|^r$. Taking expectations proves the inequality;
  $L = 0$ is immediate. Also $L = \sum_j\|f_j\|^2$.

  Choose measurable representatives of the finitely many $Af_j$. Scalar Gaussian
  moments and Tonelli give
  \[
    m_r\int_I\left(\sum_j|Af_j(t)|^2\right)^{r/2}\ud t
    =\E\left\|A\sum_j\gamma_jf_j\right\|_{L^r}^r
    \le\|A\|^rm_r\left(\sum_j\|f_j\|^2\right)^{r/2}.
  \]
  Cancel the finite positive $m_r$ and take the $r$th root. Finite tensor
  density and Bochner completeness give the extension. The zero space is
  trivial. If $Af \ne 0$ and $G$ has an infinite orthonormal sequence $(e_j)$,
  the images of the bounded sequence $f\otimes e_j$ have pairwise distance
  $\sqrt2\|Af\|_{L^r}$. They have no convergent subsequence, even when the
  scalar map $A$ is compact. Only finite auxiliary Gaussians were used in the
  estimate. This proves Lemma~\ref{L:uniform-mean-tensor}.
\end{proof}

Recall the supported identity $H_I = H_t\otimes H_{\mathrm{sp}}$, the compact
scalar covariance map $C_t:H_t \to L^r(I)$, $2 < r < 2/a$, and its completed
tensor $U = C_t\otimes I_{H_{\mathrm{sp}}}$ from
Section~\ref{S:actual-cameron-mean}. On finite smooth tensors $U$ agrees with
the earlier $\mathcal{U}$; density gives equality of their completed maps. For
block applications fix the same $r$ as in~\eqref{E:cameron-block-parameters}.
All $c_t,\varepsilon_J$ and mean norms below use that one choice. Choose a
smooth-test orthonormal basis of $H_t$ and let $\pi_J$ be its first-$J$
orthogonal projection, with $\pi_0 = 0$. Put
\[
  c_t=\|C_t\|,\qquad
  \varepsilon_J=\|C_t(I-\pi_J)\|_{H_t\to L^r(I)},\qquad
  \mathcal{T}_R=T_RU .
\]

\begin{proposition}[Uniform approximation of the completed mean]
  \label{P:uniform-mean-regulator}
  On $H_I$ the operators
  \[
    P_{\mathrm{ret}}=\pi_J\otimes P_K,\qquad
    Q_{J,K}=I-P_{\mathrm{ret}}
  \]
  are real orthogonal projections, and $\varepsilon_J \to 0$. The mean is a
  strongly measurable $L^r(I;\mathcal{K}(E))$ field and satisfies, uniformly for
  $R \ge 1$,
  \begin{equation}\label{E:uniform-completed-mean}
    \|\mathcal{T}_R\|\le M_0c_t,\qquad
    \|\mathcal{T}_RQ_{J,K}\|
    \le M_0\varepsilon_J+M_fc_tK^{-\beta}.
  \end{equation}
  These norms have domain $H_I$ and target $L^r(I;\mathcal{L}(E))$.
\end{proposition}

\begin{proof}
  The tensor projections are selfadjoint, idempotent and contractive on finite
  tensors and hence on their Hilbert completion. For the scalar tail, failure of
  convergence would supply unit $v_J$ with $\|C_t(I-\pi_J)v_J\|$ bounded below
  on a subsequence. But $(I-\pi_J)v_J$ tends weakly to zero, since its pairing
  with $h$ is bounded by $\|(I-\pi_J)h\|$. Compactness supplies a
  norm-convergent image subsequence; bounded linear scalar observables show its
  limit is zero, a contradiction. Lemma~\ref{L:uniform-mean-tensor} transfers
  this scalar tail norm to the Hilbert-valued time map without a dimension
  factor.

  For fixed $R$, $T_R$ is bounded into the separable compact-operator space on
  $E$. Composition with the Bochner field $U\Phi$ gives the stated mean and its
  first bound. The exact decomposition is
  \[
    Q_{J,K}=(I-\pi_J)\otimes I+\pi_J\otimes(I-P_K).
  \]
  The first term is bounded by $M_0\varepsilon_J$. For the second term use the
  intertwining identity
  \[
    U(\pi_J\otimes(I-P_K))
    =(I_{L^r}\otimes(I-P_K))(C_t\pi_J\otimes I).
  \]
  It holds first on finite tensors, then by continuity.
  Apply~\eqref{E:uniform-mean-frequency} pointwise, followed by
  Lemma~\ref{L:uniform-mean-tensor} for $C_t\pi_J$, whose norm is at most $c_t$.
  This proves~\eqref{E:uniform-completed-mean}. Infinite spatial rank is allowed
  throughout; neither tensor compactness nor uniform-radius compactness has been
  asserted. This proves Proposition~\ref{P:uniform-mean-regulator}.
\end{proof}

\subsection{The block Cameron embedding for the given noise}

Use $B,\delta,N,s_m,\ell_m,I_m$ and $\theta,r,p$ exactly as
in~\eqref{E:cameron-block-parameters} and its preceding block partition. Let
$i_I:H_I \to H_W$ be the isometric closed inclusion. On the full original noise
space set
\[
  P_{\mathrm{full}}=i_IP_{\mathrm{ret}}i_I^*,\qquad
  Q_{\mathrm{full}}=I-P_{\mathrm{full}} .
\]
These are orthogonal projections on $H_W$. Define the residual fast noise and
stopped $E$ operator path by
\begin{equation}\label{E:uniform-actual-residual}
  Y_m(f)=W(Q_{\mathrm{full}}T_{\delta,m}f),\qquad
  X_m(u)=g_\delta A_{m_R}(Y_m)(\min(u,\ell_m)),\qquad
  g_\delta=\delta^{(1-a)/2}.
\end{equation}
Here $T_{\delta,m}$ is the explicit original isometry
\eqref{E:raw-block-pullback}, and $A_{m_R}$ takes its $E$ restriction. Its
isometry is proved in Lemma~\ref{L:raw-block-inputs}. The symbol
$H_{\mathrm{fast}}$ below denotes this same original covariance norm in fast
coordinates, carried isometrically into $H_W$. The $E$ little H\"older space
means the closure of Lipschitz compact-operator paths starting at zero in
$C^\theta$ norm. Compact operators on separable $E$ and polygonal approximation
make this closure separable. The last interval can be empty, in which case $X_N
= 0$.

\begin{proposition}[Uniform-radius Cameron bound]
  \label{P:uniform-actual-cameron}
  The tuple $X = (X_m)_{m=0}^N$ is a centered Gaussian variable in the separable
  little H\"older compact-operator path space on $E$. Its Bochner covariance $Lh
  = \E[XW(h)]$, $h \in H_W$, is
  \begin{equation}\label{E:uniform-actual-covariance}
    (Lh)_m(u)=\sqrt\delta\int_0^{\min(u,\ell_m)}
    S_E(-v)\mathcal{T}_R(Q_{J,K}i_I^*h)(s_m+\delta v)
    S_E(v)\ud v .
  \end{equation}
  In the exact norm $\|x\|_{\delta,p} =
  (\delta\sum_m\|x_m\|_{C^\theta}^p)^{1/p}$, its Cameron embedding has the
  uniform-radius bound
  \begin{equation}\label{E:uniform-actual-cameron-bound}
    \|Lh\|_{\delta,p}
    \le\sqrt\delta\,C_{B,E}(2/B)^{1/p-1/r}
    [M_0\varepsilon_J+M_fc_tK^{-\beta}]\|h\|_{H_W},
    \qquad C_{B,E}\le18B^{3-1/r}.
  \end{equation}
  All spatial constants are independent of $R \ge 1$.
\end{proposition}

\begin{proof}
  Orthogonality and original pullback isometry give $\operatorname{Var}Y_m(f)
  \le \|f\|_{H_{\mathrm{fast}}}^2$. Gaussian transfer,
  Proposition~\ref{P:source-gaussian-transfer}, therefore constructs the
  operator paths on the given $W$, with finite path moments at each fixed
  $R,B,\delta,J,K$. Choose an admissible H\"older exponent strictly above
  $\theta$ as in Lemma~\ref{L:cameron-actual-residual}; constant stopping puts
  the paths in the separable little $C^\theta$ closure. Finite-coefficient
  constructions and their $L^2$ limits show that every continuous linear
  coordinate is in the first chaos of the given $W$.

  Every explicit observed row, pulled back to slow time, belongs to $H_I$. This
  is the supported-row approximation from Lemma~\ref{L:cameron-actual-residual},
  also valid before any projection; approximate endpoints inside $I$ against the
  integrable time kernel and locally finite spatial covariance. For such a row
  $k$, exact orthogonality gives
  \[
    \langle Q_{\mathrm{full}}T_{\delta,m}k,h\rangle_{H_W}
    =\langle T_{\delta,m}k,i_IQ_{J,K}i_I^*h\rangle_{H_W}.
  \]
  Directions perpendicular to $H_I$ disappear from this pairing. Apply the
  completed $J = 0$ identity \eqref{E:cameron-actual-covariance} to
  $Q_{J,K}i_I^*h$, and restrict its input and output to $E$. The free $F$ group
  preserves $E$ and restricts to $S_E$; the mean restricts to $\mathcal{T}_R$.
  This proves~\eqref{E:uniform-actual-covariance}. Its amplitude is exactly
  $g_\delta\delta^{a/2} = \sqrt\delta$, on the noise $W$. Arbitrary projected
  controls are covered by the completed mean identity; no smoothness or finite
  rank of $P_K$ is required. Compact operator values and strong continuity of
  $S_E$ and its adjoint make conjugation norm continuous on compact operators,
  as in Proposition~\ref{P:cameron-actual-embedding}. Thus, the integral is
  Bochner measurable. Fixed-parameter $L^2$ covariance continuity, dense states
  and rational times identify the entire operator path.

  For explicit free constants, the Bessel-unitary Fourier matrix of $S_E(v)$ is
  the matrix displayed in that proposition. For $|v| \le B$, $B \ge 1$, the
  upper off-diagonal entry is at most $\sqrt2 B$: use $|\sin(vk)| \le Bk$ for $k
  \le 1$ and $\langle k\rangle/k \le \sqrt2$ for $k \ge 1$. The lower entry is
  at most one, as are both diagonal entries; at zero the sine quotient has value
  $v$. Its Frobenius bound is $\sqrt{3+2B^2} \le 3B$. Hence, $G_{B,E} =
  \sup_{|v|\le B}\|S_E(v)\| \le 3B$. With
  \[
    \mathcal{M}_h(t)=\mathcal{T}_R(Q_{J,K}i_I^*h)(t),\qquad
    t_m=\|\mathcal{M}_h\|_{L^r(I_m;\mathcal{L}(E))},
  \]
  the primitive increment bound and the affine time change give
  \[
    \|(Lh)_m\|_{C^\theta}
    \le\sqrt\delta\,C_{B,E}\delta^{-1/r}t_m,\qquad
    C_{B,E}=G_{B,E}^2[B^{1-1/r}+B^{1-1/r-\theta}]
    \le18B^{3-1/r}.
  \]
  Here $1-1/r > \theta$, constant stopping preserves the bound, and the empty
  block contributes zero.

  Put $n = N+1 \ge 2$. The finite power-mean inequality at exponent $r/p > 1$
  with nonnegative inputs $t_m^p$ gives
  \[
    \left(\sum_m t_m^p\right)^{r/p}
    \le n^{r/p-1}\sum_m t_m^r,\qquad
    \left(\sum_m t_m^p\right)^{1/p}
    \le n^{1/p-1/r}\left(\sum_m t_m^r\right)^{1/r}.
  \]
  Raising nonnegative terms uses only positive powers, including when $t_m = 0$.
  The true partition gives $\sum_m t_m^r = \|\mathcal{M}_h\|_{L^r(I)}^r$ and
  $\delta n \le 2/B$. The exact $\delta$ weight therefore yields
  \[
    \|Lh\|_{\delta,p}
    \le\sqrt\delta\,C_{B,E}(\delta n)^{1/p-1/r}\|\mathcal{M}_h\|_{L^r}
    \le\sqrt\delta\,C_{B,E}(2/B)^{1/p-1/r}
    [M_0\varepsilon_J+M_fc_tK^{-\beta}]\|h\|_{H_W}.
  \]

  Finally, let $H_{\mathrm{seen}}$ be the closed span in $H_W$ of all
  coefficient tests $k_\ell$ with $\ell(X) = W(k_\ell)$. The first Wiener chaos
  Cameron identification in Proposition~\ref{P:cameron-actual-embedding} applies
  on $H_W$: $L$ kills $H_{\mathrm{seen}}^\perp$ and its Cameron space is
  $L(H_{\mathrm{seen}})$ with minimum norm of a preimage in the original
  covariance space. Thus, $\|Lh\|_{\mathrm{CM}} = \|P_{\mathrm{seen}}h\|_{H_W}$.
  Applying the bound to this minimum-norm representative gives precisely the
  Cameron embedding bound, without an inverse covariance or finite-dimensional
  density. This proves Proposition~\ref{P:uniform-actual-cameron}.
\end{proof}

The orthogonal $P_{\mathrm{full}}$ and $Q_{\mathrm{full}}$ Gaussian fields are
independent; residual time blocks can remain correlated. The retained spatial
head may have infinite rank. Heat filtering alone would not give this orthogonal
decomposition. These linear bounds do not bound the expected residual path norm
uniformly on growing masks, nor perform conditioning or integration over the
infinite sum. The remaining stochastic, product and pointwise upper estimates
require their own proofs.
