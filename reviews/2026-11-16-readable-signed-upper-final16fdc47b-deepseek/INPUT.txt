\section{An upper bound from a signed variational expression}
\label{S:original-signed-score-upper}

Keep all auxiliary choices fixed as in the Gaussian tube construction, including
$\alpha/4 < s_{\rm sp} < 1/2$, the admissible $\theta,\nu,r,p$, and the three
fixed even-$Q$ conditions. The mean bound $C_{\rm mean}$ is independent of
$R,B,\delta$; it may depend on those fixed auxiliary parameters and the model.
The space $H_I$ has the original covariance $\Gamma$. Its norm is not reweighted
by $g$ and is not a sum of fast block costs. Let $R_\delta = \delta^{-1}+2$ and
$M_\delta = \E\|P_\delta\|_{\mathcal{L}(E)}^2$. For every fixed integer $B$ the
Gaussian block tuple has typical radius $b_{\delta,B} =
2\E\|X_{\delta,B}\|_{\delta,p} \to 0$. Its endpoint is the same $P_\delta$ for
every $B$ by Lemma~\ref{L:tube-actual-endpoint}, not a $B$-dependent surrogate.
Write $\mathcal{M}_{\delta,B,R}$ for the normalized mean tuple
in~\eqref{E:tube-actual-mean}, with the indicated block length and radius. Write
$\mathcal{F}_{\delta,B}$ for the everywhere-defined endpoint functional
$F_\delta$ of Lemma~\ref{L:tube-actual-endpoint}.

\begin{lemma}[Signed variational expression and quantitative radial envelope]
  \label{L:signed-score-envelope}
  For $R \ge 1$ define the variational expression
  \begin{equation}\label{E:signed-raw-score}
    \mathcal{S}_R=\sup_{\Phi\in H_I}
    \{2\operatorname{Rate}_R(\Phi)-\|\Phi\|_{H_I}^2/2\},
    \qquad V_{\rm up}=\limsup_{\delta\downarrow0}\mathcal{S}_{R_\delta}.
  \end{equation}
  Then $0 \le \mathcal{S}_R,V_{\rm up} \le 2+2C_{\rm mean}^2 < \infty$. The tube
  envelope $G_{\delta,B}$ of Proposition~\ref{P:tube-coarse-envelope} satisfies,
  uniformly in $R \ge 1$ (in particular at the stochastic radius $R_\delta$),
  \begin{equation}\label{E:signed-polynomial-envelope}
    G_{\delta,B}(\rho)\le A_0+A_1B^{3\nu}\rho^\nu
    +\eta_B b_{\delta,B}^\nu ,
  \end{equation}
  for $0 < \delta < \min(1,1/B)$ and every $\rho \ge 0$, where
  \[
    \begin{aligned}
      A_0&=4C_{\rm gr},\\
      A_1&=2^{\nu+1-\nu/r}C_{\rm gr}32^\nu C_{\rm mean}^\nu,\\
      \eta_B&=2^{\nu+1-\nu/p}C_{\rm gr}B^{\nu/p}.
    \end{aligned}
  \]
  The variational expression is not identified here with a constant in the cited
  source or a variational value over nonnegative controls.
\end{lemma}

\begin{proof}
  The mean is linear, so $\mathcal{T}_R0 = 0$. Its free form has nonnegative
  gradient energy and spectral bottom zero; thus $\operatorname{Rate}_R(0) = 0$.
  Proposition~\ref{P:uniform-rate-averaging} gives $\operatorname{Rate}_R(\Phi)
  \le 1+C_{\rm mean}\|\Phi\|$. With $q = \|\Phi\| \ge 0$,
  \[
    2\operatorname{Rate}_R(\Phi)-q^2/2
    \le2+2C_{\rm mean}q-q^2/2
    =2+2C_{\rm mean}^2-(q-2C_{\rm mean})^2/2 .
  \]
  The coarser inequality also follows from half-weight arithmetic--geometric
  mean at bases $q^2,(2C_{\rm mean})^2$, with nonnegative weights and bases,
  including zeros. The zero control gives the lower bound. No optimizer, finite
  rank, positive control or convergence of $\mathcal{S}_{R_\delta}$ is assumed.
  The finite ceiling is auxiliary-parameter dependent and need not be sharp.

  For $\|a\|_{\delta,p} \le b_{\delta,B}$, $\|\Phi\| \le \rho$ set $v_m =
  \|a_m\|_\theta$ and $h_m = \|(\mathcal{M}_{\delta,B,R}\Phi)_m\|_\theta$. The
  quantitative physical bound~\eqref{E:uniform-product-growth}, already
  including the free factor, and convexity give
  \[
    \mathcal{F}_{\delta,B}(a+\mathcal{M}_{\delta,B,R}\Phi)
    \le2C_{\rm gr}B\delta n
    +2^\nu C_{\rm gr}B\delta\sum_m(v_m^\nu+h_m^\nu).
  \]
  Its upper expression is nonnegative, so it controls the maximum at one. The
  true primitive estimate and weighted power mean give
  \begin{align*}
    \delta\sum v_m^\nu&\le(2/B)^{1-\nu/p}b_{\delta,B}^\nu,\\
    \delta\sum h_m^\nu&\le(2/B)^{1-\nu/r}
    (32B^{3-1/r}C_{\rm mean}\rho)^\nu .
  \end{align*}
  The exponents are $p/\nu \ge 1$, $r/\nu > 1$, inputs nonnegative, $n = N+1 \ge
  2$, $\delta > 0$, and $\delta n \le 2/B$. Zeros and $p = \nu$ are legal. The
  first constant is $4C_{\rm gr}$. In the mean term the factors $B^{\nu/r}$ and
  $B^{-\nu/r}$ cancel exactly, leaving $B^{3\nu}$. These computations give the
  displayed coefficients and~\eqref{E:signed-polynomial-envelope}, uniformly for
  all radii. They use no unspecified $c_Y(B)$ or quantitative $F$ estimate. This
  proves Lemma~\ref{L:signed-score-envelope}.
\end{proof}

Lemma~\ref{L:signed-score-envelope} bounds the envelope $G_{\delta,B}$ at every
radius $\rho$. We now split the radial integral
in~\eqref{E:tube-coarse-envelope} at a radius $M$. Below $M$ we compare the
exponent with $\mathcal{S}_{R_\delta}$, and above $M$ we use
Lemma~\ref{L:tube-rayleigh-tail}.

\begin{proposition}[Finite-delta bulk and exact tail]
  \label{P:signed-moment-bulk-tail}
  For fixed integer $B \ge 1$, $0 < \delta < \min(1,1/B)$ and $M > 0$, define
  the deterministic number
  \[
    \mathcal{E}_{\delta,B,M}=
    \max\left(0,\sup_{\substack{R\ge1,\ \|\Phi\|\le M\\
    \|x\|_{\delta,p}\le b_{\delta,B}}}
    [\mathcal{F}_{\delta,B}(x+\mathcal{M}_{\delta,B,R}\Phi)
    -2\operatorname{Rate}_R(\Phi)]\right).
  \]
  It is finite. For every polynomial $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$,
  with $c_{\rm ball} > 0$, $\kappa_{\rm ball} \ge 0$, one has
  \begin{equation}\label{E:signed-error-limit}
    e_B:=\limsup_{\delta\downarrow0}\mathcal{E}_{\delta,B,M_B}
    \ \text{finite and nonnegative},\qquad e_B\longrightarrow0 .
  \end{equation}
  If $4A_1B^{3\nu} \le M^{2-\nu}$, then
  \begin{equation}\label{E:signed-finite-moment}
    M_\delta\le
    \frac{M^2}{2\delta}
    e^{(\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M})/\delta}
    +4e^{(A_0+\eta_Bb_{\delta,B}^\nu-M^2/4)/\delta}.
  \end{equation}
\end{proposition}

\begin{proof}
  The polynomial envelope and nonnegative rate make this supremum finite. It is
  an ordinary deterministic real number, not a random uncountable-radius
  supremum. At fixed $R$ the inner functions are continuous in controls and
  paths; no measurable maximizing selector or joint $\delta,R$ measurability is
  needed. The zero control and zero path have rate zero and
  $\mathcal{F}_{\delta,B}(0) \ge 0$, so the untruncated supremum is already
  nonnegative. The anchor $b_{\delta,B} \to 0$ at fixed $B$ satisfies
  Proposition~\ref{P:uniform-arbitrary-residual} exactly. Its polynomial-radius
  conclusion gives $\limsup_B e_B \le 0$. At fixed $B$, the polynomial envelope
  bounds the inner limsup by a finite number, and $e_B \ge 0$ then
  proves~\eqref{E:signed-error-limit}.

  The exact Gaussian-enlargement bound is
  \[
    M_\delta\le\int_0^\infty\frac{\rho}{\delta}
    e^{(G_{\delta,B}(\rho)-\rho^2/2)/\delta}\ud\rho .
  \]
  For every $\rho \le M$, every base-ball $a$ and $\|\Phi\| \le \rho$,
  \[
    \mathcal{F}_{\delta,B}(a+\mathcal{M}_{\delta,B,R_\delta}\Phi)-\rho^2/2
    \le2\operatorname{Rate}_{R_\delta}(\Phi)-\|\Phi\|^2/2
    +\mathcal{E}_{\delta,B,M}
    \le\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M}.
  \]
  Taking suprema and integrating $\rho/\delta$ over $[0,M]$ gives the first term
  of~\eqref{E:signed-finite-moment}, even if $M^2/(2\delta) < 1$. This used
  $\rho^2 \ge \|\Phi\|^2$, not equality or an optimizer. For the tail apply
  Lemma~\ref{L:tube-rayleigh-tail} with $C = A_1B^{3\nu}$ and intercept
  $A_0+\eta_Bb_{\delta,B}^\nu$. All cutoff conditions hold: $M > 0$, $\rho \ge
  M$, $C \ge 0$, $1 < \nu < 2$, $4C \le M^{2-\nu}$. Its analytic integration
  retains both the front factor four and the exponent $-M^2/4$. Adding bulk and
  tail proves the finite-delta bound. No infinite-dimensional density or Laplace
  principle is used. This proves Proposition~\ref{P:signed-moment-bulk-tail}.
\end{proof}

\begin{proposition}[Two limits on the same moment]
  \label{P:signed-operator-upper}
  The global operator moment obeys
  \[
    0\le\liminf_{\delta\downarrow0}\delta\log M_\delta
    \le\limsup_{\delta\downarrow0}\delta\log M_\delta
    \le V_{\rm up}\le2+2C_{\rm mean}^2 .
  \]
\end{proposition}

\begin{proof}
  Choose any $c_{\rm ball} > 0$ and $\kappa_{\rm ball} > 3\nu/(2-\nu)$, and set
  $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$. The cutoff in
  Proposition~\ref{P:signed-moment-bulk-tail} holds for all sufficiently large
  integers $B$. For $A_1 = 0$ this is immediate; otherwise its side ratio is
  \[
    \frac{M_B^{2-\nu}}{4A_1B^{3\nu}}
    =\frac{c_{\rm ball}^{2-\nu}}{4A_1}
    B^{\kappa_{\rm ball}(2-\nu)-3\nu}\longrightarrow\infty .
  \]
  Both constants and the strictly positive exponent are retained. At such a
  fixed $B$, a positive sum is at most twice its larger term. Taking
  $\delta\log$ of~\eqref{E:signed-finite-moment} therefore gives
  \begin{align*}
    \delta\log M_\delta\le\delta\log2+
    \max\{&
    \mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M_B}
    +\delta\log(M_B^2/(2\delta)),\\
    &A_0+\eta_Bb_{\delta,B}^\nu-M_B^2/4+\delta\log4\}.
  \end{align*}
  At fixed $B$, $b_{\delta,B} \to 0$. All logarithmic normalizers vanish,
  including $\delta\log(M_B^2/(2\delta)) = \delta\log(M_B^2/2)-\delta\log\delta
  \to 0$. The last limit follows from $te^{-t} \le 2/t$ for $t = -\log\delta >
  0$. Finite variational expression and error bounds permit limsup
  subadditivity, giving
  \[
    \limsup_{\delta\downarrow0}\delta\log M_\delta
    \le\max\{V_{\rm up}+e_B,\ A_0-M_B^2/4\}.
  \]
  The left side has no $B$: restart identifies every block product with this
  same $P_\delta$. Now let integer $B \to \infty$. We have $e_B \to 0$, $M_B \to
  \infty$ and finite $V_{\rm up} \ge 0$. This gives the asserted upper bound,
  while $M_\delta \ge 1$ gives the nonnegative lower bound. The order was
  $\delta$ first at fixed block length, ball and thresholds, then $B$; no
  diagonal or limit of $\mathcal{S}_{R_\delta}$ was assumed. This proves
  Proposition~\ref{P:signed-operator-upper}.
\end{proof}

\begin{proposition}[Pointwise upper bound from the signed variational expression]
  \label{P:original-signed-point-upper}
  For every original admissible model and every fixed $x$,
  \[
    \limsup_{T\to\infty}T^{-\kappa}

    \le\limsup_{\delta\downarrow0}\mathcal{S}_{\delta^{-1}+2}

  expression have rate zero.
\end{proposition}

  by $\operatorname{DataPref}_\delta M_\delta$, with
  $\delta\log\operatorname{DataPref}_\delta \to 0$. Its with the regularization
  removed first, full-chain and canonical law proof is the input here, not an
  original $E$ point trace. The decreasing bijection $T_\delta =
  \delta^{-1/\kappa}$ has inverse $\delta = T^{-\kappa}$. Take logs, normalized
  limsup and Proposition~\ref{P:signed-operator-upper}. Original canonical
  spatial stationarity from Proposition~\ref{P:raw-block-original-lower} gives
  every fixed $x$. The moment is at least one by its full expansion and finite
  by the bound, so no logarithm or difference of infinite quantities is hidden.

  For $\Gamma = 0$ set $C_{\rm mean} = 0$ directly. Then $H_I = \{0\}$,

  functional at zero may exceed zero. The formal $a\downarrow0$ exponent is
  $(4-\alpha)/(3-\alpha)$, with no new endpoint-in-$a$ theorem asserted. The
  Jensen and lower bounds remain valid. This proves
  Proposition~\ref{P:original-signed-point-upper}.
\end{proof}

  2$, $\delta > 0$, and $\delta n \le 2/B$. Zeros and $p = \nu$ are legal. The
  first constant is $4C_{\rm gr}$. In the mean term the factors $B^{\nu/r}$ and
  $B^{-\nu/r}$ cancel exactly, leaving $B^{3\nu}$. These computations give the
  displayed coefficients and~\eqref{E:signed-polynomial-envelope}, uniformly for
  all radii. They use no unspecified $c_Y(B)$ or quantitative $F$ estimate. This
  proves Lemma~\ref{L:signed-score-envelope}.
\end{proof}

Lemma~\ref{L:signed-score-envelope} bounds the envelope $G_{\delta,B}$ at every
radius $\rho$. We now split the radial integral
in~\eqref{E:tube-coarse-envelope} at a radius $M$. Below $M$ we compare the
exponent with $\mathcal{S}_{R_\delta}$, and above $M$ we use
Lemma~\ref{L:tube-rayleigh-tail}.

\begin{proposition}[Finite-delta bulk and exact tail]
  \label{P:signed-moment-bulk-tail}
  For fixed integer $B \ge 1$, $0 < \delta < \min(1,1/B)$ and $M > 0$, define
  the deterministic number
  \[
    \mathcal{E}_{\delta,B,M}=
    \max\left(0,\sup_{\substack{R\ge1,\ \|\Phi\|\le M\\
    \|x\|_{\delta,p}\le b_{\delta,B}}}
    [\mathcal{F}_{\delta,B}(x+\mathcal{M}_{\delta,B,R}\Phi)
    -2\operatorname{Rate}_R(\Phi)]\right).
  \]
  It is finite. For every polynomial $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$,
  with $c_{\rm ball} > 0$, $\kappa_{\rm ball} \ge 0$, one has
  \begin{equation}\label{E:signed-error-limit}
    e_B:=\limsup_{\delta\downarrow0}\mathcal{E}_{\delta,B,M_B}
    \ \text{finite and nonnegative},\qquad e_B\longrightarrow0 .
  \end{equation}
  If $4A_1B^{3\nu} \le M^{2-\nu}$, then
  \begin{equation}\label{E:signed-finite-moment}
    M_\delta\le
    \frac{M^2}{2\delta}
    e^{(\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M})/\delta}
    +4e^{(A_0+\eta_Bb_{\delta,B}^\nu-M^2/4)/\delta}.
  \end{equation}
\end{proposition}

\begin{proof}
  We first identify the vanishing deterministic error on polynomial control
  balls. We then split the exact radial integral at $M$: the bulk uses the
  signed variational score, and the tail uses the explicit subquadratic
  envelope.


  \medskip
  \noindent\textit{Step 1.~}We control the deterministic product error.
  The polynomial envelope and nonnegative rate make this supremum finite. It is
  an ordinary deterministic real number, not a random uncountable-radius
  supremum. At fixed $R$ the inner functions are continuous in controls and
  paths; no measurable maximizing selector or joint $\delta,R$ measurability is
  needed. The zero control and zero path have rate zero and
  $\mathcal{F}_{\delta,B}(0) \ge 0$, so the untruncated supremum is already
  nonnegative. The anchor $b_{\delta,B} \to 0$ at fixed $B$ satisfies
  Proposition~\ref{P:uniform-arbitrary-residual} exactly. Its polynomial-radius
  conclusion gives $\limsup_B e_B \le 0$. At fixed $B$, the polynomial envelope
  bounds the inner limsup by a finite number, and $e_B \ge 0$ then
  proves~\eqref{E:signed-error-limit}.


  \medskip
  \noindent\textit{Step 2.~}We split the radial integral and keep both
  prefactors.
  The exact Gaussian-enlargement bound is
  \[
    M_\delta\le\int_0^\infty\frac{\rho}{\delta}
    e^{(G_{\delta,B}(\rho)-\rho^2/2)/\delta}\ud\rho .
  \]
  For every $\rho \le M$, every base-ball $a$ and $\|\Phi\| \le \rho$,
  \[
    \mathcal{F}_{\delta,B}(a+\mathcal{M}_{\delta,B,R_\delta}\Phi)-\rho^2/2
    \le2\operatorname{Rate}_{R_\delta}(\Phi)-\|\Phi\|^2/2
    +\mathcal{E}_{\delta,B,M}
    \le\mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M}.
  \]

  Taking suprema and integrating $\rho/\delta$ over $[0,M]$ gives the first term
  of~\eqref{E:signed-finite-moment}, even if $M^2/(2\delta) < 1$. This used
  $\rho^2 \ge \|\Phi\|^2$, not equality or an optimizer. For the tail apply
  Lemma~\ref{L:tube-rayleigh-tail} with $C = A_1B^{3\nu}$ and intercept
  $A_0+\eta_Bb_{\delta,B}^\nu$. All cutoff conditions hold: $M > 0$, $\rho \ge
  M$, $C \ge 0$, $1 < \nu < 2$, $4C \le M^{2-\nu}$. Its analytic integration
  retains both the front factor four and the exponent $-M^2/4$. Adding bulk and
  tail proves the finite-delta bound. No infinite-dimensional density or Laplace
  principle is used. This proves Proposition~\ref{P:signed-moment-bulk-tail}.
\end{proof}

\begin{proposition}[Two limits on the same moment]
  \label{P:signed-operator-upper}
  The global operator moment obeys
  \[
    0\le\liminf_{\delta\downarrow0}\delta\log M_\delta
    \le\limsup_{\delta\downarrow0}\delta\log M_\delta
    \le V_{\rm up}\le2+2C_{\rm mean}^2 .
  \]
\end{proposition}

\begin{proof}
  We choose a polynomial cutoff large enough for the quadratic tail to
  dominate. We first take the small-parameter limit at fixed block length, and
  only then send the block length to infinity, keeping the same endpoint
  moment on the left.


  \medskip
  \noindent\textit{Step 1.~}We choose a polynomial tail cutoff and bound the
  logarithm of the sum.
  Choose any $c_{\rm ball} > 0$ and $\kappa_{\rm ball} > 3\nu/(2-\nu)$, and set
  $M_B = c_{\rm ball}B^{\kappa_{\rm ball}}$. The cutoff in
  Proposition~\ref{P:signed-moment-bulk-tail} holds for all sufficiently large
  integers $B$. For $A_1 = 0$ this is immediate; otherwise its side ratio is
  \[
    \frac{M_B^{2-\nu}}{4A_1B^{3\nu}}
    =\frac{c_{\rm ball}^{2-\nu}}{4A_1}
    B^{\kappa_{\rm ball}(2-\nu)-3\nu}\longrightarrow\infty .
  \]
  Both constants and the strictly positive exponent are retained. At such a
  fixed $B$, a positive sum is at most twice its larger term. Taking
  $\delta\log$ of~\eqref{E:signed-finite-moment} therefore gives
  \begin{align*}
    \delta\log M_\delta\le\delta\log2+
    \max\{&
    \mathcal{S}_{R_\delta}+\mathcal{E}_{\delta,B,M_B}
    +\delta\log(M_B^2/(2\delta)),\\
    &A_0+\eta_Bb_{\delta,B}^\nu-M_B^2/4+\delta\log4\}.
  \end{align*}

  \medskip
  \noindent\textit{Step 2.~}We take the small-parameter limit before the
  block-length limit.
  At fixed $B$, $b_{\delta,B} \to 0$. All logarithmic normalizers vanish,
  including $\delta\log(M_B^2/(2\delta)) = \delta\log(M_B^2/2)-\delta\log\delta
  \to 0$. The last limit follows from $te^{-t} \le 2/t$ for $t = -\log\delta >
  0$. Finite variational expression and error bounds permit limsup
  subadditivity, giving
  \[
    \limsup_{\delta\downarrow0}\delta\log M_\delta
    \le\max\{V_{\rm up}+e_B,\ A_0-M_B^2/4\}.
  \]

  The lower bound $M_\delta\ge1$ used here is~\eqref{E:point-fixed-finiteness}
  of Lemma~\ref{L:point-actual-finiteness}.
  The left side has no $B$: restart identifies every block product with this
  same $P_\delta$. Now let integer $B \to \infty$. We have $e_B \to 0$, $M_B \to
  \infty$ and finite $V_{\rm up} \ge 0$. This gives the asserted upper bound,
  while $M_\delta \ge 1$ gives the nonnegative lower bound. The order was
  $\delta$ first at fixed block length, ball and thresholds, then $B$; no
  diagonal or limit of $\mathcal{S}_{R_\delta}$ was assumed. This proves
  Proposition~\ref{P:signed-operator-upper}.
\end{proof}

\begin{proposition}[Pointwise upper bound from the signed variational expression]
  \label{P:original-signed-point-upper}
  For every original admissible model and every fixed $x$,
  \[
    \limsup_{T\to\infty}T^{-\kappa}
    \log\E|u_\Gamma(T,x)|^2
    \le\limsup_{\delta\downarrow0}\mathcal{S}_{\delta^{-1}+2}
    \le2+2C_{\rm mean}^2 .
  \]
  For zero covariance the two normalized moments and the signed variational
  expression have rate zero.
\end{proposition}

\begin{proof}
  Proposition~\ref{P:original-point-transfer} bounds the original point moment
  by $\operatorname{DataPref}_\delta M_\delta$, with
  $\delta\log\operatorname{DataPref}_\delta \to 0$. Its proof, which removes
  the regularization first, keeps the full chain and works with the canonical
  law, is the input here, not an original $E$ point trace. The decreasing
  bijection $T_\delta =
  \delta^{-1/\kappa}$ has inverse $\delta = T^{-\kappa}$. Take logs, normalized
  limsup and Proposition~\ref{P:signed-operator-upper}. Original canonical
  spatial stationarity from Proposition~\ref{P:raw-block-original-lower} gives
  every fixed $x$. The moment is at least one by its full expansion and finite
  by the bound, so no logarithm or difference of infinite quantities is hidden.

  For $\Gamma = 0$ set $C_{\rm mean} = 0$ directly. Then $H_I = \{0\}$,
  $\operatorname{Rate}_R = \mathcal{S}_R = V_{\rm up} = 0$, $u = 1$ and
  $P_\delta = S_E(\delta^{-1})$. The free lower and upper bounds give $0 \le
  \delta\log M_\delta \le 2\delta\log[C(1+\delta^{-1})] \to 0$. This direct
  proof needs no nonzero Gaussian or nondegenerate Cameron assumption; the free
  functional at zero may exceed zero. The formal $a\downarrow0$ exponent is
  $(4-\alpha)/(3-\alpha)$, with no new endpoint-in-$a$ theorem asserted. The
  Jensen and lower bounds remain valid. This proves
  Proposition~\ref{P:original-signed-point-upper}.
\end{proof}

The varying variational expression remains a signed variational value for the
original covariance. Its positive-control comparison, exact limit and
identification of the derived constant require the subsequent variational
arguments.

\section{Model and standing assumptions}\label{S:model}

We study the hyperbolic Anderson model
\cite{chen:author2026:time-dependency}*{(1.1)}
\begin{equation}\label{E:HAM}
  \frac{\partial^2 u}{\partial t^2}(t,x)=\Delta u(t,x)+\W(t,x)u(t,x),
  \qquad u(0,x)=1,\quad \frac{\partial u}{\partial t}(0,x)=0,
\end{equation}
for $(t,x) \in \R_+\times\R^d$, $d \in \{1,2,3\}$, interpreted in the
Stratonovich sense (limit of classical wave equations driven by smoothed noises
$\W_{\epsilon,\delta}$). The noise is centered Gaussian with
\begin{equation}\label{E:cov}
  \operatorname{Cov}\big(\W(s,x),\W(t,y)\big)=|s-t|^{-\alpha_0}\gamma(x-y),
  \qquad 0<\alpha_0<1,
\end{equation}
where $\gamma \ge 0$ is nonnegative definite with spectral measure $\mu$.

\begin{hypothesis}[Homogeneous covariance]\label{H:homog}
  Let $d \in \{1,2,3\}$ and $0 < \alpha_0 < 1$. The spatial covariance is a
  nonnegative, even, positive-definite tempered Radon measure $\Gamma$ on
  $\R^d$, with positive even tempered Radon spectral measure $\mu$ in the
  convention
  \[
    \Gamma=\int e^{i\xi\cdot x}\mu(\ud\xi)
    \quad\text{in the sense of tempered distributions}.
  \]
  The symbol $\gamma$ in~\eqref{E:cov} denotes this covariance as a function
  when one exists, and as a distribution otherwise. For every Borel set $A$ and
  $c > 0$, $\Gamma(cA) = c^{d-\alpha}\Gamma(A)$, and
  \begin{equation}\label{E:threshold}
    \alpha_0+\alpha<2,
  \end{equation}
  with $0 < \alpha < d$, or $\alpha = d = 1$. In particular, the last case
  includes spatial white noise $\Gamma = \delta_0$.
\end{hypothesis}

\paragraph{Zero covariance.}
Hypothesis~\ref{H:homog} allows $\Gamma = 0$ by design. In this case the
centered Gaussian noise is $\W = 0$, and the solution with initial data
$(1,0)$ is $u(t,x) = 1$. Thus $\E|u(t,x)|^p = 1$ for every $p > 0$, and
all normalized logarithmic moment rates are zero. Results asserting strict
positivity of covariance-dependent quantities impose $\Gamma \ne 0$ locally.

\section{Gaussian enlargement and the Rayleigh comparison}
\label{S:actual-gaussian-tubes}

We first state the Gaussian geometry with its exact covariance scale. Let $H$ be
real separable Hilbert, $W$ its given isonormal process, and $X$ a strongly
measurable centered Gaussian variable in a real separable Banach space
$\mathscr{B}$, with $\E\|X\|^2 < \infty$. Assume every continuous linear
coordinate has the form $\ell(X) = W(k_\ell)$, $k_\ell \in H$. Fix $0 < \delta <
1$ and write
\begin{equation}\label{E:tube-covariance-scale}
  Ch=\E[XW(h)]=\sqrt\delta\,\mathcal{M}_\delta h,\qquad
  H_{\rm seen}=\overline{\operatorname{span}\{k_\ell\}},\qquad
  E_0=\overline{C(H)} .
\end{equation}
Assume $\mathcal{M}_\delta:H \to \mathscr{B}$ is compact. All Hilbert balls
below are closed.

\subsection{The Wiener-space realization}

The Gaussian isoperimetric theorem that we import in
Section~\ref{S:gaussian-isoperimetry-input} is stated for an abstract Wiener
space. We first show that the law of $X$ on $E_0$ is of this kind, and we
identify its Cameron space.

\begin{proposition}[First Wiener chaos realization on the support]
  \label{P:tube-wiener-realization}
  The variable $X$ lies in $E_0$ almost surely, with a centered Radon Gaussian
  law of full support there. Its Cameron space and norm are
  \begin{equation}\label{E:tube-cameron-space}
    H_{\rm CM}=C(H_{\rm seen}),\qquad
    \|Ch\|_{\rm CM}=\|P_{\rm seen}h\|_H .
  \end{equation}
  This support triple is an abstract Wiener realization, including the
  measurable-norm condition.
\end{proposition}

\begin{proof}
  The zero law has $C = 0$ and $E_0 = \{0\}$ and is immediate. A countable
  norming family exists in $\mathscr{B}^*$: choose a dense sequence on the unit
  sphere and apply Hahn--Banach at each point. Approximation gives $\|x\| =
  \sup_j|\ell_j(x)|$. Norms of differences from a countable dense set show that
  these coordinates generate the Borel sets. Consequently, $X$ is measurable in
  the completion of $\sigma(W(H))$. Fix an orthonormal basis $(e_j)$ of $H$ and
  set $\mathcal{G}_j = \sigma(W(e_1),\ldots,W(e_j))$. In finite dimension these
  sums and projections stabilize at the dimension. Bochner conditional
  expectations need no Radon--Nikodym property here: define them first on
  finite-valued simple variables using scalar conditional expectations, then
  extend by the $L^2$ norm contraction. Simple approximation supplies their
  scalar testing identities. Gaussian regression and the countable norming
  family give
  \begin{equation}\label{E:tube-finite-regression}
    X_j=\E[X\mid\mathcal{G}_j]
    =\sum_{l\le j}W(e_l)Ce_l,\qquad X_j\longrightarrow X
    \quad\hbox{in }L^2(\mathscr{B}).
  \end{equation}
  To justify convergence, finite-coordinate scalar events approximate all events
  in the generated sigma field: the class of events whose indicators are
  approximable in $L^2$ is a monotone class containing the finite-coordinate
  algebra. Approximate a Bochner variable by simple variables and use the
  conditional $L^2$ contraction.

  Bounded testing gives $\ell(Ch) = \langle k_\ell,h\rangle$. Thus, $\ker C =
  H_{\rm seen}^\perp$, and $C$ is injective on $H_{\rm seen}$. The closed first
  Wiener chaos generated by $X$ is exactly $W(H_{\rm seen})$. Its covariance
  image therefore has the first Wiener chaos norm
  in~\eqref{E:tube-cameron-space}; unused directions are removed by the
  orthogonal projection, without an inverse covariance. This is the Cameron
  image under the first Wiener chaos definition. The finite sums
  in~\eqref{E:tube-finite-regression} lie in $E_0$, so $L^2$ convergence and an
  almost-sure subsequence put $X$ in that closed space almost surely.
  Functionals on $E_0$ extend to $\mathscr{B}$ by Hahn--Banach, so restriction
  preserves the scalar laws and first chaos.

  Its law is tight. For each radius $2^{-j}$ choose a finite union of balls of
  mass greater than $1-\varepsilon2^{-j-1}$, using a countable cover of the
  separable space. The intersection of the corresponding finite closed unions is
  closed and totally bounded and has mass at least $1-\varepsilon$. Completeness
  makes it compact. Metric Borel regularity on these compact sets, followed by
  tightness, gives a Radon law. To prove full support, the tail $X-X_j$ is
  independent of the finite Gaussian sum: every tail coordinate has label
  $(I-P_j)k_\ell$, orthogonal to $e_1,\ldots,e_j$. Joint Gaussianity and
  countable Borel coordinates give Banach tail independence. Given $h$ and
  $\varepsilon > 0$, choose $j$ with $\|CP_jh-Ch\| < \varepsilon/3$ and
  $\mathbb{P}(\|X-X_j\| < \varepsilon/3) > 0$. The finite sum has positive
  probability in every ball about $CP_jh$, even if its covariance is singular:
  the preimage under its finite linear map contains an open ball about the
  coefficient vector of $P_jh$. Independence proves positivity of every ball
  about $Ch$. Closure proves that the support is all of $E_0$.

  We check the measurable norm explicitly. In $H_{\rm seen}$ put $V_j =
  \operatorname{span}\{P_{\rm seen}e_l:l\le j\}$. Any finite orthonormal family
  $(h_k)$ in $V_j^\perp$ is also orthogonal to $e_1,\ldots,e_j$. The same
  regression and Bochner Jensen argument gives
  \[
    \E[X-X_j\mid W(h_k),\hbox{ all }k]
    =\sum_k W(h_k)Ch_k,\qquad
    \E\left\|\sum_kW(h_k)Ch_k\right\|^2
    \le\E\|X-X_j\|^2 .
  \]
  For any $\varepsilon > 0$ choose $j$ so that the right side is less than
  $\varepsilon^3$. Markov bounds the probability that this finite sum has norm
  exceeding $\varepsilon$ by $\varepsilon$, uniformly over every such finite
  orthonormal family. Since $C:H_{\rm seen} \to H_{\rm CM}$ is isometric, the
  excluded finite-dimensional Cameron subspace is $C(V_j)$. This is the
  measurable-norm tail condition for the dense Hilbert image, whose Banach
  completion is $E_0$. The characteristic functional of the law is $\E
  e^{i\ell(X)} = \exp(-\|k_\ell\|_H^2/2)$. These facts give the stated abstract
  Wiener realization. The zero space has the direct zero-law interpretation.
  This proves Proposition~\ref{P:tube-wiener-realization}.
\end{proof}

\subsection{Closed Gaussian enlargements and the quadratic cost}
\label{S:gaussian-isoperimetry-input}

The Gaussian isoperimetric theorem is due to Borell and, independently,
Sudakov--Tsirelson~\cites{borell:75:brunn-minkowski, sudakov.cirel-son:74:extremal}.
We use its set-enlargement form from
Friz and Hairer~\cite{friz.hairer:20:course}*{Theorem~11.6, p.~190}, with footnote~3 on
p.~190; p.~191 supplies the following context. For an abstract Wiener space, a
Borel $A$ of positive mass and its Cameron unit ball $\mathcal{K}$, it states
\[
  \mu((A+t\mathcal{K})^c)
  \le\overline{\Phi}_{\rm N}(\widehat{a}+t),\qquad
  \widehat{a}=\Phi_{\rm N}^{-1}(\mu(A)),\quad t\ge0 .
\]
Here $\Phi_{\rm N}$ is the standard normal distribution function; the same page
gives $\overline{\Phi}_{\rm N}(y) \le e^{-y^2/2}$ for $y \ge 0$. This theorem is
imported, not proved here. It concerns Gaussian input sets, not Gaussianity of a
nonlinear wave. We verify Borel measurability of every enlargement we use.

\begin{lemma}[Compact images and attained tube radii]
  \label{L:tube-closed-geometry}
  Suppose the law of $X$ is nonzero and put
  \[
    b_\delta=2\E\|X\|,\qquad
    A_\delta=\{x\in E_0:\|x\|\le b_\delta\},\qquad
    \mathcal{T}_r=A_\delta+\mathcal{M}_\delta(\overline{B}_H(r)).
  \]
  Every $\mathcal{T}_r$ is closed. The extended distance $d_\delta(x) =
  \inf\{r\ge0:x\in\mathcal{T}_r\}$ satisfies
  \begin{equation}\label{E:tube-attained-radius}
    \{d_\delta\le r\}=\mathcal{T}_r,\qquad
    \mathbb{P}(d_\delta(X)>r)\le e^{-r^2/(2\delta)},\quad r\ge0 .
  \end{equation}
  In particular, $d_\delta$ is lower semicontinuous and finite almost surely.
\end{lemma}

\begin{proof}
  A compact map from a separable Hilbert space sends a closed bounded ball to a
  compact set, including closedness of the image. Indeed, diagonal orthonormal
  coordinates and Fatou give a weakly convergent subsequence of any bounded
  sequence, with limit still in the ball. Compactness gives norm-convergent
  image subsequences, and bounded scalar testing identifies every limit with the
  image of that weak limit. This proves sequential compactness. Closed
  $A_\delta$ plus a compact image is closed: along a converging sequence of
  sums, extract the compact component and pass the residual into the closed base
  set.

  Membership in $\mathcal{T}_r$ implies $d_\delta \le r$. Conversely, if
  $d_\delta(x) \le r$, choose $h_j$ with $\|h_j\| \le r+1/j$ and
  $x-\mathcal{M}_\delta h_j \in A_\delta$. A weak subsequence has limit $h$ of
  norm at most $r$; compactness gives $\mathcal{M}_\delta h_j \to
  \mathcal{M}_\delta h$. Closedness of the residual proves membership in
  $\mathcal{T}_r$. This also covers $r = 0$ and proves least finite-radius
  attainment. Closed sublevels give lower semicontinuity and Borel
  measurability.

  The nonzero law has $b_\delta > 0$, and Markov gives $\mu(A_\delta) \ge 1/2$.
  Projection onto $H_{\rm seen}$ removes all unused directions in the original
  covariance space, so~\eqref{E:tube-cameron-space} gives the exact set
  identities
  \[
    \mathcal{K}=C(\overline{B}_H(1))
    =\sqrt\delta\,\mathcal{M}_\delta(\overline{B}_H(1)),\qquad
    \mathcal{T}_r=A_\delta+(r/\sqrt\delta)\mathcal{K} .
  \]
  These are Borel enlargements by the preceding closedness. The imported theorem
  applies to the realization in Proposition~\ref{P:tube-wiener-realization}.
  Since $\widehat{a} \ge 0$, it yields~\eqref{E:tube-attained-radius}. The value
  $\widehat{a} = +\infty$ is allowed when $\mu(A_\delta) = 1$. The tail tends to
  zero, proving almost-sure finiteness. The Cameron embedding does not change
  the factor one half in the Gaussian cost. This proves
  Lemma~\ref{L:tube-closed-geometry}.
\end{proof}

\input{notes/figures/gaussian-enlargement}

Lemma~\ref{L:tube-closed-geometry} bounds the tail of the tube distance
$d_\delta(X)$. We now convert this tail bound into a bound for
$\E e^{F(X)/\delta}$, for a functional $F$ that satisfies $F \le G(r)$ on each
tube $\mathcal{T}_r$.

\LeanMarker{P:tube-rayleigh-transfer}
\begin{proposition}[Comparison with the Rayleigh integral, including jumps and zero law]
  \label{P:tube-rayleigh-transfer}
  Let $F:E_0 \to \R$ be Borel and let $G:[0,\infty) \to \R$ be Borel and
  nondecreasing. If $F(x) \le G(r)$ for every $x \in \mathcal{T}_r$ and every $r
  \ge 0$, then
  \begin{equation}\label{E:tube-rayleigh-transfer}
    \E e^{F(X)/\delta}
    \le\int_0^\infty\frac r\delta
    \exp\left(\frac{G(r)-r^2/2}{\delta}\right)\ud r .
  \end{equation}
  For zero law use $E_0 = \{0\}$, $C = \mathcal{M}_\delta = 0$; the same
  conclusion holds when $F(0) \le G(0)$.
\end{proposition}

\begin{proof}
  The auxiliary Rayleigh variable $\mathcal{R}_\delta$ with density
  $(r/\delta)e^{-r^2/(2\delta)}$, $r > 0$, has mass one and survival function
  $e^{-r^2/(2\delta)}$ by direct integration. It is a scalar stochastic
  dominator, not a norm of a Gaussian variable in the covariance Hilbert space.
  For every level $t \ge 0$, the upper level set of $e^{G(r)/\delta}$ is an
  upper interval, possibly empty or the whole halfline. An open endpoint is
  controlled by~\eqref{E:tube-attained-radius}. At a positive closed endpoint
  $c$, let $s\uparrow c$:
  \[
    \mathbb{P}(d_\delta(X)\ge c)
    =\lim_{s\uparrow c}\mathbb{P}(d_\delta(X)>s)
    \le e^{-c^2/(2\delta)}
    =\mathbb{P}(\mathcal{R}_\delta\ge c).
  \]
  Rayleigh has no atoms, and at the closed endpoint zero both probability bounds
  are one. On the null event $\{d_\delta(X)=\infty\}$ define
  $e^{G(d_\delta(X))/\delta}$ to be one. No value of $G$ at infinity is needed
  and no expectation changes. Layer cake and Tonelli therefore order the
  expectations of $e^{G(d_\delta(X))/\delta}$ and
  $e^{G(\mathcal{R}_\delta)/\delta}$, even when $G$ jumps or expectations are
  infinite. Least-radius attainment gives $F(X) \le G(d_\delta(X))$ almost
  surely. No measurable selection of a control or residual is needed. This
  proves~\eqref{E:tube-rayleigh-transfer}. For zero law, $X = 0$ and $F(0) \le
  G(0) \le G(r)$; normalization of the Rayleigh density proves the claim
  directly. In particular, $F(0)$ need not be zero. This proves
  Proposition~\ref{P:tube-rayleigh-transfer}.
\end{proof}

\subsection{The wave tuple and its globally defined endpoint}

Retain all original covariances and amplitudes. Use the exponents and fixed even
$Q$ in~\eqref{E:radius-anchor-parameters}--\eqref{E:radius-anchor-Q}; put $\nu =
1/\theta$ and choose $(1-\theta)^{-1} < r_t < 2/a$, $\nu \le p < 2 < r_t$. All
these intervals are nonempty by the strict subcritical assumptions. Fix integer
$B \ge 1$ and the true partition $\delta,N,s_m,\ell_m$ from
Section~\ref{S:actual-cameron-mean}. Set $L_\delta = \delta^{-1}$ and eventually
$R_\delta = L_\delta+2$. On the same given slow $W$ define the full, unprojected
input
\begin{equation}\label{E:tube-actual-input}
  W_m(f)=W(T_{\delta,m}f),\qquad
  X_m(u)=g_\delta M_R(W_m)(\min(u,\ell_m)),\qquad
  g_\delta=\delta^{(1-a)/2}.
\end{equation}
Write $P_{\delta,E}^R$ for the physical $E$ flow driven by $g_\delta M_R(W\circ
T_{\delta,0})$ on the global fast horizon $[0,L_\delta]$. Let
$\mathscr{B}_\delta$ be the finite product of little
$C^\theta([0,B];\mathcal{K}(E))$ spaces of paths starting at zero, with exact
norm
\[
  \|x\|_{\delta,p}=\left(\delta\sum_{m=0}^N
  \|x_m\|_{C^\theta}^p\right)^{1/p}.
\]

\begin{proposition}[Eligibility of the E law]\label{P:tube-actual-law}
  The tuple $X_\delta = (X_m)$ satisfies the preceding Gaussian hypotheses with
  $H = H_I$ and
  \begin{align}
    (\mathcal{M}_\delta\Phi)_m(u)
    &=\int_0^{\min(u,\ell_m)}
    S_E(-v)\mathcal{T}_R\Phi(s_m+\delta v)S_E(v)\ud v,
    \label{E:tube-actual-mean}\\
    \E[X_\delta W(\Phi)]&=\sqrt\delta\,\mathcal{M}_\delta\Phi .
    \label{E:tube-actual-covariance}
  \end{align}
  Its support is $E_{0,\delta} = \overline{C_\delta(H_I)} \subset
  \mathscr{B}_\delta$, where $C_\delta\Phi = \E[X_\delta W(\Phi)] =
  \sqrt\delta\,\mathcal{M}_\delta\Phi$. The map $\mathcal{M}_\delta$ is compact
  at each fixed radius and parameters. At $R = R_\delta$,
  \begin{equation}\label{E:tube-typical-radius}
    b_\delta=2\E\|X_\delta\|_{\delta,p}
    \le C_{Q,B,p}\delta^{(1-a)/2-d/Q}\longrightarrow0 .
  \end{equation}
\end{proposition}

\begin{proof}
  Compact operators on separable Hilbert $E$ form a separable operator-norm
  space: finite-rank rational matrices in an orthonormal basis are dense.
  Polygonal paths give separability of the little H\"older closure. The
  construction is a $C^\theta$ limit of Lipschitz finite coefficient paths
  starting at zero; stopping at $\ell_m$ and extending constantly preserves this
  space and gives zero for an empty final interval. Gaussian transfer retains
  the given $W$ for all finitely many blocks. Every finite collection of linear
  statistics is an $L^2$ limit of joint first Wiener chaos Gaussians, hence
  jointly centered Gaussian. The finite tuple has finite $L^2$ norm by the
  fixed-parameter path moments in Corollary~\ref{C:radius-covariance-residuals}.
  The original Hilbert space $H_I$ is separable by
  Lemma~\ref{L:cameron-supported-tensor}. No separability of all
  $\mathcal{L}(E)$ or full $C^\theta$ is claimed.

  The $E$ mean field $\mathcal{T}_R$ is the compression of
  Proposition~\ref{P:cameron-full-mean}; it is compact into
  $L^{r_t}(I;\mathcal{K}(E))$ at fixed $R$.
  Proposition~\ref{P:uniform-mean-regulator} gives $\|\mathcal{T}_R\| \le C_{\rm
  mean} = M_0\|C_t\|$ independently of $R \ge 1$. Compact conjugation, affine
  time pullback and $1-1/r_t > \theta$ make the primitive
  in~\eqref{E:tube-actual-mean} bounded from that field space to
  $\mathscr{B}_\delta$, with constant stopping. Composition proves compactness
  of $\mathcal{M}_\delta$. There is no uniform-radius compactness assertion.

  The rank-zero identity~\eqref{E:cameron-actual-covariance}, compressed to $E$
  as in Proposition~\ref{P:symplectic-block-mean},
  gives~\eqref{E:tube-actual-covariance}. Its scale is exactly
  $g_\delta\delta^{a/2} = \sqrt\delta$. Every stopped row's slow pullback lies
  in $H_I$ by the inside-$I$ endpoint approximation of
  Lemma~\ref{L:cameron-actual-residual}. Countable matrix coordinates and
  rational times identify the tuple in the closed first Wiener chaos $W(H_I)$,
  so every linear statistic has its label there. Equivalently, apply that lemma
  at $J = 0$ on $F$ and compose any $E$-tuple statistic with the bounded
  $E$-compression. This is same-$W$ covariance, without a block-independence
  assumption or a time-indicator multiplier on the covariance space.
  Proposition~\ref{P:tube-wiener-realization} therefore applies to this exact
  law on its support. Finally, Proposition~\ref{P:radius-mean-anchor}, with zero
  residual projection, gives~\eqref{E:tube-typical-radius}. The third fixed-$Q$
  condition $Q > 2d/(1-a)$ makes this exponent positive; the first two supply
  the path moments and chaining. Fast directions used for the symplectic
  baseline do not replace this global $H_I$ action or get summed as separate
  costs. Zero covariance gives zero operator paths and means directly. This
  proves Proposition~\ref{P:tube-actual-law}.
\end{proof}

\begin{lemma}[The endpoint functional]\label{L:tube-actual-endpoint}
  For every $x \in \mathscr{B}_\delta$ define
  \[
    \begin{aligned}
      P_m(x_m)&=S_E(\ell_m)U_{x_m}(\ell_m,0),\qquad
      \mathcal{P}_\delta(x)=P_N(x_N)\cdots P_0(x_0),\\
      F_\delta(x)&=2\delta\log\max(1,\|\mathcal{P}_\delta(x)\|).
    \end{aligned}
  \]
  This is a finite continuous functional on all tuples. For the input it
  satisfies
  \begin{equation}\label{E:tube-global-endpoint}
    \mathcal{P}_\delta(X_\delta)
    =P_{\delta,E}^R(L_\delta,0),\qquad
    e^{F_\delta(X_\delta)/\delta}
    =\|P_{\delta,E}^R(L_\delta,0)\|^2 .
  \end{equation}
  These are almost-sure identities for each fixed admissible $\delta,R,B$,
  without a simultaneous event over all parameter choices.
\end{lemma}

\begin{proof}
  Generic Young evolution is globally defined and locally Lipschitz in every
  $C^\theta$ operator path. Finite products, fixed-time free factors, norm and
  $\log\max(1,\cdot)$ preserve continuity. Thus, $F_\delta$ is defined even off
  every Gaussian full-measure set. Every $x_m$ in this ambient space is a
  compact-valued $C^\theta$ path starting at zero. The finite-product assertion
  of Lemma~\ref{L:symplectic-free} gives, for every such tuple,
  \[
    \|\mathcal{P}_\delta(x)\|
    \ge\left\|S_E\Big(\sum_{m=0}^N\ell_m\Big)\right\|\ge1 .
  \]

  Lemma~\ref{L:raw-block-restart}, in its case without the deterministic mean,
  applies to this same full input with base physical profile $\eta^2$. To check
  the exact objects, use its single regularized slow noise preserving the given
  noise and pull it back by $T_{\delta,0}$ and $T_{\delta,m}$. At fixed $\delta$
  the global fast source random field is $\delta^{a/2}Y^\rho(\delta u,x)$ and
  its local restriction is its time shift by $mB$. Their ordinary Duhamel flows
  have the same massless free groups, amplitude $g_\delta$ and mask $m_R$.
  Source limits are taken on the fixed finite extended horizon $(N+1)B$ and on
  each local $B$ before composing the finitely many factors. The original
  cocycle from that lemma therefore gives the ordered physical $F$ product with
  the true remainder. The full operator paths and free groups intertwine through
  the closed isometry $i:E \to F$, including the source limits. Restricting the
  product and using injectivity of $i$ gives the first identity
  in~\eqref{E:tube-global-endpoint}. Both fast velocity and displacement remain
  in each factor. No mask was commuted through a free group, and the time blocks
  remain correlated on the same $W$. For zero covariance every input is zero and
  the ordered factors are $S_E(\ell_m)$, whose product is $S_E(L_\delta)$.

  The lower bound already established for every tuple removes the maximum at
  $X_\delta$. Together with the first identity, this proves the second identity
  in~\eqref{E:tube-global-endpoint}. No fixed-radius Gaussian-operator rate or
  source-to-point result is an input to this restart argument. This proves
  Lemma~\ref{L:tube-actual-endpoint}.
\end{proof}

\subsection{Uniform coarse envelope and the exact radial integral}

To apply Proposition~\ref{P:tube-rayleigh-transfer} to the endpoint functional
$F_\delta$ of Lemma~\ref{L:tube-actual-endpoint}, we need a nondecreasing Borel
function that dominates $F_\delta$ on every tube. We take the supremum of
$F_\delta$ over each tube and bound its growth in the radius, uniformly in
$\delta$ and $R \ge 1$.

\begin{proposition}[Coarse tube envelope and operator moment]
  \label{P:tube-coarse-envelope}
  On the support $E_{0,\delta}$ let $A_\delta$ be its closed ball of radius
  $b_\delta$ and put
  \[
    G_\delta(r)=
    \sup_{\substack{a\in A_\delta\\\|\Phi\|_{H_I}\le r}}
    F_\delta(a+\mathcal{M}_\delta\Phi).
  \]
  This is finite, nondecreasing and Borel. There is $C_{\rm env} < \infty$,
  independent of $\delta$ and $R \ge 1$, such that
  \begin{equation}\label{E:tube-coarse-envelope}
    G_\delta(r)\le C_{\rm env}(1+b_\delta^\nu+r^\nu),\qquad
    \E\|P_{\delta,E}^R(L_\delta,0)\|^2
    \le\int_0^\infty\frac r\delta
    e^{(G_\delta(r)-r^2/2)/\delta}\ud r .
  \end{equation}
\end{proposition}

\begin{proof}
  Put $G_B = \sup_{|v|\le B}\|S_E(v)\|$ and $C_{\rm pr} =
  G_B^2[B^{1-1/r_t}+B^{1-1/r_t-\theta}]$. The lower bound $G_B \ge 1$ follows
  directly from $S_E(0) = I$. Proposition~\ref{P:uniform-actual-cameron} gives
  $1 \le G_B \le 3B$, $C_{\rm pr} \le 18B^{3-1/r_t}$. Its primitive estimate,
  now without the covariance factor $\sqrt\delta$, gives
  \begin{equation}\label{E:tube-primitive-sum}
    \delta\sum_m h_m^{r_t}
    \le(C_{\rm pr}C_{\rm mean})^{r_t}\|\Phi\|_{H_I}^{r_t},
    \qquad h_m=\|(\mathcal{M}_\delta\Phi)_m\|_{C^\theta}.
  \end{equation}
  Indeed, each primitive norm is at most $C_{\rm pr}\delta^{-1/r_t}
  \|\mathcal{T}_R\Phi\|_{L^{r_t}(I_m)}$. Raising to $r_t$ cancels the $\delta$
  weight, and the true partition sums to the full field norm, including an empty
  cell.

  The generic Banach Young bound supplies $c_Y = c_Y(B,\theta) > 0$ with
  $\log\|U_d(u,0)\| \le c_Y(1+\|d\|_{C^\theta}^{\nu})$ for $u \le B$. This
  follows from~\eqref{E:young-upper-refined-growth}
  and~\eqref{E:constant-young-growth} by $d \le 1+d^\nu$ and $(1+d)^\nu \le
  2^{\nu-1}(1+d^\nu)$. It applies on $\mathcal{L}(E)$ with multiplication norm
  one, without Gaussian, positivity or Hamiltonian assumptions on $d$. For $a
  \in A_\delta$ put $v_m = \|a_m\|_{C^\theta}$ and $n = N+1$. Convexity and
  submultiplicativity give
  \[
    F_\delta(a+\mathcal{M}_\delta\Phi)
    \le2\delta n(\log G_B+c_Y)
    +2^\nu c_Y\delta\sum_m(v_m^\nu+h_m^\nu).
  \]
  The upper expression is nonnegative, so it also bounds the logarithm with the
  maximum at one. The exact finite weighted power-mean inequalities are
  \begin{align*}
    \delta\sum_m v_m^\nu
    &\le(\delta n)^{1-\nu/p}\|a\|_{\delta,p}^\nu,\\
    \delta\sum_m h_m^\nu
    &\le(\delta n)^{1-\nu/r_t}
    \left(\delta\sum_m h_m^{r_t}\right)^{\nu/r_t}.
  \end{align*}
  They use exponents $p/\nu \ge 1$, $r_t/\nu > 1$ and nonnegative inputs
  $v_m^\nu,h_m^\nu$, followed by positive powers. Zero norms and empty cells
  cause no negative powers; $\delta > 0$ and $n \ge 2$, including the case $p =
  \nu$. Since $\delta n \le 2/B$, these bounds and~\eqref{E:tube-primitive-sum}
  give $F_\delta(a+\mathcal{M}_\delta\Phi) \le D_0+D_1b_\delta^\nu+D_2r^\nu$,
  where
  \begin{align*}
    D_0&=2(2/B)(\log G_B+c_Y),\\
    D_1&=2^\nu c_Y(2/B)^{1-\nu/p},\\
    D_2&=2^\nu c_Y(2/B)^{1-\nu/r_t}
    (C_{\rm pr}C_{\rm mean})^\nu .
  \end{align*}
  Hence, $C_{\rm env} = 1+D_0+D_1+D_2$ is independent of $\delta,R$ and proves
  the coarse bound for every admissible $\delta$, not just when $b_\delta$ is
  small. The supremum sets are nonempty and this bound makes their suprema
  finite. Monotonicity gives Borel measurability without asserting attainment of
  these suprema. The definition bounds $F_\delta$ on every closed tube. Apply
  Proposition~\ref{P:tube-rayleigh-transfer} and~\eqref{E:tube-global-endpoint}
  to obtain the operator-moment bound. For zero law its direct case applies,
  with no assumption that the free endpoint norm is one. This proves
  Proposition~\ref{P:tube-coarse-envelope}.
\end{proof}

\LeanMarker{L:tube-rayleigh-tail}
\begin{lemma}[Exact Rayleigh tail bound]\label{L:tube-rayleigh-tail}
  Suppose $G$ is Borel and $G(r) \le A_0+Cr^\nu$, $C \ge 0$, $1 < \nu < 2$. If
  $M > 0$ and $4C \le M^{2-\nu}$, then
  \begin{equation}\label{E:tube-rayleigh-tail}
    \int_M^\infty\frac r\delta
    e^{(G(r)-r^2/2)/\delta}\ud r
    \le4e^{(A_0-M^2/4)/\delta}.
  \end{equation}
  In particular, the operator-moment bound in~\eqref{E:tube-coarse-envelope} is
  finite at fixed parameters.
\end{lemma}

\begin{proof}
  The finite scalar cutoff inequality gives, for $r \ge M$, $Cr^\nu-r^2/2 \le
  -M^2/8-r^2/8$. Its exact conditions are $M > 0$, $r \ge M$, $C \ge 0$, $1 <
  \nu < 2$ and $4C \le M^{2-\nu}$. Indeed, $r^{2-\nu} \ge M^{2-\nu} \ge 4C$
  gives $Cr^\nu \le r^2/4$, while $r^2 \ge M^2$ splits the remaining $-r^2/4$
  into the two displayed negative terms. Analytically, $-\frac{\ud}{\ud
  r}[4e^{-r^2/(8\delta)}] = (r/\delta)e^{-r^2/(8\delta)}$. Integrating gives the
  front factor $4e^{-M^2/(8\delta)}$; multiplication by $e^{(A_0-M^2/8)/\delta}$
  proves \eqref{E:tube-rayleigh-tail}. Such a cutoff exists since $2-\nu > 0$.
  On $[0,M]$ the growth bound gives $G(r) \le A_0+CM^\nu$, so the integral is
  finite there as well. Apply this to $A_0 = C_{\rm env}(1+b_\delta^\nu)$ and $C
  = C_{\rm env}$ in the preceding proposition. No infinite-dimensional Gaussian
  density is used. This proves Lemma~\ref{L:tube-rayleigh-tail}.
\end{proof}

\LeanMarker{L:tube-half-action}
\begin{lemma}[Supremum with the Gaussian quadratic cost]\label{L:tube-half-action}
  For finite $J:H \to \R$ bounded above on every original ball, let $g_J(r) =
  \sup_{\|h\|\le r}J(h)$. In the extended reals,
  \[
    \sup_{r\ge0}\{g_J(r)-r^2/2\}
    =\sup_{h\in H}\{J(h)-\|h\|_H^2/2\}.
  \]
\end{lemma}

\begin{proof}
  For the lower inequality take $r = \|h\|$. For the upper inequality, every
  $\|h\| \le r$ gives $J(h)-r^2/2 \le J(h)-\|h\|^2/2$. Take first the ball
  supremum and then the radius supremum. Each ball is nonempty with finite
  supremum, so no infinity-minus-infinity expression occurs. No compactness,
  rank or density assumption is needed. This proves
  Lemma~\ref{L:tube-half-action}.
\end{proof}

The Gaussian comparison retains one half of the squared norm of the single
control. The coarse envelope is not the sharp spectral/product envelope. Its
application to the original canonical point moment, and the sharp envelope and
variational comparison, require their separate proofs.

\section{Uniform spectral rates and phase products}
\label{S:uniform-physical-products}

Retain every original covariance and amplitude. Set $E = H^{1/2}\times
H^{-1/2}$, $F = E\oplus\R^2$, and $m_R = \eta(\cdot/R)^2$, $R \ge 1$, with the
same fixed cutoff profile. Choose
\[
  \begin{aligned}
    &\frac12<\theta<\min(1-a/2,(3-a-\alpha)/2),\qquad \nu=1/\theta,\\
    &(1-\theta)^{-1}<r<2/a,\qquad \nu\le p<2<r .
  \end{aligned}
\]
Choose auxiliary exponents $0 < \gamma < \min(1/2,(2-\alpha)/2)$ and $\theta <
\beta < \min(1-a/2,(3-a-\alpha)/2)$ for the phase construction. The original
strict threshold makes these intervals nonempty. Write $i_I:H_I \to H_W$ for the
supported closed inclusion. For nonzero covariance the field is
\[
  \mathcal{T}_R=T_R(C_t\otimes I_{H_{\rm sp}}),\qquad
  \|\mathcal{T}_R\|_{H_I\to L^r(I;\mathcal{K}(E))}
  \le C_{\rm mean}=M_0\|C_t\| ,
\]
by Proposition~\ref{P:uniform-mean-regulator}. For zero covariance set $H_{\rm
sp} = H_I = \{0\}$ and $M_0 = C_{\rm mean} = 0$. Means and their rates are then
zero, and mean flows are free; no nonzero-noise estimate is needed for that
case. The residual comparisons below still act on the nonzero phase $E$.

\subsection{The operator class on E and its spectral constants}

For real smooth compact $q$ let $V_q = m_R(\Gamma*q)$, $M_q:H^{1/2} \to
H^{-1/2}$ be its multiplier and $L_q^E(u,v) = (0,M_qu)$. Let $\mathcal{V}_{E,R}$
be their operator-norm closure in $\mathcal{K}(E)$.

\begin{lemma}[Unique potential and fixed-radius lift]\label{L:uniform-E-lift}
  Each $L \in \mathcal{V}_{E,R}$ has the form $L(u,v) = (0,Mu)$, with $\|L\| =
  \|M\| = :k$, and determines a unique real $V \in H^{-1/2}$ supported in
  $\operatorname{supp}m_R$. The symmetric pairing satisfies
  \begin{equation}\label{E:uniform-E-pairing}
    |v_V(u,w)|\le k\|u\|_{H^{1/2}}\|w\|_{H^{1/2}} .
  \end{equation}
  It has a lift $L_F$ in the exact class of Section~\ref{S:signed-static-wave},
  with $L_Fi = iL$. Lift constants may depend on $R$ and are used only for
  identification. Here $v_V(u,w) = \langle Vu,w\rangle_{H^{-1/2},H^{1/2}}$ on
  real states, with its Hermitian extension on complex states.
\end{lemma}

\begin{proof}
  For zero covariance, $L = M = V = L_F = 0$ gives the claims directly. If
  $L_n^E \to L$, the zero columns persist and $M_n \to M$ in
  $\mathcal{L}(H^{1/2},H^{-1/2})$, with the stated norm equality. At fixed $R$
  choose real compact smooth $\chi_R = 1$ near $\operatorname{supp}m_R$. Since
  $M_n\chi_R = V_n$, $\|V_n-V_j\|_{H^{-1/2}} \le
  \|\chi_R\|_{H^{1/2}}\|M_n-M_j\|$. Thus, $V_n \to V$, with the same support by
  testing outside it. Smooth multiplication on $H^{-1/2}$ identifies $Mu = Vu$
  on compact tests, then on $H^{1/2}$ by density. The identity $M\chi_R = V$
  gives uniqueness; symmetry and~\eqref{E:uniform-E-pairing} pass to the limit.

  The original $F$ operators are $(u,v,c_1,c_2)\mapsto(0,M_nu+c_1V_n,0,0)$.
  Their differences are bounded by $(1+\|\chi_R\|_{H^{1/2}})\|M_n-M_j\|$. They
  therefore converge in the old $F$ class and give the lift. This bound is
  allowed to grow with $R$; it will enter no uniform estimate. Dense smooth
  original controls and bounded completion of $T_R$ also show that all completed
  mean fields lie in $\mathcal{V}_{E,R}$ almost everywhere. Bochner averages
  preserve this closed linear class. Complexification preserves $k$ by the
  Hilbert real/imaginary splitting. The real symmetric pairing becomes Hermitian
  and its diagonal splits into the two real diagonals. This proves
  Lemma~\ref{L:uniform-E-lift}.
\end{proof}

\begin{proposition}[Spectral rate in E]\label{P:uniform-E-spectral}
  For $L \in \mathcal{V}_{E,R}$ put
  \[
    \beta(V)=\inf_{\substack{u\in H^1\\\|u\|_2=1}}
    \{\|\nabla u\|_2^2-v_V(u,u)\},\qquad \lambda(L)=\sqrt{-\beta(V)} .
  \]
  If $K \ge \max(1,k)$, the physical wave satisfies, for $t \ge 0$,
  \begin{equation}\label{E:uniform-E-static-bound}
    -K^2\le\beta(V)=\inf\operatorname{spec}H_V\le0,\qquad
    \|P_V^E(t)\|\le8(1+K)^2(1+t)e^{\lambda(L)t}.
  \end{equation}
  If two such operators have norms at most $K$, then
  \begin{equation}\label{E:uniform-E-rate-continuity}
    |\beta(V)-\beta(W)|\le(K+2)\|L_V-L_W\|,\qquad
    |\lambda(L_V)-\lambda(L_W)|
    \le\sqrt{(K+2)\|L_V-L_W\|}.
  \end{equation}
  Every quantitative constant here uses the $E$ norm alone.
\end{proposition}

\begin{proof}
  We derive the bounds without transferring a quantitative $F$ bound. With $b =
  \|u\|_2$, $z = \|u\|_{H^1}$, Fourier Cauchy--Schwarz gives $\|u\|_{H^{1/2}}^2
  \le bz$. Equation~\eqref{E:uniform-E-pairing} and the half-weight
  arithmetic--geometric mean inequality give $|v_V(u,u)| \le Kbz \le
  z^2/2+K^2b^2/2$. The weights are $1/2,1/2$, bases $z^2,(Kb)^2 \ge 0$; positive
  powers include zero norms. For $q_V = \|\nabla u\|_2^2-v_V(u,u)$ and $c =
  2+K^2$,
  \begin{equation}\label{E:uniform-E-form-budget}
    \tfrac12z^2+(1+K^2/2)b^2
    \le q_V(u)+cb^2
    \le\tfrac32z^2+(1+3K^2/2)b^2
    \le3(1+K^2)z^2 .
  \end{equation}
  This is a densely defined closed semibounded Hermitian form on full $H^1$: its
  shifted norm is equivalent to the complete $H^1$ norm. The lower bound
  dominates $2b^2$, since $z \ge b$ and $K \ge 1$.
  From~\cite{teschl:14:mathematical}*{Theorem~2.14, p.~81, (2.63)} and
  Theorem~3.6, p.~109, (3.51)--(3.53), it represents a unique self-adjoint $H_V$
  with $A = H_V+c \ge 2$ and $D(A^{1/2}) = H^1$. Closedness uses the definition
  on that book's p.~80. The spectral meaning of $A \ge 2$ follows from bounded
  spectral slices: a nonzero slice below two would give a unit vector in $D(A)$
  contradicting the form lower bound. The bottom argument below uses the same
  representation.

  Put $B_0 = 1-\Delta$ and $L_K = 3(1+K^2)$. The common forms obey $\tfrac12B_0
  \le A \le L_KB_0$. For positive self-adjoint $T,S \ge 1$ with ordered forms,
  completing the square gives
  \[
    \langle f,(T+t)^{-1}f\rangle
    =\sup_u\{2\operatorname{Re}\langle f,u\rangle-q_T(u)-t\|u\|^2\}
    \ge\langle f,(S+t)^{-1}f\rangle .
  \]
  The identity $\sqrt x = \pi^{-1}\int_0^\infty t^{-1/2}x/(x+t)\ud t$ follows by
  $t = xv^2$. Positive spectral integration and Tonelli give square-root form
  order, with domain inclusions, from $I-t(T+t)^{-1} \le I-t(S+t)^{-1}$. Apply
  this to $(B_0,2A)$ and $(A,L_KB_0)$, both pairs at least one:
  \[
    D(A^{1/4})=H^{1/2},\qquad
    2^{-1/4}\|u\|_{H^{1/2}}\le\|A^{1/4}u\|_2
    \le L_K^{1/4}\|u\|_{H^{1/2}} .
  \]
  Under the $L^2$ pairing the dual norm is $\|A^{-1/4}v\|_2$, by Cauchy--Schwarz
  and the equality test $u = A^{-1/2}v$. Completion gives $H^{-1/2}$. Thus, with
  $M_K = L_K^{1/4}$, the phase norm $\|(u,v)\|_{E_A}^2 =
  \|A^{1/4}u\|_2^2+\|A^{-1/4}v\|_2^2$ lies between $M_K^{-2}\|(u,v)\|_E^2$ and
  $M_K^2\|(u,v)\|_E^2$. The condition cost is $M_K^2 = \sqrt{3(1+K^2)} \le
  2(1+K)$.

  For unit $L^2$ data, $b = 1$ and $z \ge 1$. The form bound gives $q_V \ge
  z^2/2-1-K^2/2 \ge -K^2$. A dilated compact unit bump translated wholly off the
  potential support has zero potential pairing and kinetic energy tending to
  zero, so $\beta \le 0$. Complex form infima agree with real ones: dividing
  each complex diagonal by its squared norm is the convex combination of the two
  normalized real diagonals, omitting zero parts. The spectral form formulas
  identify $\beta$ with the spectral bottom by bounded spectral slices. There
  are support points because scalar spectral measures of unit vectors have mass
  one; their arbitrarily small bounded slices give approximate eigenvectors and
  the reverse form bound. If $\Sigma$ denotes the closed PVM support and real
  $z\notin\Sigma$, the Borel function $f_z(h) = (h-z)^{-1}$ on $\Sigma$, zero
  outside, is bounded, as is $hf_z(h)$. Its calculus has range in $D(H_V)$ and
  supplies both inverse identities for $H_V-z$ on their respective domains.
  Thus, no such $z$ is spectral; the approximate eigenvectors give the other
  inclusion. Hence, $\lambda \le K$ and $h+c \ge 2$ on the spectrum.

  Define $\operatorname{Cos}_t(h),\operatorname{Sin}_t(h)$ by $\cos(t\sqrt
  h),\sin(t\sqrt h)/\sqrt h$ for $h > 0$, by $1,t$ at zero and by
  $\cosh(t\sqrt{-h}),\sinh(t\sqrt{-h})/\sqrt{-h}$ for $h < 0$. In the $E_A$
  isometry the matrix is
  \[
    \begin{pmatrix}
      \operatorname{Cos}_t(h)&\sqrt{h+c}\operatorname{Sin}_t(h)\\
      -h\operatorname{Sin}_t(h)/\sqrt{h+c}&\operatorname{Cos}_t(h)
    \end{pmatrix}.
  \]
  For $h \ge 0$, its off-diagonals have bounds $1+\sqrt c\,t$ and one; at zero
  their values are $\sqrt c\,t,0$. For $\beta \le h < 0$, the entries are
  bounded using $|\operatorname{Cos}_t| \le e^{\lambda t}$,
  $|\operatorname{Sin}_t| \le te^{\lambda t}$, upper off-diagonal at most $\sqrt
  c\,te^{\lambda t}$, and lower at most $Ke^{\lambda t}$. Since $\sqrt c \le
  2K$, each entry is at most $2(1+K)(1+t)e^{\lambda t}$. Set these entries to
  zero off the closed spectral support before applying bounded Borel calculus.
  From~\cite{teschl:14:mathematical}*{Theorems~3.1--3.2, pp.~102--104,
  (3.18)--(3.33)}, their matrix norm is at most twice that entry bound, and the
  condition cost gives \eqref{E:uniform-E-static-bound}. Spectral dominated
  convergence and scalar addition give the group.

  Denote the spectral matrix evolution by $P_V^E(t)$. It is the physical $E$
  wave. The fixed-$R$ lift in Lemma~\ref{L:uniform-E-lift} belongs to the
  precise class in Proposition~\ref{P:signed-physical-wave}. Its form is this
  same $q_V$ and has the same self-adjoint representative by uniqueness,
  regardless of the lift's larger shift. On dense bounded spectral cutoffs the
  two waves have the same $\operatorname{Cos},\operatorname{Sin}$ functions of
  $H_V$. Boundedness extends equality to all $E$. The old graph-core Duhamel
  identification therefore applies; no lift norm or $\chi_R$ constant entered
  the bounds above. Conjugation commutes with the real structures, so real
  restrictions are the phases. For compact $L$, finite-rank approximation and
  strong continuity of the free group and its adjoint make $v\mapsto
  S_E(-v)LS_E(v)$ norm continuous on finite intervals. Its primitive is
  Lipschitz. Its Young equation is therefore the ordinary bounded-interaction
  equation by Proposition~\ref{P:constant-young-linear}. This identifies the
  averaged static baseline without transferring any lift norm into the uniform
  bounds.

  Finally, take a unit $\varepsilon$-minimizer for $W$, $0 < \varepsilon \le 1$.
  Since $\beta(W) \le 0$, it has $q_W \le 1$. Thus, $z^2-1-Kz \le 1$, which
  forces $z \le K+2$. The multiplier difference changes its form by at most
  $\|L_V-L_W\|\|u\|_{H^{1/2}}^2 \le (K+2)\|L_V-L_W\|$. Take infima, let
  $\varepsilon\downarrow0$ and reverse $V,W$. The square-root inequality on
  nonnegative $-\beta$ gives \eqref{E:uniform-E-rate-continuity}. For zero
  covariance the class is $\{0\}$, $H_V = -\Delta$ and $\beta = \lambda = 0$;
  the free wave gives the claims directly. This proves
  Proposition~\ref{P:uniform-E-spectral}.
\end{proof}

\subsection{Scalar-time averaging and mean bounds}

Proposition~\ref{P:uniform-E-spectral} bounds the wave of one fixed operator
$L \in \mathcal{V}_{E,R}$ through the rate $\lambda(L)$. The mean field
$\mathcal{T}_R\Phi$ depends on the time variable. We now average it over the
cells of the partition and bound the resulting change in the integrated rate,
uniformly in $R \ge 1$.

\begin{proposition}[Uniform completed-field averaging]\label{P:uniform-rate-averaging}
  Use the true slow partition $I_m$ with mesh at most $\delta B$. Let
  $\mathcal{A}_\delta$ average over each positive-length cell; an empty final
  cell is ignored. Put $\omega_\delta = \|(\mathcal{A}_\delta-I)C_t\|_{H_t\to
  L^r(I)}$. At fixed $B$, $\omega_\delta \to 0$. For $M \ge 0$, if
  $\|\Phi\|_{H_I} \le M$, $W_R = \mathcal{T}_R\Phi$, $\overline{W}_R =
  \mathcal{A}_\delta W_R$ and $Z = C_{\rm mean}M$, then
  \begin{align}
    \|W_R\|_{L^r},\ \|\overline{W}_R\|_{L^r}&\le Z,\qquad
    \|\overline{W}_R-W_R\|_{L^r}\le M_0M\omega_\delta=:\Omega_\delta(M),
    \label{E:uniform-field-averaging}\\
    \operatorname{Rate}_R(\Phi)&=\int_I\lambda(\mathcal{T}_R\Phi(s))\ud s,
    \label{E:uniform-mean-rate}\\
    \int_I|\lambda(\overline{W}_R)-\lambda(W_R)|\ud s
    &\le\sqrt{3+2Z}\sqrt{\Omega_\delta(M)}=:\Lambda_\delta(M)\to0 .
    \label{E:uniform-rate-averaging}
  \end{align}
  These errors are uniform in $R \ge 1$. The rate is finite, nonnegative and
  continuous on $H_I$.
\end{proposition}

\begin{proof}
  Bochner averaging is a contraction on $L^r(I;\mathscr{Z})$ for any Banach
  $\mathscr{Z}$, by Jensen. For continuous fields on $[0,1]$, its error is
  bounded by the modulus of continuity at the mesh. Simple-field approximation,
  followed by continuous approximation of scalar indicators in $L^r$, proves
  density of those fields. The contraction gives strong convergence for every
  Bochner field. In the scalar case it is uniform on the compact closure of
  $C_t$'s unit-ball image, by a finite net and $\|\mathcal{A}_\delta-I\| \le 2$.
  This proves $\omega_\delta \to 0$.

  On finite tensors, integration commutes with the fixed bounded $T_R$:
  \[
    \mathcal{A}_\delta^{\mathcal{K}(E)}
    T_R(C_t\otimes I)\Phi
    =T_R((\mathcal{A}_\delta C_t)\otimes I)\Phi .
  \]
  The dimension-free Hilbert tensor bound in Lemma~\ref{L:uniform-mean-tensor},
  and $\|T_R\| \le M_0$, extend this identity to the completed $H_I$. Applying
  it to the difference gives~\eqref{E:uniform-field-averaging}. Averages remain
  in the same closed $E$ operator class. Only scalar-time compactness was used;
  the tensor map is not claimed compact. Nor does this assert noncompactness of
  a composite spatial family: each fixed-radius completed mean is compact.

  For any two such fields, apply~\eqref{E:uniform-E-rate-continuity} with the
  pointwise bound
  \[
    K(s)=\max(1,\|W_1(s)\|,\|W_2(s)\|).
  \]
  Cauchy--Schwarz gives
  \[
    \int_I|\lambda(W_1)-\lambda(W_2)|
    \le\sqrt{3+\|W_1\|_{L^1}+\|W_2\|_{L^1}}\,
    \sqrt{\|W_1-W_2\|_{L^1}} .
  \]
  Continuity of $\lambda$ on the class makes these quantities measurable. Since
  $|I| = 1$, this proves~\eqref{E:uniform-rate-averaging}. It also proves rate
  continuity, while $\lambda(L) \le \max(1,\|L\|)$ gives
  $\operatorname{Rate}_R(\Phi) \le 1+C_{\rm mean}\|\Phi\|$. For zero covariance
  the fields and rate errors are zero directly. This proves
  Proposition~\ref{P:uniform-rate-averaging}.
\end{proof}

For the remaining proofs use the full parameter choices and the
$B,\delta,N,\ell_m,I_m$ partition of Section~\ref{S:actual-cameron-mean}. For
$W_R = \mathcal{T}_R\Phi$ define
\[
  H_m(u)=\int_0^{\min(u,\ell_m)}
  S_E(-v)W_R(s_m+\delta v)S_E(v)\ud v,\qquad
  P_m^{\rm mean}=S_E(\ell_m)U_{H_m}(\ell_m,0).
\]

\begin{lemma}[Explicit physical primitive and growth bounds]
  \label{L:uniform-product-budgets}
  Write $G_B = \sup_{|v|\le B}\|S_E(v)\|$ and $C_B =
  G_B^2[B^{1-1/r}+B^{1-1/r-\theta}]$. Then $1 \le G_B \le 3B$, $C_B \le
  18B^{3-1/r} \le 32B^{3-1/r}$ and $C_BB^{1/r} \ge B$. The stopped primitives
  lie in the little H\"older space of paths starting at zero and
  \begin{equation}\label{E:uniform-product-primitive}
    \|H_m\|_\theta\le a_m:=C_B\delta^{-1/r}\|W_R\|_{L^r(I_m)},
    \qquad\delta\sum_m a_m^r\le C_B^r\|W_R\|_{L^r}^r .
  \end{equation}
  There is a fixed $C_{\rm gr} = C_{\rm gr}(\theta) > 0$ such that
  \begin{equation}\label{E:uniform-product-growth}
    \log\|S_E(\ell)U_D(\ell,0)\|
    \le C_{\rm gr}B(1+\|D\|_\theta^\nu),\qquad 0\le\ell\le B .
  \end{equation}
  Here $D$ is any $C^\theta([0,B];\mathcal{L}(E))$ path starting at zero; no
  Gaussian or Hamiltonian hypothesis is imposed on this growth bound. The mean
  endpoints have $\|P_m^{\rm mean}\| \ge 1$.
\end{lemma}

\begin{proof}
  The balanced free matrix has upper entry $\langle k\rangle\sin(vk)/k$ and
  lower $-k\sin(vk)/\langle k\rangle$. Splitting $k \le 1$ and $k \ge 1$ bounds
  the upper by $\sqrt2 B$, while the lower and diagonals have bound one. At zero
  the off-diagonals are $v,0$. The Frobenius bound gives $G_B \le \sqrt{3+2B^2}
  \le 3B$; $S_E(0) = I$ gives $G_B \ge 1$. The stated $C_B$ inequalities follow,
  including $C_BB^{1/r} = G_B^2(B+B^{1-\theta}) \ge B$. H\"older in time with
  $1-1/r > \theta$, the affine time change, compact conjugation and constant
  stopping prove \eqref{E:uniform-product-primitive}; the true partition sums
  the $r$th powers, with zero on an empty cell.

  For the explicit Young bound put $d = \|D\|_\theta$ and $c_\theta =
  2\max(1,C_{\rm Young})$, as in the local estimate proving
  Lemma~\ref{L:young-upper-growth}. If $c_\theta dB^\theta \le 1/4$,
  whole-interval contraction gives $\log\sup\|U_D\| \le \log(4/3) < 1/2$.
  Otherwise at most $2B(4c_\theta d)^\nu$ contraction pieces have norm at most
  $4/3$. Combine their logarithms with $\log G_B \le \log(4B)$. For example
  $C_{\rm gr} = 4+\log4+2(4c_\theta)^\nu$ gives
  \eqref{E:uniform-product-growth}, since $B \ge 1$. This is the generic algebra
  argument on $\mathcal{L}(E)$ with multiplication norm one, not an unspecified
  radius-dependent bound.

  Finally, Proposition~\ref{P:uniform-mean-regulator} gives $W_R \in
  L^r(I;\mathcal{K}(E))$ for each fixed $R,\Phi$. The compact conjugation and
  stopped Bochner primitive used above therefore make each $H_m$ a
  compact-valued $C^\theta$ path starting at zero. Lemma~\ref{L:symplectic-free}
  gives $\|P_m^{\rm mean}\| \ge \|S_E(\ell_m)\| \ge 1$, deterministically for
  each $\Phi$. This norm bound needs no fast covariance direction or action
  estimate. An empty endpoint is $I$; zero covariance gives the free endpoint.
  This proves Lemma~\ref{L:uniform-product-budgets}.
\end{proof}

\subsection{Full blocks, the true last cell and polynomial control balls}

\begin{proposition}[Finite-mesh physical mean-product bound]
  \label{P:uniform-mean-product}
  Fix $B$ as above, $M \ge 0$ and $K \ge 1$, and put $Z = C_{\rm mean}M$.
  Uniformly for $R \ge 1$, $\|\Phi\| \le M$,
  \begin{align}
    \delta\sum_{m=0}^N\log\|P_m^{\rm mean}\|
    -\operatorname{Rate}_R(\Phi)
    \le{}&\frac{L_{B,K}}B+
    C_{\rm gr}32^r Z^rB^{3r}(K^{-r}+K^{\nu-r})\notag\\
    &+\Lambda_\delta(M)+C(\theta,B,K)E_\delta+
    \operatorname{Last}_\delta,\label{E:uniform-mean-product}
  \end{align}
  where $L_{B,K} = \log16+4\log(1+K)+3\log(1+B)$ and
  \begin{align*}
    E_\delta&=(2/B)^{1-1/r}C_B\Omega_\delta(M)
    +(2/B)^{1-\nu/r}C_B^\nu\Omega_\delta(M)^\nu,\\
    \operatorname{Last}_\delta&=
    C_{\rm gr}B[\delta+C_B^\nu Z^\nu\delta^{1-\nu/r}].
  \end{align*}
  The last three terms vanish as $\delta\downarrow0$ at fixed $B,K,M$.
\end{proposition}

\begin{proof}
  On each positive-length cell use its own Bochner average $\overline{M}_m$; set
  it to zero on an empty cell and form the constant-field primitive
  $\overline{H}_m$. Put $e_m = C_B\delta^{-1/r}
  \|W_R-\overline{W}_R\|_{L^r(I_m)}$. Jensen
  and~\eqref{E:uniform-product-primitive} give
  \[
    \|\overline{H}_m\|_\theta\le a_m,\quad
    \|H_m-\overline{H}_m\|_\theta\le e_m,\quad
    \delta\sum e_m^r\le C_B^r\Omega_\delta(M)^r,\quad
    \delta\sum a_m^r\le D_M:=C_B^rZ^r .
  \]
  Finite power mean at exponents $r$ and $r/\nu > 1$, with nonnegative inputs
  $e_m,e_m^\nu$, gives $\delta\sum_m(e_m+e_m^\nu) \le E_\delta$. Here $n = N+1
  \ge 2$, $\delta > 0$ and $\delta n \le 2/B$; all subsequent powers are
  positive and zero errors are allowed.

  On full cells only, $\|\overline{M}_m\| \le (\delta
  B)^{-1/r}\|W_R\|_{L^r(I_m)} = a_m/(C_BB^{1/r}) \le a_m/B$. Split full indices
  $m < N$ into good $a_m \le K$ and bad $a_m > K$. On a good block,
  Proposition~\ref{P:young-upper-multiplicative} applies with base
  $\overline{H}_m$, perturbation $H_m-\overline{H}_m$ and the left multiplier
  $S_E(B)$. The baseline is the static wave of $\overline{M}_m$ by
  Proposition~\ref{P:uniform-E-spectral}. Here $\|\overline{M}_m\| \le K/B \le
  K$, whereas $\|\overline{H}_m\|_\theta \le a_m \le K$. The spectral estimate
  gives
  \[
    \|S_E(B)U_{\overline{H}_m}(B,0)\|
    \le8(1+K)^2(1+B)e^{B\lambda(\overline{M}_m)}.
  \]
  Consequently,
  \[
    \log\|P_m^{\rm mean}\|
    \le L_{B,K}+B\lambda(\overline{M}_m)
    +C(\theta,B,K)(e_m+e_m^\nu).
  \]
  This sufficient prefactor follows from the bound in $E$, $8(1+K)^2(1+B)$, by
  enlarging it. Since $\delta N \le 1/B$, the prefactors sum to at most
  $L_{B,K}/B$. Nonnegativity of $\lambda$ gives $\delta\sum_{\rm
  good}B\lambda(\overline{M}_m) \le \int_I\lambda(\overline{W}_R) \le
  \operatorname{Rate}_R(\Phi)+\Lambda_\delta(M)$. Bad and last rate cells can be
  dropped in this upper bound.

  On bad full cells~\eqref{E:uniform-product-growth} gives $C_{\rm
  gr}B(1+a_m^\nu)$. Because $a_m > K \ge 1$ and $r > \nu$, $1+a_m^\nu \le
  (K^{-r}+K^{\nu-r})a_m^r$. Their weighted sum is bounded by $C_{\rm
  gr}BC_B^rZ^r(K^{-r}+K^{\nu-r})$. The inequality $C_B \le 32B^{3-1/r}$ yields
  the displayed bad-block term.

  If $\ell_N = 0$, the last endpoint is $I$ and $\log\|I\| = 0$. Otherwise use
  the stopped path, not its frozen full-cell formula. Since $a_N \le
  C_B\delta^{-1/r}Z$, its weighted logarithm is at most
  $\operatorname{Last}_\delta$ by~\eqref{E:uniform-product-growth}. This tends
  to zero because $\nu < r$, even for arbitrarily short last cells; no division
  by that cell length occurs. Combining these estimates
  proves~\eqref{E:uniform-mean-product}. All constants are independent of $R$.
  This proves Proposition~\ref{P:uniform-mean-product}.
\end{proof}

\begin{corollary}[Every polynomial control radius]\label{C:uniform-polynomial-balls}
  For $c_{\rm ball} > 0$, $\kappa_{\rm ball} \ge 0$ put $M_B = c_{\rm
  ball}B^{\kappa_{\rm ball}}$. Then
  \[
    \limsup_{\substack{B\to\infty\\B\in\mathbb{N}}}
    \limsup_{\delta\downarrow0}
    \sup_{\substack{R\ge1\\\|\Phi\|\le M_B}}
    \left\{\delta\sum_m\log\|P_m^{\rm mean}\|
    -\operatorname{Rate}_R(\Phi)\right\}\le0 .
  \]
\end{corollary}

\begin{proof}
  Choose $s > r(3+\kappa_{\rm ball})/(r-\nu)$ and $K_B = B^s$. At each fixed
  $B$, the inner $\delta$ limit has a fixed finite radius and threshold, so the
  last three errors in~\eqref{E:uniform-mean-product} vanish first. Its bad
  terms become
  \[
    C_{\rm gr}32^r C_{\rm mean}^r c_{\rm ball}^r
    \{B^{r(3+\kappa_{\rm ball})-sr}
    +B^{r(3+\kappa_{\rm ball})+s(\nu-r)}\}.
  \]
  Both exponents are negative: the second by choice and the first because $s >
  3+\kappa_{\rm ball}$. Also $L_{B,K_B}/B \le [\log16+7\log2+(4s+3)\log B]/B \to
  0$. Thus, the outer integer-$B$ limit gives the claim. No diagonal choice
  $B_\delta,M_\delta$ is asserted. At fixed parameters all primitive maps are
  bounded linear into separable little H\"older products. Young local dependence
  and positive endpoint norms make the displayed logarithms continuous; the rate
  is continuous by Proposition~\ref{P:uniform-rate-averaging}. Growth bounds
  make each control-ball supremum finite. A countable dense subset of each
  closed Hilbert ball has the same supremum. Supremum over $R$ is an ordinary
  deterministic real supremum, not a random measurability or compact-family
  claim. This proves Corollary~\ref{C:uniform-polynomial-balls}.
\end{proof}

\subsection{Arbitrary small residual paths and the ordered product}

Corollary~\ref{C:uniform-polynomial-balls} compares the sum of the logarithms
$\log\|P_m^{\rm mean}\|$ with the rate, for the mean primitives $H_m$ alone. We
now add compact-operator paths $x_m$ to the $H_m$ and bound the norm of the
ordered product of the perturbed endpoints when these paths are small.

\begin{proposition}[Residual comparison and the integrated rate]
  \label{P:uniform-arbitrary-residual}
  Let $x_m$ be arbitrary little $C^\theta$ compact-operator paths starting at
  zero on $[0,B]$, put $v_m = \|x_m\|_\theta$ and $Y = (\delta\sum_m
  v_m^p)^{1/p}$. Define
  \[
    P_m^{\rm pert}=S_E(\ell_m)U_{H_m+x_m}(\ell_m,0),\qquad
    z_{\rm rat}=\prod_m\frac{\|P_m^{\rm pert}\|}{\|P_m^{\rm mean}\|}.
  \]
  For fixed integer $B \ge 1$, $M \ge 0$ and every $K_{\rm res} \ge 1$,
  \begin{align}
    \delta\log\max(1,z_{\rm rat})\le{}&
    C_{K_{\rm res}}[(2/B)^{1-1/p}Y+(2/B)^{1-\nu/p}Y^\nu]\notag\\
    &+c_BD_M(K_{\rm res}^{-r}+K_{\rm res}^{\nu-r}),
    \label{E:uniform-residual-ratio}
  \end{align}
  uniformly in $R \ge 1$, $\|\Phi\| \le M$. The finite constants may be written
  $C_{K_{\rm res}} = C(\theta,B,K_{\rm res})$ and $c_B = 2^{\nu-1}C_{\rm gr}B$.
  Here $D_M = C_B^r(C_{\rm mean}M)^r$. If $b_\delta \ge 0$ is any deterministic
  function of $\delta$ tending to zero at each fixed $B$, $M_B$ is as in
  Corollary~\ref{C:uniform-polynomial-balls}, and $\mathcal{F} =
  2\delta\log\max(1,\|P_N^{\rm pert}\cdots P_0^{\rm pert}\|)$, then
  \begin{equation}\label{E:uniform-residual-integrated-rate}
    \limsup_{\substack{B\to\infty\\B\in\mathbb{N}}}
    \limsup_{\delta\downarrow0}
    \sup_{\substack{R\ge1,\ \|\Phi\|\le M_B\\Y\le b_\delta}}
    \{\mathcal{F}-2\operatorname{Rate}_R(\Phi)\}\le0 .
  \end{equation}
  No Gaussian, Cameron or Hamiltonian condition on the residuals is imposed.
\end{proposition}

\begin{proof}
  All endpoint norms are positive by the Young and free inverses. On every cell
  with $a_m \le K_{\rm res}$ apply
  Proposition~\ref{P:young-upper-multiplicative} with base $H_m$, perturbation
  $x_m$ and left multiplier $S_E(\ell_m)$. It gives logarithmic ratio at most
  $C(\theta,B,K_{\rm res})(v_m+v_m^\nu)$. On cells with $a_m > K_{\rm res}$ use
  \eqref{E:uniform-product-growth}, $(a_m+v_m)^\nu \le
  2^{\nu-1}(a_m^\nu+v_m^\nu)$ and the mean denominator $\|P_m^{\rm mean}\| \ge
  1$. The logarithmic ratio is at most $c_B(1+a_m^\nu+v_m^\nu)$, with $c_B$
  independent of $K_{\rm res},R,\delta,\Phi,M$. Increase $C_{K_{\rm res}}$ to
  dominate both good and bad residual constants. Summation gives a nonnegative
  bound
  \[
    \log\max(1,z_{\rm rat})
    \le C_{K_{\rm res}}\sum_m(v_m+v_m^\nu)
    +c_B\sum_{a_m>K_{\rm res}}(1+a_m^\nu).
  \]
  The bad sum times $\delta$ is at most $D_M(K_{\rm res}^{-r}+K_{\rm
  res}^{\nu-r})$. Finite power mean at exponents $p \ge 1$ and $p/\nu \ge 1$,
  with nonnegative inputs $v_m,v_m^\nu$, gives $\delta\sum v_m \le
  (2/B)^{1-1/p}Y$ and $\delta\sum v_m^\nu \le (2/B)^{1-\nu/p}Y^\nu$. This
  includes $p = \nu$, zero entries and the true empty block, with no singular
  powers. It proves~\eqref{E:uniform-residual-ratio}.

  The product $A_{\rm prod} = \prod_m\|P_m^{\rm mean}\|$ is at least one.
  Submultiplicativity, without commutation, gives
  \[
    \begin{aligned}
      \max(1,\|P_N^{\rm pert}\cdots P_0^{\rm pert}\|)
      &\le A_{\rm prod}\max(1,z_{\rm rat}),\\
      \mathcal{F}-2\delta\sum_m\log\|P_m^{\rm mean}\|
      &\le2\delta\log\max(1,z_{\rm rat}).
    \end{aligned}
  \]
  At fixed $B,M,K_{\rm res}$ the residual terms vanish on $Y \le b_\delta$. Let
  $\delta\downarrow0$ first and then $K_{\rm res} \to \infty$. Since $r > \nu$,
  the residual excess has limsup at most zero, uniformly in $R,\Phi$. Now take
  $M = M_B$ at each fixed $B$ and combine with
  Corollary~\ref{C:uniform-polynomial-balls}; a supremum of sums is no larger
  than the sum of the suprema. The outer integer-$B$ limit
  proves~\eqref{E:uniform-residual-integrated-rate}. The endpoint functional is
  continuous on all tuples by generic Young local dependence, including
  nonphysical residuals. Nothing was multiplied as scalar displacement moments.
  This proves Proposition~\ref{P:uniform-arbitrary-residual}.
\end{proof}

These are uniform deterministic $E$-phase product estimates. Their Gaussian
Rayleigh application, original point transfer and signed-variational value over
nonnegative controls comparison are proved separately in
Sections~\ref{S:actual-gaussian-tubes}--\ref{S:intrinsic-second-moment}.

\section{The original point moment and the growing-cone phase}
\label{S:original-point-transfer}

Retain all original covariances and amplitudes. Write $a = \alpha_0$, $\kappa =
(4-\alpha-a)/(3-\alpha) > 1$. Fix the auxiliary parameters of the preceding
Gaussian and product sections once, in their strict admissible gaps. For $0 <
\delta < 1$ put
\[
  L=\delta^{-1},\quad R=L+2,\quad T_\delta=\delta^{-1/\kappa},
  \quad g=\delta^{(1-a)/2},\quad \Gamma_\delta=g^2\Gamma .
\]
Use the same real even root $\eta$, set $\psi = \eta(\cdot/R)$, $m = \psi^2$, $z
= (\psi,0) \in E$, $D = B_1(0)$ and $v_d = |D| > 0$. Let $W_{\rm fast} = W\circ
T_{\delta,0}$ on the given slow probability space, and let $P_\delta$ be the
physical $E$ endpoint at time $L$ driven by $gM_R(W_{\rm fast})$, with physical
noise mask $m$ once. The datum $\psi$ is not the auxiliary cutoff used to extend
the operator path to constant columns on $F$.

\begin{lemma}[Restriction and fixed-parameter finiteness]
  \label{L:point-actual-finiteness}
  For nonzero covariance, the $E$ flow and its source flows are the closed
  isometric restrictions of their $F$ extensions. Their operator norms and
  differences are measurable, and
  \begin{equation}\label{E:point-fixed-finiteness}
    1\le\E\|P_\delta\|_{\mathcal{L}(E)}^2<\infty .
  \end{equation}
\end{lemma}

\begin{proof}
  Let $A_F = gA_{F,R}(W_{\rm fast})$ be the scaled full operator path. Linearity
  and Proposition~\ref{P:constant-phase-driver} give $A_Fi = iD_E$, with $D_E =
  gM_R(W_{\rm fast})$, and $S_F(t)i = iS_E(t)$. The extension uses a cutoff
  $\widetilde{\psi} = 1$ near its larger localization, not the datum $\psi$.
  Applying $i$ to the state Young sums and using uniqueness gives $U_F(t,s)i =
  iU_E(t,s)$, then $P_F(t,s)i = iP_E(t,s)$. The same argument applies to
  sources. Closedness and isometry imply $\|P_E\| \le \|P_F\|$ and the analogous
  difference bound; density of $E$ in $F$ is neither true nor used. The
  continuous Young map on its separable path span, followed by bounded physical
  conjugation and restriction, gives measurable endpoint norms and uniform
  time-triangle difference norms. No operator-norm continuity of the free group
  is assumed. Young Neumann pieces and the free inverse also give invertibility.

  We check a direct finiteness input, without a fixed-mask asymptotic rate. Take
  $B = 1$ in Propositions~\ref{P:tube-actual-law}
  and~\ref{P:tube-coarse-envelope} and Lemma~\ref{L:tube-rayleigh-tail}. Every
  $0 < \delta < 1$ satisfies their clock condition. The global endpoint
  in~\eqref{E:tube-global-endpoint} is exactly this $P_\delta$: the same given
  $W_{\rm fast}$, amplitude $g$, root $\eta$, one mask $m$, massless $E$ phase,
  and horizon $\delta^{-1}$. We use the same admissible $\theta$ and operator
  path version. Explicit row identities, rational times and Young uniqueness
  identify versions if a smaller common admissible exponent is used. Thus, the
  finite radial bound applies to this endpoint, not merely an equal-law copy. At
  these fixed parameters the operator path $gM_R(W_{\rm fast})$ is a
  compact-valued $C^\theta([0,L];\mathcal{K}(E))$ path starting at zero on its
  path event. Lemma~\ref{L:symplectic-free} gives $\|P_\delta\| \ge \|S_E(L)\|
  \ge 1$ almost surely, proving~\eqref{E:point-fixed-finiteness}. This use
  requires no $F$-norm rate along growing radii. Source convergence and source
  uniform integrability are checked next, separately from this original endpoint
  finiteness. This proves Lemma~\ref{L:point-actual-finiteness}.
\end{proof}

\begin{lemma}[The same regularized fast noises preserving the given noise]\label{L:point-fast-sources}
  At each fixed $\delta$, take one regularized slow noise preserving the given
  noise $Y_{\epsilon,\tau}$ and define
  \[
    Y_{\rm fast}(t,x)=\delta^{a/2}Y_{\epsilon,\tau}(\delta t,x).
  \]
  Its physical $E$ flows $P_E^{\epsilon,\tau}$ with coefficient $m gY_{\rm
  fast}$ satisfy
  \begin{equation}\label{E:point-source-flow-limit}
    \E\sup_{0\le s\le t\le L}
    \|P_E^{\epsilon,\tau}(t,s)-P_E(t,s)\|^2\longrightarrow0
    \quad\hbox{as }(\epsilon,\tau)\to(0,0).
  \end{equation}
  The unmasked field $gY_{\rm fast}$ has the canonical source law for covariance
  $\Gamma_\delta$, spatial width $\epsilon$ and temporal width $\tau/\delta$.
\end{lemma}

\begin{proof}
  Testwise this field is $Y_{\epsilon,\tau}(T_{\delta,0}f)$. The original
  isometry and source domination give source and residual variances bounded by
  $\|f\|_{H_{\rm fast}}^2$, with residual variance tending to zero at fixed
  $\delta$ on every smooth compact test. The Gaussian transfer
  Proposition~\ref{P:source-gaussian-transfer} therefore gives operator-path
  convergence in every finite $L^pC^\theta$, retaining $W_{\rm fast}$ literally.
  Multiplication by fixed $g$ and the bounded $F$ extension preserve it. The
  transfer bound and Proposition~\ref{P:actual-operator-driver} also give
  uniform source $\sqrt p$ path moments at these fixed $\delta,R,L$. The
  moment-series argument yields square-exponential path norms. Young local
  dependence gives probability convergence on ball of operator pathss; the
  growth exponent $1/\theta < 2$ and those tails give uniform flow moments of
  every finite order and uniform integrability. The proof of
  Proposition~\ref{P:constant-gaussian-flow} applies to this very pullback
  family. Restriction from $F$ then proves~\eqref{E:point-source-flow-limit}. No
  uniform-in-$\delta$ source bound or exchange of limits is asserted.

  This source is also the ordinary response for all $E$ data. On Schwartz states
  its first primitive has row $gY_{\rm fast}(h_R)$. The source indicator
  approximation and bounded point variances at positive widths identify it with
  the statewise integral $\int_0^t S_E(-r)B(mgY_{\rm fast}(r))S_E(r)\ud r$.
  Fubini uses $\E|ZY| \le \|Z\|_2\|Y\|_2$ when Gaussian testing is needed.
  Countable dense states, rational times and continuity give the full
  ordinary-primitive identity, as in
  Proposition~\ref{P:source-ordinary-response}.
  Lemma~\ref{L:constant-source-solution} and
  Proposition~\ref{P:symplectic-source} then identify the Young equation with
  the ordinary interaction equation on $E$.

  In the temporal kernel $K_\tau(v) = \Gamma_{\rm E}(a)^{-1}
  \int_0^{\tau^{-1}}e^{-\lambda|v|}\lambda^{a-1}\ud\lambda$, substitution
  $\lambda' = \delta\lambda$ gives
  \[
    \delta^aK_\tau(\delta v)=K_{\tau/\delta}(v).
  \]
  Hence, $gY_{\rm fast}$ has covariance
  $K_{\tau/\delta}(t-s)(p_{2\epsilon}*\Gamma_\delta)(x-y)$. The shared-width
  maximum and spatial heat-sum conventions persist for two sources. Positive
  multiplication by $g^2$ preserves every original covariance class, including
  spatial white noise. At fixed $\delta$, $(\epsilon,\tau/\delta) \to (0,0)$.
  The full source point-moment limit in
  Proposition~\ref{P:masked-canonical-transfer} thus applies to this
  single-source law and its continuous version; no newly reconstructed auxiliary
  family for $\Gamma_\delta$ is required. This proves
  Lemma~\ref{L:point-fast-sources}.
\end{proof}

\begin{lemma}[Compact-data source identification]\label{L:point-compact-source}
  For fixed $\delta,\epsilon,\tau$, the displacement $v_{\epsilon,\tau}(t,x) =
  (\operatorname{pr}_1P_E^{\epsilon,\tau}(t,0)z)(x)$ has a jointly measurable
  continuous version and satisfies the ordinary scalar mild equation with
  coefficient $b = mgY_{\rm fast}$ and initial data $(\psi,0)$.
\end{lemma}

\begin{proof}
  On the fixed source regularity event, $b$ is smooth in space, has common
  compact support in $B_{2R}$, and is continuous in time in every bounded
  spatial derivative norm on $[0,L]$. Choose integer $k > d/2+1$ and $E_k =
  H^k\times H^{k-1}$. Its Bessel-unitary free matrix is
  \[
    \begin{pmatrix}
      \cos(t|\xi|)&\langle\xi\rangle\sin(t|\xi|)/|\xi|\\
      -|\xi|\sin(t|\xi|)/\langle\xi\rangle&\cos(t|\xi|)
    \end{pmatrix}.
  \]
  The zero sine quotient is $t$; finite-horizon bounds and dominated Fourier
  convergence give a strongly continuous group on $E_k$. Leibniz and the
  derivative-norm equivalence, obtained by expanding $(1+|\xi|^2)^{k-1}$ and
  Parseval, give
  \[
    \|b(t)f\|_{H^{k-1}}
    \le C_k\max_{|j|\le k-1}\|\partial^jb(t)\|_\infty
    \|f\|_{H^{k-1}} .
  \]
  Thus, $B(b(t))$ is a norm-continuous bounded force on $E_k$, with a finite
  uniform sample bound $B_L$. For $Z_0 = S(t)z$ and $Z_{n+1}(t) =
  \int_0^tS(t-r)B(b(r))Z_n(r)\ud r$,
  \[
    \sup_{t\le L}\|Z_n(t)\|_{E_k}
    \le C_L^{n+1}B_L^nL^n\|z\|_{E_k}/n! .
  \]
  This series converges in $C_tE_k$. Iterating a homogeneous Volterra difference
  proves uniqueness. Its continuous embedding into $E$ and ordinary $E$
  uniqueness identify the sum with $P_E^{\epsilon,\tau}(t,0)z$.

  Fourier Cauchy--Schwarz gives $\|\widehat{f}\|_1 \le \|f\|_{H^k}
  (\int\langle\xi\rangle^{-2k}\ud\xi)^{1/2} < \infty$. Fourier inversion and
  dominated convergence imply continuous bounded displacement, and the same
  holds for velocity since $k-1 > d/2$. The source derivatives on the compact
  support form a measurable $C_tC_x^{k-1}$ random element, by countable
  evaluations and continuity. Picard iteration and local norm estimates make its
  phase sum measurable in $C_tE_k$; the Fourier embedding gives joint
  measurability. These high norms are used only at positive source widths.

  The first component of Duhamel is, distributionally,
  \[
    v(t)=\cos(t\sqrt{-\Delta})\psi+
    \int_0^t G_d(t-r)*[b(r)v(r)]\ud r .
  \]
  Here $G_d(h)$ is the positive Green measure of mass $h$ and support in $B_h$,
  including the sphere measure for $d = 3$. The product $bv$ is continuous
  compactly supported, with finite uniform $L^1$ and supremum bounds. Against a
  compact smooth $\phi$, absolute Fubini is bounded by
  $\int_0^t\|\phi\|_\infty\|b(r)v(r)\|_1(t-r)\ud r < \infty$. Writing $G_d(h)$
  as $h$ times the dilation of its unit-time probability measure proves
  continuity of the convolution term. The free term is continuous by the Fourier
  embedding. Thus, the distributional identity agrees everywhere with the
  pointwise continuous mild versions. No Green density in $d = 3$ or original
  Sobolev point trace is used. This proves Lemma~\ref{L:point-compact-source}.
\end{proof}

The displacement $v_{\epsilon,\tau}$ of Lemma~\ref{L:point-compact-source} has
the masked coefficient $mgY_{\rm fast}$ and the data $(\psi,0)$. The unmasked
source solution has the coefficient $gY_{\rm fast}$ and the data $(1,0)$. We now
show that the two agree on a cone that contains $D$ at time $L$.

\begin{proposition}[Cone identity and an average of squares]
  \label{P:point-source-cone}
  Let $u_{\epsilon,\tau}^{\rm full}$ be the unmasked source solution with
  coefficient $gY_{\rm fast}$ and data $(1,0)$. At the fixed parameters, almost
  surely,
  \[
    v_{\epsilon,\tau}=u_{\epsilon,\tau}^{\rm full}
    \quad\hbox{on }\mathcal{C}=
    \{(t,x):0\le t\le L,\ |x|\le1+L-t\}.
  \]
  Consequently,
  \begin{equation}\label{E:point-source-square-average}
    \E\|\operatorname{pr}_1P_E^{\epsilon,\tau}(L,0)z\|_{L^2(D)}^2
    =v_d\,\E|u_{\epsilon,\tau}^{\rm full}(L,0)|^2<\infty .
  \end{equation}
\end{proposition}

\begin{proof}
  Proposition~\ref{P:masked-source-moments}, applied through the exact source
  law of Lemma~\ref{L:point-fast-sources}, gives the locally continuous
  measurable full solution and its $L^2$ point Picard sum. The pathwise cone
  argument uses the realized pullback field; the point second moments depend
  only on its Gaussian covariance through those Bochner chain sums. We place no
  global unmasked field in $E$ or $F$. The cone is causally closed: a
  predecessor at distance at most $t-s$ has norm at most $1+L-s$. Its spatial
  projection lies a positive distance inside the plateau of both $\psi$ and $m$.

  The total initial difference $(\psi-1,0)$ is represented in $E$ plus constants
  by $(\psi,0)+(-1,0)$. Every entry of the full free phase kernel is a compactly
  supported distribution in $B_t$, and the top-left is $\partial_tG_d(t)$. Since
  $|x|+t \le 1+L = R-1$ on the cone, its entire translated kernel support lies
  in the region where this total datum vanishes.
  Lemma~\ref{L:constant-phase-free} therefore gives
  $\cos(t\sqrt{-\Delta})\psi(x) = 1$ there. Testing the $E$ and constant
  summands separately would not give this total finite-propagation statement.

  The two mild equations have the same coefficient on the cone. For their
  difference $w$, causal closure gives the homogeneous Green integral equation
  there. Put $H = \sup_{\mathcal{C}}|w|$ and $M = \sup_{\mathcal{C}}|gY_{\rm
  fast}|$, both finite sample constants. Iteration with the positive Green
  measures gives $|w(t,x)| \le HM^nt^{2n}/(2n)!$. Indeed, the mass recursion is
  $f_0 = 1$, $f_{n+1}(t) = \int_0^t(t-r)f_n(r)\ud r = t^{2n+2}/(2n+2)!$. The
  factorial ratio forces this bound to zero on the fixed cone. No positivity of
  the sample coefficient or expectation of a global source supremum is required.

  At time $L$ the cone contains $D$. Tonelli and spatial stationarity of the
  unmasked single-source law give~\eqref{E:point-source-square-average}.
  Finiteness is the full source point-moment bound. This is an integral of
  squares, not a squared average or a covariance comparison with a masked model.
  Events are for each fixed $\delta,\epsilon,\tau$. This proves
  Proposition~\ref{P:point-source-cone}.
\end{proof}

Agreement on the terminal observation set alone would not suffice:
both initial phase components must agree on its backward cone.
No-go~\ref{P:nogo-31} gives a smooth free-wave counterexample when
the velocity guard is omitted, and a separate phase example explaining
why operator growth alone does not imply growth for every initial state.


Proposition~\ref{P:point-source-cone} holds at fixed positive
$\epsilon,\tau$ and for the scaled covariance $\Gamma_\delta$. We now remove the
regularization and undo the scaling to reach the original point moment
$\E|u_\Gamma(T_\delta,0)|^2$.

\begin{proposition}[Exact original point transfer and polynomial cost]
  \label{P:original-point-transfer}
  For every original model, including zero covariance,
  \begin{align}
    \E|u_\Gamma(T_\delta,0)|^2
    &=v_d^{-1}\E\|\operatorname{pr}_1P_\delta z\|_{L^2(D)}^2,
    \label{E:original-point-transfer}\\
    &\le v_d^{-1}\|\psi\|_{H^{1/2}}^2\E\|P_\delta\|^2,\notag\\
    \limsup_{T\to\infty}T^{-\kappa}\log\E|u_\Gamma(T,x)|^2
    &\le\limsup_{\delta\downarrow0}\delta\log\E\|P_\delta\|^2
    \quad\hbox{for every fixed }x .
    \label{E:original-point-rate-reduction}
  \end{align}
\end{proposition}

\begin{proof}
  Assume nonzero covariance until the last paragraph, and fix $\delta$.
  Equation~\eqref{E:point-source-flow-limit} gives convergence of the source
  displacement in $L^2(\Omega;L^2(D))$, since $\|\operatorname{pr}_1e\|_{L^2(D)}
  \le \|e\|_E$. Expected squared norms converge by Cauchy--Schwarz applied to
  their difference times their sum. Separately, the full source point-moment
  limit \eqref{E:masked-unmasked-moment-limit} at covariance $\Gamma_\delta$
  gives convergence of the right side of~\eqref{E:point-source-square-average}
  to $v_d\E|u_{\Gamma_\delta}(L,0)|^2$. Both cutoffs $(\epsilon,\tau/\delta)$
  tend to zero at fixed $\delta$. Thus, the two limits in the equal prelimit
  moments give
  \[
    \E|u_{\Gamma_\delta}(L,0)|^2
    =v_d^{-1}\E\|\operatorname{pr}_1P_\delta z\|_{L^2(D)}^2 .
  \]

  The law normalization is the consequence of
  Lemma~\ref{L:exact-canonical-scaling} and the canonical-series time change
  proved in Proposition~\ref{P:raw-block-point-comparison}. Explicitly, for
  $\delta = T^{-\kappa}$ and $r_T = T\delta$, $V_\delta$ has the law of
  $u_\Gamma(Tt,r_Tx)$ and equation \eqref{E:exact-fast-equation}. At positive
  source widths, $U(t,x) = V_\delta(\delta t,x)$ has coefficient
  $\sqrt\delta\,Y_{\epsilon,\tau}(\delta t,x) = gY_{\rm fast}(t,x)$. The
  fixed-order and absolutely convergent canonical-series argument passes this
  time change to the point moment at fixed $\delta$. Hence,
  $\E|u_\Gamma(T_\delta,0)|^2 = \E|u_{\Gamma_\delta}(L,0)|^2$. This is a
  law/moment identity, not a pathwise identity on the same noise. It preserves
  all mixed, unequal, self and empty Stratonovich chains and the
  Green-before-noise convention in $d = 3$. The shrinking $r_T$ is not the mask
  radius $R$; no spatial Jacobian occurs at point zero. The phase/operator
  inequality now proves~\eqref{E:original-point-transfer}.

  Parseval and the inhomogeneous Bessel weights give, for $R \ge 1$,
  \[
    \|\psi\|_{H^{1/2}}^2\le\|\psi\|_{H^1}^2
    =R^d\|\eta\|_2^2+R^{d-2}\|\nabla\eta\|_2^2
    \le C_\eta R^d,\quad C_\eta=\|\eta\|_2^2+\|\nabla\eta\|_2^2>0.
  \]
  This includes the negative $d-2$ power in $d = 1$. For
  $\operatorname{DataPref}_\delta = v_d^{-1}C_\eta(\delta^{-1}+2)^d$,
  \[
    \delta\log\operatorname{DataPref}_\delta
    =\delta\log(C_\eta/v_d)+d\delta[\log(1+2\delta)-\log\delta]\to0 .
  \]
  The nontrivial term follows by $t = -\log\delta$ and $te^{-t} \to 0$, since
  $e^t \ge t^2/2$. The original point moment is at least one: its nonnegative
  full pairing expansion contains the unit empty pair of chains. It is finite
  by~\eqref{E:original-point-transfer} and~\eqref{E:point-fixed-finiteness}.
  Thus, logs are defined without subtracting infinite quantities. Since $\alpha
  < 2 < 3$, $a < 1$, we have $\kappa > 1$ and $\delta\mapsto T_\delta$ is a
  decreasing bijection $(0,1) \to (1,\infty)$ with inverse $\delta =
  T^{-\kappa}$. Taking logs and this change of parameter proves
  \eqref{E:original-point-rate-reduction} at zero. Original canonical spatial
  stationarity, proved in Proposition~\ref{P:raw-block-original-lower}, extends
  it to every fixed $x$; no stationarity of the masked operator is assumed.

  For zero covariance, $u = 1$, all operator paths vanish and $P_\delta =
  S_E(L)$. The same total free cone argument gives the spatial-square identity.
  Lemma~\ref{L:symplectic-free} gives $\|S_E(L)\| \ge 1$, and its Fourier matrix
  gives $\|S_E(L)\| \le C(1+L)$. Thus, $0 \le \delta\log\|S_E(\delta^{-1})\|^2
  \to 0$ and both claims hold directly without a nonzero-noise import. This
  proves Proposition~\ref{P:original-point-transfer}.
\end{proof}

\section{Time blocks for the original noise and the pointwise lower bound}
\label{S:actual-raw-block-lower}

Fix a nonzero covariance from Hypothesis~\ref{H:homog} and any real nonnegative
nonzero $\Psi \in C_c^\infty((0,1)\times\R^d)$. Retain the \emph{original} $Z =
W(\Psi)$, $v = v_\Psi > 0$ and mean $h_s$ of Lemma~\ref{L:compact-raw-profile}.
Choose one finite even mask $m_R \in [0,1]$, equal to one on $B_R$ and supported
in $\overline{B}_{2R}$, as in Proposition~\ref{P:compact-original-common-trial}.
Then $V_s = m_Rh_s$ has the ground family, rate $\lambda_s$ and full $F$
projection of Section~\ref{S:compact-mean-family}. Set
\[
  I_R=\int_0^1\lambda_s\ud s,\qquad
  Q_2=\liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,0)|^2.
\]
We will prove $Q_2 \ge 2I_R-v/2$, for this same control and covariance. Fix a
real nonnegative smooth compact $\phi$ of integral one, the bounded observable
$O_\phi$, and the data $e_0 = (0,1,0) \in F$. Both displacement and velocity are
retained. This is the same physical profile $m_R = \zeta_R$ used in the operator

\subsection{The common original noise and its exact Gaussian shift}

Choose an integer $B \ge 1$, later fixed by the cone estimate, and take $0 <
\delta < \min(1,1/B)$. Define
\begin{equation}\label{E:raw-block-clocks}
  N=\left\lfloor\frac1{\delta B}\right\rfloor\ge1,\quad
  s_m=m\delta B\ (0\le m\le N),\quad
  \tau=\delta^{-1}-NB\in[0,B),\quad
  g_\delta=\delta^{(1-a)/2}.
\end{equation}
For a smooth compact positive-fast-time test $f$, extended by zero, put
\begin{equation}\label{E:raw-block-pullback}
  T_{\delta,m}f(s,x)
  =\delta^{a/2-1}f((s-s_m)/\delta,x),\qquad
  W_{\delta,m}(f)=W(T_{\delta,m}f).
\end{equation}
Write $W_{\mathrm{fast}} = W_{\delta,0}$. Use the same admissible
$\gamma,\theta,\beta$ as~\eqref{E:driver-parameters}. Write $A_R(W_{\delta,m})$
for the $F$ integrated operator obtained by the bounded extension
\eqref{E:constant-phase-driver}, on the local horizon $[0,B]$, and set
\begin{equation}\label{E:raw-block-drivers}
  X_m=g_\delta A_R(W_{\delta,m}),\qquad
  H_m(u)=\int_0^uS_F(-r)iB_F(V_{s_m+\delta r})S_F(r)\ud r.
\end{equation}
The convolution formula for $h$ extends smoothly to $s \in [0,2]$, covering
these means. Ground templates will only use $s_m \in [0,1]$.

\LeanMarker{L:raw-block-inputs}
\begin{lemma}[Coupled Gaussian block inputs]\label{L:raw-block-inputs}
  Each $T_{\delta,m}$ is an isometry of the original test completion, respecting
  covariance-null tests. All $W_{\delta,m}$ and $Z$ are on the same given noise
  space, without an independence assertion. Let $\mathcal{K}(F)$ have the
  operator norm and let $\mathscr{B}_0 = h_0^\theta([0,B];\mathcal{K}(F))$ be
  the $C^\theta$ closure of zero-initial polygonal compact-operator paths. It is
  a separable Banach space. In $\mathscr{B}_\delta = \mathscr{B}_0^{N+1}$ with
  maximum norm, $(X,Z)$, $X = (X_m)_{m=0}^N$, is jointly centered Gaussian and
  \begin{equation}\label{E:raw-block-covariance}
    b:=\E[XZ]=\sqrt\delta\,H,\qquad H=(H_m)_{m=0}^N.
  \end{equation}
  Moreover,
  \begin{equation}\label{E:raw-block-moments}
    \|X_m\|_{L^p\mathscr{B}_0}
    \le C_{R,B}\sqrt p\,g_\delta,\qquad p\ge1.
  \end{equation}
\end{lemma}

\begin{proof}
  Substituting $s = s_m+\delta u$, $t = s_m+\delta r$ in the original covariance
  gives the factor $\delta^{a-2}\delta^2\delta^{-a} = 1$. Thus, the maps extend
  isometrically through the null quotient. Direct test substitution gives
  $W_{\delta,m}(f) = W_{\mathrm{fast}}(f(\cdot-mB,\cdot))$. These are explicit
  common-space identities.

  Finite-rank rational matrices in a countable real Hilbert basis are dense in
  $\mathcal{K}(F)$: for compact $T$, $P_nTP_n \to T$ in norm by compactness of
  $T,T^*$, and finite entries admit rational approximation. Polygonal paths with
  rational breakpoints and such matrices give separability of $\mathscr{B}_0$.
  To approximate breakpoints, use uniform convergence, uniform Lipschitz bounds
  and $[f]_\theta \le (2\|f\|_\infty)^{1-\theta}\operatorname{Lip}(f)^\theta$.
  The same inequality puts every zero-initial Lipschitz compact-valued path in
  $\mathscr{B}_0$ by polygonal approximation. The finite original coefficient
  paths $iA_jL_\psi$ have these properties. Their Gaussian sums converge in
  $L^pC^\theta$ and hence in the closed space $\mathscr{B}_0$. Their laws and
  the parent Gaussian norm bound give~\eqref{E:raw-block-moments}. Every finite
  joint coefficient list with $Z$ is Gaussian on the same $W$. Scalar continuous
  functionals pass through the $L^2$ limits, preserving Gaussianity by
  convergence of variances and characteristic functions. Thus, $(X,Z)$ is the
  asserted Banach Gaussian variable. Cauchy--Schwarz gives $\E\|XZ\| \le
  \|X\|_2\sqrt v < \infty$, so its Bochner covariance exists.

  The original covariance and the test pullback give
  \begin{equation}\label{E:raw-block-mean-jacobian}
    \E[W_{\delta,m}(f)Z]
    =\delta^{a/2}\int f(u,x)h_{s_m+\delta u}(x)\ud x\ud u.
  \end{equation}
  Absolute Fubini uses compact supports, local Radon mass and $a < 1$. This
  identity extends to the explicit matrix-element tests for the operator path on
  $E$: their time indicator is approximated by smooth bounded tests on one
  compact support, converging in the original covariance norm by dominated
  convergence against $|u-r|^{-a}\Gamma$; their mean pairings converge as well.
  The Sobolev adjoint is the one in~\eqref{E:driver-covariance-row}. Hence, the
  path covariance is $g_\delta\delta^{a/2}H_m = \sqrt\delta\,H_m$ on real
  Schwartz $E$ pairs. For constant inputs the two columns are the same $L_\psi$
  inputs $(\psi,0)$ and $(0,\psi)$, and total finite propagation identifies
  their deterministic response. The span of $E$ and both constant coordinates is
  dense in $F$; $E$ alone is not. Boundedness, rational times and path
  continuity prove~\eqref{E:raw-block-covariance} on all $F$. In particular, $H
  = b/\sqrt\delta \in \mathscr{B}_\delta$. This proves
  Lemma~\ref{L:raw-block-inputs}.
\end{proof}

The next identity is the covariance-direction Cameron--Martin formula;
see Janson~\cite{janson:97:gaussian}*{Theorem~14.1(v), pp.~216--217}.
We prove it on the norm-Borel space used here to verify the measurable
realization and its deterministic translate.

\begin{proposition}[Change of law on the whole norm-Borel space]
  \label{P:raw-block-gaussian-law}
  For every nonnegative norm-Borel $j$ on all of $\mathscr{B}_\delta$,
  \begin{equation}\label{E:raw-block-gaussian-law}
    \E j(X)=
    \E\left[e^{-Z/\sqrt\delta-v/(2\delta)}j(X+H)\right].
  \end{equation}
  The equality is valid also as an extended nonnegative integral.
\end{proposition}

\begin{proof}
  For real $s$, the density $e^{-sZ-s^2v/2}$ defines an equivalent probability
  measure $Q_s$ by completing the square in the $N(0,v)$ density. For any finite
  list $\ell$ of continuous real coordinates on $\mathscr{B}_\delta$, the
  Gaussian vector $(\ell(X),Z)$ is equal in law to $(TG,z\cdot G)$ for a finite
  standard Gaussian vector $G$. A positive covariance square root gives this
  representation even when singular; no inverse is used. Here $Tz = \ell(b)$ and
  $|z|^2 = v$. Substitution $G' = G+sz$ in its density gives
  $\E_{Q_s}f(\ell(X+sb)) = \E f(\ell(X))$ for every bounded Borel $f$.

  The matrix entries of each block in a real Hilbert basis at rational times,
  together with $0,B$, generate the norm-Borel sigma-field. To see the
  operator-norm point, for a countable dense set $\mathcal{D}$ of unit vectors
  obtained from normalized rational finite basis combinations, $\|T\| =
  \sup_{u,w\in\mathcal{D}}|\langle Tu,w\rangle|$. These pairings are finite
  linear combinations of matrix entries. Supremum and H\"older norms of
  continuous paths are suprema over rational times and pairs. Thus, distance to
  every fixed element of $\mathscr{B}_\delta$ is coordinate-measurable. Open
  balls are measurable by rational closed-ball approximation, and separability
  makes every open set a countable union of balls from dense centers. Conversely
  the coordinates are continuous. The cylinder sets form a generating pi-system;
  uniqueness of probability measures extends the finite-coordinate equality to
  all norm-Borel sets. Simple functions and monotone convergence give the
  identity for nonnegative $j$. Finally, take $s = \delta^{-1/2}$ and use $sb =
  H$ from~\eqref{E:raw-block-covariance}. The map and its deterministic
  translate are defined on the entire input space; there is no translated
  exceptional-domain assumption. This proves
  Proposition~\ref{P:raw-block-gaussian-law}.
\end{proof}

The identity retains the random shifted input $X+H$. Replacing its
expected contribution by evaluation at the deterministic mean can lose
the exponential constant, even in a scalar Gaussian calculation; see
No-go~\ref{P:nogo-16}. Positive semidefinite residual covariance does not
justify discarding that residual: No-go~\ref{P:nogo-19} gives an admissible
parent model whose rank-one projection has a larger small-coupling wave mean.


\subsection{Small noisy blocks and a cone in the displacement-velocity space}

For any $x \in \mathscr{B}_0$ define $D_x(u) = S_F(u)U_x(u,0)$ by the
everywhere-defined linear Young map of
Proposition~\ref{P:constant-young-linear}. Write $D_m^0 = D_{X_m}$ and $D_m^+ =
D_{X_m+H_m}$. The latter retains both the mean and the residual noise.

The shifted blocks $D_m^+$ are random. We now show that their largest distance
from the deterministic $P_{s_m}$ of~\eqref{E:compact-F-block}, over all blocks
$m \le N$, tends to zero in every finite $L^q$.

\begin{lemma}[Uniform block error]\label{L:raw-block-uniform-error}
  Let $M_\delta = \max_{m\le N}\|X_m\|_{\mathscr{B}_0}$, including its factor
  $g_\delta$. Then $M_\delta \to 0$ in every finite $L^q$, and, for some $c >
  0$,
  \begin{equation}\label{E:raw-block-maximum-exponential}
    \sup_\delta\E e^{cM_\delta^2}<\infty.
  \end{equation}
  With the deterministic $P_s$ in~\eqref{E:compact-F-block},
  \begin{equation}\label{E:raw-block-operator-error}
    E_{\mathrm{op},\delta}:=
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m^+(u)-P_{s_m}(u)\|_{\mathcal{B}(F)}
    \longrightarrow0
    \quad\hbox{in every finite }L^q.
  \end{equation}
\end{lemma}

\begin{proof}
  The maximum and~\eqref{E:raw-block-moments}, with $N+1 \le 2/\delta$, give
  \[
    \|M_\delta\|_p
    \le (N+1)^{1/p}C\sqrt p\,g_\delta
    \le C'\sqrt p\,\delta^{(1-a)/2-1/p}.
  \]
  For fixed $q$, choose $p > \max(q,2/(1-a))$ to prove the first limit. No
  independence is needed. Choose an integer $n_0$ with $n_0(1-a) \ge 1$. For $n
  \ge n_0$,
  \[
    \E M_\delta^{2n}
    \le2C^{2n}(2n)^n\delta^{n(1-a)-1}
    \le2C^{2n}(2n)^n.
  \]
  For $1 \le n < n_0$, Lyapunov bounds the moment by the appropriate power of
  the $2n_0$ moment. These finitely many terms are bounded. For the remaining
  exponential series, $n! \ge (n/e)^n$ gives a geometric tail $2\sum_{n\ge
  n_0}(2ecC^2)^n$ for small $c$. This
  proves~\eqref{E:raw-block-maximum-exponential}; no such assertion is made for
  the unscaled maximum.

  Let $H_m^{\mathrm{stat}}$ be the primitive for constant potential $V_{s_m}$.
  Compact mean smoothness on $[0,2]$ and the statewise primitive bound give
  \[
    \sup_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \sup_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C\delta.
  \]
  Indeed, the potential difference is at most $C\delta B$ in $L^\infty$ and
  $L^2$; integration of the bounded free groups gives its Lipschitz primitive
  bound on this fixed horizon. On $\{M_\delta\le1\}$, local Young stability on
  the fixed ball of radius $H_0+1$ gives $E_{\mathrm{op},\delta} \le
  C(M_\delta+\delta)$. This proves convergence in probability. The global Young
  bound also gives
  \[
    \max_m\sup_{u\le B}\|D_m^+(u)\|
    \le C_B\exp\big(C_B(1+H_0+M_\delta)^{1/\theta}\big).
  \]
  Since $1/\theta < 2$, every finite power of this bound is dominated by a
  constant times the exponential in~\eqref{E:raw-block-maximum-exponential}.
  Uniform higher moments yield uniform integrability, proving
  \eqref{E:raw-block-operator-error}. Suprema are measurable: on a countable
  dense set of unit states each difference is continuous in $u$, so rational
  times and those states suffice. Operator-norm continuity of the physical free
  group is not used. Only $s_m \le 1$ supplies the ground templates. This proves
  Lemma~\ref{L:raw-block-uniform-error}.
\end{proof}

Lemma~\ref{L:raw-block-uniform-error} bounds the distance of each shifted block
from $P_{s_m}$. It says nothing about the product of the blocks. The next lemma
is deterministic and holds for every $N \ge 1$. If each factor $D_m$ is within
$\epsilon_{\mathrm{op}}$ of $P_{\sigma_m}$, and adjacent indices $\sigma_m$ are
within $\omega$, then the product satisfies the lower bound
\eqref{E:raw-block-positive-product}.

\LeanMarker{L:raw-block-indexed-cone}
\begin{lemma}[Indexed cone and terminal displacement]
  \label{L:raw-block-indexed-cone}
  Fix $0 < \varrho < 1/2$. There are an integer $B \ge 1$ and positive
  $\omega,\epsilon_{\mathrm{op}},c_*$ such that the following holds for every $N
  \ge 1$. If $\sigma_0,\ldots,\sigma_N \in I$ have adjacent distances at most
  $\omega$ and real bounded operator families $D_m(u)$, $0 \le u \le B$, satisfy
  \[
    \max_{m\le N}\sup_{0\le u\le B}
    \|D_m(u)-P_{\sigma_m}(u)\|\le\epsilon_{\mathrm{op}},
  \]
  then, for $0 \le u \le B$,
  \begin{equation}\label{E:raw-block-positive-product}
    O_\phi D_N(u)D_{N-1}(B)\cdots D_0(B)e_0
    \ge c_*(1-\varrho)^{N-1}
    \exp\left(B\sum_{m<N}\lambda_{\sigma_m}
    +u\lambda_{\sigma_N}\right)>0.
  \end{equation}
  No positivity, composition or independence assumption on the $D_m$ is
  required.
\end{lemma}

\begin{proof}
  From~\eqref{E:compact-normalized-complement} fix $C_{\mathrm{gap}} \ge 1$,
  $\eta_{\mathrm{gap}} > 0$ so the normalized complementary norm is at most
  $C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}t}$. Define positive margins and
  finite bounds
  \[
    a_*=\inf_I\ell_s(e_0),\quad j_*=\inf_I O_\phi(e_s),\quad
    M_e=\max(1,\sup_I\|e_s\|),\quad
    M_\ell=\max(1,\sup_I\|\ell_s\|).
  \]
  Also put $M_P = \max(1,\sup_I\|\Pi_s\|)$, $M_Q = 1+M_P$, $M_z = M_e+1$, $E_0 =
  \|e_0\| = 1$ and $H_{\mathrm{obs}} = \max(1,\|O_\phi\|)$. Choose the aperture
  \emph{first}: $\vartheta = \min(1,j_*/(16H_{\mathrm{obs}}C_{\mathrm{gap}})) >
  0$. Next choose integer $B \ge 1$ with
  \[
    q_B:=C_{\mathrm{gap}}e^{-\eta_{\mathrm{gap}}B}
    \le\min\left(\frac18,\frac{\vartheta a_*}{8M_QE_0}\right).
  \]
  Choose $\nu > 0$ such that $\nu M_z \le \min(\varrho/2,\vartheta,1/4)$.
  Uniform continuity of $\ell_s,\Pi_s$ supplies $\omega > 0$ for which adjacent
  distances at most $\omega$ give both norm differences at most $\nu$. Finally,
  choose $\epsilon_{\mathrm{op}} > 0$ satisfying all five smallness conditions
  \begin{align*}
    M_\ell\epsilon_{\mathrm{op}}M_z&\le\varrho/2,&
    M_Q\epsilon_{\mathrm{op}}M_z&\le\vartheta/4,\\
    M_\ell\epsilon_{\mathrm{op}}E_0&\le a_*/2,&
    M_Q\epsilon_{\mathrm{op}}E_0&\le\vartheta a_*/8,&
    H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z&\le j_*/8.
  \end{align*}
  All denominators are positive. This order of choices uses the normalized
  complement rather than a growing terminal norm.

  The cone at $s$ consists of $z = Ae_s+r$, with $A = \ell_s(z) > 0$, $\ell_s(r)
  = 0$ and $\|r\|_F \le \vartheta A$. In particular, $r = Q_sz$ and $\|z\| \le
  M_zA$. For entrance, $z_1 = D_0(B)e_0$, with $b = \sigma_0$, has normalized
  amplitude at least $a_*-M_\ell\epsilon_{\mathrm{op}}E_0 \ge a_*/2$. Its
  normalized residual is at most $q_BM_QE_0+M_Q\epsilon_{\mathrm{op}}E_0 \le
  \vartheta a_*/4$. Thus, it enters the cone with residual ratio at most
  $\vartheta/2$ and amplitude at least $e^{\lambda_bB}a_*/2$.

  For an old-cone state $z = Ae_b+r$ and an adjacent index $c$, rebase before
  using the complementary estimate:
  \[
    \ell_c(z)\ge(1-\nu M_z)A,\qquad
    \|Q_cz\|\le\|Q_bz\|+\|(\Pi_b-\Pi_c)z\|
    \le(\vartheta+\nu M_z)A.
  \]
  For the next full block $D$ close to $P_c(B)$, put $x =
  e^{-\lambda_cB}\ell_c(Dz)$ and $y = e^{-\lambda_cB}\|Q_cDz\|$. Since
  $\lambda_c > 0$, the perturbation and smallness conditions give
  \[
    x\ge(1-\nu M_z-M_\ell\epsilon_{\mathrm{op}}M_z)A
    \ge(1-\varrho)A>A/2,
  \]
  and
  \[
    y\le
    [q_B(\vartheta+\nu M_z)+M_Q\epsilon_{\mathrm{op}}M_z]A
    \le\vartheta A/2.
  \]
  The last inequality uses $q_B \le 1/8$, $0 \le \nu M_z \le \vartheta$ and
  $M_Q\epsilon_{\mathrm{op}}M_z \le \vartheta/4$. Together with $\varrho < 1/2$,
  these give $y \le \vartheta x$. These are precisely the finite residual and
  cone-conversion margins; the old residual was not assumed to lie in the new
  kernel. Each full block therefore preserves the new cone and multiplies its
  amplitude by at least $e^{\lambda_cB}(1-\varrho)$.

  For the terminal $0 \le u \le B$, rebase once more to $c = \sigma_N$. The same
  bounds give $\ell_c(z) \ge 3A/4$ and $\|Q_cz\| \le 2\vartheta A$.
  Consequently,
  \[
    e^{-\lambda_cu}O_\phi(P_c(u)z)/A
    \ge3j_*/4-2H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta.
  \]
  The perturbation changes this by at most
  $H_{\mathrm{obs}}\epsilon_{\mathrm{op}}M_z \le j_*/8$. Since
  $16H_{\mathrm{obs}}C_{\mathrm{gap}}\vartheta \le j_*$, the result is at least
  $j_*/2$. Combine entrance, all $N-1$ intermediate full blocks and this
  terminal estimate to obtain~\eqref{E:raw-block-positive-product}, with $c_* =
  a_*j_*/4$. The induction is empty for $N = 1$ and the estimate includes $u =
  0$. This proves Lemma~\ref{L:raw-block-indexed-cone}.
\end{proof}

\subsection{Source restart and comparison with the original point}

So far the blocks $D_m^0$ and $D_m^+$ are defined one at a time, from the local
operator paths $X_m$ and $X_m+H_m$. We now show that each family forms the
successive pieces of its corresponding global flow, and that its finite
products are source limits in $L^2$ at fixed $\delta,N$. The proof of
Proposition~\ref{P:raw-block-point-comparison} uses this to verify the
source-limit premise of the masked comparison.

\LeanMarker{L:raw-block-restart}
\begin{lemma}[Composition over time blocks and regularized products]
  \label{L:raw-block-restart}
  Fix $\delta,B,N$ and let $L = \delta^{-1}$, $L_{\mathrm{ext}} = (N+1)B \ge L$.
  On this global fast horizon define $A_{\mathrm{glob}} = g_\delta
  A_R(W_{\mathrm{fast}})$. Its physical flow obeys
  \begin{equation}\label{E:raw-block-restart}
    P_{\mathrm{glob}}(mB+u,mB)=D_m^0(u),\qquad 0\le u\le B.
  \end{equation}
  The analogous identity holds with the deterministic mean $V_{\delta t}$ added
  globally and local operator paths $X_m+H_m$. Their finite block products,
  including the last remainder $\tau$, are source limits in $L^2$, at this fixed
  $\delta,N$.
\end{lemma}

\begin{proof}
  Take one regularized slow noise preserving the given noise $Y_{\epsilon,\eta}$
  on the extension of the given $W$ constructed in
  Section~\ref{S:source-driver-convergence}. Define the fast and local noises by
  \[
    Y_{\mathrm{fast}}(f)=Y_{\epsilon,\eta}(T_{\delta,0}f),\qquad
    Y_m(f)=Y_{\epsilon,\eta}(T_{\delta,m}f).
  \]
  The pullback isometry and~\eqref{E:source-two-dominations} give source and
  residual variance bounds by $\|f\|_{H_W}^2$. The residual tends to zero on
  each transformed smooth compact test at fixed $\delta,m$. Joint Gaussianity is
  retained. Proposition~\ref{P:source-gaussian-transfer} therefore gives the
  $L^pC^\theta$ operator-path limits on $[0,L_{\mathrm{ext}}]$ and on $[0,B]$,
  including both constant columns.

  Their random fields are $Y_{\mathrm{fast}}(u,x) =
  \delta^{a/2}Y_{\epsilon,\eta}(\delta u,x)$ and $Y_m(u,x) =
  Y_{\mathrm{fast}}(mB+u,x)$. Indeed, these follow by change of variables in
  their test integrals. The temporal law is $K_{\eta/\delta}$ because
  $\delta^aK_\eta(\delta r) = K_{\eta/\delta}(r)$; the spatial heat law and all
  joint width rules are unchanged. They are continuous in time and smooth in
  space. Their ordinary primitives equal their Gaussian operator paths: their
  matrix entries are $Y_m(h_R)$, by the indicator approximation used
  in~\eqref{E:raw-block-mean-jacobian} and the point-test integral identity.
  Total finite propagation extends this identity to both constant columns. This
  does not assume a separately chosen extension preserving the given noise
  automatically commutes with time dilation.

  Changing the ordinary source integration variable and composing the free
  groups gives the exact identity on $F$
  \[
    S_F(mB)\big[A_{\mathrm{glob}}^{\mathrm{src}}(mB+u)
    -A_{\mathrm{glob}}^{\mathrm{src}}(mB)\big]
    S_F(-mB)=X_m^{\mathrm{src}}(u).
  \]
  No physical mask is commuted through a wave group. Restriction, difference and
  conjugation are bounded maps on these fixed-horizon $C^\theta$ spaces. Passing
  the source limits gives the same identity for $A_{\mathrm{glob}},X_m$. For the
  finitely many $m$, a convergent subsequence or rational times and path
  continuity gives simultaneous identities in $u$. Conjugation norms need only
  be finite at this fixed $\delta$.

  Conjugate the global interaction Young equation on $[mB,mB+u]$ by $S_F(mB)$.
  Bounded linearity of its Riemann sums and that original identity give the
  local equation driven by $X_m$. Uniqueness identifies it with $U_{X_m}$,
  proving \eqref{E:raw-block-restart}. The ordinary deterministic primitive for
  $V_{\delta t}$ has the identical cocycle with local $H_m$. Composition now
  identifies the full product, with terminal $\tau$.

  For completeness, source convergence of these finite products requires no
  growing-product stability assertion. At fixed $\delta,N$ let $C \ge 1$ be the
  maximum norm of their source and original factors, and let $e_{\mathrm{src}}$
  be the maximum factor difference. The parent Young convergence gives every
  finite moment of $C$ uniformly in source widths and $e_{\mathrm{src}} \to 0$
  in every finite $L^p$. If $d(n)$ is the difference of the first $n$ ordered
  products, then
  \[
    d(n+1)\le C d(n)+C^n e_{\mathrm{src}},\qquad d(0)=0.
  \]
  Extend both finite sequences by identities so the recurrence holds for all
  natural $n$. The finite error-accumulation estimate gives $d(N+1) \le
  (N+1)C^{N+1}e_{\mathrm{src}}$. H\"older, using $C$ in $L^{4(N+1)}$ and the
  error in $L^4$, proves $L^2$ convergence. This is only a fixed-$\delta$ source
  limit. Shifted source products may use the fixed original mean plus the
  regularized noise; no finite-source Cameron mean is substituted. This proves
  Lemma~\ref{L:raw-block-restart}.
\end{proof}

\begin{proposition}[The square observable and canonical comparison]
  \label{P:raw-block-point-comparison}
  Define the signed averages without and with the deterministic mean
  \[
    U_{\delta,B}
    =O_\phi D_N^0(\tau)D_{N-1}^0(B)\cdots D_0^0(B)e_0,
    \qquad
    Y_{\delta,B}
    =O_\phi D_N^+(\tau)D_{N-1}^+(B)\cdots D_0^+(B)e_0.
  \]
  They have all finite unweighted moments, and
  \begin{align}
    \E U_{\delta,B}^2
    &=\E\left[e^{-Z/\sqrt\delta-v/(2\delta)}
    Y_{\delta,B}^2\right],
    \label{E:raw-block-square-identity}\\
    \E U_{\delta,B}^2
    &\le\E|V_\delta(1,0)|^2
    =\E|u(T,0)|^2,\qquad \delta=T^{-\kappa}.
    \label{E:raw-block-original-comparison}
  \end{align}
  The variance $v$ in the density is the original slow original variance.
\end{proposition}

\begin{proof}
  On all of $\mathscr{B}_\delta$ put
  \[
    j_\delta(x)=
    |O_\phi D_{x_N}(\tau)D_{x_{N-1}}(B)\cdots D_{x_0}(B)e_0|^2.
  \]
  The everywhere-defined continuous Young map, fixed-time bounded free groups,
  finite products and bounded $O_\phi$ make $j_\delta$ continuous and
  nonnegative on the whole input space. Apply
  Proposition~\ref{P:raw-block-gaussian-law} directly to these original paths.
  Since $X+H$ is its exact translate, this proves
  \eqref{E:raw-block-square-identity} with the single original $Z = W(\Psi)$ and
  its variance $v$. Finite products of the local Young flows have every finite
  unweighted moment by their Gaussian path bounds and H\"older, at fixed
  $\delta,N$. The Gaussian identity itself did not require prior integrability
  of $j_\delta$. No source convergence under its exponential density is used.

  To apply the masked comparison at the amplitude, put $\Gamma_\delta =
  g_\delta^2\Gamma$ and $W' = g_\delta W_{\mathrm{fast}}$. This is a nonzero
  member of the same admissible covariance class, with unchanged degree $\alpha$
  and temporal parameter $a$. Its integrated operator is $g_\delta$ times the
  original one: the original matrix forms are linear in noise, and
  basis-independent Gaussian transfer and the bounded $F$ extension retain both
  constant columns. The $n$th source chain is multiplied by $g_\delta^n$. Thus,
  this is exactly the model without the deterministic mean at that amplitude.

  We verify the source-limit premise of the masked comparison. Keep $\delta,B,N$
  fixed and use the single regularized noise preserving the given noise of
  Lemma~\ref{L:raw-block-restart}. For $m < N$ let $F_m = D_m^0(B)$ and put $F_N
  = D_N^0(\tau)$; write $F_{m,\rho}$ for the corresponding ordinary source
  factors, where both source widths in $\rho$ tend to zero.
  Lemma~\ref{L:raw-block-restart} and Proposition~\ref{P:constant-gaussian-flow}
  give factor convergence in every finite $L^q$ and uniform bounds for every
  finite moment of their norms. These statements are on the fixed finite
  horizons $[0,(N+1)B]$ and $[0,B]$, and include both constant-response columns.
  Set $n = N+1$ and
  \[
    C_\rho=\max\left(1,\max_{m\le N}\|F_{m,\rho}\|,
    \max_{m\le N}\|F_m\|\right),\qquad
    e_\rho=\max_{m\le N}\|F_{m,\rho}-F_m\|.
  \]
  Then $\sup_\rho\|C_\rho\|_q < \infty$ for every finite $q$, and $\|e_\rho\|_q
  \to 0$. Telescoping the ordered products gives
  \[
    \|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\|
    \le n C_\rho^n e_\rho.
  \]
  Therefore, H\"older with exponents $4n$ and $4$ gives
  \[
    \big\|F_{N,\rho}\cdots F_{0,\rho}-F_N\cdots F_0\big\|_{L^2}
    \le n\|C_\rho\|_{L^{4n}}^n\|e_\rho\|_{L^4}
    \longrightarrow0.
  \]
  No uniform estimate in the growing number $N$ is needed: this source limit is
  completed at each fixed $\delta,N$. The physical source identification in
  Lemma~\ref{L:constant-source-solution}, with the same $m_R$ and initial state
  $e_0$, identifies the source product with the tested ordinary masked solution
  at fast time $L = 1/\delta$. Boundedness of $O_\phi$ and $\|e_0\| = 1$ thus
  identify $U_{\delta,B}$ as its $L^2$ limit. The sources $g_\delta
  Y_{\mathrm{fast}}$ have the correct law, contraction and original-test limit
  by Lemma~\ref{L:raw-block-restart}.

  At positive source widths, $0 \le m_R \le 1$ multiplies each nonnegative
  displacement Wick integrand by the product of its vertex masks. Positive Green
  measures and covariance retain this comparison for unequal, self and empty
  chains as well. Proposition~\ref{P:masked-source-moments} gives the tested
  comparison, and the just-proved $L^2$ convergence discharges the premise of
  Proposition~\ref{P:masked-canonical-transfer} at $\Gamma_\delta$. Its
  canonical unmasked moment limit uses the fixed-order joint source limits and
  the absolutely summable order bounds. Since $\phi \ge 0$ and $\int\phi = 1$,
  these facts give
  \[
    \E U_{\delta,B}^2
    \le\E|u_{\Gamma_\delta}(L,0)|^2.
  \]
  Only the source products without the deterministic mean were needed for this
  passage. The shifted original product is already the everywhere-defined
  $j_\delta(X+H)$ in the Gaussian identity; no tilted source limit or change of
  the original slow $\Psi,Z,v$ has been made.

  The ordinary change from fast $u$ to slow $t = \delta u$, with phase
  $(V,\delta V_t)$, transforms the unmasked source equation into the regularized
  version of~\eqref{E:exact-fast-equation}, with slow terminal time one. Indeed,
  $g_\delta Y_{\mathrm{fast}}(u) = \sqrt\delta\,Y_{\epsilon,\eta}(\delta u)$,
  and its fast cutoff is $\eta/\delta$. The canonical fixed-order and absolutely
  convergent series construction passes this law identity at fixed $\delta$ to
  the original moment, preserving every mixed, unequal, self and empty chain.
  Lemma~\ref{L:exact-canonical-scaling} gives the final equality
  in~\eqref{E:raw-block-original-comparison}. The physical scale $r_T = T\delta$
  is distinct from the fixed mask radius $R$. The mask need not contain the
  growing backward cone: the comparison is a moment inequality, not equality of
  masked and full fields. All source limits preceded this large-time comparison.
  This proves Proposition~\ref{P:raw-block-point-comparison}.
\end{proof}

Proposition~\ref{P:raw-block-point-comparison} bounds $\E|u(T,0)|^2$ from below
by the weighted second moment of $Y_{\delta,B}$
in~\eqref{E:raw-block-square-identity}. We still need a lower bound for
$Y_{\delta,B}$ itself. The proof below obtains one on an event whose probability
tends to one. It uses the estimates of Lemma~\ref{L:raw-block-uniform-error} and
the cone bound of Lemma~\ref{L:raw-block-indexed-cone}.

\LeanMarker{P:raw-block-original-lower}
\begin{proposition}[Original point lower rate at the Gaussian quadratic cost]
  \label{P:raw-block-original-lower}
  For every original positive $\Psi$ and eligible fixed mask $R$ chosen above,
  and every fixed $x \in \R^d$,
  \begin{equation}\label{E:raw-block-original-lower}
    \liminf_{T\to\infty}T^{-\kappa}\log\E|u(T,x)|^2
    \ge 2\int_0^1\lambda_s\ud s-\frac{v_\Psi}{2}.
  \end{equation}
\end{proposition}

\begin{proof}
  Fix $e_{\mathrm{rate}} > 0$. Choose $0 < \varrho < 1/2$ with $\log(1-\varrho)
  > -e_{\mathrm{rate}}/3$, then choose the cone constants, including an integer
  $B$ fixed independently of $\delta$, from
  Lemma~\ref{L:raw-block-indexed-cone}. The control $\Psi$ and eligible mask $R$
  remain fixed throughout.

  Choose a finite $p > 2/(1-a)$ and put $\gamma_p = p(1-a)/2-1 > 0$.
  From~\eqref{E:raw-block-moments} and $N+1 \le 2/\delta$, Markov and the union
  bound give, for $\eta > 0$,
  \[
    \begin{split}
      \mathbb{P}(M_\delta>\eta)
      &\le\eta^{-p}\sum_{m=0}^N
      \E\|X_m\|_{C^\theta}^p\\
      &\le A_p\eta^{-p}\delta^{\gamma_p},\qquad
      A_p=2(C_{R,B}\sqrt p)^p<\infty.
    \end{split}
  \]
  This estimate concerns the coupled blocks; no independence is needed. At these
  fixed data $A_p$ is independent of $\delta$.

  We also specify the deterministic tolerance used with this bound. The
  primitive estimates in Lemma~\ref{L:raw-block-uniform-error} give, for the
  constant-potential primitives $H_m^{\mathrm{stat}}$,
  \[
    \max_m\max(\|H_m\|_{C^\theta},
    \|H_m^{\mathrm{stat}}\|_{C^\theta})\le H_0,\qquad
    \max_m\|H_m-H_m^{\mathrm{stat}}\|_{C^\theta}\le C_B\delta.
  \]
  Apply Proposition~\ref{P:constant-young-linear} on the fixed ball of operator
  paths of radius $H_0+1$ and use the bounded free groups on $[0,B]$. There is
  $L_B \ge 1$, depending on these fixed $R,B,\Psi$ and the admissible exponents,
  such that on $\{M_\delta\le1\}$,
  \[
    E_{\mathrm{op},\delta}\le L_B(M_\delta+\delta).
  \]
  Set $\eta = \min(1,\epsilon_{\mathrm{op}}/(2L_B)) > 0$. If $\delta \le
  \epsilon_{\mathrm{op}}/(2L_B)$, then $M_\delta \le \eta$ implies
  $E_{\mathrm{op},\delta} \le \epsilon_{\mathrm{op}}$. Thus, for the measurable
  event on the original common space
  \[
    E_\delta=\{E_{\mathrm{op},\delta}\le\epsilon_{\mathrm{op}}\}
    \cap\{|Z|\le\delta^{-1/4}\},\qquad
    p_\delta=\mathbb{P}(E_\delta),
  \]
  Chebyshev for the same single $Z = W(\Psi)$, with $\E Z^2 = v$, gives
  \[
    p_\delta\ge1-A_p\eta^{-p}\delta^{\gamma_p}
    -v\delta^{1/2}\longrightarrow1.
  \]
  This is one union bound, with no independence between blocks or from $Z$ and
  no product of block probabilities.

  Let $C_\lambda$ be a Lipschitz bound for $s\mapsto\lambda_s$. Take $\delta$
  small enough, in addition to $0 < \delta < \min(1,1/B)$, that
  \[
    \begin{gathered}
      \delta B\le\omega,\qquad
      \delta\le\epsilon_{\mathrm{op}}/(2L_B),\qquad
      C_\lambda\delta B\le e_{\mathrm{rate}}/3,\\
      \delta|\log c_*|\le e_{\mathrm{rate}}/3,\qquad
      A_p\eta^{-p}\delta^{\gamma_p}+v\delta^{1/2}\le1/2.
    \end{gathered}
  \]
  Every constant has been fixed before this choice. The condition involving
  $C_\lambda$ is vacuous if that constant is zero. All source limits in
  Proposition~\ref{P:raw-block-point-comparison} have already been completed at
  each such fixed $\delta$.

  The indices $s_m = m\delta B$ lie in $I$ with adjacent gaps at most $\omega$.
  On $E_\delta$ the shifted full-$F$ blocks therefore satisfy
  Lemma~\ref{L:raw-block-indexed-cone}, retaining both phase entries and the
  final remainder $\tau$. In particular, $Y_{\delta,B} > 0$, and
  \[
    \delta\log Y_{\delta,B}
    \ge\delta\log c_*+\delta(N-1)\log(1-\varrho)
    +\delta B\sum_{m<N}\lambda_{s_m}
    +\delta\tau\lambda_{s_N}.
  \]
  Since $s_N+\delta\tau = 1$, the last two terms are the left Riemann sum for
  $I_R$ on the entire interval, including its last interval of length
  $\delta\tau$, with mesh at most $\delta B$. Its error is at most
  $C_\lambda\delta B$. Also $0 \le \delta(N-1) \le 1/B \le 1$, so the
  logarithmic loss is at least $\log(1-\varrho) > -e_{\mathrm{rate}}/3$. The two
  other losses are each at most $e_{\mathrm{rate}}/3$ by the choices above.
  Hence,
  \[
    Y_{\delta,B}\ge
    \exp((I_R-e_{\mathrm{rate}})/\delta)\quad\hbox{on }E_\delta,
  \]
  including the $N = 1$ and $\tau = 0$ cases of the cone estimate.

  The integrand in~\eqref{E:raw-block-square-identity} is nonnegative. On the
  same $E_\delta$ its linear Gaussian term satisfies $-Z/\sqrt\delta \ge
  -\delta^{-3/4}$. Restriction to this event
  and~\eqref{E:raw-block-original-comparison} therefore give the explicit bound
  for the original point moment
  \[
    \E|u(T,0)|^2\ge\E U_{\delta,B}^2
    \ge p_\delta\exp\left(
    \frac{2I_R-2e_{\mathrm{rate}}-v/2}{\delta}
    -\delta^{-3/4}\right),\qquad \delta=T^{-\kappa}.
  \]
  Here $p_\delta \ge 1/2$ for small $\delta$, so $-\delta\log2 \le \delta\log
  p_\delta \le 0$. Multiplying the logarithm of the preceding inequality by
  $\delta$ yields
  \[
    \delta\log\E|u(T,0)|^2
    \ge2I_R-2e_{\mathrm{rate}}-v/2
    -\delta^{1/4}+\delta\log p_\delta.
  \]
  Thus, the normalized linear penalty is $\delta\delta^{-3/4} = \delta^{1/4} \to
  0$, and $\delta\log p_\delta \to 0$. The quadratic Gaussian cost is precisely
  half the original variance $v$. Since $\kappa > 0$, $\delta = T^{-\kappa}$ is
  a bijection between all sufficiently large $T$ and all sufficiently small
  $\delta$. This gives the full continuous-time liminf $Q_2 \ge
  2I_R-2e_{\mathrm{rate}}-v/2$. For each $e_{\mathrm{rate}}$ the argument first
  fixed its own $B = B(e_{\mathrm{rate}})$ and then took $\delta\downarrow0$.
  The original point rate $Q_2$ is independent of $B$; only now let
  $e_{\mathrm{rate}}\downarrow0$. No choice $B = B(\delta)$ is made.

  Finally, spatial translation of all positive-width vertices preserves the
  original noise law, Green kernels and constant initial data. The fixed-order
  $L^2$ and absolute-series limits preserve equality of point second moments.
  Hence, zero can be replaced by every fixed $x$
  in~\eqref{E:raw-block-original-lower}. Every control and mask were fixed
  before asymptotics; a supremum over them may follow their individual
  inequalities. This proves Proposition~\ref{P:raw-block-original-lower}.
\end{proof}
