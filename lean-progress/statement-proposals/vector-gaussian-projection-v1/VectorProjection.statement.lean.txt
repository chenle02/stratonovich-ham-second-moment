import GaussianChaos.ChaosDensity

set_option autoImplicit false

noncomputable section
namespace StratonovichHAMAudit.VectorProjection

open GaussianChaos.GaussianProduct GaussianChaos.ChaosDensity
open MeasureTheory

universe u v w

variable {Mode : Type u} {E : Type v}
variable [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]

/-- An actual vector-valued Gaussian L2 element, not an arbitrary polynomial representative. -/
def atom (ν : Mode →₀ ℕ) (x : E) : Lp E 2 (noise Mode) :=
  (ContinuousLinearMap.toSpanSingleton ℝ x).compLp (hermiteNormalized ν)

/-- The finite vector-valued Hermite polynomial in the actual probability L2 space. -/
def realize (f : (Mode →₀ ℕ) →₀ E) : Lp E 2 (noise Mode) :=
  f.sum (fun ν x => atom ν x)

/-- The closed L2 subspace generated by all selected modes with arbitrary vector coefficients. -/
def cutoffSpace (p : (Mode →₀ ℕ) → Prop) : ClosedSubmodule ℝ (Lp E 2 (noise Mode)) :=
  (Submodule.span ℝ {z | ∃ (ν : Mode →₀ ℕ) (x : E), p ν ∧ z = atom ν x}).closure

/-- Coefficient deletion is a definition; its identification with orthogonal projection is the target. -/
def cut (p : (Mode →₀ ℕ) → Prop) (f : (Mode →₀ ℕ) →₀ E) : (Mode →₀ ℕ) →₀ E := by
  classical
  exact f.filter p

/-- Each coordinate is a Gaussian mode. The first component labels its color. -/
def colorDegree {Color : Type w} {Index : Type u} [DecidableEq Color]
    (ν : (Color × Index) →₀ ℕ) (c : Color) : ℕ :=
  ν.sum (fun ci n => if ci.1 = c then n else 0)

def colorLow {Color : Type w} {Index : Type u} [DecidableEq Color]
    (N : Color → ℕ) (ν : (Color × Index) →₀ ℕ) : Prop :=
  ∀ c, colorDegree ν c ≤ N c

-- PROPOSED CLOSED PROP, TYPECHECK ONLY. No theorem/proof of projection equality supplied.
#check (∀ {Mode : Type u} {E : Type v}
    [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
    (p : (Mode →₀ ℕ) → Prop) (f : (Mode →₀ ℕ) →₀ E),
    (cutoffSpace (E := E) p).toSubmodule.starProjection (realize f) = realize (cut p f))

-- The manuscript's per-color total-degree predicate is a literal specialization.
#check (∀ {Color : Type w} {Index : Type u} {E : Type v}
    [DecidableEq Color] [NormedAddCommGroup E] [InnerProductSpace ℝ E] [CompleteSpace E]
    (N : Color → ℕ) (f : ((Color × Index) →₀ ℕ) →₀ E),
    (cutoffSpace (E := E) (colorLow N)).toSubmodule.starProjection (realize f) =
      realize (cut (colorLow N) f))

end StratonovichHAMAudit.VectorProjection
